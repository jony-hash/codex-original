def _row_indices(cursor: int) -> torch.Tensor:
    _require(0 <= cursor < UPDATE_COUNT, "small-bank update cursor is out of range")
    batch = cursor % FIT_BATCHES
    return torch.arange(batch * BATCH_SIZE, (batch + 1) * BATCH_SIZE, dtype=torch.long)


def _expected_row_counts(cursor: int) -> list[int]:
    _require(0 <= cursor <= UPDATE_COUNT, "small-bank recovery cursor is out of range")
    counts = [0] * FIT_ROWS
    for update in range(cursor):
        start = (update % FIT_BATCHES) * BATCH_SIZE
        for index in range(start, start + BATCH_SIZE):
            counts[index] += 1
    return counts


def _classification(labels: torch.Tensor, predictions: torch.Tensor) -> dict[str, Any]:
    labels = torch.as_tensor(labels, dtype=torch.long, device="cpu").reshape(-1)
    predictions = torch.as_tensor(predictions, dtype=torch.long, device="cpu").reshape(-1)
    _require(labels.shape == predictions.shape and labels.numel() > 0, "classification row vectors differ")
    _require(bool(((labels >= 0) & (labels < 10) & (predictions >= 0) & (predictions < 10)).all().item()), "classification contains an invalid digit")
    confusion = torch.bincount(labels * 10 + predictions, minlength=100).reshape(10, 10)
    counts = confusion.sum(1)
    _require(bool((counts > 0).all().item()), "classification population omits a class")
    recalls = confusion.diag().to(torch.float64) / counts
    histogram = confusion.sum(0)
    return {
        "rows": int(labels.numel()),
        "macro_recall": float(recalls.mean().item()),
        "per_class_recall": recalls.tolist(),
        "minimum_class_recall": float(recalls.min().item()),
        "confusion_matrix": confusion.tolist(),
        "prediction_histogram": histogram.tolist(),
        "class8_prediction_mass": float(histogram[8].to(torch.float64).item() / labels.numel()),
        "true_labels": labels.tolist(),
        "prediction_labels": predictions.tolist(),
    }


def _class_balanced_positions(labels: torch.Tensor, *, rows_per_class: int) -> torch.Tensor:
    labels = torch.as_tensor(labels, dtype=torch.long, device="cpu").reshape(-1)
    _require(rows_per_class > 0 and bool(((labels >= 0) & (labels < 10)).all().item()), "S1 probe labels are invalid")
    positions: list[torch.Tensor] = []
    for digit in range(10):
        rows = torch.nonzero(labels == digit, as_tuple=False).flatten()
        _require(rows.numel() >= rows_per_class, "S1 probe population lacks every requested class at the fixed allocation")
        positions.append(rows[:rows_per_class])
    return torch.cat(positions)


def _resource_summary_from_entries(entries: Sequence[Mapping[str, Any]]) -> dict[str, Any]:
    _require(bool(entries), "resource telemetry is empty")
    required = {
        "global_used_fraction",
        "global_used_bytes",
        "global_total_bytes",
        "torch_peak_allocated_bytes",
        "torch_peak_reserved_bytes",
    }
    for index, entry in enumerate(entries):
        missing = required.difference(entry)
        _require(not missing, f"resource telemetry entry {index} is missing fields: {sorted(missing)}")
    global_peak = max(entries, key=lambda entry: float(entry["global_used_fraction"]))
    return {
        "max_global_used_fraction": float(global_peak["global_used_fraction"]),
        "peak_global_used_bytes": int(global_peak["global_used_bytes"]),
        "device_total_bytes": int(global_peak["global_total_bytes"]),
        "peak_torch_allocated_bytes": max(int(entry["torch_peak_allocated_bytes"]) for entry in entries),
        "peak_torch_reserved_bytes": max(int(entry["torch_peak_reserved_bytes"]) for entry in entries),
        "resource_measurements_are_nonadditive": True,
    }


def _preflight_within_window(
    *,
    qualification_epoch: float,
    qualification_elapsed_before_seconds: float = 0.0,
    now: float | None = None,
) -> bool:
    current = time.time() if now is None else float(now)
    qualification_elapsed = float(qualification_elapsed_before_seconds) + current - float(qualification_epoch)
    return qualification_elapsed <= MAX_PREFLIGHT_SECONDS

def _bootstrap_scores(
    labels: torch.Tensor,
    predictions: torch.Tensor,
    *,
    generator: torch.Generator,
    replicates: int,
) -> torch.Tensor:
    labels = torch.as_tensor(labels, dtype=torch.long, device="cpu").reshape(-1)
    predictions = torch.as_tensor(predictions, dtype=torch.long, device="cpu").reshape(-1)
    class_scores: list[torch.Tensor] = []
    for digit in range(10):
        positions = torch.nonzero(labels == digit, as_tuple=False).flatten()
        _require(positions.numel() > 0, f"bootstrap population has no class {digit}")
        draws = torch.randint(positions.numel(), (replicates, positions.numel()), generator=generator)
        selected = positions.index_select(0, draws.reshape(-1)).reshape(replicates, -1)
        class_scores.append((predictions.index_select(0, selected.reshape(-1)).reshape(replicates, -1) == digit).to(torch.float64).mean(1))
    return torch.stack(class_scores, dim=1).mean(1)


def _paired_bootstrap(
    labels: torch.Tensor,
    before: torch.Tensor,
    after: torch.Tensor,
    *,
    replicates: int = BOOTSTRAP_REPLICATES,
    seed: int = BOOTSTRAP_SEED,
) -> dict[str, Any]:
    labels = torch.as_tensor(labels, dtype=torch.long, device="cpu")
    before = torch.as_tensor(before, dtype=torch.long, device="cpu")
    after = torch.as_tensor(after, dtype=torch.long, device="cpu")
    _require(labels.shape == before.shape == after.shape, "paired bootstrap vectors differ")
    generator = torch.Generator(device="cpu").manual_seed(seed)
    per_class: list[torch.Tensor] = []
    for digit in range(10):
        positions = torch.nonzero(labels == digit, as_tuple=False).flatten()
        _require(positions.numel() > 0, f"paired bootstrap population has no class {digit}")
        draws = torch.randint(positions.numel(), (replicates, positions.numel()), generator=generator)
        selected = positions.index_select(0, draws.reshape(-1)).reshape(replicates, -1)
        per_class.append(
            (after.index_select(0, selected.reshape(-1)).reshape(replicates, -1) == digit).to(torch.float64).mean(1)
            - (before.index_select(0, selected.reshape(-1)).reshape(replicates, -1) == digit).to(torch.float64).mean(1)
        )
    deltas = torch.stack(per_class, dim=1).mean(1)
    return {
        "method": "class-stratified paired bootstrap of macro recall",
        "replicates": replicates,
        "seed": seed,
        "delta": float(_classification(labels, after)["macro_recall"] - _classification(labels, before)["macro_recall"]),
        "ci95": [float(x) for x in torch.quantile(deltas, torch.tensor([0.025, 0.975], dtype=torch.float64)).tolist()],
        "lower_bound_above_zero": bool(torch.quantile(deltas, torch.tensor(0.025, dtype=torch.float64)).item() > 0.0),
    }


def _independent_bootstrap(
    labels: torch.Tensor,
    before: torch.Tensor,
    after: torch.Tensor,
    *,
    replicates: int = BOOTSTRAP_REPLICATES,
    seed: int = BOOTSTRAP_SEED,
) -> dict[str, Any]:
    labels = torch.as_tensor(labels, dtype=torch.long, device="cpu")
    before = torch.as_tensor(before, dtype=torch.long, device="cpu")
    after = torch.as_tensor(after, dtype=torch.long, device="cpu")
    _require(labels.shape == before.shape == after.shape, "independent bootstrap vectors differ")
    generator = torch.Generator(device="cpu").manual_seed(seed)
    before_scores = _bootstrap_scores(labels, before, generator=generator, replicates=replicates)
    after_scores = _bootstrap_scores(labels, after, generator=generator, replicates=replicates)
    deltas = after_scores - before_scores
    ci = torch.quantile(deltas, torch.tensor([0.025, 0.975], dtype=torch.float64))
    return {
        "method": "independent class-stratified bootstrap; no row pairing asserted",
        "replicates": replicates,
        "seed": seed,
        "delta": float(_classification(labels, after)["macro_recall"] - _classification(labels, before)["macro_recall"]),
        "ci95": [float(x) for x in ci.tolist()],
        "lower_bound_above_zero": bool(ci[0].item() > 0.0),
    }


def _classify_gate(*, fixed: bool, transfer: bool, safety: bool) -> tuple[str, str]:
    if not safety:
        return SAFETY_FAILURE, RUN_VALID_FAILED
    if fixed and transfer:
        return CASE_BOTH, RUN_COMPLETED
    if fixed:
        return CASE_FIXED, RUN_VALID_FAILED
    if transfer:
        return CASE_TRANSFER, RUN_VALID_FAILED
    return CASE_NEITHER, RUN_VALID_FAILED


def _parent_report_refs(report: Mapping[str, Any], names: Sequence[str]) -> dict[str, dict[str, Any]]:
    artifacts = report.get("artifacts")
    _require(isinstance(artifacts, Mapping), "accepted step1200 report lacks artifact identities")
    result: dict[str, dict[str, Any]] = {}
    for name in names:
        record = artifacts.get(name)
        _require(isinstance(record, Mapping) and isinstance(record.get("sha256"), str), f"accepted step1200 report lacks {name} identity")
        result[name] = dict(record)
    return result


def _validate_fixed_population_identity(
    context: Any,
    payload: Mapping[str, Any],
    *,
    expected_fit: torch.Tensor,
    expected_observation: torch.Tensor,
) -> dict[str, torch.Tensor]:
    fit_indices = _tensor(payload.get("fit_indices"), "V2 fit indices").to(dtype=torch.long, device="cpu")
    observation_indices = _tensor(payload.get("observation_indices"), "V2 observation indices").to(dtype=torch.long, device="cpu")
    fit_labels = _tensor(payload.get("fit_labels"), "V2 fit labels").to(dtype=torch.long, device="cpu")
    observation_labels = _tensor(payload.get("observation_labels"), "V2 observation labels").to(dtype=torch.long, device="cpu")
    _require(
        fit_indices.shape == (FIT_ROWS,)
        and observation_indices.shape == (OBSERVATION_ROWS,)
        and torch.equal(fit_indices, expected_fit)
        and torch.equal(observation_indices, expected_observation)
        and not bool(torch.isin(fit_indices, observation_indices).any().item())
        and bool(torch.isin(fit_indices, context.train_indices).all().item())
        and bool(torch.isin(observation_indices, context.train_indices).all().item()),
        "V2 fit/observation row identities or TRAIN role changed",
    )
    _require(
        torch.equal(fit_labels, context.labels.index_select(0, fit_indices))
        and torch.equal(observation_labels, context.labels.index_select(0, observation_indices)),
        "fixed fit/observation labels differ from canonical TRAIN rows",
    )
    class_order = torch.arange(10, dtype=torch.long)
    _require(
        torch.equal(fit_labels.reshape(FIT_BATCHES, 10), class_order.expand(FIT_BATCHES, 10))
        and torch.equal(observation_labels.reshape(OBSERVATION_ROWS // 10, 10), class_order.expand(OBSERVATION_ROWS // 10, 10)),
        "fixed populations are not in the accepted class-interleaved order",
    )
    fit_images = context.images.index_select(0, fit_indices).contiguous()
    observation_images = context.images.index_select(0, observation_indices).contiguous()
    fit_targets = F.adaptive_avg_pool2d(fit_images.clamp(0.0, 1.0), (28, 28)).contiguous()
    observation_targets = F.adaptive_avg_pool2d(observation_images.clamp(0.0, 1.0), (28, 28)).contiguous()
    _require(
        torch.equal(fit_targets, _tensor(payload.get("fit_targets_28x28"), "V2 fit targets")),
        "fixed fit target-image identity changed",
    )
    return {
        "fit_indices": fit_indices,
        "observation_indices": observation_indices,
        "fit_labels": fit_labels,
        "observation_labels": observation_labels,
        "fit_images": fit_images,
        "observation_images": observation_images,
        "fit_targets": fit_targets,
        "observation_targets": observation_targets,
    }


def _balanced_dev_probe(context: Any, *, rows_per_class: int = 2) -> torch.Tensor:
    indices = torch.as_tensor(context.development_indices, dtype=torch.long, device="cpu")
    labels = context.labels.index_select(0, indices).to(dtype=torch.long, device="cpu")
    groups = [indices[labels == digit][:rows_per_class] for digit in range(10)]
    _require(all(group.numel() == rows_per_class for group in groups), "DEV qualification probe cannot supply every class")
    return torch.cat(groups).contiguous()

def _validate_historical_preflight(path: Path, payload: Mapping[str, Any]) -> None:
    budget_admission = path.name == "preflight_08.json"
    prior_ready = path.name == "preflight_09.json"
    previous_auth = path.name in {"preflight_09.json", "preflight_10.json", "preflight_11.json"}
    previous_budget_block = path.name == "preflight_11.json"
    expected_status = (
        "PREFLIGHT_BLOCKED(resource)"
        if budget_admission
        else "PREFLIGHT_BLOCKED(resource)"
        if previous_budget_block
        else PREFLIGHT_READY
        if prior_ready
        else RUN_INVALID_HARNESS
    )
    phase = payload.get("phase")
    valid_phase = (
        phase == "cumulative_preflight_budget_admission"
        if budget_admission
        else phase == "remaining_total_budget_admission"
        if previous_budget_block
        else phase is None
        if prior_ready
        else phase == "resolve_authorities_and_fixed_populations"
        if path.name in {"preflight_01.json", "preflight_10.json"}
        else phase in CUDA_PREFLIGHT_PHASES
    )
    _require(
        payload.get("schema") == PREFLIGHT_SCHEMA
        and payload.get("task_id") == TASK_ID
        and payload.get("outer_gate") == OUTER_GATE
        and payload.get("authorization_id") == (PREVIOUS_AUTHORIZATION_ID if previous_auth else HISTORICAL_AUTHORIZATION_ID)
        and payload.get("status") == expected_status
        and payload.get("attempt_created") is False
        and _has_zero_scientific_updates(payload)
        and valid_phase,
        f"prior {path.name} is not a pre-attempt record under its historical task identity",
    )


def _has_zero_scientific_updates(payload: Mapping[str, Any]) -> bool:
    if payload.get("scientific_optimizer_updates") == 0:
        return True
    checks = payload.get("checks")
    if not isinstance(checks, Mapping):
        return False
    if checks.get("scientific_optimizer_updates") == 0:
        return True
    correctness = checks.get(CORRECTNESS_INVARIANTS)
    return isinstance(correctness, Mapping) and correctness.get("scientific_optimizer_updates") == 0


def _validate_failed_attempt_without_updates(payload: Mapping[str, Any], *, source_manifest_id: str) -> None:
    _require(
        payload.get("schema") == REPORT_SCHEMA
        and payload.get("task_id") == TASK_ID
        and payload.get("outer_gate") == OUTER_GATE
        and payload.get("authorization_id") == PREVIOUS_AUTHORIZATION_ID
        and payload.get("source_manifest_id") == source_manifest_id
        and payload.get("state") == RUN_INVALID_HARNESS
        and payload.get("phase") == "STEP0_MEASUREMENT"
        and payload.get("local_completed_updates") == 0
        and payload.get("parent_global_step") == PARENT_STEP
        and payload.get("failure") == "CanaryError: accepted DEV labels is not a tensor",
        "attempt_01 is not the recorded zero-update DEV-label harness failure",
    )


def _accepted_dev_vectors(parent_dev: Mapping[str, Any]) -> tuple[torch.Tensor, torch.Tensor]:
    ed = parent_dev.get("E_D")
    _require(isinstance(ed, Mapping), "accepted step1200 DEV E_D measurement is missing")
    labels = torch.as_tensor(ed.get("true_labels"), dtype=torch.long, device="cpu")
    predictions = torch.as_tensor(ed.get("prediction_labels"), dtype=torch.long, device="cpu")
    _require(
        labels.shape == predictions.shape == (DEV_ROWS,)
        and bool(((labels >= 0) & (labels < 10) & (predictions >= 0) & (predictions < 10)).all().item()),
        "accepted step1200 DEV label or prediction vectors are malformed",
    )
    return labels, predictions


def _validate_step0_dev_reproduction(
    step0_measurement: Mapping[str, Any],
    accepted_dev: Mapping[str, Any],
) -> dict[str, Any]:
    accepted_labels, accepted_predictions = _accepted_dev_vectors(accepted_dev)
    measured_labels = _tensor(step0_measurement.get("dev_labels"), "saved step1200 DEV labels").to(device="cpu", dtype=torch.long)
    measured_predictions = _tensor(step0_measurement.get("dev_predictions"), "saved step1200 DEV predictions").to(device="cpu", dtype=torch.long)
    metrics = step0_measurement.get("dev_metrics")
    accepted_ed = accepted_dev.get("E_D")
    _require(isinstance(metrics, Mapping) and isinstance(metrics.get("metrics"), Mapping) and isinstance(accepted_ed, Mapping), "step1200 DEV metrics are malformed")
    reconstruction_delta = abs(float(metrics["L_rec"]) - float(accepted_dev["L_rec"]))
    macro_delta = abs(float(metrics["metrics"]["macro_recall"]) - float(accepted_ed["macro_recall"]))
    predictions_equal = torch.equal(measured_predictions, accepted_predictions)
    _require(
        measured_labels.shape == measured_predictions.shape == (DEV_ROWS,)
        and torch.equal(measured_labels, accepted_labels)
        and predictions_equal
        and reconstruction_delta <= 1.0e-6
        and macro_delta <= 1.0e-12,
        "full step1200 baseline DEV predictions or reconstruction did not reproduce the accepted authority",
    )
    return {
        "prediction_vector_equal": predictions_equal,
        "labels_equal": True,
        "reconstruction_absolute_delta": reconstruction_delta,
        "macro_recall_absolute_delta": macro_delta,
    }


def _source_manifest_compatible(current: Mapping[str, Any], ready: Mapping[str, Any]) -> bool:

        "relevant_source_identity",
        "relevant_source_paths",

        all(key in current and key in ready for key in required)
        and all(current[key] == ready[key] for key in required),
        "current READY source semantics or input identity changed",
    )
    return True

def _build_runtime(root: Path, inputs: Mapping[str, Any], *, device: torch.device) -> tuple[Any, torch.optim.Optimizer, tuple[torch.nn.Module, ...]]:
    context = inputs["context"]
    external_references, external_weights = readout._step600_external_weights(root, context.data_identity)
    consumer = readout._build_consumer(
        inputs["checkpoint"],
        device=device,
        expected_data_identity=context.data_identity,
        external_references=external_references,
        external_weights=external_weights,
        expected_frozen_runtime_state_sha256=inputs["checkpoint"]["frozen_runtime_state_sha256"],
    )
    params = dict(consumer.runtime.named_parameters())
    _require(tuple(params) == ("encoder.phase", "generator.phase"), "trainable parameter groups changed")
    for parameter in params.values():
        parameter.requires_grad_(True)
        parameter.grad = None
    optimizer = consumer.create_optimizer()
    optimizer.load_state_dict(copy.deepcopy(inputs["checkpoint"]["optimizer_state"]))
    _require(consumer.optimizer_manifest(optimizer) == inputs["checkpoint"]["optimizer_configuration"], "step1200 Adam configuration changed")
    _require(_optimizer_steps(optimizer) == [PARENT_STEP, PARENT_STEP], "restored scientific Adam does not start at step1200")
    evaluators = augmented._load_training_evaluators(root, torch.device("cpu"))
    _require(augmented._frozen_evaluators_clear(evaluators), "E_S/E_I/E_C are not frozen")
    for evaluator in evaluators:
        _require(all(parameter.device.type == "cpu" for parameter in evaluator.parameters()), "canonical evaluator must start on CPU")
    _require(
        all(not parameter.requires_grad and parameter.grad is None for parameter in context.independent_evaluator.parameters()),
        "E_D is trainable or has a gradient before any scientific update",
    )
    bogt.restore_rng_state(inputs["checkpoint"]["rng_state"])
    _require(consumer.frozen_state_sha256 == inputs["checkpoint"]["frozen_runtime_state_sha256"], "step1200 frozen optical runtime identity changed")
    return consumer, optimizer, evaluators


def _record_resource(recorder: Any, phase: str, *, step: int | None = None, started: float | None = None) -> dict[str, Any]:
    return recorder.sample(phase, batch_index=step, progress=None if step is None else {"completed_updates": step}, started=started)


def _measure_images(
    consumer: Any,
    context: Any,
    indices: torch.Tensor,
    *,
    role: str,
    phase: str,
    batch_size: int,
    recorder: Any,
    task_started_epoch: float,
    keep_raw: bool,
    qualification_started_epoch: float | None = None,
    qualification_elapsed_before_seconds: float = 0.0,
) -> dict[str, Any]:
    device = torch.device("cuda:0")
    indices = torch.as_tensor(indices, dtype=torch.long, device="cpu").contiguous()
    labels = context.labels.index_select(0, indices).to(torch.long).cpu()
    raw_parts: list[torch.Tensor] = []
    prediction_parts: list[torch.Tensor] = []
    reconstruction_sum = 0.0
    power_sum: torch.Tensor | None = None
    power_count = 0
    finite_images = zero_images = constant_images = 0
    raw_min = float("inf")
    raw_max = float("-inf")
    foreground_values: list[torch.Tensor] = []
    background_values: list[torch.Tensor] = []
    started = time.perf_counter()
    evaluator = context.independent_evaluator
    evaluator.eval()
    _require(all(not p.requires_grad and p.grad is None for p in evaluator.parameters()), "E_D gradient firewall was already dirty")
    for start in range(0, int(indices.numel()), batch_size):
        if qualification_started_epoch is not None:
            _check_preflight_deadlines(
                phase=phase,
                qualification_epoch=qualification_started_epoch,
                qualification_elapsed_before_seconds=qualification_elapsed_before_seconds,
            )
        batch_indices = indices[start : start + batch_size]
        images = context.images.index_select(0, batch_indices).to(device)
        with torch.no_grad():
            result = consumer.runtime.forward_from_image(images)
            raw = result.raw_intensity
            target_intensity = result.image_input.slm_intensity
            _require(raw.shape == (batch_indices.numel(), 1, 200, 200), "real-image raw CMOS geometry changed")
            _require(torch.isfinite(raw).all().item(), f"{phase} raw CMOS is nonfinite")
            reconstruction_sum += float((raw - target_intensity).square().sum().cpu())
            powers = bogt.contracted_plane_powers(result).detach().to(torch.float64).cpu()
            raw_cpu = raw.detach().to(torch.float32).cpu().contiguous()
        _require(not raw.requires_grad and raw.grad_fn is None, f"{phase} raw CMOS retained an autograd graph")
        flat = raw_cpu.reshape(raw_cpu.shape[0], -1)
        finite_images += int(torch.isfinite(flat).all(1).sum().item())
        row_min, row_max = flat.amin(1), flat.amax(1)
        zero_images += int((row_max == 0.0).sum().item())
        constant_images += int((row_min == row_max).sum().item())
        raw_min = min(raw_min, float(raw_cpu.min().item()))
        raw_max = max(raw_max, float(raw_cpu.max().item()))
        power_sum = powers.sum(0) if power_sum is None else power_sum + powers.sum(0)
        power_count += int(powers.shape[0])
        view = F.adaptive_avg_pool2d(raw_cpu.clamp(0.0, 1.0), (28, 28))
        targets = F.adaptive_avg_pool2d(context.images.index_select(0, batch_indices).clamp(0.0, 1.0), (28, 28))
        logits = evaluator(view)
        _require(torch.isfinite(logits).all().item() and not logits.requires_grad and logits.grad_fn is None, f"{phase} E_D logits are invalid or graph-attached")
        prediction_parts.append(logits.argmax(1).to(device="cpu"))
        _, foreground, background = render_v2.v1._canonical_pixel_loss(view, targets)
        foreground_values.append(foreground.detach().to(torch.float64).cpu())
        background_values.append(background.detach().to(torch.float64).cpu())
        if keep_raw:
            raw_parts.append(raw_cpu)
        context.ledger.record(role, int(batch_indices.numel()))
        del result, raw, target_intensity, powers, raw_cpu, flat, view, targets, logits, images
        if start % 100 == 0 or start + batch_size >= indices.numel():
            _record_resource(recorder, phase, step=start + int(batch_indices.numel()), started=started)
    predictions = torch.cat(prediction_parts)
    metrics = _classification(labels, predictions)
    foreground_mse = float(torch.stack(foreground_values).to(torch.float64).mean().item())
    background_mse = float(torch.stack(background_values).to(torch.float64).mean().item())
    powers_mean = (power_sum / max(power_count, 1)).tolist() if power_sum is not None else []
    _require(power_sum is not None and torch.isfinite(power_sum).all().item(), f"{phase} optical power is nonfinite")
    return {
        "labels": labels,
        "predictions": predictions,
        "metrics": metrics,
        "L_rec": reconstruction_sum / (int(indices.numel()) * 200 * 200),
        "pixel": {
            "foreground_mse": foreground_mse,
            "background_mse": background_mse,
            "loss": 0.5 * (foreground_mse + background_mse),
        },
        "raw_validity": {
            "rows": int(indices.numel()),
            "finite_image_count": finite_images,
            "zero_image_count": zero_images,
            "constant_image_count": constant_images,
            "all_finite": finite_images == indices.numel(),
            "all_nonzero": zero_images == 0,
            "all_nonconstant": constant_images == 0,
            "minimum": raw_min,
            "maximum": raw_max,
        },
        "optical_power": {"finite": True, "mean_by_plane": powers_mean},
        "data_role": role,
        "source_indices_sha256": tensor_sha256(indices),
        "elapsed_seconds": time.perf_counter() - started,
        "raw_cmos": torch.cat(raw_parts) if keep_raw else None,
    }


def _measure_ols(
    consumer: Any,
    coefficients: torch.Tensor,
    labels: torch.Tensor,
    context: Any,
    *,
    phase: str,
    recorder: Any,
    task_started_epoch: float,
    keep_raw: bool = True,
    qualification_started_epoch: float | None = None,
    qualification_elapsed_before_seconds: float = 0.0,
) -> dict[str, Any]:
    device = torch.device("cuda:0")
    count = int(coefficients.shape[0]) if coefficients.ndim == 2 else 0
    _require(count > 0 and count % 10 == 0 and coefficients.shape == (count, 100) and torch.is_complex(coefficients), "S1 OLS sample bank shape changed")
    labels = torch.as_tensor(labels, dtype=torch.long, device="cpu").contiguous()
    _require(labels.shape == (count,), "S1 label vector differs from the OLS sample bank")
    predictions: list[torch.Tensor] = []
    raw_parts: list[torch.Tensor] = []
    power_sum: torch.Tensor | None = None
    finite_images = zero_images = constant_images = 0
    raw_min = float("inf")
    raw_max = float("-inf")
    evaluator = context.independent_evaluator
    started = time.perf_counter()
    for start in range(0, count, 10):
        if qualification_started_epoch is not None:
            _check_preflight_deadlines(
                phase=phase,
                qualification_epoch=qualification_started_epoch,
                qualification_elapsed_before_seconds=qualification_elapsed_before_seconds,
            )
        context.ledger.record(bogt.TRAIN_ROLE, 10)
        z = coefficients[start : start + 10].to(device)
        with torch.inference_mode():
            result = consumer.runtime.forward_from_ols(z)
            raw = result.raw_intensity
            _require(raw.shape == (z.shape[0], 1, 200, 200), "S1 raw CMOS geometry changed")
            raw_cpu = raw.detach().to(torch.float32).cpu().contiguous()
            power = m0._generated_path_plane_powers(result).detach().to(torch.float64).cpu()
            view = F.adaptive_avg_pool2d(raw.clamp(0.0, 1.0), (28, 28))
            logits = evaluator(view.to("cpu"))
        _require(torch.isfinite(raw_cpu).all().item() and torch.isfinite(power).all().item(), f"{phase} raw CMOS or optical power is nonfinite")
        _require(not logits.requires_grad and logits.grad_fn is None and torch.isfinite(logits).all().item(), f"{phase} E_D result is invalid")
        flat = raw_cpu.reshape(raw_cpu.shape[0], -1)
        finite_images += int(torch.isfinite(flat).all(1).sum().item())
        row_min, row_max = flat.amin(1), flat.amax(1)
        zero_images += int((row_max == 0.0).sum().item())
        constant_images += int((row_min == row_max).sum().item())
        raw_min = min(raw_min, float(raw_cpu.min().item()))
        raw_max = max(raw_max, float(raw_cpu.max().item()))
        power_sum = power.sum(0) if power_sum is None else power_sum + power.sum(0)
        predictions.append(logits.argmax(1).cpu())
        if keep_raw:
            raw_parts.append(raw_cpu)
        del z, result, raw, raw_cpu, power, view, logits, flat
        if start % 100 == 0 or start + 10 >= count:
            _record_resource(recorder, phase, step=start + 10, started=started)
    predicted = torch.cat(predictions)
    metrics = _classification(labels, predicted)
    return {
        "labels": labels,
        "predictions": predicted,
        "metrics": metrics,
        "raw_validity": {
            "rows": count,
            "finite_image_count": finite_images,
            "zero_image_count": zero_images,
            "constant_image_count": constant_images,
            "all_finite": finite_images == count,
            "all_nonzero": zero_images == 0,
            "all_nonconstant": constant_images == 0,
            "minimum": raw_min,
            "maximum": raw_max,
        },
        "optical_power": {"finite": bool(power_sum is not None and torch.isfinite(power_sum).all().item()), "mean_by_plane": (power_sum / count).tolist() if power_sum is not None else []},
        "raw_cmos": torch.cat(raw_parts) if keep_raw else None,
        "elapsed_seconds": time.perf_counter() - started,
    }


def _save_artifact(path: Path, payload: Mapping[str, Any], *, source_id: str, artifact_kind: str) -> dict[str, Any]:
    sidecar = path.with_suffix(path.suffix + ".identity.json")
    if path.exists():
        _require(sidecar.is_file(), f"artifact exists without its identity record: {path.name}")
        record = _json_read(sidecar)
        _require(
            record.get("schema") == ARTIFACT_SCHEMA
            and record.get("source_manifest_id") == source_id
            and record.get("artifact_kind") == artifact_kind
            and record.get("sha256") == file_sha256(path)
            and record.get("size_bytes") == path.stat().st_size,
            f"existing artifact identity differs: {path.name}",
        )
        saved = safe_torch_load(path, expected_sha256=record["sha256"], map_location="cpu")
        _require(isinstance(saved, Mapping) and saved.get("source_manifest_id") == source_id and saved.get("artifact_kind") == artifact_kind, f"existing artifact payload identity differs: {path.name}")
        return {"path": path.name, "sha256": record["sha256"], "size_bytes": record["size_bytes"], "reused": True, "payload": dict(saved)}
    full_payload = {"schema": ARTIFACT_SCHEMA, "artifact_kind": artifact_kind, "source_manifest_id": source_id, **dict(payload)}
    atomic_torch_save_new(path, full_payload, artifact_kind=artifact_kind)
    record = {
        "schema": ARTIFACT_SCHEMA,
        "artifact_kind": artifact_kind,
        "source_manifest_id": source_id,
        "path": path.name,
        "sha256": file_sha256(path),
        "size_bytes": path.stat().st_size,
    }
    atomic_write_new_json(sidecar, record, artifact_kind=f"{artifact_kind} identity")
    return {"path": path.name, "sha256": record["sha256"], "size_bytes": record["size_bytes"], "reused": False, "payload": full_payload}

def _batch_update(
    consumer: Any,
    optimizer: torch.optim.Optimizer,
    evaluators: Sequence[torch.nn.Module],
    context: Any,
    fit_images: torch.Tensor,
    cursor: int,
    *,
    recorder: Any,
) -> dict[str, Any]:
    rows = _row_indices(cursor)
    device = torch.device("cuda:0")
    batch = fit_images.index_select(0, rows).to(device)
    context.ledger.record(bogt.TRAIN_ROLE, BATCH_SIZE)
    vectors = augmented._compute_gradient_vectors(
        consumer,
        batch,
        smooth=evaluators[0],
        independent=evaluators[1],
        ec=evaluators[2],
        recorder=recorder,
        batch_index=cursor,
        phase_prefix="small_bank_training",
    )
    update = augmented._apply_update_vectors(
        consumer,
        optimizer,
        G_BOGT=vectors["G_BOGT"],
        G_canonical=vectors["G_canonical"],
        kappa=SCIENTIFIC_KAPPA,
        recorder=recorder,
        batch_index=cursor,
        phase_prefix="small_bank_training",
    )
    optimizer.zero_grad(set_to_none=True)
    _require(all(parameter.grad is None for parameter in consumer.runtime.parameters()), "training gradients were retained across updates")
    _require(augmented._frozen_evaluators_clear(evaluators), "canonical evaluator gradients were retained")
    _require(all(parameter.grad is None and not parameter.requires_grad for parameter in context.independent_evaluator.parameters()), "E_D entered the training gradient path")
    record = {
        "local_update": cursor + 1,
        "parent_global_step": PARENT_STEP + cursor + 1,
        "batch_index": cursor % FIT_BATCHES,
        "fit_row_indices": rows.tolist(),
        "fit_row_indices_sha256": tensor_sha256(rows),
        **vectors["losses"],
        "G_BOGT_norm": update["G_BOGT_norm"],
        "kappa_G_canonical_norm": update["kappa_G_canonical_norm"],
        "cosine_G_BOGT_G_canonical": update["cosine_G_BOGT_G_canonical"],
        "group_update_norms": dict(update["group_update_norms"]),
    }
    del batch, vectors, update
    return record

        "parent_checkpoint_sha256": inputs["checkpoint_sha256"],

        "row_update_counts": list(row_counts),
        "training_history": [dict(record) for record in history],
        "read_ledger": inputs["context"].ledger.to_dict(),
        "frozen_runtime_state_sha256": consumer.frozen_state_sha256,
        "elapsed_seconds_from_first_cuda_preflight": time.time() - first_cuda_epoch,
    }


def _validate_recovery_payload(payload: Mapping[str, Any], *, source_id: str, parent_hash: str, data_identity: Mapping[str, Any]) -> int:
    cursor = payload.get("cursor")
    _require(
        payload.get("schema") == CHECKPOINT_SCHEMA
        and payload.get("task_id") == TASK_ID
        and payload.get("outer_gate") == OUTER_GATE
        and payload.get("authorization_id") == AUTHORIZATION_ID
        and payload.get("source_manifest_id") == source_id
        and payload.get("parent_checkpoint_sha256") == parent_hash
        and payload.get("data_identity") == dict(data_identity)
        and isinstance(cursor, int)
        and 0 <= cursor <= UPDATE_COUNT
        and payload.get("parent_global_step") == PARENT_STEP + cursor
        and len(payload.get("row_update_counts", [])) == FIT_ROWS
        and payload.get("row_update_counts") == _expected_row_counts(cursor)
        and len(payload.get("training_history", [])) == cursor
        and payload.get("optimizer_step_vector") == [PARENT_STEP + cursor, PARENT_STEP + cursor],
        "small-bank recovery identity, row schedule, cursor, history, or Adam step is invalid",
    )
    return cursor


def _restore_recovery(
    payload: Mapping[str, Any],
    consumer: Any,
    optimizer: torch.optim.Optimizer,
    inputs: Mapping[str, Any],
    *,
    source_id: str,
) -> tuple[int, list[dict[str, Any]], list[int]]:
    cursor = _validate_recovery_payload(
        payload,
        source_id=source_id,
        parent_hash=inputs["checkpoint_sha256"],
        data_identity=inputs["context"].data_identity,
    )
    consumer.load_trainable_state_dict(payload["trainable_parameter_state"])
    optimizer.load_state_dict(copy.deepcopy(payload["optimizer_state"]))
    _require(consumer.optimizer_manifest(optimizer) == payload.get("optimizer_configuration"), "recovered Adam configuration changed")
    bogt.restore_rng_state(payload["rng_state"])
    _require(_optimizer_steps(optimizer) == [PARENT_STEP + cursor, PARENT_STEP + cursor], "recovered Adam and small-bank cursor differ")
    _require(consumer.frozen_state_sha256 == payload.get("frozen_runtime_state_sha256"), "recovered frozen optical identity changed")
    for parameter in consumer.runtime.parameters():
        parameter.grad = None
    optimizer.zero_grad(set_to_none=True)
    read_ledger = payload.get("read_ledger")
    _require(isinstance(read_ledger, Mapping), "recovery lacks its data-role ledger")
    inputs["context"].ledger.counts = {str(k): int(v) for k, v in read_ledger.items()}
    _require(all(parameter.grad is None for parameter in consumer.runtime.parameters()), "recovery restoration left trainable gradients")
    return cursor, [dict(record) for record in payload["training_history"]], list(payload["row_update_counts"])


def _apply_counterfactual(
    consumer: Any,
    optimizer: torch.optim.Optimizer,
    bogt_gradient: Sequence[torch.Tensor],
    canonical_gradient: Sequence[torch.Tensor],
    *,
    include_bogt: bool,
    include_canonical: bool,
) -> dict[str, torch.Tensor]:
    params = augmented._parameter_tuple(consumer)
    optimizer.zero_grad(set_to_none=True)
    vectors: list[torch.Tensor] = []
    for left, right in zip(bogt_gradient, canonical_gradient, strict=True):
        value = (left if include_bogt else torch.zeros_like(left)) + (SCIENTIFIC_KAPPA * right if include_canonical else torch.zeros_like(right))
        vectors.append(value)
    augmented._set_gradients(params, vectors)
    optimizer.step()
    after = augmented._trainable_state(consumer)
    optimizer.zero_grad(set_to_none=True)
    return after


def _validate_starting_optimizer(checkpoint: Mapping[str, Any], optimizer: torch.optim.Optimizer) -> dict[str, Any]:
    before = _optimizer_steps(optimizer)
    _require(before == [PARENT_STEP, PARENT_STEP], "first scientific update must begin from the accepted Adam step 1200")
    return {
        "parent_global_step_before_first_update": PARENT_STEP,
        "adam_step_before_first_update": before,
        "learning_rate": checkpoint.get("optimizer_configuration", {}).get("learning_rate", 0.001),
        "optimizer_configuration": checkpoint.get("optimizer_configuration"),
    }


def _counterfactual_tests(consumer: Any, optimizer: torch.optim.Optimizer, evaluators: Sequence[torch.nn.Module], context: Any, fit_images: torch.Tensor, recorder: Any) -> dict[str, Any]:
    parameters_before = augmented._trainable_state(consumer)
    optimizer_before = copy.deepcopy(optimizer.state_dict())
    rng_before = bogt._rng_state_payload()
    rows = _row_indices(0)
    batch = fit_images.index_select(0, rows).to(consumer.device)
    vectors = augmented._compute_gradient_vectors(
        consumer,
        batch,
        smooth=evaluators[0],
        independent=evaluators[1],
        ec=evaluators[2],
        recorder=recorder,
        batch_index=0,
        phase_prefix="preflight_negative_control",
    )
    result_states: dict[str, dict[str, torch.Tensor]] = {}
    for name, include_bogt, include_canonical in (
        ("full", True, True),
        ("bogt_off", False, True),
        ("canonical_off", True, False),
    ):
        consumer.load_trainable_state_dict(parameters_before)
        optimizer.load_state_dict(copy.deepcopy(optimizer_before))
        bogt.restore_rng_state(rng_before)
        optimizer.zero_grad(set_to_none=True)
        result_states[name] = _apply_counterfactual(
            consumer,
            optimizer,
            vectors["G_BOGT"],
            vectors["G_canonical"],
            include_bogt=include_bogt,
            include_canonical=include_canonical,
        )
    differences = {
        control: {
            name: float(torch.linalg.vector_norm((result_states["full"][name] - state[name]).to(torch.float64)).item())
            for name in result_states["full"]
        }
        for control, state in (("bogt_off", result_states["bogt_off"]), ("canonical_off", result_states["canonical_off"]))
    }
    _require(all(math.isfinite(v) and v > 0.0 for by_group in differences.values() for v in by_group.values()), "turning off either BOGT or canonical direction did not change both real parameter updates")
    consumer.load_trainable_state_dict(parameters_before)
    optimizer.load_state_dict(optimizer_before)
    bogt.restore_rng_state(rng_before)
    optimizer.zero_grad(set_to_none=True)
    _require(_tree_equal(parameters_before, augmented._trainable_state(consumer)), "negative control restoration changed parent parameters")
    _require(_tree_equal(optimizer_before, optimizer.state_dict()), "negative control restoration changed parent Adam state")
    _require(_tree_equal(rng_before, bogt._rng_state_payload()), "negative control restoration changed RNG")
    del batch, vectors, result_states
    return {"status": "PASS", "full_vs_bogt_off_parameter_delta_norms": differences["bogt_off"], "full_vs_canonical_off_parameter_delta_norms": differences["canonical_off"], "optimizer_restored": True, "RNG_restored": True}


def _preflight(root: Path) -> dict[str, Any]:
    _require(not PREFLIGHT_PATH.exists() and not ATTEMPT_DIR.exists(), "the write-once preflight or attempt path already exists")
    RUNS_DIR.mkdir(parents=True, exist_ok=True)
    _load_authorization(root, create_if_missing=True)
    invocation_started_epoch = time.time()
    qualification_started_epoch = invocation_started_epoch
    historical_budget_record = _json_read(root / PREVIOUS_READY_PREFLIGHT_PATH)
    previous_preflight_record = _json_read(root / PREVIOUS_PREFLIGHT_PATH)
    _require(
        historical_budget_record.get("status") == PREFLIGHT_READY
        and historical_budget_record.get("authorization_id") == PREVIOUS_AUTHORIZATION_ID
        and historical_budget_record.get("attempt_created") is False
        and _has_zero_scientific_updates(historical_budget_record),
        "preflight_09 is not the expected source-bound READY with no scientific updates",
    )
    _validate_historical_preflight(PREVIOUS_PREFLIGHT_PATH, previous_preflight_record)
    historical_preflight_seconds = float(previous_preflight_record["elapsed_seconds"])
    qualification_elapsed_before_seconds = 0.0
    device = torch.device("cuda:0")
    recorder = augmented.PhaseResourceRecorder(
        device,
        telemetry_path=root / PREFLIGHT_TELEMETRY_PATH,

            "resource_stop": PREFLIGHT_STOP_PATH.as_posix() if (root / PREFLIGHT_STOP_PATH).exists() else None,
            "elapsed_seconds": historical_preflight_seconds + elapsed,
            "historical_preflight_elapsed_seconds": historical_preflight_seconds,
            "authorization_03_preflight_elapsed_seconds": qualification_elapsed,
            "authorization_03_preflight_limit_seconds": MAX_PREFLIGHT_SECONDS,
            "current_preflight_elapsed_seconds": elapsed,
            "first_cuda_preflight_epoch_seconds": started_epoch,
            "repair_qualification_started_epoch_seconds": qualification_started_epoch,
            "source_manifest": locals().get("source"),
            "source_manifest_id": locals().get("source", {}).get("source_manifest_id") if isinstance(locals().get("source"), Mapping) else None,
            "execution_source_semantics_id": locals().get("source", {}).get("execution_source_semantics_id") if isinstance(locals().get("source"), Mapping) else None,
            "resource": locals().get("resource"),
            "attempt_created": False,
            "scientific_optimizer_updates": 0,
            "resource_phase_records": list(getattr(error, "resource_phase_records", recorder.entries)),
        }
        if not PREFLIGHT_PATH.exists():
            atomic_write_new_json(root / PREFLIGHT_PATH, record, artifact_kind="M0 small-bank resource-blocked preflight")
        return record
    except Exception as error:
        elapsed = time.time() - qualification_started_epoch
        qualification_elapsed = qualification_elapsed_before_seconds + elapsed
        record = {
            "schema": PREFLIGHT_SCHEMA,
            "task_id": TASK_ID,
            "outer_gate": OUTER_GATE,
            "authorization_id": AUTHORIZATION_ID,
            "status": RUN_INVALID_HARNESS,
            "phase": phase,
            "reason": f"{type(error).__name__}: {error}",
            "elapsed_seconds": historical_preflight_seconds + elapsed,
            "historical_preflight_elapsed_seconds": historical_preflight_seconds,
            "authorization_03_preflight_elapsed_seconds": qualification_elapsed,
            "authorization_03_preflight_limit_seconds": MAX_PREFLIGHT_SECONDS,
            "current_preflight_elapsed_seconds": elapsed,
            "first_cuda_preflight_epoch_seconds": started_epoch,
            "repair_qualification_started_epoch_seconds": qualification_started_epoch,
            "source_manifest": locals().get("source"),
            "source_manifest_id": locals().get("source", {}).get("source_manifest_id") if isinstance(locals().get("source"), Mapping) else None,
            "execution_source_semantics_id": locals().get("source", {}).get("execution_source_semantics_id") if isinstance(locals().get("source"), Mapping) else None,
            "resource": locals().get("resource"),
            "attempt_created": False,
            "scientific_optimizer_updates": 0,
        }
        if not PREFLIGHT_PATH.exists():
            atomic_write_new_json(root / PREFLIGHT_PATH, record, artifact_kind="M0 small-bank invalid-harness preflight")
        return record
    finally:
        del consumer, optimizer, evaluators, inputs
        gc.collect()
        if torch.cuda.is_available():
            torch.cuda.empty_cache()


def _validate_preflight(
    root: Path,
    inputs: Mapping[str, Any],
    source: Mapping[str, Any],
    *,
    allow_existing_attempt: bool = False,
) -> tuple[dict[str, Any], PreflightReceipt, float]:
    preflight = _json_read(root / PREFLIGHT_PATH)
    ready_source = preflight.get("source_manifest")
    _require(
        preflight.get("schema") == PREFLIGHT_SCHEMA
        and preflight.get("task_id") == TASK_ID
        and preflight.get("outer_gate") == OUTER_GATE
        and preflight.get("authorization_id") == AUTHORIZATION_ID
        and preflight.get("authorization_token") == AUTHORIZATION_TOKEN
        and preflight.get("status") == PREFLIGHT_READY
        and isinstance(ready_source, Mapping)
        and preflight.get("source_manifest_id") == ready_source.get("source_manifest_id")
        and preflight.get("execution_source_semantics_id") == ready_source.get("execution_source_semantics_id")
        and _source_manifest_compatible(source, ready_source)
        and (root / SOURCE_MANIFEST_PATH).is_file()
        and _json_read(root / SOURCE_MANIFEST_PATH) == ready_source
        and (allow_existing_attempt or not ATTEMPT_DIR.exists()),
        "current source-bound READY receipt does not authorize a clean attempt_01",
    )
    receipt = PreflightReceipt.from_mapping(preflight["receipt"])
    _require(receipt.status == PREFLIGHT_READY and receipt.source_semantics_id == ready_source["execution_source_semantics_id"], "preflight receipt is not source-bound READY")
    _require(receipt.authorization_id == AUTHORIZATION_ID, "preflight authorization identity changed")
    started_epoch = float(preflight["first_cuda_preflight_epoch_seconds"])
    _load_authorization(root, create_if_missing=False)
    _require(inputs["checkpoint_sha256"] == preflight["checks"]["parent_checkpoint_and_fixed_population_identity"]["step1200_checkpoint"], "accepted step1200 checkpoint changed after READY")
    return preflight, receipt, started_epoch


def _write_status(path: Path, payload: Mapping[str, Any]) -> None:
    atomic_write_json(path, dict(payload))


def _checkpoint(
    manager: RecoveryCheckpointManager,
    consumer: Any,
    optimizer: torch.optim.Optimizer,
    inputs: Mapping[str, Any],
    source: Mapping[str, Any],
    cursor: int,
    row_counts: Sequence[int],
    history: Sequence[Mapping[str, Any]],
    *,
    phase: str,
    started_epoch: float,
) -> Path:
    state = _recovery_payload(
        consumer,
        optimizer,
        inputs,
        source,
        cursor,
        row_counts,
        history,
        phase=phase,
        first_cuda_epoch=started_epoch,
    )
    path = manager.save({"scientific_state": state}, step=cursor)
    return path


def _load_latest_recovery(
    manager: RecoveryCheckpointManager,
    consumer: Any,
    optimizer: torch.optim.Optimizer,
    inputs: Mapping[str, Any],
    source: Mapping[str, Any],
    *,
    attempt_status: Mapping[str, Any],
) -> tuple[int, list[dict[str, Any]], list[int], str]:
    latest = manager.latest()
    _require(latest is not None, "interrupted attempt has no production recovery checkpoint")
    wrapper = manager.load(latest)
    _require(
        wrapper.get("source_manifest_id") == source["source_manifest_id"]
        and wrapper.get("authorization_id") == AUTHORIZATION_ID
        and wrapper.get("step") == wrapper.get("payload", {}).get("scientific_state", {}).get("cursor"),
        "latest recovery wrapper source, authorization, or cursor changed",
    )
    state = wrapper["payload"]["scientific_state"]
    cursor = _validate_recovery_payload(
        state,
        source_id=str(source["source_manifest_id"]),
        parent_hash=inputs["checkpoint_sha256"],
        data_identity=inputs["context"].data_identity,
    )
    old_phase = str(state.get("phase"))
    status_cursor = attempt_status.get("local_completed_updates", 0)
    _require(
        isinstance(status_cursor, int)
        and max(0, cursor - CHECKPOINT_INTERVAL) <= status_cursor <= cursor,
        "stale RUNNING status cursor is not covered by the most recent 25-step recovery",
    )
    control_path = ATTEMPT_DIR / "run.lock"
    _require(control_path.is_file(), "attempt lock record is missing during recovery")
    restored_cursor, history, counts = _restore_recovery(state, consumer, optimizer, inputs, source_id=str(source["source_manifest_id"]))
    return restored_cursor, history, counts, old_phase


def _sidecar_payload(path: Path, *, source_id: str, kind: str) -> Mapping[str, Any] | None:
    if not path.exists():
        return None
    sidecar = path.with_suffix(path.suffix + ".identity.json")
    _require(sidecar.is_file(), f"saved artifact has no identity sidecar: {path.name}")
    record = _json_read(sidecar)
    _require(
        record.get("schema") == ARTIFACT_SCHEMA
        and record.get("source_manifest_id") == source_id
        and record.get("artifact_kind") == kind
        and record.get("sha256") == file_sha256(path)
        and record.get("size_bytes") == path.stat().st_size,
        f"saved artifact identity does not match this attempt: {path.name}",
    )
    payload = safe_torch_load(path, expected_sha256=record["sha256"], map_location="cpu")
    _require(isinstance(payload, Mapping) and payload.get("source_manifest_id") == source_id and payload.get("artifact_kind") == kind, f"saved artifact payload identity changed: {path.name}")
    return payload

    identity: ResearchRunIdentity,
    receipt: PreflightReceipt,
    source: Mapping[str, Any],
) -> dict[str, Any]:

        control.mark_preflight_ready(receipt)
        effective = control.open_scientific_attempt(attempt, ATTEMPT_ID, receipt)

        )

def _collect_train_ols(
    consumer: Any,
    context: Any,
    attempt: Path,
    inputs: Mapping[str, Any],
    source: Mapping[str, Any],
    recorder: Any,
    task_started_epoch: float,
) -> tuple[torch.Tensor, dict[str, Any]]:
    bank_path = attempt / "train_ols_step2200.pt"
    existing = _sidecar_payload(bank_path, source_id=str(source["source_manifest_id"]), kind="full TRAIN OLS bank at step2200")
    if existing is not None:
        bank = _tensor(existing.get("z"), "saved step2200 TRAIN OLS bank")
        _require(
            bank.shape == (55_000, 100)
            and torch.is_complex(bank)
            and bank.dtype == torch.complex64
            and existing.get("data_identity") == context.data_identity
            and existing.get("selected_step") == END_STEP
            and existing.get("data_role") == bogt.TRAIN_ROLE
            and existing.get("parent_checkpoint_sha256") == inputs["checkpoint_sha256"]
            and torch.isfinite(bank.real).all().item()
            and torch.isfinite(bank.imag).all().item(),
            "saved step2200 TRAIN OLS bank is incomplete or has the wrong lineage",
        )
        return bank, {"rows": 55_000, "chunks": 55, "reused": True, "sha256": file_sha256(bank_path)}
    chunk_dir = attempt / "train_ols_chunks"
    chunk_dir.mkdir(exist_ok=True)
    pieces: list[torch.Tensor] = []
    started = time.perf_counter()
    for start in range(0, 55_000, 1000):
        path = chunk_dir / f"train_ols_chunk_{start:05d}.pt"
        old = _sidecar_payload(path, source_id=str(source["source_manifest_id"]), kind=f"TRAIN OLS chunk {start}:{start + 1000}")
        if old is not None:
            values = _tensor(old.get("z"), f"TRAIN OLS chunk {start}")
            _require(
                old.get("start_row") == start
                and old.get("data_identity") == context.data_identity
                and old.get("selected_step") == END_STEP
                and values.shape == (1000, 100)
                and values.dtype == torch.complex64
                and torch.isfinite(values.real).all().item()
                and torch.isfinite(values.imag).all().item(),
                f"TRAIN OLS chunk {start} lineage or shape changed",
            )
            pieces.append(values.contiguous())
            continue
        rows: list[torch.Tensor] = []
        for offset in range(start, start + 1000, 10):
            indices = context.train_indices[offset : offset + 10]
            rows.append(readout._encode_batch(consumer, context, indices))
        values = torch.cat(rows, 0).contiguous()
        _require(values.shape == (1000, 100) and values.dtype == torch.complex64, f"TRAIN OLS chunk {start} shape or dtype changed")
        _require(torch.isfinite(values.real).all().item() and torch.isfinite(values.imag).all().item(), f"TRAIN OLS chunk {start} is nonfinite")
        chunk_result = _save_artifact(
            path,
            {"z": values, "start_row": start, "data_identity": dict(context.data_identity), "selected_step": END_STEP, "data_role": bogt.TRAIN_ROLE, "parent_checkpoint_sha256": inputs["checkpoint_sha256"]},
            source_id=str(source["source_manifest_id"]),
            artifact_kind=f"TRAIN OLS chunk {start}:{start + 1000}",
        )
        pieces.append(_tensor(chunk_result["payload"]["z"], f"saved TRAIN OLS chunk {start}").contiguous())
        del rows, values
        gc.collect()
        torch.cuda.empty_cache()
        recorder.sample("scientific_full_train_ols_collection", progress={"completed_rows": start + 1000, "target_rows": 55_000}, started=started)
    bank = torch.cat(pieces, 0).contiguous()
    _require(bank.shape == (55_000, 100) and bank.dtype == torch.complex64 and torch.isfinite(bank.real).all().item() and torch.isfinite(bank.imag).all().item(), "full step2200 TRAIN OLS bank is invalid")
    record = _save_artifact(
        bank_path,
        {"z": bank, "data_identity": dict(context.data_identity), "selected_step": END_STEP, "data_role": bogt.TRAIN_ROLE, "row_order": "increasing canonical TRAIN source row order", "parent_checkpoint_sha256": inputs["checkpoint_sha256"], "rows": 55_000},
        source_id=str(source["source_manifest_id"]),
        artifact_kind="full TRAIN OLS bank at step2200",
    )
    return bank, {"rows": 55_000, "chunks": 55, "reused": record["reused"], "sha256": record["sha256"], "elapsed_seconds": time.perf_counter() - started, "graph_retained": False}


def _artifact_registry(attempt: Path, source_id: str) -> dict[str, dict[str, Any]]:
    output: dict[str, dict[str, Any]] = {}
    for path in sorted(attempt.rglob("*")):
        if not path.is_file() or path.name.endswith(".identity.json") or path.name == "artifact_manifest.json":
            continue
        sidecar = path.with_suffix(path.suffix + ".identity.json")
        if sidecar.is_file():
            record = _json_read(sidecar)
            _require(record.get("source_manifest_id") == source_id and record.get("sha256") == file_sha256(path), f"artifact manifest found an inconsistent file: {path.name}")
            output[path.relative_to(attempt).as_posix()] = {"sha256": record["sha256"], "size_bytes": record["size_bytes"], "artifact_kind": record["artifact_kind"]}
        else:
            output[path.relative_to(attempt).as_posix()] = {"sha256": file_sha256(path), "size_bytes": path.stat().st_size}
    return output


def _save_contact_sheet(path: Path, groups: Sequence[tuple[str, torch.Tensor, torch.Tensor]], *, samples_per_class: int = 10) -> dict[str, Any]:
    # Every panel shares the evaluator's fixed [0,1] display scale.
    columns = sum(samples_per_class for _, _, _ in groups)
    canvas = Image.new("L", (columns * 200, 10 * 220), color=0)
    draw = ImageDraw.Draw(canvas)
    column_offset = 0
    for title, images, labels in groups:
        draw.text((column_offset * 200 + 3, 2), title, fill=255)
        for digit in range(10):
            positions = torch.nonzero(labels.cpu() == digit, as_tuple=False).flatten()[:samples_per_class]
            _require(positions.numel() == samples_per_class, f"contact panel {title} lacks class {digit}")
            draw.text((2, digit * 220 + 3), str(digit), fill=255)
            for local_column, position in enumerate(positions.tolist()):
                image = images[position, 0].detach().cpu().to(torch.float32).clamp(0.0, 1.0)
                tile = Image.fromarray((image * 255.0).round().to(torch.uint8).numpy(), mode="L")
                canvas.paste(tile, ((column_offset + local_column) * 200, digit * 220 + 20))
        column_offset += samples_per_class
    with tempfile.NamedTemporaryFile(prefix=f".{path.name}.", suffix=".tmp", dir=path.parent, delete=False) as handle:
        temp_path = Path(handle.name)
        canvas.save(handle, format="PNG", optimize=True)
        handle.flush()
        os.fsync(handle.fileno())
    if path.exists():
        _require(file_sha256(path) == file_sha256(temp_path), f"existing contact sheet differs: {path.name}")
        temp_path.unlink(missing_ok=True)
    else:
        os.replace(temp_path, path)
    return {"path": path.name, "sha256": file_sha256(path), "size_bytes": path.stat().st_size, "common_vmin": 0.0, "common_vmax": 1.0, "per_image_rescaling": False}


def _artifact_report(path: Path) -> dict[str, Any]:
    return {"path": path.name, "sha256": file_sha256(path), "size_bytes": path.stat().st_size}


class _RunLock:
    def __init__(self, path: Path):
        self.path = path
        self.stream: Any = None

    def __enter__(self) -> "_RunLock":
        import msvcrt

        self.path.parent.mkdir(parents=True, exist_ok=True)
        self.stream = self.path.open("a+b")
        self.stream.seek(0, os.SEEK_END)
        if self.stream.tell() == 0:
            self.stream.write(b"0")
            self.stream.flush()
        self.stream.seek(0)
        try:
            msvcrt.locking(self.stream.fileno(), msvcrt.LK_NBLCK, 1)
        except OSError as error:
            self.stream.close()
            raise CanaryError("another process still owns the scientific attempt lock") from error
        payload = json.dumps({"pid": os.getpid(), "instance": str(uuid.uuid4()), "acquired_utc": datetime.now(timezone.utc).isoformat()}).encode("utf-8")
        self.stream.seek(1)
        self.stream.write(payload)
        self.stream.truncate()
        self.stream.flush()
        os.fsync(self.stream.fileno())
        return self

    def __exit__(self, exc_type: Any, exc: Any, traceback: Any) -> None:
        import msvcrt

        if self.stream is not None:
            self.stream.seek(0)
            msvcrt.locking(self.stream.fileno(), msvcrt.LK_UNLCK, 1)
            self.stream.close()


def _persist_training_checkpoint(
    attempt: Path,
    inputs: Mapping[str, Any],
    source: Mapping[str, Any],
    consumer: Any,
    optimizer: torch.optim.Optimizer,
    row_counts: Sequence[int],
    history: Sequence[Mapping[str, Any]],
    *,
    cursor: int,
) -> dict[str, Any]:
    path = attempt / f"scientific_step_{PARENT_STEP + cursor:06d}.pt"
    payload = _recovery_payload(
        consumer,
        optimizer,
        inputs,
        source,
        cursor,
        row_counts,
        history,
        phase="TRAINING_COMPLETE" if cursor == UPDATE_COUNT else "TRAINING",
        first_cuda_epoch=float(_json_read(_root() / PREFLIGHT_PATH)["first_cuda_preflight_epoch_seconds"]),
    )
    checkpoint = {"schema": CHECKPOINT_SCHEMA, "checkpoint_kind": "SCIENTIFIC_TRAINING_STATE", **payload}
    if path.exists():
        saved = safe_torch_load(path, map_location="cpu")
        _require(_tree_equal(saved, checkpoint), f"existing scientific checkpoint differs: {path.name}")
    else:
        atomic_torch_save_new(path, checkpoint, artifact_kind="M0 small-bank scientific checkpoint")
    return _artifact_report(path)


def _run(root: Path, *, resume: bool) -> dict[str, Any]:
    device = torch.device("cuda:0")
    inputs = _load_inputs(root)
    source = _source_manifest(root, inputs)
    preflight, receipt, started_epoch = _validate_preflight(root, inputs, source, allow_existing_attempt=resume)
    source = dict(preflight["source_manifest"])
    _require(torch.cuda.is_available() and torch.cuda.device_count() >= 1, "CUDA became unavailable after READY", augmented.ResourceBudgetExceeded)
    if resume:
        _require(ATTEMPT_DIR.is_dir(), "resume requested but attempt_01 does not exist")
        attempt_status = _json_read(ATTEMPT_DIR / "status.json")
        _require(attempt_status.get("state") in {RUNNING, RUN_INTERRUPTED}, "only a stale RUNNING or interrupted attempt can resume")
        _require(_json_read(ATTEMPT_DIR / "source_manifest.json") == source, "attempt source manifest differs from current READY")
        persisted_epoch = float(attempt_status.get("first_cuda_preflight_epoch_seconds", -1))
        _require(persisted_epoch == started_epoch, "resume lost the original preflight timing anchor")
        initial_state = RUN_INTERRUPTED if attempt_status.get("state") == RUN_INTERRUPTED else RUNNING
    else:
        _require(not ATTEMPT_DIR.exists(), "attempt_01 already exists; use --resume only for its interrupted state")
        attempt_status = {}
        initial_state = RUN_PREPARING
    identity = _identity(source, inputs)
    control = ResearchRunControl(
        identity,
        AUTHORIZATION_ID,
        preflight_status=PREFLIGHT_READY,
        state=initial_state,
        receipt=receipt,
    )
    if resume:
        effective_identity = control.resume_scientific_attempt(ATTEMPT_DIR, ATTEMPT_ID, receipt)
    else:
        effective_identity = control.open_scientific_attempt(ATTEMPT_DIR, ATTEMPT_ID, receipt)
    attempt = ATTEMPT_DIR
    if not (attempt / "source_manifest.json").exists():
        atomic_write_new_json(attempt / "source_manifest.json", source, artifact_kind="M0 small-bank attempt source manifest")
    _require(_json_read(attempt / "source_manifest.json") == source, "attempt source manifest changed after opening")
    recorder = augmented.PhaseResourceRecorder(
        device,
        telemetry_path=attempt / "resource_telemetry.jsonl",
        hard_limit=MAX_DEVICE_MEMORY_FRACTION,
        allow_existing_telemetry=resume,
    )
    consumer = optimizer = evaluators = None
    latest_phase = "PREPARING"
    lock_acquired = False
    process_instance = {"pid": os.getpid(), "nonce": str(uuid.uuid4())}
    try:
        with _RunLock(attempt / "run.lock"):
            lock_acquired = True
            if resume:
                old_instance = attempt_status.get("process_instance")
                _require(
                    isinstance(old_instance, Mapping)
                    and isinstance(old_instance.get("pid"), int)
                    and isinstance(old_instance.get("nonce"), str)
                    and bool(old_instance["nonce"]),
                    "stale RUNNING attempt lacks a process-instance record",
                )
            consumer, optimizer, evaluators = _build_runtime(root, inputs, device=device)
            adam_audit = _validate_starting_optimizer(inputs["checkpoint"], optimizer)
            parent_parameters = augmented._trainable_state(consumer)
            require_start_hash = {name: tensor_sha256(value) for name, value in parent_parameters.items()}
            _require(all(parameter.grad is None for parameter in consumer.runtime.parameters()), "scientific runtime began with gradient buffers")
            _require(consumer.frozen_state_sha256 == inputs["checkpoint"]["frozen_runtime_state_sha256"], "scientific runtime start frozen-state identity changed")
            _require(_optimizer_steps(optimizer) == [1200, 1200], "scientific Adam did not start at step1200")
            manager = RecoveryCheckpointManager(attempt / "recovery", effective_identity, source_manifest_id=str(source["source_manifest_id"]), authorization_id=AUTHORIZATION_ID, keep=3)
            heartbeat = HeartbeatWriter(attempt, RUN_ID, total_steps=UPDATE_COUNT)
            if resume:
                cursor, history, row_counts, saved_phase = _load_latest_recovery(
                    manager,
                    consumer,
                    optimizer,
                    inputs,
                    source,
                    attempt_status=attempt_status,
                )
                _require(attempt_status.get("parent_global_step") in {None, PARENT_STEP + int(attempt_status.get("local_completed_updates", 0))}, "stale attempt global step and local cursor disagree")
                if saved_phase in {"TRAINING_COMPLETE", "FINALIZING", "MEASUREMENT_COMPLETE", "ARTIFACTS_COMPLETE"}:
                    latest_phase = saved_phase

def _parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser(description=__doc__)
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--preflight", action="store_true")
    mode.add_argument("--run", action="store_true")
    mode.add_argument("--resume", action="store_true")
    parser.add_argument("--authorization-token", default=None)
    return parser.parse_args()


def main() -> int:
    args = _parse_args()
    root = _root()
    if args.authorization_token != AUTHORIZATION_TOKEN:
        raise SystemExit("explicit authorization token is required")
    if args.preflight:
        result = _preflight(root)
        print(json.dumps({k: result.get(k) for k in ("task_id", "status", "phase", "reason", "elapsed_seconds", "source_manifest_id")}, ensure_ascii=False, indent=2))
        return 0 if result.get("status") == PREFLIGHT_READY else 2
    try:
        result = _run(root, resume=args.resume)
        print(json.dumps({"task_id": TASK_ID, "status": result["result"], "classification": result["classification"], "report": (ATTEMPT_DIR / "report.json").as_posix()}, ensure_ascii=False, indent=2))
        return 0
    except augmented.ResourceBudgetExceeded as error:
        print(json.dumps({"task_id": TASK_ID, "status": RUN_INTERRUPTED, "reason": str(error)}, ensure_ascii=False, indent=2))
        return 2
    except Exception as error:
        print(json.dumps({"task_id": TASK_ID, "status": RUN_INVALID_HARNESS, "reason": f"{type(error).__name__}: {error}"}, ensure_ascii=False, indent=2))
        return 2


if __name__ == "__main__":
    raise SystemExit(main())
