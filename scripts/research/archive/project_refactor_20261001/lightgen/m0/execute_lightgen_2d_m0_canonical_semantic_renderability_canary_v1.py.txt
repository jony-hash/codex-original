class CanaryError(RuntimeError):
    """An authority, execution, or measurement invariant failed."""


class ResourceBudgetExceeded(CanaryError):
    """The fixed canary exceeded its time or device-memory envelope."""


def _require(condition: bool, message: str) -> None:
    if not condition:
        raise CanaryError(message)


def _mapping(value: object, name: str) -> Mapping[str, Any]:
    _require(isinstance(value, Mapping), f"{name} must be a mapping")
    return value


def _tensor(value: object, name: str) -> torch.Tensor:
    _require(isinstance(value, torch.Tensor), f"{name} must be a tensor")
    return value


def _finite(value: torch.Tensor, name: str) -> None:
    _require(bool(torch.isfinite(value).all().item()), f"{name} is nonfinite")


def _root() -> Path:
    return Path(__file__).resolve().parents[4]


def _relative(root: Path, path: Path) -> str:
    return path.resolve().relative_to(root.resolve()).as_posix()


def _json_read(path: Path) -> dict[str, Any]:
    _require(path.is_file(), f"required JSON authority is missing: {path}")
    try:
        value = json.loads(path.read_text(encoding="utf-8"))
    except (OSError, UnicodeDecodeError, json.JSONDecodeError) as error:
        raise CanaryError(f"required JSON authority cannot be read: {path}") from error
    return dict(_mapping(value, f"JSON authority {path.name}"))


def _json_equal(left: Any, right: Any) -> bool:
    if isinstance(left, Mapping) and isinstance(right, Mapping):
        return set(left) == set(right) and all(_json_equal(left[key], right[key]) for key in left)
    if isinstance(left, (list, tuple)) and isinstance(right, (list, tuple)):
        return len(left) == len(right) and all(_json_equal(a, b) for a, b in zip(left, right, strict=True))
    return left == right


def _tree_equal(left: Any, right: Any) -> bool:
    if isinstance(left, torch.Tensor) or isinstance(right, torch.Tensor):
        if not isinstance(left, torch.Tensor) or not isinstance(right, torch.Tensor):
            return False
        return (
            left.device.type == right.device.type
            and left.dtype == right.dtype
            and left.shape == right.shape
            and torch.equal(left, right)
        )
    if isinstance(left, Mapping) and isinstance(right, Mapping):
        return set(left) == set(right) and all(_tree_equal(left[key], right[key]) for key in left)
    if isinstance(left, (list, tuple)) and isinstance(right, (list, tuple)):
        return len(left) == len(right) and all(_tree_equal(a, b) for a, b in zip(left, right, strict=True))
    return left == right


def _tree_close(left: Any, right: Any, *, rtol: float, atol: float) -> bool:
    if isinstance(left, torch.Tensor) or isinstance(right, torch.Tensor):
        if not isinstance(left, torch.Tensor) or not isinstance(right, torch.Tensor):
            return False
        return (
            left.device.type == right.device.type
            and left.dtype == right.dtype
            and left.shape == right.shape
            and bool(torch.allclose(left, right, rtol=rtol, atol=atol, equal_nan=False))
        )
    if isinstance(left, Mapping) and isinstance(right, Mapping):
        return set(left) == set(right) and all(_tree_close(left[key], right[key], rtol=rtol, atol=atol) for key in left)
    if isinstance(left, (list, tuple)) and isinstance(right, (list, tuple)):
        return len(left) == len(right) and all(_tree_close(a, b, rtol=rtol, atol=atol) for a, b in zip(left, right, strict=True))
    if isinstance(left, numbers.Real) and isinstance(right, numbers.Real) and not isinstance(left, bool) and not isinstance(right, bool):
        return math.isclose(float(left), float(right), rel_tol=rtol, abs_tol=atol)
    return left == right


def _write_json_once(path: Path, payload: Mapping[str, Any], *, artifact_kind: str) -> None:
    if path.exists():
        _require(_json_equal(_json_read(path), payload), f"existing {artifact_kind} differs from recovered result")
        return
    atomic_write_new_json(path, payload, artifact_kind=artifact_kind)


def _write_torch_once(path: Path, payload: Mapping[str, Any], *, artifact_kind: str) -> None:
    if path.exists():
        existing = safe_torch_load(path, map_location="cpu")
        _require(_tree_equal(existing, payload), f"existing {artifact_kind} differs from recovered result")
        return
    atomic_torch_save_new(path, payload, artifact_kind=artifact_kind)


def _evaluator_view(raw: torch.Tensor) -> torch.Tensor:
    _require(raw.ndim == 4 and raw.shape[1] == 1 and tuple(raw.shape[-2:]) == (200, 200), "raw CMOS geometry changed")
    _finite(raw, "raw CMOS")
    return F.adaptive_avg_pool2d(raw.clamp(0.0, 1.0), (28, 28))


def _evaluator_predictions(
    labels: torch.Tensor,
    predictions: torch.Tensor,
    *,
    label_source: str,
) -> dict[str, Any]:
    labels = labels.detach().to(device="cpu", dtype=torch.long)
    predictions = predictions.detach().to(device="cpu", dtype=torch.long)
    _require(labels.ndim == predictions.ndim == 1 and labels.shape == predictions.shape, "prediction vectors are malformed")
    _require(bool(((labels >= 0) & (labels < 10)).all().item()), "labels are outside digit classes")
    _require(bool(((predictions >= 0) & (predictions < 10)).all().item()), "predictions are outside digit classes")
    confusion = torch.bincount(labels * 10 + predictions, minlength=100).reshape(10, 10)
    counts = torch.bincount(labels, minlength=10)
    _require(bool((counts > 0).all().item()), "classification population omits a class")
    recall = confusion.diag().to(torch.float64) / counts.to(torch.float64)
    return {
        "rows": int(labels.numel()),
        "accuracy": float((labels == predictions).to(torch.float64).mean()),
        "macro_recall": float(recall.mean()),
        "per_class_recall": [float(value) for value in recall],
        "minimum_class_recall": float(recall.min()),
        "confusion_matrix_true_rows_predicted_columns": confusion.tolist(),
        "prediction_histogram": torch.bincount(predictions, minlength=10).tolist(),
        "class8_prediction_mass": float((predictions == 8).to(torch.float64).mean()),
        "label_source": label_source,
    }


def _prediction_histogram(predictions: Sequence[int]) -> list[int]:
    values = torch.as_tensor(predictions, dtype=torch.long).reshape(-1)
    _require(
        values.numel() > 0 and bool(((values >= 0) & (values < 10)).all().item()),
        "prediction rows are empty or outside the ten-class range",
    )
    return torch.bincount(values, minlength=10).tolist()


def _raw_summary(chunks: list[torch.Tensor]) -> dict[str, Any]:
    _require(bool(chunks), "raw CMOS summary has no chunks")
    rows = torch.cat([chunk.detach().to(device="cpu", dtype=torch.float32) for chunk in chunks], dim=0)
    _finite(rows, "raw CMOS summary")
    flat = rows.to(torch.float64).flatten(1)
    std = flat.std(dim=1, unbiased=False)
    return {
        "shape": list(rows.shape),
        "finite": True,
        "finite_fraction": 1.0,
        "nonconstant_fraction": float((std > 0.0).to(torch.float64).mean()),
        "zero_image_count": int((flat.abs().sum(dim=1) == 0.0).sum()),
        "minimum": float(rows.min()),
        "maximum": float(rows.max()),
        "mean": float(rows.mean()),
        "maximum_above_one_fraction": float((rows > 1.0).to(torch.float64).mean()),
    }


def _per_sample_pixel_errors(view: torch.Tensor, target: torch.Tensor) -> tuple[torch.Tensor, torch.Tensor]:
    _require(view.shape == target.shape and view.ndim == 4 and view.shape[1] == 1, "canonical image tensors have different shapes")
    target_flat = target.flatten(1)
    squared = (view - target).square().flatten(1)
    foreground = target_flat > 0.1
    background = ~foreground
    _require(bool(foreground.any(dim=1).all().item()) and bool(background.any(dim=1).all().item()), "a target has an empty foreground or background")
    foreground_error = (squared * foreground).sum(dim=1) / foreground.sum(dim=1)
    background_error = (squared * background).sum(dim=1) / background.sum(dim=1)
    return foreground_error, background_error


def _fixed_ols_power_vector(result: Any, encoder_plane_powers: torch.Tensor) -> torch.Tensor:
    batch_size = int(result.generator_input_field.tensor.shape[0])
    _require(
        encoder_plane_powers.shape == (batch_size, 2)
        and not encoder_plane_powers.requires_grad
        and encoder_plane_powers.grad_fn is None,
        "fixed-bank encoder power records are malformed or attached to autograd",
    )
    _require(len(result.generator_records) == 2, "fixed OLS forward lacks the accepted two generator planes")
    generator_fields = (result.generator_input_field, *result.generator_records)
    generator_powers = torch.stack([physical_plane_power(field) for field in generator_fields], dim=1)
    powers = torch.cat((encoder_plane_powers.to(device=generator_powers.device, dtype=generator_powers.dtype), generator_powers), dim=1)
    _require(powers.shape == (batch_size, 5), "fixed OLS power vector does not match the accepted five-plane objective")
    _finite(powers, "fixed OLS five-plane power vector")
    return powers


def _encoder_plane_powers_for_fixed_population(
    consumer: Any,
    images: torch.Tensor,
    coefficients: torch.Tensor,
    *,
    population_name: str,
) -> torch.Tensor:
    _require(images.ndim == 4 and coefficients.shape == (images.shape[0], 100), f"{population_name} image/OLS binding is malformed")
    chunks: list[torch.Tensor] = []
    with torch.no_grad():
        for start in range(0, int(images.shape[0]), BATCH_SIZE):
            stop = min(start + BATCH_SIZE, int(images.shape[0]))
            result = consumer.runtime.forward_from_image(images[start:stop])
            observed_coefficients = result.z_ols.coefficients
            expected_coefficients = coefficients[start:stop]
            _require(
                torch.allclose(observed_coefficients, expected_coefficients, rtol=1.0e-5, atol=1.0e-7),
                f"{population_name} TRAIN image does not reproduce its fixed accepted OLS coefficients",
            )
            _require(len(result.encoder_records) == 2, f"{population_name} image path lacks two encoder planes")
            plane_powers = torch.stack([physical_plane_power(field) for field in result.encoder_records], dim=1)
            _finite(plane_powers, f"{population_name} encoder plane powers")
            chunks.append(plane_powers.detach())
            del result, observed_coefficients, plane_powers
    return torch.cat(chunks, dim=0).detach()


def _component_bundle_from_ols(
    consumer: Any,
    smooth: torch.nn.Module,
    independent: torch.nn.Module,
    ec: torch.nn.Module,
    coefficients: torch.Tensor,
    images_28: torch.Tensor,
    images_slm: torch.Tensor,
    encoder_plane_powers: torch.Tensor,
    references: Any,
) -> dict[str, Any]:
    _require(coefficients.ndim == 2 and torch.is_complex(coefficients), "fixed OLS batch is malformed")
    result = consumer.runtime.forward_from_ols(coefficients)
    raw = result.raw_intensity
    _require(raw.shape == (coefficients.shape[0], 1, 200, 200), "fixed-OLS output geometry changed")
    _finite(raw, "fixed-OLS raw CMOS")
    _require(images_28.shape == (coefficients.shape[0], 1, 28, 28), "canonical target-image shape changed")
    _require(images_slm.shape == raw.shape, "aligned SLM target shape changed")
    raw_view = _evaluator_view(raw)
    target_view = F.adaptive_avg_pool2d(images_28.clamp(0.0, 1.0), (28, 28))
    s_features, _ = smooth(raw_view)
    i_features, _ = independent(raw_view)
    c_features, _ = ec(raw_view)
    with torch.no_grad():
        s_target, _ = smooth(target_view)
        i_target, _ = independent(target_view)
        c_target, _ = ec(target_view)
    losses = {
        "L_S": three._normalized_feature_loss(s_features, s_target, "L_S"),
        "L_I": three._normalized_feature_loss(i_features, i_target, "L_I"),
        "L_C": three._normalized_feature_loss(c_features, c_target, "L_C"),
        "L_rec": (raw - images_slm).square().mean(),
        "L_op": three.optical_power_loss_from_powers(
            _fixed_ols_power_vector(result, encoder_plane_powers), references.power_reference
        ),
    }
    for name, value in losses.items():
        _finite(value, name)
    return {
        "result": result,
        "raw": raw,
        "raw_view": raw_view,
        "target_view": target_view,
        "losses": losses,
        "features": {"E_S": s_features, "E_I": i_features, "E_C": c_features},
        "target_features": {"E_S": s_target.detach(), "E_I": i_target.detach(), "E_C": c_target.detach()},
    }


def _canonical_direction_from_ols(
    consumer: Any,
    smooth: torch.nn.Module,
    independent: torch.nn.Module,
    ec: torch.nn.Module,
    coefficients: torch.Tensor,
    images_28: torch.Tensor,
    images_slm: torch.Tensor,
    encoder_plane_powers: torch.Tensor,
    references: Any,
) -> dict[str, Any]:
    generator = dict(consumer.runtime.named_parameters())["generator.phase"]
    bundle = _component_bundle_from_ols(
        consumer, smooth, independent, ec, coefficients, images_28, images_slm, encoder_plane_powers, references
    )
    losses = bundle["losses"]
    g_s = torch.autograd.grad(losses["L_S"], generator, retain_graph=True, allow_unused=False)[0]
    g_i = torch.autograd.grad(losses["L_I"], generator, retain_graph=True, allow_unused=False)[0]
    g_c = torch.autograd.grad(losses["L_C"], generator, retain_graph=True, allow_unused=False)[0]
    g_rec = torch.autograd.grad(losses["L_rec"], generator, retain_graph=True, allow_unused=False)[0]
    g_op = torch.autograd.grad(losses["L_op"], generator, retain_graph=False, allow_unused=False)[0]
    for name, value in (("g_S", g_s), ("g_I", g_i), ("g_C", g_c), ("g_rec", g_rec), ("g_op", g_op)):
        _finite(value, name)
    norms = {name: three._norm(value, name) for name, value in (("g_S", g_s), ("g_I", g_i), ("g_C", g_c))}
    u_s, u_i, u_c = g_s / norms["g_S"], g_i / norms["g_I"], g_c / norms["g_C"]
    v_sic = u_s + u_i + u_c
    v_norm = three._norm(v_sic, "v_SIC", allow_zero=True)
    _require(math.isfinite(v_norm) and v_norm > three.JOINT_DEGENERACY_TOLERANCE, "canonical evaluator direction is degenerate")
    v_rms = three._rms(v_sic, "v_SIC")
    q_sic = v_sic / v_rms
    g_sem = three.DELTA_SIC * q_sic
    g_rec_w = three.BETA * g_rec
    g_op_w = three.GAMMA * g_op
    g_rec_safe, rec_projection = three._project_auxiliary_gradient(g_rec_w, g_sem, "g_rec_w")
    g_op_safe, op_projection = three._project_auxiliary_gradient(g_op_w, g_sem, "g_op_w")
    g_total = g_sem + g_rec_safe + g_op_safe
    _finite(g_total, "g_total")
    _require(three._dot(g_total, g_sem) > 0.0, "combined gradient has no positive canonical component")
    return {
        "bundle": bundle,
        "g_S": g_s,
        "g_I": g_i,
        "g_C": g_c,
        "g_rec": g_rec,
        "g_op": g_op,
        "u_S": u_s,
        "u_I": u_i,
        "u_C": u_c,
        "v_SIC": v_sic,

def _eval_raw_population(
    raw_cpu: torch.Tensor,
    images_28: torch.Tensor,
    labels: torch.Tensor,
    smooth: torch.nn.Module,
    independent: torch.nn.Module,
    ec: torch.nn.Module,
    ed: torch.nn.Module,
    *,
    include_ed: bool,
    label_source: str,
) -> dict[str, Any]:
    _require(raw_cpu.device.type == "cpu", "measurement raw CMOS must be staged on CPU")
    _require(raw_cpu.shape == (labels.numel(), 1, 200, 200), "measurement population geometry changed")
    device = next(smooth.parameters()).device
    predictions: dict[str, list[torch.Tensor]] = {name: [] for name in ("E_S", "E_I", "E_C", "E_D")}
    target_losses = {name: 0.0 for name in ("E_S", "E_I", "E_C")}
    foreground_errors: list[torch.Tensor] = []
    background_errors: list[torch.Tensor] = []
    with torch.no_grad():
        for start in range(0, int(labels.numel()), BATCH_SIZE):
            stop = min(start + BATCH_SIZE, int(labels.numel()))
            raw = raw_cpu[start:stop].to(device)
            target = images_28[start:stop].to(device)
            view = _evaluator_view(raw)
            target_view = F.adaptive_avg_pool2d(target.clamp(0.0, 1.0), (28, 28))
            fg, bg = _per_sample_pixel_errors(view, target_view)
            foreground_errors.append(fg.cpu())
            background_errors.append(bg.cpu())
            for name, evaluator in (("E_S", smooth), ("E_I", independent), ("E_C", ec)):
                generated_features, logits = evaluator(view)
                target_features, _ = evaluator(target_view)
                loss = three._normalized_feature_loss(generated_features, target_features, f"{name}_measure_loss")
                target_losses[name] += float(loss.cpu()) * (stop - start)
                predictions[name].append(logits.argmax(dim=1).cpu())
            if include_ed:
                logits = ed(view)
                _require(not logits.requires_grad and logits.grad_fn is None, "E_D measurement output retained a graph")
                predictions["E_D"].append(logits.argmax(dim=1).cpu())
    result: dict[str, Any] = {
        "rows": int(labels.numel()),
        "training_evaluator_feature_losses": {
            name: target_losses[name] / int(labels.numel()) for name in target_losses
        },
        "descriptive_mean_feature_loss": sum(target_losses.values()) / (3.0 * int(labels.numel())),
        "descriptive_mean_is_not_the_optimized_scalar": True,
        "foreground_mse": float(torch.cat(foreground_errors).to(torch.float64).mean()),
        "background_mse": float(torch.cat(background_errors).to(torch.float64).mean()),
        "raw_cmos": _raw_summary([raw_cpu]),
        "E_S": _evaluator_predictions(labels, torch.cat(predictions["E_S"]), label_source=label_source),
        "E_I": _evaluator_predictions(labels, torch.cat(predictions["E_I"]), label_source=label_source),
        "E_C": _evaluator_predictions(labels, torch.cat(predictions["E_C"]), label_source=label_source),
        "E_D_measured": include_ed,
    }
    if include_ed:
        result["E_D"] = _evaluator_predictions(labels, torch.cat(predictions["E_D"]), label_source=label_source)
    result["row_vectors"] = {
        "labels": labels.detach().cpu().to(torch.long).tolist(),
        "E_S_predictions": torch.cat(predictions["E_S"]).tolist(),
        "E_I_predictions": torch.cat(predictions["E_I"]).tolist(),
        "E_C_predictions": torch.cat(predictions["E_C"]).tolist(),
        "E_D_predictions": torch.cat(predictions["E_D"]).tolist() if include_ed else None,
        "foreground_mse": torch.cat(foreground_errors).tolist(),
        "background_mse": torch.cat(background_errors).tolist(),
    }
    return result


def _paired_pixel_endpoint_comparison(
    pixel: Mapping[str, Any],
    canonical: Mapping[str, Any],
    expected_labels: torch.Tensor,
) -> dict[str, Any]:
    pixel_rows = _mapping(pixel.get("row_vectors"), "pixel endpoint row vectors")
    canonical_rows = _mapping(canonical.get("row_vectors"), "canonical endpoint row vectors")
    labels = expected_labels.detach().cpu().to(torch.long).tolist()
    _require(pixel_rows.get("labels") == labels and canonical_rows.get("labels") == labels, "pixel and canonical endpoints do not bind to the exact same observation rows")
    pixel_prediction = _tensor(torch.tensor(pixel_rows.get("E_D_predictions")), "pixel E_D predictions")
    canonical_prediction = _tensor(torch.tensor(canonical_rows.get("E_D_predictions")), "canonical E_D predictions")
    _require(pixel_prediction.shape == canonical_prediction.shape == expected_labels.shape, "paired endpoint E_D prediction vectors differ in shape")
    pixel_correct = pixel_prediction.eq(expected_labels.cpu())
    canonical_correct = canonical_prediction.eq(expected_labels.cpu())
    transition = torch.bincount(pixel_prediction * 10 + canonical_prediction, minlength=100).reshape(10, 10)
    pixel_foreground = torch.tensor(pixel_rows["foreground_mse"], dtype=torch.float64)
    canonical_foreground = torch.tensor(canonical_rows["foreground_mse"], dtype=torch.float64)
    pixel_background = torch.tensor(pixel_rows["background_mse"], dtype=torch.float64)
    canonical_background = torch.tensor(canonical_rows["background_mse"], dtype=torch.float64)
    _require(pixel_foreground.shape == canonical_foreground.shape == expected_labels.shape, "paired foreground error vectors differ in shape")
    _require(pixel_background.shape == canonical_background.shape == expected_labels.shape, "paired background error vectors differ in shape")
    return {
        "exact_row_pairing": True,
        "rows": int(expected_labels.numel()),
        "E_D": {
            "prediction_transition_matrix_pixel_rows_canonical_columns": transition.tolist(),
            "changed_prediction_rows": int((pixel_prediction != canonical_prediction).sum()),
            "correct_after_wrong_before": int((canonical_correct & ~pixel_correct).sum()),
            "wrong_after_correct_before": int((pixel_correct & ~canonical_correct).sum()),
            "class8_prediction_mass_delta": float((canonical_prediction == 8).to(torch.float64).mean() - (pixel_prediction == 8).to(torch.float64).mean()),
            "per_class_recall_delta": [
                float(canonical_correct[expected_labels.cpu() == digit].to(torch.float64).mean() - pixel_correct[expected_labels.cpu() == digit].to(torch.float64).mean())
                for digit in range(10)
            ],
        },
        "foreground_mse": {
            "pixel_mean": float(pixel_foreground.mean()),
            "canonical_mean": float(canonical_foreground.mean()),
            "paired_mean_delta": float((canonical_foreground - pixel_foreground).mean()),
        },
        "background_mse": {
            "pixel_mean": float(pixel_background.mean()),
            "canonical_mean": float(canonical_background.mean()),
            "paired_mean_delta": float((canonical_background - pixel_background).mean()),
        },
    }


def _fixed_bank_measurement(
    consumer: Any,
    coefficients: torch.Tensor,
    images_28: torch.Tensor,
    images_slm: torch.Tensor,
    encoder_plane_powers: torch.Tensor,
    labels: torch.Tensor,
    smooth: torch.nn.Module,
    independent: torch.nn.Module,
    ec: torch.nn.Module,
    ed: torch.nn.Module,
    references: Any,
    *,
    include_ed: bool,
    retain_raw: bool,
    label_source: str,
) -> tuple[dict[str, Any], torch.Tensor | None]:
    _require(coefficients.shape == (labels.numel(), 100), "fixed OLS measurement bank shape changed")
    _require(images_28.shape == (labels.numel(), 1, 28, 28), "fixed measurement targets changed")
    _require(images_slm.shape == (labels.numel(), 1, 200, 200), "fixed optical target geometry changed")
    raw_chunks: list[torch.Tensor] = []
    losses_sum = {name: 0.0 for name in ("L_S", "L_I", "L_C", "L_rec", "L_op")}
    fg_chunks: list[torch.Tensor] = []
    bg_chunks: list[torch.Tensor] = []
    power_finite = True
    device = consumer.device
    with torch.no_grad():
        for start in range(0, int(labels.numel()), BATCH_SIZE):
            stop = min(start + BATCH_SIZE, int(labels.numel()))
            z = coefficients[start:stop].to(device)
            image = images_28[start:stop].to(device)
            slm_target = images_slm[start:stop].to(device)
            result = consumer.runtime.forward_from_ols(z)
            raw = result.raw_intensity
            _require(raw.shape == (stop - start, 1, 200, 200), "fixed-bank raw CMOS shape changed")
            _finite(raw, "fixed-bank raw CMOS")
            view = _evaluator_view(raw)
            target_view = F.adaptive_avg_pool2d(image.clamp(0.0, 1.0), (28, 28))
            fg, bg = _per_sample_pixel_errors(view, target_view)
            fg_chunks.append(fg.cpu())
            bg_chunks.append(bg.cpu())
            for name, evaluator in (("E_S", smooth), ("E_I", independent), ("E_C", ec)):
                current_features, _ = evaluator(view)
                target_features, _ = evaluator(target_view)
                loss = three._normalized_feature_loss(current_features, target_features, f"{name}_measure_loss")
                losses_sum[f"L_{name[-1]}"] += float(loss.cpu()) * (stop - start)
            reconstruction = (raw - slm_target).square().mean()
            power_loss = three.optical_power_loss_from_powers(
                _fixed_ols_power_vector(result, encoder_plane_powers[start:stop].to(device)), references.power_reference
            )
            losses_sum["L_rec"] += float(reconstruction.cpu()) * (stop - start)
            losses_sum["L_op"] += float(power_loss.cpu()) * (stop - start)
            power_finite = power_finite and bool(torch.isfinite(power_loss).item())
            if include_ed:
                logits = ed(view)
                _require(not logits.requires_grad and logits.grad_fn is None, "E_D output retained a graph")
                # Store labels/predictions through the shared measurement function below.
            if retain_raw:
                raw_chunks.append(raw.detach().to(device="cpu", dtype=torch.float32))
            del raw, result, view, target_view
    raw_cpu = torch.cat(raw_chunks, dim=0) if retain_raw else None
    _require(power_finite, "fixed-bank power measurement is nonfinite")
    summary: dict[str, Any] = {
        "rows": int(labels.numel()),
        "training_evaluator_feature_losses": {
            "E_S": losses_sum["L_S"] / int(labels.numel()),
            "E_I": losses_sum["L_I"] / int(labels.numel()),
            "E_C": losses_sum["L_C"] / int(labels.numel()),
        },
        "descriptive_mean_feature_loss": sum(losses_sum[name] for name in ("L_S", "L_I", "L_C")) / (3.0 * int(labels.numel())),
        "descriptive_mean_is_not_the_optimized_scalar": True,
        "L_rec": losses_sum["L_rec"] / int(labels.numel()),
        "L_op": losses_sum["L_op"] / int(labels.numel()),
        "foreground_mse": float(torch.cat(fg_chunks).to(torch.float64).mean()),
        "background_mse": float(torch.cat(bg_chunks).to(torch.float64).mean()),
        "power_finite": power_finite,
        "E_D_measured": include_ed,
    }
    if raw_cpu is not None:
        summary["raw_cmos"] = _raw_summary([raw_cpu])
        eval_result = _eval_raw_population(
            raw_cpu,
            images_28.detach().cpu(),
            labels.detach().cpu(),
            smooth,
            independent,
            ec,
            ed,
            include_ed=include_ed,
            label_source=label_source,
        )
        for name in ("E_S", "E_I", "E_C"):
            summary[name] = eval_result[name]
            summary["training_evaluator_feature_losses"][name] = eval_result["training_evaluator_feature_losses"][name]
        if include_ed:
            summary["E_D"] = eval_result["E_D"]
        summary["foreground_mse"] = eval_result["foreground_mse"]
        summary["background_mse"] = eval_result["background_mse"]
        if labels.numel() == OBSERVATION_ROWS:
            summary["row_vectors"] = eval_result["row_vectors"]
    return summary, raw_cpu


def _measure_dev(
    consumer: Any,
    images: torch.Tensor,
    labels: torch.Tensor,
    indices: torch.Tensor,
    references: Any,
    ed: torch.nn.Module,
    *,
    step: int,
) -> tuple[dict[str, Any], dict[str, torch.Tensor]]:
    labels_dev = labels.index_select(0, indices).detach().cpu().to(torch.long)
    predictions: list[torch.Tensor] = []
    raw_min = math.inf
    raw_max = -math.inf
    raw_sum = 0.0
    raw_pixel_count = 0
    raw_above_one_count = 0
    nonconstant_count = 0
    zero_image_count = 0
    rec_weighted = 0.0
    op_weighted = 0.0
    rows = int(indices.numel())
    with torch.no_grad():
        for start in range(0, rows, DEV_BATCH_SIZE):
            selected = indices[start : start + DEV_BATCH_SIZE]
            batch = images.index_select(0, selected).to(consumer.device)
            result = consumer.runtime.forward_from_image(batch)
            raw = result.raw_intensity
            _require(result.image_input is not None, "DEV image adapter did not return its target image")
            target_slm = result.image_input.slm_intensity
            _finite(raw, "DEV raw CMOS")
            _finite(target_slm, "DEV SLM target")
            rec = (raw - target_slm).square().mean()
            op = three.optical_power_loss_from_powers(
                three.contracted_plane_powers(result), references.power_reference
            )
            _finite(rec, "DEV reconstruction")
            _finite(op, "DEV optical power")
            view = _evaluator_view(raw)
            logits = ed(view)
            _require(not logits.requires_grad and logits.grad_fn is None, "DEV E_D output retained a graph")
            predictions.append(logits.argmax(dim=1).detach().cpu())
            raw64 = raw.detach().to(torch.float64)
            flat = raw64.flatten(1)
            nonconstant_count += int((flat.std(dim=1, unbiased=False) > 0.0).sum().item())
            zero_image_count += int((flat.abs().sum(dim=1) == 0.0).sum().item())
            raw_min = min(raw_min, float(raw64.min().cpu()))
            raw_max = max(raw_max, float(raw64.max().cpu()))
            raw_sum += float(raw64.sum().cpu())
            raw_above_one_count += int((raw > 1.0).sum().item())
            raw_pixel_count += raw.numel()
            count = int(batch.shape[0])
            rec_weighted += float(rec.cpu()) * count
            op_weighted += float(op.cpu()) * count
            del raw, result, view, logits
    prediction = torch.cat(predictions)
    raw_summary = {
        "shape": [rows, 1, 200, 200],
        "finite": True,
        "finite_fraction": 1.0,
        "nonconstant_fraction": nonconstant_count / rows,
        "zero_image_count": zero_image_count,
        "minimum": raw_min,
        "maximum": raw_max,
        "mean": raw_sum / raw_pixel_count,
        "maximum_above_one_fraction": raw_above_one_count / raw_pixel_count,
        "retained_raw_tensor": False,
    }
    measurement = {
        "step": step,
        "data_role": "DEV",
        "rows": rows,
        "L_rec": rec_weighted / rows,
        "L_op": op_weighted / rows,
        "optical_power_finite": math.isfinite(op_weighted),
        "E_D": _evaluator_predictions(labels_dev, prediction, label_source="canonical DEV label"),
        "raw_cmos": raw_summary,
        "E_D_in_backward": False,
    }
    vectors = {"labels": labels_dev, "predictions": prediction, "correct": prediction.eq(labels_dev)}
    return measurement, vectors

def _paired_bootstrap(
    labels: torch.Tensor,
    before_correct: torch.Tensor,
    after_correct: torch.Tensor,
    *,
    replicates: int = BOOTSTRAP_REPLICATES,
    seed: int = BOOTSTRAP_SEED,
) -> dict[str, Any]:
    labels = labels.detach().cpu().to(torch.long)
    before_correct = before_correct.detach().cpu().to(torch.bool)
    after_correct = after_correct.detach().cpu().to(torch.bool)
    _require(labels.shape == before_correct.shape == after_correct.shape and labels.ndim == 1, "DEV paired bootstrap vectors differ")
    generator = torch.Generator(device="cpu").manual_seed(seed)
    indices_by_class = [torch.nonzero(labels == digit, as_tuple=False).flatten() for digit in range(10)]
    _require(all(indices.numel() > 0 for indices in indices_by_class), "paired bootstrap has an empty class")
    draws = torch.empty(replicates, dtype=torch.float64)
    for replicate in range(replicates):
        class_deltas: list[torch.Tensor] = []
        for indices in indices_by_class:
            sampled = indices[torch.randint(indices.numel(), (indices.numel(),), generator=generator)]
            class_deltas.append(
                after_correct[sampled].to(torch.float64).mean()
                - before_correct[sampled].to(torch.float64).mean()
            )
        draws[replicate] = torch.stack(class_deltas).mean()
    observed = float(after_correct.to(torch.float64).mean() - before_correct.to(torch.float64).mean())
    return {
        "method": "paired class-stratified row bootstrap, percentile interval",
        "replicates": replicates,
        "seed": seed,
        "observed_macro_recall_delta": observed,
        "ci95": [float(value) for value in torch.quantile(draws, torch.tensor([0.025, 0.975], dtype=torch.float64))],
        "class_recall_delta": [
            float(after_correct[indices].to(torch.float64).mean() - before_correct[indices].to(torch.float64).mean())
            for indices in indices_by_class
        ],
        "bootstrap_delta_values": draws.tolist(),
    }


def _classify(
    fit0: Mapping[str, Any],
    fit300: Mapping[str, Any],
    observation0: Mapping[str, Any],
    observation300: Mapping[str, Any],
    dev_bootstrap: Mapping[str, Any],
    reconstruction: Mapping[str, Any],
    optical_validity: bool,
) -> str:
    losses0 = _mapping(fit0.get("training_evaluator_feature_losses"), "fit step0 evaluator losses")
    losses300 = _mapping(fit300.get("training_evaluator_feature_losses"), "fit step300 evaluator losses")
    fit_all_improved = all(float(losses300[name]) < float(losses0[name]) for name in ("E_S", "E_I", "E_C"))
    obs0 = _mapping(observation0, "observation step0")
    obs300 = _mapping(observation300, "observation step300")
    obs_all_improved = all(
        float(_mapping(obs300[name], name)["macro_recall"]) > float(_mapping(obs0[name], name)["macro_recall"])
        for name in ("E_S", "E_I", "E_C")
    )
    delta = float(dev_bootstrap["observed_macro_recall_delta"])
    interval = dev_bootstrap.get("ci95")
    _require(isinstance(interval, (list, tuple)) and len(interval) == 2, "DEV bootstrap interval is malformed")
    lower = float(interval[0])
    practical_transfer = delta >= PRACTICAL_TRANSFER_DELTA and lower > 0.0
    safety = bool(reconstruction.get("passed")) and optical_validity
    if practical_transfer and not safety:
        return CASE_E
    if practical_transfer and safety and fit_all_improved:
        return CASE_A
    if fit_all_improved and delta <= 0.0:
        return CASE_C
    if fit_all_improved or obs_all_improved:
        return CASE_B
    fit_any_improved = any(float(losses300[name]) < float(losses0[name]) for name in ("E_S", "E_I", "E_C"))
    obs_any_improved = any(
        float(_mapping(obs300[name], name)["macro_recall"]) > float(_mapping(obs0[name], name)["macro_recall"])
        for name in ("E_S", "E_I", "E_C")
    )
    if not fit_any_improved and not obs_any_improved:
        return CASE_D
    return CASE_F


def _runtime_snapshot(consumer: Any) -> dict[str, torch.Tensor]:
    return {name: value.detach().cpu().clone() for name, value in consumer.runtime.state_dict().items()}


def _evaluator_snapshot(models: Mapping[str, torch.nn.Module]) -> dict[str, dict[str, torch.Tensor]]:
    return {
        name: {key: value.detach().cpu().clone() for key, value in model.state_dict().items()}
        for name, model in models.items()
    }


def _device_snapshot(device: torch.device) -> dict[str, int]:
    torch.cuda.synchronize(device)
    free_bytes, total_bytes = torch.cuda.mem_get_info(device)
    allocated = torch.cuda.memory_allocated(device)
    reserved = torch.cuda.memory_reserved(device)
    return {
        "total_bytes": int(total_bytes),
        "free_bytes": int(free_bytes),
        "device_used_bytes": int(total_bytes - free_bytes),
        "torch_allocated_bytes": int(allocated),
        "torch_reserved_bytes": int(reserved),
        "torch_peak_allocated_bytes": int(torch.cuda.max_memory_allocated(device)),
        "torch_peak_reserved_bytes": int(torch.cuda.max_memory_reserved(device)),
    }


def _check_resource(snapshot: Mapping[str, int], phase: str) -> None:
    total = max(1, int(snapshot["total_bytes"]))
    observed = max(
        int(snapshot["device_used_bytes"]),
        int(snapshot["torch_peak_reserved_bytes"]),
    )
    if observed > int(total * MAX_DEVICE_MEMORY_FRACTION):
        raise ResourceBudgetExceeded(
            f"CUDA resource envelope exceeded in {phase}: global_used={snapshot['device_used_bytes']}, "
            f"torch_peak_reserved={snapshot['torch_peak_reserved_bytes']}, total={total}, "
            f"limit={MAX_DEVICE_MEMORY_FRACTION}"
        )


@contextmanager
def _phase_meter(device: torch.device, phase: str):
    # Device discovery alone does not initialize the CUDA context on Windows.
    # reset_peak_memory_stats(device) can otherwise fail with "Invalid device
    # argument" before the first measured phase begins.
    torch.cuda.current_device()
    with torch.cuda.device(device):
        torch.cuda.reset_peak_memory_stats()
        started = time.perf_counter()
        before = _device_snapshot(device)
        record: dict[str, Any] = {"phase": phase, "started_at": time.time(), "device_before": before}
        try:
            yield record
        finally:
            after = _device_snapshot(device)
            record["elapsed_seconds"] = time.perf_counter() - started
            record["device_after"] = after
            record["device_global_peak_observed_bytes"] = max(before["device_used_bytes"], after["device_used_bytes"])
            record["torch_peak_allocated_bytes"] = after["torch_peak_allocated_bytes"]
            record["torch_peak_reserved_bytes"] = after["torch_peak_reserved_bytes"]
            _check_resource(after, phase)


def _objective_authority(root: Path) -> tuple[dict[str, Any], dict[str, Any]]:
    report_path = root / THREE_EVALUATOR_REPORT
    source_path = root / THREE_EVALUATOR_SOURCE
    _require(file_sha256(report_path) == THREE_EVALUATOR_REPORT_SHA256, "accepted three-evaluator report identity changed")
    report = _json_read(report_path)
    source = _json_read(source_path)
    _require(
        report.get("result") == "VALID_COMPLETED"
        and report.get("outer_gate") == three.OUTER_GATE
        and report.get("classification") == three.CLASSIFICATION_SUPPORTED
        and report.get("outer_gate_passed") is True,
        "three-evaluator semantic objective is not accepted",
    )
    _require(report.get("source_manifest") == source, "accepted three-evaluator source manifest differs from report")
    _require(
        source.get("source_manifest_id") == THREE_EVALUATOR_SOURCE_MANIFEST_ID
        and source.get("execution_source_semantics_id") == report.get("source_manifest", {}).get("execution_source_semantics_id"),
        "accepted three-evaluator source identity changed",
    )
    sic = _mapping(report.get("sic_gradient_contract"), "accepted SIC gradient contract")
    aux = _mapping(report.get("auxiliary_contract"), "accepted auxiliary objective contract")
    _require(
        sic.get("g_S") == "grad_generator(L_S)"
        and sic.get("g_I") == "grad_generator(L_I)"
        and sic.get("g_C") == "grad_generator(L_C)"
        and sic.get("u_k") == "g_k/||g_k||_2"
        and sic.get("v_SIC") == "u_S+u_I+u_C"
        and sic.get("q_SIC") == "+v_SIC/RMS(v_SIC)"
        and sic.get("E_D_in_backward") is False
        and math.isclose(float(sic.get("delta_SIC", -1)), three.DELTA_SIC, rel_tol=0.0, abs_tol=1.0e-18),
        "accepted three-evaluator semantic direction no longer matches its implementation",
    )
    _require(
        math.isclose(float(aux.get("beta", -1)), three.BETA, rel_tol=0.0, abs_tol=1.0e-12)
        and math.isclose(float(aux.get("gamma", -1)), three.GAMMA, rel_tol=0.0, abs_tol=1.0e-12)
        and math.isclose(float(aux.get("eps_proj", -1)), three.EPS_PROJ, rel_tol=0.0, abs_tol=1.0e-18)
        and aux.get("projection") == "remove only opposing component relative to g_sem",
        "accepted stabilizer weights or projection rule changed",
    )
    evaluators = _mapping(report.get("evaluators"), "accepted evaluator identities")
    expected = {
        "E_S": robust.EXPECTED_SMOOTH_CHECKPOINT_SHA256,
        "E_I": robust.EXPECTED_INDEPENDENT_CHECKPOINT_SHA256,
        "E_C": robust.EXPECTED_EC_CHECKPOINT_SHA256,
    }
    for name, expected_hash in expected.items():
        item = _mapping(evaluators.get(name), name)
        _require(item.get("checkpoint_sha256") == expected_hash, f"accepted {name} identity is ambiguous")
    _require(evaluators["E_C"].get("role") == "training component and DEV diagnostic", "E_C role changed")
    _require(evaluators.get("E_D", {}).get("role") == "independent confirmation only", "E_D is not the independent holdout")
    return report, source

def _freeze_models(consumer: Any, smooth: torch.nn.Module, independent: torch.nn.Module, ec: torch.nn.Module, ed: torch.nn.Module) -> torch.optim.Adam:
    params = dict(consumer.runtime.named_parameters())
    _require(set(params) == {"encoder.phase", "generator.phase"}, "runtime trainable parameter surface changed")
    params["encoder.phase"].requires_grad_(False)
    params["generator.phase"].requires_grad_(True)
    for parameter in consumer.runtime.parameters():
        parameter.grad = None
    models = {"E_S": smooth, "E_I": independent, "E_C": ec, "E_D": ed}
    for model in models.values():
        model.eval()
        for parameter in model.parameters():
            parameter.requires_grad_(False)
            parameter.grad = None
    consumer.runtime.eval()
    optimizer = render_v1._adam([params["generator.phase"]])
    group = optimizer.param_groups[0]
    _require(
        len(optimizer.param_groups) == 1
        and group["params"] == [params["generator.phase"]]
        and group["lr"] == render_v2.LEARNING_RATE
        and tuple(group["betas"]) == tuple(render_v2.ADAM_BETAS)
        and group["eps"] == render_v1.ADAM_EPSILON
        and group["weight_decay"] == render_v1.WEIGHT_DECAY
        and group["amsgrad"] is False
        and not optimizer.state,
        "fresh-zero Adam semantics differ from the accepted V2 pixel arm",
    )
    return optimizer


def _fresh_attempt_optimizer(consumer: Any, smooth: torch.nn.Module, independent: torch.nn.Module, ec: torch.nn.Module, ed: torch.nn.Module) -> torch.optim.Adam:
    """Create the scientific optimizer after preflight smoke mutations."""
    optimizer = _freeze_models(consumer, smooth, independent, ec, ed)
    _require(not optimizer.state, "preflight optimizer state leaked into the scientific attempt")
    return optimizer

    ed: torch.nn.Module,

def _optimizer_internal_step(optimizer: torch.optim.Optimizer, parameter: torch.nn.Parameter) -> int:
    state = optimizer.state.get(parameter)
    if state is None:
        return 0
    step = state.get("step")
    _require(isinstance(step, torch.Tensor) and step.numel() == 1, "Adam internal step is missing or malformed")
    value = float(step.detach().cpu().item())
    _require(math.isfinite(value) and value >= 0.0 and value.is_integer(), "Adam internal step is not a nonnegative integer")
    return int(value)


def _reconstruct_scientific_state(
    root: Path,
    inputs: Mapping[str, Any],
    device: torch.device,
    preflight_references: Mapping[str, weakref.ReferenceType[Any]],
) -> tuple[tuple[Any, torch.nn.Module, torch.nn.Module, torch.nn.Module, torch.nn.Module, dict[str, torch.Tensor], torch.optim.Adam, dict[str, torch.Tensor]], dict[str, Any]]:
    gc.collect()
    live_preflight_objects = [name for name, reference in preflight_references.items() if reference() is not None]
    _require(not live_preflight_objects, f"preflight mutable state is still alive: {live_preflight_objects}")
    if device.type == "cuda":
        torch.cuda.synchronize(device)
        torch.cuda.empty_cache()

    consumer, smooth, independent, ec, ed, frozen, optimizer = _load_run_models(root, inputs, device)
    runtime_data = _fixed_runtime_data(inputs, consumer, device)
    expected_state = _mapping(inputs["authorities"]["checkpoint"].get("trainable_parameter_state"), "accepted step600 trainable state")
    params = dict(consumer.runtime.named_parameters())
    _require(set(expected_state) == {"encoder.phase", "generator.phase"}, "accepted step600 trainable state names changed")
    _require(
        all(torch.equal(params[name].detach().cpu(), _tensor(expected_state[name], f"accepted {name}")) for name in expected_state),
        "scientific runtime does not reproduce the accepted step600 trainable parameters",
    )
    _require(not optimizer.state and _optimizer_internal_step(optimizer, params["generator.phase"]) == 0, "scientific Adam is not fresh at step 0")
    three._restore_rng(inputs["authorities"]["checkpoint"]["rng_state"])
    resource_snapshot = _device_snapshot(device) if device.type == "cuda" else None
    if resource_snapshot is not None:
        _check_resource(resource_snapshot, "independent scientific runtime reconstruction")
    state = (consumer, smooth, independent, ec, ed, frozen, optimizer, runtime_data)
    evidence = {
        "preflight_mutable_state_destroyed": True,
        "preflight_object_count_released": len(preflight_references),
        "scientific_runtime_reconstructed_from_accepted_step600": True,
        "scientific_generator_matches_step600": True,
        "scientific_adam_step_before_attempt": 0,
        "scientific_adam_state_empty_before_attempt": not optimizer.state,
        "resource_snapshot": resource_snapshot,
    }
    return state, evidence

def _fit_batch_indices(cursor: int, device: torch.device) -> torch.Tensor:
    _require(0 <= cursor < UPDATE_COUNT, "Generator update cursor is out of range")
    return (cursor * BATCH_SIZE + torch.arange(BATCH_SIZE, device=device)) % 200


def _train_batch(
    consumer: Any,
    optimizer: torch.optim.Optimizer,
    smooth: torch.nn.Module,
    independent: torch.nn.Module,
    ec: torch.nn.Module,
    ed: torch.nn.Module,
    data: Mapping[str, torch.Tensor],
    references: Any,
    cursor: int,
    clean_generator: torch.Tensor,
) -> dict[str, Any]:
    indices = _fit_batch_indices(cursor, consumer.device)
    optimizer.zero_grad(set_to_none=True)
    three._clear_runtime_grads(consumer)
    generator = dict(consumer.runtime.named_parameters())["generator.phase"]
    generator_before = generator.detach().clone()
    direction = _canonical_direction_from_ols(
        consumer,
        smooth,
        independent,
        ec,
        data["fit_ols"].index_select(0, indices),
        data["fit_targets_28"].index_select(0, indices),
        data["fit_targets_slm"].index_select(0, indices),
        data["fit_encoder_plane_powers"].index_select(0, indices),
        references,
    )
    generator.grad = direction["g_total"].detach().clone()
    _finite(generator.grad, "canonical Generator gradient")
    for name, parameter in consumer.runtime.named_parameters():
        if name != "generator.phase":
            _require(parameter.grad is None, f"non-Generator parameter received gradient: {name}")
    optimizer.step()
    _finite(generator, "updated Generator phase")
    phase_update = generator.detach() - generator_before
    update_rms = three._rms(phase_update, "phase update")
    _require(update_rms > 0.0, "Generator phase did not update")
    step_record = three._composition_record(
        direction,
        step=cursor + 1,
        phase_displacement=three._norm(generator.detach() - clean_generator, "step600 phase displacement"),
        phase_update_rms=update_rms,
    )
    _require(step_record["trainable_parameter_names"] == ["generator.phase"], "training parameter scope changed")
    _require(step_record["E_D_in_backward"] is False, "E_D entered the objective backward path")
    for evaluator_name, model in (("E_S", smooth), ("E_I", independent), ("E_C", ec), ("E_D", ed)):
        _require(all(parameter.grad is None for parameter in model.parameters()), f"{evaluator_name} parameter gradient was retained")
    return step_record


def _rng_snapshot() -> dict[str, Any]:
    return {
        "cpu": torch.random.get_rng_state().clone(),
        "cuda": [value.clone() for value in torch.cuda.get_rng_state_all()],
    }


def _restore_rng(value: Mapping[str, Any]) -> None:
    cpu = _tensor(value.get("cpu"), "recovery CPU RNG")
    cuda = value.get("cuda")
    _require(isinstance(cuda, list) and cuda and all(isinstance(item, torch.Tensor) for item in cuda), "recovery CUDA RNG state is malformed")
    torch.random.set_rng_state(cpu.cpu())
    torch.cuda.set_rng_state_all([item.cpu() for item in cuda])

def _restore_recovery(
    consumer: Any,
    optimizer: torch.optim.Optimizer,
    checkpoint: Mapping[str, Any],
    config: Mapping[str, Any],
    expected_encoder_phase: torch.Tensor,
) -> tuple[int, list[dict[str, Any]], dict[str, torch.Tensor]]:
    _require(checkpoint.get("schema") == CHECKPOINT_SCHEMA, "recovery checkpoint schema changed")
    _require(checkpoint.get("data_binding") == _data_binding(config), "recovery fixed-bank identity changed")
    cursor = int(checkpoint.get("cursor", -1))
    _require(0 <= cursor <= UPDATE_COUNT, "recovery cursor is outside the fixed update range")
    params = dict(consumer.runtime.named_parameters())
    generator = _tensor(checkpoint.get("generator_phase"), "recovery Generator phase")
    encoder = _tensor(checkpoint.get("encoder_phase"), "recovery Encoder phase")
    _require(generator.shape == params["generator.phase"].shape and generator.dtype == params["generator.phase"].dtype, "recovery Generator shape or dtype changed")
    _require(encoder.shape == params["encoder.phase"].shape and encoder.dtype == params["encoder.phase"].dtype, "recovery Encoder shape or dtype changed")
    _require(
        torch.equal(encoder, expected_encoder_phase.detach().cpu()),
        "recovery Encoder differs from the accepted frozen state",
    )
    params["generator.phase"].data.copy_(generator.to(params["generator.phase"].device))
    params["encoder.phase"].data.copy_(encoder.to(params["encoder.phase"].device))
    optimizer.load_state_dict(copy.deepcopy(_mapping(checkpoint.get("optimizer_state"), "recovery Adam state")))
    trace = [dict(row) for row in checkpoint.get("trace", [])]
    _require(len(trace) == cursor, "recovery trace and update cursor disagree")
    raw_snapshots = _mapping(checkpoint.get("measurement_phase_snapshots"), "recovery measurement phase snapshots")
    expected_snapshot_steps = {str(step) for step in CHECKPOINT_MEASUREMENTS if step <= cursor}
    _require(set(raw_snapshots) == expected_snapshot_steps, "recovery measurement phase snapshot set differs from cursor")
    measurement_phase_snapshots: dict[str, torch.Tensor] = {}
    for step, raw_snapshot in raw_snapshots.items():
        snapshot = _tensor(raw_snapshot, f"recovery phase snapshot {step}")
        _require(snapshot.shape == params["generator.phase"].shape and snapshot.dtype == params["generator.phase"].dtype, f"recovery phase snapshot {step} geometry changed")
        _finite(snapshot, f"recovery phase snapshot {step}")
        measurement_phase_snapshots[step] = snapshot.cpu().clone()
    if cursor:
        state = optimizer.state.get(params["generator.phase"])
        _require(isinstance(state, Mapping) and float(_tensor(state.get("step"), "Adam recovery step").item()) == float(cursor), "recovery Adam step differs from cursor")
        _finite(_tensor(state.get("exp_avg"), "Adam first moment"), "Adam first moment")
        _finite(_tensor(state.get("exp_avg_sq"), "Adam second moment"), "Adam second moment")
    _restore_rng(_mapping(checkpoint.get("rng_state"), "recovery RNG state"))
    return cursor, trace, measurement_phase_snapshots


def _resource_projection(
    update_seconds: float,
    measurement_seconds: float,
    dev_seconds: float,
    write_seconds: float,
) -> dict[str, float]:
    projected_training = update_seconds * UPDATE_COUNT * 1.25
    projected_measurements = measurement_seconds * 4.0
    projected_dev = dev_seconds * 1.25
    projected_write = write_seconds * 1.5
    total = projected_training + projected_measurements + projected_dev + projected_write
    return {
        "projected_training_seconds": projected_training,
        "projected_fit_observation_measurement_seconds": projected_measurements,
        "projected_full_dev_confirmation_seconds": projected_dev,
        "projected_artifact_write_seconds": projected_write,
        "projected_total_seconds": total,
    }


def _smoke_lifecycle(
    identity: ResearchRunIdentity,
    receipt: PreflightReceipt,
    sample_summary: Mapping[str, Any],
    temp_root: Path,
) -> dict[str, Any]:
    control = ResearchRunControl(identity, AUTHORIZATION_ID)
    control.mark_preflight_ready(receipt)
    control.start_execution()
    attempt_name = identity.scientific_attempt_id.rsplit(":", 1)[-1]
    attempt = temp_root / attempt_name
    effective = control.open_scientific_attempt(attempt, ATTEMPT_ID, receipt)
    heartbeat = HeartbeatWriter(attempt, effective.run_id, total_steps=UPDATE_COUNT)
    heartbeat.update(state=RUN_PREPARING, phase="PREPARING", step=0, force=True)
    heartbeat.update(state=RUNNING, phase="TRAINING", step=3, force=True)
    write_started = time.perf_counter()
    atomic_write_new_json(attempt / "smoke_report.json", {"smoke_only": True, "summary": dict(sample_summary)}, artifact_kind="canary lifecycle smoke report")
    heartbeat.update(state=RUNNING, phase="FINALIZING", step=3, force=True)
    control.complete()
    atomic_write_new_json(
        attempt / "status.json",
        {"state": RUN_COMPLETED, "phase": "COMPLETED", "smoke_only": True, "attempt_id": effective.scientific_attempt_id},
        artifact_kind="canary lifecycle smoke status",
    )
    _require((attempt / "smoke_report.json").is_file() and (attempt / "status.json").is_file(), "production lifecycle smoke did not finalize")
    return {"status": "PASS", "heartbeat_phases": ["PREPARING", "TRAINING", "FINALIZING"], "report_written": True, "artifact_write_seconds": time.perf_counter() - write_started, "terminal_state": control.state}


def _objective_parity_check(
    consumer: Any,
    smooth: torch.nn.Module,
    independent: torch.nn.Module,
    ec: torch.nn.Module,
    data: Mapping[str, torch.Tensor],
    references: Any,
) -> dict[str, Any]:
    indices = torch.arange(BATCH_SIZE, device=consumer.device)
    images = data["fit_images"].index_select(0, indices)
    generator = dict(consumer.runtime.named_parameters())["generator.phase"]
    old_direction = three._sic_direction(consumer, smooth, independent, ec, images, references)
    old_gradient = old_direction["g_total"].detach().clone()
    new_direction = _canonical_direction_from_ols(
        consumer,
        smooth,
        independent,
        ec,
        data["fit_ols"].index_select(0, indices),
        data["fit_targets_28"].index_select(0, indices),
        data["fit_targets_slm"].index_select(0, indices),
        data["fit_encoder_plane_powers"].index_select(0, indices),
        references,
    )
    comparisons: dict[str, Any] = {}
    for name in ("L_S", "L_I", "L_C", "L_rec", "L_op"):
        left = float(old_direction["bundle"]["losses"][name].detach().cpu())
        right = float(new_direction["bundle"]["losses"][name].detach().cpu())
        _require(math.isclose(left, right, rel_tol=1.0e-5, abs_tol=1.0e-7), f"fixed-OLS canonical {name} differs from accepted objective semantics")
        comparisons[name] = {"accepted_path": left, "fixed_OLS_path": right, "absolute_difference": abs(left - right)}
    max_difference = float((old_gradient - new_direction["g_total"]).abs().max().detach().cpu())
    cosine = three._cosine(old_gradient, new_direction["g_total"], "accepted/fixed-OLS canonical gradient")
    _require(torch.allclose(old_gradient, new_direction["g_total"], rtol=1.0e-5, atol=1.0e-8), "fixed-OLS semantic direction differs from accepted objective semantics")
    semantic_disabled = new_direction["g_rec_safe"] + new_direction["g_op_safe"]
    disabled_distance = three._norm(new_direction["g_total"] - semantic_disabled, "disabled semantic control delta")
    _require(disabled_distance > 0.0, "disabled-semantic negative control did not change the gradient")
    _require(generator.grad is None, "objective parity probe populated Generator .grad")
    return {
        "status": "PASS",
        "losses": comparisons,
        "accepted_gradient_cosine": cosine,
        "gradient_max_absolute_difference": max_difference,
        "semantic_disabled_negative_control": {"changed_direction": True, "delta_norm": disabled_distance},
        "E_D_in_backward": False,
    }


def _preflight(
    root: Path,
    inputs: Mapping[str, Any],
    config: Mapping[str, Any],
    source: Mapping[str, Any],
    identity: ResearchRunIdentity,
    consumer: Any,
    optimizer: torch.optim.Optimizer,
    smooth: torch.nn.Module,
    independent: torch.nn.Module,
    ec: torch.nn.Module,
    ed: torch.nn.Module,
    runtime_data: Mapping[str, torch.Tensor],
    *,
    task_started: float,
    initial_resource_stages: list[dict[str, Any]],
) -> tuple[dict[str, Any], PreflightReceipt]:
    start = time.perf_counter()
    device = consumer.device
    resource_stages = list(initial_resource_stages)
    evaluator_states = _evaluator_snapshot({"E_S": smooth, "E_I": independent, "E_C": ec, "E_D": ed})
    runtime_state = _runtime_snapshot(consumer)
    generator = dict(consumer.runtime.named_parameters())["generator.phase"]
    clean_generator = generator.detach().clone()
    fit_start = inputs["fit_start_raw"].to(device)
    # Verify the fixed-bank production path on the registered first batch; the
    # complete step0 measurement runs inside the authorized attempt.
    with _phase_meter(device, "fixed_bank_step600_path_and_objective") as stage:
        fixed_start, fixed_start_raw = _fixed_bank_measurement(
            consumer,
            runtime_data["fit_ols"][:BATCH_SIZE],
            runtime_data["fit_targets_28"][:BATCH_SIZE],
            runtime_data["fit_targets_slm"][:BATCH_SIZE],
            runtime_data["fit_encoder_plane_powers"][:BATCH_SIZE],
            runtime_data["fit_labels"][:BATCH_SIZE],
            smooth,
            independent,
            ec,
            ed,
            consumer.references,
            include_ed=False,
            retain_raw=True,
            label_source="fixed TRAIN fit-bank label",
        )
        _require(fixed_start_raw is not None, "step600 fixed-bank raw measurement is missing")
        _require(torch.allclose(fixed_start_raw, fit_start.cpu()[:BATCH_SIZE], rtol=1.0e-5, atol=1.0e-7), "step600 fixed-input baseline raw CMOS does not reproduce V2")
        stage["fit_rows"] = BATCH_SIZE
        resource_stages.append(stage)
    with _phase_meter(device, "step600_observation_and_pixel_endpoint_measurement") as stage:
        obs_start, obs_start_raw = _fixed_bank_measurement(
            consumer,
            runtime_data["observation_ols"][:BATCH_SIZE],
            runtime_data["observation_targets_28"][:BATCH_SIZE],
            runtime_data["observation_targets_slm"][:BATCH_SIZE],
            runtime_data["observation_encoder_plane_powers"][:BATCH_SIZE],
            runtime_data["observation_labels"][:BATCH_SIZE],
            smooth,
            independent,
            ec,
            ed,
            consumer.references,
            include_ed=True,
            retain_raw=True,
            label_source="fixed TRAIN observation-bank label",
        )
        _require(obs_start_raw is not None, "step600 observation raw measurement is missing")
        _require(torch.allclose(obs_start_raw, inputs["observation_start_raw"][:BATCH_SIZE], rtol=1.0e-5, atol=1.0e-7), "step600 observation raw CMOS does not reproduce V2")
        pixel_observation = _eval_raw_population(
            inputs["pixel_observation_raw"][:BATCH_SIZE],
            runtime_data["observation_targets_28"][:BATCH_SIZE].detach().cpu(),
            runtime_data["observation_labels"][:BATCH_SIZE].detach().cpu(),
            smooth,
            independent,
            ec,
            ed,
            include_ed=True,
            label_source="fixed TRAIN observation-bank label",
        )
        stage["observation_rows"] = BATCH_SIZE
        stage["pixel_endpoint_rows"] = BATCH_SIZE
        resource_stages.append(stage)
    _require(obs_start["E_D"]["rows"] == BATCH_SIZE, "V2 step600 E_D observation smoke is incomplete")
    v2_pixel_prediction = inputs["v2_report"]["generator_only_arm"]["observation_end"]["per_sample"]["E_D_prediction"]
    _require(len(v2_pixel_prediction) == 100, "V2 pixel endpoint predictions do not cover the fixed observation bank")
    _require(pixel_observation["E_D"]["prediction_histogram"] == _prediction_histogram(v2_pixel_prediction[:BATCH_SIZE]), "V2 pixel endpoint E_D smoke histogram differs")
    _require(pixel_observation["row_vectors"]["E_D_predictions"] == v2_pixel_prediction[:BATCH_SIZE], "V2 pixel endpoint per-row E_D smoke predictions differ")
    parity = _objective_parity_check(consumer, smooth, independent, ec, runtime_data, consumer.references)
    with tempfile.TemporaryDirectory(prefix="m0-canonical-canary-preflight-") as temp_dir:
        smoke_recovery = RecoveryCheckpointManager(
            Path(temp_dir) / "recovery",
            identity,
            source_manifest_id=str(source["source_manifest_id"]),
            authorization_id=AUTHORIZATION_ID,
        )
        trace: list[dict[str, Any]] = []
        step_seconds: list[float] = []
        smoke_phase_snapshots = {"0": clean_generator.detach().cpu().clone()}
        with _phase_meter(device, "three_real_generator_updates_and_recovery") as stage:
            for cursor in range(3):
                update_started = time.perf_counter()
                trace.append(
                    _train_batch(
                        consumer,
                        optimizer,
                        smooth,
                        independent,
                        ec,
                        ed,
                        runtime_data,
                        consumer.references,
                        cursor,
                        clean_generator,
                    )
                )
                torch.cuda.synchronize(device)
                step_seconds.append(time.perf_counter() - update_started)
            params = dict(consumer.runtime.named_parameters())
            checkpoint_phase = params["generator.phase"].detach().cpu().clone()
            checkpoint_optimizer = copy.deepcopy(optimizer.state_dict())
            _require(not torch.equal(params["generator.phase"].detach(), clean_generator), "three-step smoke did not update Generator")
            _require(torch.equal(params["encoder.phase"].detach().cpu(), runtime_state["encoder.phase"]), "three-step smoke changed Encoder")
            _require(_tree_equal(three._frozen_runtime_state(consumer), {k: v for k, v in runtime_state.items() if k != "generator.phase"}), "three-step smoke changed frozen runtime")
            next_checkpoint = _recovery_payload(consumer, optimizer, 3, trace, config, smoke_phase_snapshots)
            smoke_recovery.save(next_checkpoint, step=3)
            expected_next_batch = _fit_batch_indices(3, device).detach().cpu()
            expected_next = _train_batch(
                consumer,
                optimizer,
                smooth,
                independent,
                ec,
                ed,
                runtime_data,
                consumer.references,
                3,
                clean_generator,
            )
            expected_phase = params["generator.phase"].detach().cpu().clone()
            expected_optimizer = copy.deepcopy(optimizer.state_dict())
            saved_wrapper = smoke_recovery.load(smoke_recovery.latest())
            cursor, trace, smoke_phase_snapshots = _restore_recovery(
                consumer,
                optimizer,
                saved_wrapper["payload"],
                config,
                runtime_state["encoder.phase"],
            )
            _require(cursor == 3 and torch.equal(_fit_batch_indices(cursor, device).cpu(), expected_next_batch), "resume scheduler did not restore next batch")
            _require(torch.equal(params["generator.phase"].detach().cpu(), checkpoint_phase), "recovery did not restore the exact pre-next-update Generator state")
            _require(_tree_equal(optimizer.state_dict(), checkpoint_optimizer), "recovery did not restore the exact pre-next-update Adam state")
            resumed_next = _train_batch(
                consumer,
                optimizer,
                smooth,
                independent,
                ec,
                ed,
                runtime_data,
                consumer.references,
                cursor,
                clean_generator,
            )
            phase_after_resume = params["generator.phase"].detach().cpu()
            phase_difference = (phase_after_resume - expected_phase).abs()
            recovery_next_update_close = _tree_close(
                {
                    "phase": phase_after_resume,
                    "optimizer": optimizer.state_dict(),
                    "trace": resumed_next,
                },
                {
                    "phase": expected_phase,
                    "optimizer": expected_optimizer,
                    "trace": expected_next,
                },
                rtol=RECOVERY_UPDATE_RTOL,
                atol=RECOVERY_UPDATE_ATOL,
            )
            _require(
                recovery_next_update_close,
                f"recovery next-update state differs beyond rtol={RECOVERY_UPDATE_RTOL}, atol={RECOVERY_UPDATE_ATOL}: max_phase_abs_difference={float(phase_difference.max().item())}, differing_phase_values={int((phase_difference != 0).sum().item())}",
            )
            stage["updates"] = 3
            stage["next_batch_after_resume"] = expected_next_batch.tolist()
            stage["checkpoint_restore_exact"] = True
            stage["next_update_numerical_equivalence"] = {
                "passed": True,
                "rtol": RECOVERY_UPDATE_RTOL,
                "atol": RECOVERY_UPDATE_ATOL,
                "max_phase_abs_difference": float(phase_difference.max().item()),
                "differing_phase_values": int((phase_difference != 0).sum().item()),
            }
            resource_stages.append(stage)
        with torch.no_grad():
            smoke_raw = consumer.runtime.forward_from_ols(runtime_data["observation_ols"][:BATCH_SIZE]).raw_intensity.detach()
            smoke_logits = ed(_evaluator_view(smoke_raw))
            _require(not smoke_raw.requires_grad and smoke_raw.grad_fn is None, "observation smoke raw CMOS retained a graph")
            _require(not smoke_logits.requires_grad and smoke_logits.grad_fn is None, "E_D smoke output retained a graph")
        smoke_summary = _eval_raw_population(
            smoke_raw.cpu(),
            runtime_data["observation_targets_28"][:BATCH_SIZE].cpu(),
            runtime_data["observation_labels"][:BATCH_SIZE].cpu(),
            smooth,
            independent,
            ec,
            ed,
            include_ed=True,
            label_source="fixed TRAIN observation-bank label",
        )
        with _phase_meter(device, "fixed_train_image_E_D_forward_resource_probe") as image_stage:
            with torch.no_grad():
                image_result = consumer.runtime.forward_from_image(runtime_data["fit_images"][:BATCH_SIZE])
                image_raw = image_result.raw_intensity
                _require(image_result.image_input is not None, "TRAIN image smoke did not return its target image")
                image_target_slm = image_result.image_input.slm_intensity
                image_reconstruction = (image_raw - image_target_slm).square().mean()
                image_power = three.optical_power_loss_from_powers(
                    three.contracted_plane_powers(image_result), consumer.references.power_reference
                )
                image_logits = ed(_evaluator_view(image_raw))
            _finite(image_raw, "TRAIN image smoke raw CMOS")
            _finite(image_reconstruction, "TRAIN image smoke reconstruction")
            _finite(image_power, "TRAIN image smoke power")
            _finite(image_logits, "TRAIN image smoke E_D logits")
            _require(not image_logits.requires_grad and image_logits.grad_fn is None, "TRAIN image E_D smoke retained a graph")
            image_stage["rows"] = BATCH_SIZE
            image_stage["forecast_rows"] = 5000
            image_stage["data_role"] = "TRAIN"
            del image_result, image_raw, image_target_slm, image_reconstruction, image_power, image_logits
            resource_stages.append(image_stage)
        smoke_lifecycle = _smoke_lifecycle(
            identity,
            PreflightReceipt(
                source_semantics_id=str(source["execution_source_semantics_id"]),
                authorization_id=AUTHORIZATION_ID,
                checks={CORRECTNESS_INVARIANTS: {"status": "READY"}, RESOURCE_ENVELOPE: {"status": "READY"}, IMPLEMENTATION_SELECTION: {"status": "READY"}},
                selected_engine="CUDA",
                resource={"preflight_smoke_only": True},
                status=PREFLIGHT_READY,
            ),
            {"rows": 10, "E_D_forward_only": smoke_summary["E_D"], "raw_graph_free": True},
            Path(temp_dir),
        )
        artifact_write_started = time.perf_counter()
        atomic_torch_save_new(
            Path(temp_dir) / "measurement_write_probe.pt",
            {"fit_step0_raw_cmos": fixed_start_raw, "observation_step0_raw_cmos": obs_start_raw},
            artifact_kind="canary measurement write probe",
        )
        atomic_write_new_json(
            Path(temp_dir) / "report_write_probe.json",
            {"source_semantics_id": source["execution_source_semantics_id"], "summary": dict(smoke_summary)},
            artifact_kind="canary report write probe",
        )
        smoke_lifecycle["artifact_write_seconds"] += time.perf_counter() - artifact_write_started
    # Restore the exact accepted model/optimizer and RNG before READY.
    consumer.load_trainable_state_dict(dict(inputs["authorities"]["checkpoint"]["trainable_parameter_state"]))
    optimizer = _freeze_models(consumer, smooth, independent, ec, ed)
    three._restore_rng(inputs["authorities"]["checkpoint"]["rng_state"])
    _require(_tree_equal(_runtime_snapshot(consumer), runtime_state), "preflight did not restore accepted step600 runtime state")
    _require(_tree_equal(_evaluator_snapshot({"E_S": smooth, "E_I": independent, "E_C": ec, "E_D": ed}), evaluator_states), "preflight changed evaluator state")
    _require(all(parameter.grad is None for model in (smooth, independent, ec, ed) for parameter in model.parameters()), "preflight retained evaluator gradients")
    if time.perf_counter() - task_started > MAX_PREFLIGHT_SECONDS:
        raise ResourceBudgetExceeded("preflight exceeded its fixed 10-minute budget")
    if time.perf_counter() - task_started > MAX_TOTAL_SECONDS:
        raise ResourceBudgetExceeded("preflight consumed the total canary budget")
    update_seconds = sum(step_seconds) / len(step_seconds)
    by_phase = {str(stage["phase"]): stage for stage in resource_stages}
    measurement_seconds = float(by_phase["fixed_bank_step600_path_and_objective"]["elapsed_seconds"])
    measurement_seconds += float(by_phase["step600_observation_and_pixel_endpoint_measurement"]["elapsed_seconds"])
    dev_seconds = float(by_phase["fixed_train_image_E_D_forward_resource_probe"]["elapsed_seconds"]) * (5000.0 / BATCH_SIZE)
    write_probe_seconds = float(smoke_lifecycle["artifact_write_seconds"])
    projection = _resource_projection(update_seconds, measurement_seconds, dev_seconds, write_probe_seconds)
    if projection["projected_total_seconds"] + (time.perf_counter() - task_started) > MAX_TOTAL_SECONDS:
        raise ResourceBudgetExceeded("projected canary exceeds the two-hour budget")
    preflight = {
        "schema": PREFLIGHT_SCHEMA,
        "status": PREFLIGHT_READY,
        "task_id": TASK_ID,
        "outer_gate": OUTER_GATE,
        "authorization_id": AUTHORIZATION_ID,
        "source_manifest": dict(source),
        "source_semantics_id": source["execution_source_semantics_id"],
        "source_manifest_id": source["source_manifest_id"],
        "input_authority_hashes": dict(inputs["input_authority_hashes"]),
        "checks": {
            "accepted_step600_and_data_identity": "PASS",
            "exact_V2_fit_and_observation_rows_and_targets": "PASS",
            "pixel_Generator_optimizer_parity": "PASS",
            "accepted_E_S_E_I_E_C_objective_identity": "PASS",
            "E_D_holdout_firewall": "PASS",
            "step0_full_DEV_E_D_reproduction": "REQUIRED_BEFORE_TRAINING_IN_ATTEMPT",
            "fixed_OLS_objective_semantics_parity": "PASS",
            "three_step_real_CUDA_update": "PASS",
            "disabled_semantic_negative_control": "PASS",
            "frozen_runtime_and_evaluators": "PASS",
            "observation_graph_free": "PASS",
            "recovery_next_update_equivalence": "PASS",
            "production_lifecycle_smoke": smoke_lifecycle["status"],
            "resource_envelope": "PASS",
            "write_once_attempt_path_absent": "PASS",
        },
        "objective_parity": parity,
        "step600_fit_measurement": fixed_start,
        "step600_observation_measurement": obs_start,
        "pixel_generator_observation_endpoint": pixel_observation,
        "three_step_updates": {"updates": 3, "trace": trace, "persistent_scientific_updates": 0},
        "recovery_equivalence": {
            "status": "PASS",
            "cursor": 3,
            "next_batch": expected_next_batch.tolist(),
            "checkpoint_restore_exact": True,
            "next_update_numerically_equivalent": True,
            "rtol": RECOVERY_UPDATE_RTOL,
            "atol": RECOVERY_UPDATE_ATOL,
        },
        "production_lifecycle_smoke": smoke_lifecycle,
        "resource_stages": resource_stages,
        "timing": {
            "elapsed_seconds": time.perf_counter() - start,
            "cumulative_elapsed_seconds": time.perf_counter() - task_started,
            "mean_generator_update_seconds": update_seconds,
            "projected": projection,
            "maximum_seconds": MAX_TOTAL_SECONDS,
            "maximum_preflight_seconds": MAX_PREFLIGHT_SECONDS,
        },
        "data_firewall": {"fit_train_rows": 200, "observation_train_rows": 100, "dev_rows_preflight": 0, "dev_rows_at_step0_and_300_only": 5000, "protected": 0, "official_test": 0, "sealed": 0},
    }
    receipt = PreflightReceipt(
        source_semantics_id=str(source["execution_source_semantics_id"]),
        authorization_id=AUTHORIZATION_ID,
        checks={
            CORRECTNESS_INVARIANTS: {key: value for key, value in preflight["checks"].items() if key not in {"resource_envelope"}},
            RESOURCE_ENVELOPE: {"status": "READY", "timing": projection, "stages": resource_stages},
            IMPLEMENTATION_SELECTION: {"status": "READY", "engine": "CUDA", "optimizer": config["optimizer"]},
        },
        selected_engine="CUDA",
        resource={"projection": projection, "stages": resource_stages},
        status=PREFLIGHT_READY,
    )
    preflight["preflight_receipt"] = receipt.to_dict()
    return preflight, receipt


def _contact_sheet(
    path: Path,
    *,
    labels: torch.Tensor,
    targets: torch.Tensor,
    baseline_raw: torch.Tensor,
    pixel_raw: torch.Tensor,
    canonical_raw: torch.Tensor,
    title: str,
) -> None:
    _require(targets.shape[0] == labels.numel() == baseline_raw.shape[0] == pixel_raw.shape[0] == canonical_raw.shape[0], "contact-sheet row identity differs")
    scale = 1.0
    target_view = F.adaptive_avg_pool2d(targets.detach().cpu().clamp(0.0, 1.0), (28, 28))
    views = [
        target_view,
        _evaluator_view(baseline_raw.detach().cpu()),
        _evaluator_view(pixel_raw.detach().cpu()),
        _evaluator_view(canonical_raw.detach().cpu()),
    ]
    count = int(labels.numel())
    tile = 30
    header = 24
    left = 32
    canvas = Image.new("L", (left + tile * 4, header + count * tile), 248)
    draw = ImageDraw.Draw(canvas)
    font = ImageFont.load_default()
    draw.text((left, 4), "TARGET", fill=0, font=font)
    draw.text((left + tile, 4), "STEP600", fill=0, font=font)
    draw.text((left + tile * 2, 4), "PIXEL", fill=0, font=font)
    draw.text((left + tile * 3, 4), "CANONICAL", fill=0, font=font)
    for row in range(count):
        top = header + row * tile
        draw.text((2, top + 9), f"{int(labels[row])}", fill=0, font=font)
        for column, view in enumerate(views):
            image = view[row, 0].clamp(0.0, scale).mul(255.0 / scale).round().to(torch.uint8).numpy()
            tile_image = Image.fromarray(image, mode="L").resize((28, 28), resample=Image.Resampling.NEAREST)
            canvas.paste(tile_image, (left + column * tile, top))
    with tempfile.NamedTemporaryFile(prefix=f".{path.stem}-", suffix=".png", dir=path.parent, delete=False) as handle:
        temp_path = Path(handle.name)
    try:
        canvas.save(temp_path, format="PNG", optimize=False)
        if path.exists():
            _require(file_sha256(path) == file_sha256(temp_path), f"existing {title} contact sheet differs")
        else:
            os.replace(temp_path, path)
    finally:
        temp_path.unlink(missing_ok=True)


def _attempt_status(path: Path, phase: str, **values: Any) -> None:
    prior = _json_read(path) if path.is_file() else {}
    atomic_write_json(
        path,
        {
            **prior,
            "schema": "m0_canonical_semantic_renderability_canary_status.v1",
            "task_id": TASK_ID,
            "outer_gate": OUTER_GATE,
            "authorization_id": AUTHORIZATION_ID,
            "state": values.pop("state", RUNNING),
            "phase": phase,
            "updated_at": time.time(),
            **values,
        },
    )

        path,
        "INVALID_HARNESS",
        state=RUN_INVALID_HARNESS,
        result=RUN_INVALID_HARNESS,
        classification=None,
        scientific_classification=None,
        failure_type=failure_type,
        failure=failure,
        **values,
    )


def _preflight_elapsed_fields(*, prior_seconds: float, run_started: float, receipt_started: float) -> dict[str, float]:
    now = time.perf_counter()
    return {
        "elapsed_seconds": now - receipt_started,
        "cumulative_preflight_elapsed_seconds": prior_seconds + now - run_started,
    }

def _read_completed_report_for_recovery(
    source: Mapping[str, Any],
    inputs: Mapping[str, Any],
    identity: ResearchRunIdentity,
) -> dict[str, Any] | None:
    path = ATTEMPT_DIR / "report.json"
    if not path.is_file():
        return None
    report = _json_read(path)
    _require(
        report.get("schema") == REPORT_SCHEMA
        and report.get("task_id") == TASK_ID
        and report.get("outer_gate") == OUTER_GATE
        and report.get("result") == "VALID_COMPLETED"
        and report.get("outer_gate_passed") is True
        and report.get("classification") in {CASE_A, CASE_B, CASE_C, CASE_D, CASE_E, CASE_F},
        "recovery final report identity or completion state is invalid",
    )
    _require(report.get("source_manifest") == source, "recovery final report source manifest differs")
    _require(report.get("run_identity") == identity.to_dict(), "recovery final report run identity differs")
    _require(report.get("input_authorities") == inputs["input_authority_hashes"], "recovery final report authority identities differ")
    _require(report.get("starting_authority", {}).get("sha256") == m0_rescue.START_CHECKPOINT_SHA256, "recovery final report step600 authority differs")
    updates = _mapping(report.get("training_updates"), "recovery completed training updates")
    adam_step_audit = report.get("adam_step_audit")
    _require(
        isinstance(adam_step_audit, Mapping)
        and
        updates.get("generator_phase") == UPDATE_COUNT
        and updates.get("completed_updates") == UPDATE_COUNT
        and updates.get("optimizer_state_internal_step") == float(UPDATE_COUNT)
        and adam_step_audit.get("step_before_scientific_update") == 0
        and adam_step_audit.get("step_after_first_real_update") == 1
        and adam_step_audit.get("final_step") == UPDATE_COUNT
        and updates.get("adam_step_audit") == dict(adam_step_audit)
        and updates.get("trainable_groups") == ["generator.phase"],
        "recovery final report update accounting is incomplete",
    )
    artifacts = _mapping(report.get("artifacts"), "recovery completed artifacts")
    _require(set(artifacts) == {"measurements", "step0_dev_baseline", "step300_checkpoint", "fit_contact_sheet", "observation_contact_sheet"}, "recovery final artifact set changed")
    attempt_root = ATTEMPT_DIR.resolve()
    for name, raw_record in artifacts.items():
        record = _mapping(raw_record, f"recovery artifact {name}")
        relative = Path(str(record.get("path", "")))
        _require(not relative.is_absolute() and ".." not in relative.parts, f"recovery artifact path is unsafe: {name}")
        artifact_path = (attempt_root / relative).resolve()
        _require(artifact_path.is_relative_to(attempt_root) and artifact_path.is_file(), f"recovery artifact is missing or escapes attempt: {name}")
        _require(file_sha256(artifact_path) == record.get("sha256"), f"recovery artifact hash differs: {name}")
        _require(artifact_path.stat().st_size == record.get("size_bytes"), f"recovery artifact size differs: {name}")
    firewall = _mapping(report.get("data_firewall"), "recovery report data firewall")
    _require(firewall.get("protected") == firewall.get("official_test") == firewall.get("sealed") == 0, "recovery report data firewall is not clean")
    return report

    }

def _load_run_models(root: Path, inputs: Mapping[str, Any], device: torch.device) -> tuple[Any, torch.nn.Module, torch.nn.Module, torch.nn.Module, torch.nn.Module, dict[str, torch.Tensor], torch.optim.Adam]:
    consumer, _parent_optimizer, smooth, independent, ec, ed = m0_rescue._load_models(root, inputs["authorities"], device)
    optimizer = _freeze_models(consumer, smooth, independent, ec, ed)
    _require(consumer.references is not None, "step600 frozen references are not bound")
    _require(consumer.weights is not None, "step600 frozen weights are not bound")
    frozen_runtime = three._frozen_runtime_state(consumer)
    evaluator_states = _evaluator_snapshot({"E_S": smooth, "E_I": independent, "E_C": ec, "E_D": ed})
    return consumer, smooth, independent, ec, ed, {"runtime": frozen_runtime, "evaluators": evaluator_states}, optimizer


def _check_frozen_state(consumer: Any, smooth: torch.nn.Module, independent: torch.nn.Module, ec: torch.nn.Module, ed: torch.nn.Module, frozen: Mapping[str, Any]) -> dict[str, bool]:
    _require(_tree_equal(frozen["runtime"], three._frozen_runtime_state(consumer)), "frozen encoder, OLS, W_OLS, or optics changed")
    models = {"E_S": smooth, "E_I": independent, "E_C": ec, "E_D": ed}
    after = _evaluator_snapshot(models)
    _require(_tree_equal(frozen["evaluators"], after), "one or more frozen evaluators changed")
    _require(all(parameter.grad is None for model in models.values() for parameter in model.parameters()), "frozen evaluator gradients are not empty")
    trainable = [name for name, parameter in consumer.runtime.named_parameters() if parameter.requires_grad]
    _require(trainable == ["generator.phase"], "trainable parameter scope changed")
    return {"encoder_frozen": True, "ols_w_ols_geometry_frozen": True, "evaluators_frozen": True, "E_D_in_backward": False, "trainable_parameters": trainable}


def _run_attempt(
    root: Path,
    inputs: Mapping[str, Any],
    config: Mapping[str, Any],
    source: Mapping[str, Any],
    identity: ResearchRunIdentity,
    receipt: PreflightReceipt,
    preflight: Mapping[str, Any],
    consumer: Any,
    optimizer: torch.optim.Optimizer,
    smooth: torch.nn.Module,
    independent: torch.nn.Module,
    ec: torch.nn.Module,
    ed: torch.nn.Module,
    frozen: Mapping[str, Any],
    runtime_data: Mapping[str, torch.Tensor],
    *,
    state_isolation: Mapping[str, Any],
    task_started: float,
    resume: bool,
) -> dict[str, Any]:
    generator = dict(consumer.runtime.named_parameters())["generator.phase"]
    accepted_generator = inputs["authorities"]["checkpoint"]["trainable_parameter_state"]["generator.phase"]
    _require(
        torch.equal(generator.detach().cpu(), _tensor(accepted_generator, "accepted step600 Generator phase")),
        "attempt_02 Generator start differs from accepted step600",
    )
    if not resume:
        _require(
            state_isolation.get("preflight_mutable_state_destroyed") is True
            and state_isolation.get("preflight_adam_object_released") is True
            and state_isolation.get("scientific_runtime_reconstructed_from_accepted_step600") is True
            and state_isolation.get("scientific_generator_matches_step600") is True
            and state_isolation.get("scientific_adam_step_before_attempt") == 0
            and not optimizer.state
            and _optimizer_internal_step(optimizer, generator) == 0,
            "attempt_02 cannot open without proven preflight/scientific state isolation and Adam step 0",
        )
    if resume:
        _require(ATTEMPT_DIR.is_dir(), "resume requires the existing canary attempt")
        old_status = _json_read(ATTEMPT_DIR / "status.json")
        _require(
            old_status.get("task_id") == TASK_ID
            and old_status.get("outer_gate") == OUTER_GATE
            and old_status.get("state") in {RUNNING, RUN_INTERRUPTED}
            and old_status.get("source_manifest_id") == source["source_manifest_id"],
            "stale attempt status identity is not resumable",
        )
        elapsed_before_resume = float(old_status.get("cumulative_elapsed_seconds", 0.0))
        _require(math.isfinite(elapsed_before_resume) and elapsed_before_resume >= 0.0, "stale attempt elapsed-time record is invalid")
        saved_source = _json_read(ATTEMPT_DIR / "source_manifest.json")
        _require(saved_source == source, "stale attempt source manifest differs")
        saved_preflight = _json_read(ATTEMPT_DIR / "preflight.json")
        _require(saved_preflight == preflight, "stale attempt preflight differs")
        control = ResearchRunControl(identity, AUTHORIZATION_ID, preflight_status=PREFLIGHT_READY, state=RUN_INTERRUPTED, receipt=receipt)
        effective_identity = control.resume_scientific_attempt(ATTEMPT_DIR, ATTEMPT_ID, receipt)
    else:
        _require(not ATTEMPT_DIR.exists(), "attempt_02 already exists and cannot be reopened as a new attempt")
        control = ResearchRunControl(identity, AUTHORIZATION_ID)
        control.mark_preflight_ready(receipt)
        control.start_execution()
        effective_identity = control.open_scientific_attempt(ATTEMPT_DIR, ATTEMPT_ID, receipt)
        _write_json_once(ATTEMPT_DIR / "source_manifest.json", source, artifact_kind="source manifest")
        _write_json_once(ATTEMPT_DIR / "preflight.json", preflight, artifact_kind="preflight receipt")
        execution_manifest = {
            "schema": "m0_canonical_semantic_renderability_canary_execution_manifest.v1",
            "task_id": TASK_ID,
            "outer_gate": OUTER_GATE,
            "authorization": {"id": AUTHORIZATION_ID, "token": AUTHORIZATION_TOKEN, "scope": "one fixed-bank Generator-only 300-update canary"},
            "input_authorities": dict(inputs["input_authority_hashes"]),
            "source_manifest": dict(source),
            "run_identity": effective_identity.to_dict(),
            "run_config": dict(config),
            "preflight": {
                "path": "preflight.json",
                "run_record_path": config["run_preflight_record_path"],
                "source_semantics_id": receipt.source_semantics_id,
                "status": receipt.status,
            },
            "scope": {"trainable": ["generator.phase"], "frozen": ["encoder.phase", "OLS", "W_OLS", "optical geometry", "E_S", "E_I", "E_C", "E_D"], "E_D_in_backward": False},
            "preflight_state_isolation": dict(state_isolation),
        }
        _write_json_once(ATTEMPT_DIR / "execution_manifest.json", execution_manifest, artifact_kind="execution manifest")
    control.state = RUN_SCIENTIFIC_ACTIVE
    step0_dev_path = ATTEMPT_DIR / "step0_dev_measurement.json"
    try:
        if step0_dev_path.exists():
            step0_dev_record = _json_read(step0_dev_path)
            step0_dev, step0_dev_vectors, step0_dev_resource_stage = _validate_step0_dev_record(step0_dev_record, source, inputs)
        else:
            _attempt_status(
                ATTEMPT_DIR / "status.json",
                "STEP0_DEV_BASELINE",
                state=RUNNING,
                source_manifest_id=source["source_manifest_id"],
                completed_updates=0,
                cumulative_elapsed_seconds=time.perf_counter() - task_started,
            )
            with _phase_meter(consumer.device, "attempt_step0_full_DEV_E_D_baseline") as step0_dev_resource_stage:
                step0_dev, step0_dev_vectors = _measure_dev(
                    consumer,
                    inputs["context"].images,
                    inputs["context"].labels,
                    inputs["context"].development_indices,
                    consumer.references,
                    ed,
                    step=0,
                )
                step0_dev_resource_stage["dev_rows"] = 5000
            step0_dev_record = _step0_dev_record(step0_dev, step0_dev_vectors, source, inputs, step0_dev_resource_stage)
            _write_json_once(step0_dev_path, step0_dev_record, artifact_kind="step0 DEV baseline")
            step0_dev, step0_dev_vectors, step0_dev_resource_stage = _validate_step0_dev_record(step0_dev_record, source, inputs)
        _require(
            abs(float(step0_dev["E_D"]["macro_recall"]) - EXPECTED_STEP600_DEV_E_D_MACRO) <= STEP600_DEV_E_D_TOLERANCE,
            f"step600 DEV E_D baseline does not reproduce 0.1318: {step0_dev['E_D']['macro_recall']}",
        )
    except Exception as error:
        control.invalid_harness()
        _mark_invalid_attempt_status(
            ATTEMPT_DIR / "status.json",
            completed_updates=0,
            cumulative_elapsed_seconds=time.perf_counter() - task_started,
            failure_type=type(error).__name__,
            failure=str(error),
        )
        return {"result": RUN_INVALID_HARNESS, "phase": "STEP0_DEV_BASELINE", "failure_type": type(error).__name__, "failure": str(error)}
    references = consumer.references
    generator = dict(consumer.runtime.named_parameters())["generator.phase"]
    clean_generator = inputs["authorities"]["checkpoint"]["trainable_parameter_state"]["generator.phase"].to(consumer.device)
    initial_rng = inputs["authorities"]["checkpoint"]["rng_state"]
    recovery = RecoveryCheckpointManager(
        ATTEMPT_DIR / "recovery",
        effective_identity,
        source_manifest_id=str(source["source_manifest_id"]),
        authorization_id=AUTHORIZATION_ID,
    )
    heartbeat = HeartbeatWriter(ATTEMPT_DIR, RUN_ID, total_steps=UPDATE_COUNT)
    cursor = 0
    trace: list[dict[str, Any]] = []
    measurement_phase_snapshots = {"0": clean_generator.detach().cpu().clone()}
    adam_step_audit: dict[str, int | None] = {"step_before_scientific_update": 0, "step_after_first_real_update": None, "final_step": None}
    if resume:
        latest = recovery.latest()
        _require(latest is not None, "stale attempt has no recovery checkpoint")
        wrapper = recovery.load(latest)
        _require(wrapper.get("source_manifest_id") == source["source_manifest_id"] and wrapper.get("step") == wrapper.get("payload", {}).get("cursor"), "recovery wrapper source or cursor changed")
        cursor, trace, measurement_phase_snapshots = _restore_recovery(
            consumer,
            optimizer,
            _mapping(wrapper.get("payload"), "recovery payload"),
            config,
            inputs["authorities"]["checkpoint"]["trainable_parameter_state"]["encoder.phase"],
        )
        persisted_adam_audit = _json_read(ATTEMPT_DIR / "status.json").get("adam_step_audit")
        if persisted_adam_audit is not None:
            adam_step_audit = dict(_mapping(persisted_adam_audit, "persisted Adam step audit"))
        _require(
            cursor == 0 or adam_step_audit.get("step_after_first_real_update") == 1,
            "resumed attempt lacks the durable first-update Adam step audit",
        )
    else:
        three._restore_rng(initial_rng)
        cursor = 0
        trace = []
    if resume:
        task_started = time.perf_counter() - elapsed_before_resume
        completed_report = _read_completed_report_for_recovery(source, inputs, effective_identity)
        if completed_report is not None:
            completed_artifacts = _mapping(completed_report.get("artifacts"), "recovered completed artifacts")
            control.complete()
            _attempt_status(
                ATTEMPT_DIR / "status.json",
                "COMPLETE",
                state=RUN_COMPLETED,
                completed_updates=UPDATE_COUNT,
                cumulative_elapsed_seconds=time.perf_counter() - task_started,
                classification=completed_report["classification"],
                result="VALID_COMPLETED",
                report_sha256=file_sha256(ATTEMPT_DIR / "report.json"),
                measurement_sha256=completed_artifacts["measurements"]["sha256"],
                terminal_at=time.time(),
            )
            _write_completed_heartbeat(heartbeat, classification=completed_report["classification"], step=UPDATE_COUNT)
            return completed_report
    _attempt_status(
        ATTEMPT_DIR / "status.json",
        "PREPARING",
        state=RUNNING,
        source_manifest_id=source["source_manifest_id"],
        execution_source_semantics_id=source["execution_source_semantics_id"],
        completed_updates=cursor,
        cumulative_elapsed_seconds=time.perf_counter() - task_started,
        preflight_status=PREFLIGHT_READY,
        authorization_id=AUTHORIZATION_ID,
        preflight_state_isolation=dict(state_isolation),
        adam_step_audit=dict(adam_step_audit),
    )
    deadline = task_started + MAX_TOTAL_SECONDS

    def recovery_payload(step: int) -> Mapping[str, Any]:
        return _recovery_payload(consumer, optimizer, step, trace, config, measurement_phase_snapshots)

    interrupt = InterruptController(
        heartbeat=heartbeat,
        recovery=recovery,
        payload_factory=recovery_payload,
        phase="TRAINING",
        total_steps=UPDATE_COUNT,
    )
    training_stage: dict[str, Any]
    execution_resource_stages: list[dict[str, Any]] = [step0_dev_resource_stage]
    with _phase_meter(consumer.device, "canonical_generator_training") as training_stage:
        heartbeat.update(state=RUN_PREPARING, phase="PREPARING", step=cursor, force=True)
        with interrupt:
            for update_index in range(cursor, UPDATE_COUNT):
                if time.perf_counter() >= deadline:
                    control.interrupted()
                    _attempt_status(
                        ATTEMPT_DIR / "status.json",
                        "INTERRUPTED",
                        state=RUN_INTERRUPTED,
                        completed_updates=update_index,
                        cumulative_elapsed_seconds=time.perf_counter() - task_started,
                        stop_reason="TOTAL_WALL_CLOCK_BUDGET_REACHED",
                        scientific_classification=None,
                    )
                    heartbeat.update(state=RUN_INTERRUPTED, phase="INTERRUPTED", step=update_index, force=True)
                    return {"result": "INTERRUPTED", "step": update_index, "reason": "TOTAL_WALL_CLOCK_BUDGET_REACHED"}
                trace.append(
                    _train_batch(
                        consumer,
                        optimizer,
                        smooth,
                        independent,
                        ec,
                        ed,
                        runtime_data,
                        references,
                        update_index,
                        clean_generator,
                    )
                )
                completed = update_index + 1
                current_adam_step = _optimizer_internal_step(optimizer, generator)
                _require(current_adam_step == completed, f"scientific Adam step differs from update count at update {completed}")
                if update_index == 0:
                    adam_step_audit["step_after_first_real_update"] = current_adam_step
                    _attempt_status(
                        ATTEMPT_DIR / "status.json",
                        "FIRST_SCIENTIFIC_UPDATE_VERIFIED",
                        state=RUNNING,
                        completed_updates=completed,
                        cumulative_elapsed_seconds=time.perf_counter() - task_started,
                        adam_step_audit=dict(adam_step_audit),
                    )
                if completed in CHECKPOINT_MEASUREMENTS:
                    measurement_phase_snapshots[str(completed)] = generator.detach().cpu().clone()
                if completed % CHECKPOINT_INTERVAL == 0 or completed == UPDATE_COUNT:
                    torch.cuda.synchronize(consumer.device)
                    recovery.save(recovery_payload(completed), step=completed)
                    resource_snapshot = _device_snapshot(consumer.device)
                    _check_resource(resource_snapshot, "canonical Generator training")
                    _attempt_status(
                        ATTEMPT_DIR / "status.json",
                        "TRAINING",
                        state=RUNNING,
                        completed_updates=completed,
                        cumulative_elapsed_seconds=time.perf_counter() - task_started,
                        recovery_checkpoint=recovery.latest().name if recovery.latest() else None,
                        resource_snapshot=resource_snapshot,
                        latest_losses={name: trace[-1][name] for name in ("L_S", "L_I", "L_C", "L_rec", "L_op")},
                    )
                    torch.cuda.reset_peak_memory_stats(consumer.device)
                if completed % 10 == 0:
                    heartbeat.update(state=RUNNING, phase="TRAINING", step=completed, latest_losses={name: trace[-1][name] for name in ("L_S", "L_I", "L_C", "L_rec", "L_op")})
                interrupted = interrupt.handle_safe_boundary(step=completed, latest_losses={name: trace[-1][name] for name in ("L_S", "L_I", "L_C", "L_rec", "L_op")})
                if interrupted is not None:
                    control.interrupted()
                    _attempt_status(ATTEMPT_DIR / "status.json", "INTERRUPTED", state=RUN_INTERRUPTED, completed_updates=completed, cumulative_elapsed_seconds=time.perf_counter() - task_started, recovery_checkpoint=interrupted.name)
                    return {"result": "INTERRUPTED", "step": completed, "recovery_checkpoint": interrupted.name}
        training_stage["completed_updates"] = len(trace)
    execution_resource_stages.append(training_stage)
    _require(len(trace) == UPDATE_COUNT, "canonical Generator update count is not exactly 300")
    _require(
        _optimizer_internal_step(optimizer, generator) == UPDATE_COUNT,
        "scientific Adam state did not begin at zero and advance exactly 300 times",
    )
    adam_step_audit["final_step"] = _optimizer_internal_step(optimizer, generator)
    _require(
        adam_step_audit.get("step_before_scientific_update") == 0
        and adam_step_audit.get("step_after_first_real_update") == 1
        and adam_step_audit.get("final_step") == UPDATE_COUNT,
        "Adam step audit does not show the required 0 to 1 to 300 progression",
    )
    _require(not torch.equal(generator.detach(), clean_generator), "canonical arm did not update generator.phase")
    _require(
        set(measurement_phase_snapshots) == {str(step) for step in CHECKPOINT_MEASUREMENTS},
        "canonical arm lacks a fixed descriptive measurement phase snapshot",
    )
    frozen_checks = _check_frozen_state(consumer, smooth, independent, ec, ed, frozen)

    trajectories: dict[str, Any] = {}
    fit_raw_by_step: dict[str, torch.Tensor] = {}
    observation_raw_by_step: dict[str, torch.Tensor] = {}
    observation_measurements: dict[str, Any] = {}
    fit_measurements: dict[str, Any] = {}
    dev_records: dict[str, Any] = {}
    dev_vectors: dict[str, dict[str, torch.Tensor]] = {}
    dev_vectors["0"] = step0_dev_vectors
    dev_records["0"] = step0_dev
    _attempt_status(ATTEMPT_DIR / "status.json", "POST_TRAIN_MEASUREMENTS", state=RUNNING, completed_updates=UPDATE_COUNT, finalization_phase="MEASUREMENTS_PENDING")
    heartbeat.update(state=RUNNING, phase="POST_TRAIN_COLLECTION", step=UPDATE_COUNT, force=True)
    for step in CHECKPOINT_MEASUREMENTS:
        phase_snapshot = measurement_phase_snapshots[str(step)]
        generator.data.copy_(phase_snapshot.to(device=generator.device))
        _require(torch.equal(generator.detach().cpu(), phase_snapshot), f"step{step} measurement did not restore its exact Generator phase")
        if time.perf_counter() >= deadline:
            raise ResourceBudgetExceeded("two-hour total canary budget reached before finalization")
        if step == 0:
            with _phase_meter(consumer.device, "step0_fixed_fit_and_observation_measurement") as stage:

        ATTEMPT_DIR / "observation_bank_comparison_contact_sheet.png",

def run_canary(*, repo_root: Path | None = None, resume: bool = False) -> dict[str, Any]:
    root = _root() if repo_root is None else repo_root.resolve()
    started = time.perf_counter()
    _require(torch.cuda.is_available() and torch.cuda.device_count() >= 1, "formal canonical semantic canary requires CUDA")
    device = torch.device("cuda:0")
    if resume:
        _require((root / ATTEMPT_DIR).is_dir(), "resume requires existing attempt_02")
    else:
        _require(not (root / ATTEMPT_DIR).exists(), "attempt_02 already exists; use explicit same-attempt recovery only")
    # A fresh run's preflight path is write-once. A resume consumes its existing
    # READY record and never creates a second scientific attempt.
    if resume:
        _require((root / ATTEMPT_DIR).is_dir(), "resume requires an existing attempt_02")
    initial_resource_stages: list[dict[str, Any]] = []
    with _phase_meter(device, "input_authority_and_fixed_bank_resolution") as stage:
        inputs = _load_inputs(root, resume=resume)
        optimizer_parity = _validate_pixel_optimizer_parity(inputs)
        stage["fit_rows"] = FIT_ROWS
        stage["observation_rows"] = OBSERVATION_ROWS
        initial_resource_stages.append(stage)
    with _phase_meter(device, "source_identity_and_run_configuration") as stage:
        config = _run_config(inputs)
        config["optimizer_parity_verified"] = optimizer_parity
        source = _create_source(root, inputs, config)
        identity = _identity(source, config, inputs["context"].data_identity)
        stage["source_semantics_id"] = source["execution_source_semantics_id"]
        initial_resource_stages.append(stage)
    with _phase_meter(device, "step600_runtime_and_frozen_evaluator_load") as stage:
        consumer, smooth, independent, ec, ed, frozen, optimizer = _load_run_models(root, inputs, device)
        stage["evaluator_names"] = ["E_S", "E_I", "E_C", "E_D"]
        initial_resource_stages.append(stage)
    with _phase_meter(device, "fixed_train_bank_runtime_preparation") as stage:
        runtime_data = _fixed_runtime_data(inputs, consumer, device)
        stage["fit_rows"] = FIT_ROWS
        stage["observation_rows"] = OBSERVATION_ROWS
        initial_resource_stages.append(stage)
    prior_preflight_seconds = sum(float(item["elapsed_seconds"]) for item in inputs["preflight_repair_history"] or [])
    task_started = started - prior_preflight_seconds
    preflight_path = _active_preflight_path(root, config, resume=resume)
    if resume:
        preflight = _json_read(preflight_path)
        _require(preflight.get("status") == PREFLIGHT_READY and preflight.get("source_semantics_id") == source["execution_source_semantics_id"], "resume READY receipt is not source-bound")
        receipt = PreflightReceipt.from_mapping(preflight["preflight_receipt"])
        execution_manifest = _json_read(root / ATTEMPT_DIR / "execution_manifest.json")
        state_isolation = dict(_mapping(execution_manifest.get("preflight_state_isolation"), "attempt_02 state-isolation evidence"))
        _require(
            state_isolation.get("preflight_mutable_state_destroyed") is True
            and state_isolation.get("scientific_runtime_reconstructed_from_accepted_step600") is True
            and state_isolation.get("scientific_generator_matches_step600") is True
            and state_isolation.get("scientific_adam_step_before_attempt") == 0,
            "attempt_02 state-isolation evidence is incomplete",
        )
    else:
        _require(not preflight_path.exists(), "write-once active preflight already exists")
        (root / RUNS_ROOT).mkdir(parents=True, exist_ok=True)
        preflight_started = time.perf_counter()
        try:
            preflight, receipt = _preflight(
                root,
                inputs,
                config,
                source,
                identity,
                consumer,
                optimizer,
                smooth,
                independent,
                ec,
                ed,
                runtime_data,
                task_started=task_started,
                initial_resource_stages=initial_resource_stages,
            )
        except Exception as error:
            resource_failure = isinstance(error, ResourceBudgetExceeded) or (
                isinstance(error, RuntimeError) and "out of memory" in str(error).casefold()
            )
            elapsed_fields = _preflight_elapsed_fields(
                prior_seconds=prior_preflight_seconds,
                run_started=started,
                receipt_started=preflight_started,
            )
            current_resource: dict[str, Any] | None = None
            try:
                current_resource = _device_snapshot(device)
            except Exception:
                pass
            blocked = {
                "schema": PREFLIGHT_SCHEMA,
                "status": PREFLIGHT_BLOCKED if resource_failure else RUN_INVALID_HARNESS,
                "classification": "PREFLIGHT_BLOCKED(resource)" if resource_failure else "INVALID_HARNESS",
                "task_id": TASK_ID,
                "outer_gate": OUTER_GATE,
                "authorization_id": AUTHORIZATION_ID,
                "run_record_path": config["run_preflight_record_path"],
                "source_manifest": source,
                "source_semantics_id": source["execution_source_semantics_id"],
                "failure_type": type(error).__name__,
                "failure": str(error),
                "exception_traceback": traceback.format_exc(limit=8),
                "elapsed_seconds": elapsed_fields["elapsed_seconds"],
                "prior_preflight_elapsed_seconds": prior_preflight_seconds,
                "cumulative_preflight_elapsed_seconds": elapsed_fields["cumulative_preflight_elapsed_seconds"],
                "resource_snapshot_at_failure": current_resource,
                "scientific_attempt_opened": False,
                "protected_test_sealed_reads": {"protected": 0, "official_test": 0, "sealed": 0},
            }
            atomic_write_new_json(preflight_path, blocked, artifact_kind="canary blocked preflight")
            return blocked
        elapsed_fields = _preflight_elapsed_fields(
            prior_seconds=prior_preflight_seconds,
            run_started=started,
            receipt_started=preflight_started,
        )
        _require(elapsed_fields["cumulative_preflight_elapsed_seconds"] <= MAX_PREFLIGHT_SECONDS, "preflight exceeded the approved cap")
        preflight["run_record_path"] = config["run_preflight_record_path"]
        preflight["prior_preflight_elapsed_seconds"] = prior_preflight_seconds
        preflight.update(elapsed_fields)
        atomic_write_new_json(preflight_path, preflight, artifact_kind="canary READY preflight")
    _require(preflight.get("status") == PREFLIGHT_READY and receipt.status == PREFLIGHT_READY, "canary preflight is not READY")
    _require(time.perf_counter() - task_started <= MAX_TOTAL_SECONDS, "preflight and execution exceed total budget")
    if not resume:
        preflight_generator = dict(consumer.runtime.named_parameters())["generator.phase"]
        preflight_adam_step = _optimizer_internal_step(optimizer, preflight_generator)
        preflight_references = _preflight_state_references(
            consumer,
            optimizer,
            smooth,
            independent,
            ec,
            ed,
            runtime_data,
        )
        _require(
            preflight_adam_step >= 3,
            "production preflight did not exercise and mutate its disposable Adam optimizer",
        )
        del consumer, smooth, independent, ec, ed, frozen, optimizer, runtime_data, preflight_generator
        scientific_state, state_isolation = _reconstruct_scientific_state(
            root,
            inputs,
            device,
            preflight_references,
        )
        consumer, smooth, independent, ec, ed, frozen, optimizer, runtime_data = scientific_state
        state_isolation["preflight_optimizer_internal_step_discarded"] = preflight_adam_step
        state_isolation["preflight_adam_object_released"] = True
        state_isolation["scientific_optimizer_is_distinct_fresh_object"] = True
        generator = dict(consumer.runtime.named_parameters())["generator.phase"]
        _require(not optimizer.state and _optimizer_internal_step(optimizer, generator) == 0, "scientific attempt must begin with fresh-zero Adam state")
    else:
        generator = dict(consumer.runtime.named_parameters())["generator.phase"]
        _require(not optimizer.state, "resumed scientific Adam must be fresh before its recovery checkpoint is loaded")
        state_isolation["scientific_runtime_reconstructed_from_accepted_step600"] = True
        state_isolation["scientific_adam_step_before_recovery"] = 0
    with render_v2._attempt_execution_lock(root):
        return _run_attempt(
            root,
            inputs,
            config,
            source,
            identity,
            receipt,
            preflight,
            consumer,
            optimizer,
            smooth,
            independent,
            ec,
            ed,
            frozen,
            runtime_data,
            state_isolation=state_isolation,
            task_started=started,
            resume=resume,
        )


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--resume", action="store_true", help="resume only the existing interrupted attempt_02")
    args = parser.parse_args()
    try:
        result = run_canary(resume=args.resume)
    except ResourceBudgetExceeded as error:
        if ATTEMPT_DIR.exists():
            try:
                _attempt_status(
                    ATTEMPT_DIR / "status.json",
                    "INTERRUPTED",
                    state=RUN_INTERRUPTED,
                    result=RUN_INTERRUPTED,
                    classification=None,
                    scientific_classification=None,
                    stop_reason="RESOURCE_BUDGET_EXCEEDED",
                    failure_type=type(error).__name__,
                    failure=str(error),
                )
            except Exception:
                pass
        print(json.dumps({"status": "RESOURCE_BUDGET_STOPPED", "failure_type": type(error).__name__, "failure": str(error), "attempt_exists": ATTEMPT_DIR.exists()}, indent=2))
        return 2
    except Exception as error:
        if ATTEMPT_DIR.exists():
            try:
                _mark_invalid_attempt_status(
                    ATTEMPT_DIR / "status.json",
                    failure_type=type(error).__name__,
                    failure=str(error),
                )
            except Exception:
                pass
        print(json.dumps({"status": "INVALID_HARNESS", "failure_type": type(error).__name__, "failure": str(error), "attempt_exists": ATTEMPT_DIR.exists()}, indent=2))
        return 2
    print(json.dumps(result, indent=2))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
