def _joint_norm(vectors: Sequence[torch.Tensor]) -> float:
    _require(bool(vectors), "joint vector is empty")
    squared = sum(float(value.detach().to(torch.float64).square().sum().cpu()) for value in vectors)
    result = math.sqrt(squared)
    _require(math.isfinite(result), "joint vector norm is nonfinite")
    return result


def _joint_rms(vectors: Sequence[torch.Tensor]) -> float:
    count = sum(value.numel() for value in vectors)
    _require(count > 0, "joint vector has no elements")
    squared = sum(float(value.detach().to(torch.float64).square().sum().cpu()) for value in vectors)
    result = math.sqrt(squared / count)
    _require(math.isfinite(result) and result > 0.0, "joint canonical direction RMS is zero or nonfinite")
    return result


def _joint_canonical_gradient(
    evaluator_gradients: Mapping[str, Sequence[torch.Tensor]], *, delta_sic: float
) -> tuple[tuple[torch.Tensor, ...], dict[str, Any]]:
    """Apply the accepted SIC operator over one concatenated joint vector."""
    _require(set(evaluator_gradients) == {"E_S", "E_I", "E_C"}, "canonical operator requires exactly E_S/E_I/E_C")
    groups = [tuple(evaluator_gradients[name]) for name in ("E_S", "E_I", "E_C")]
    _require(all(len(group) == len(groups[0]) and len(group) > 0 for group in groups), "canonical evaluator gradient parameter layouts differ")
    template = groups[0]
    for name, group in zip(("E_S", "E_I", "E_C"), groups, strict=True):
        _require(all(left.shape == right.shape and left.dtype == right.dtype for left, right in zip(template, group, strict=True)), f"{name} gradient layout differs from the joint trainable vector")
        _require(all(bool(torch.isfinite(value).all().item()) for value in group), f"{name} gradient is nonfinite")
    norms = [_joint_norm(group) for group in groups]
    _require(all(value > three.JOINT_DEGENERACY_TOLERANCE for value in norms), "accepted canonical evaluator gradient is zero or degenerate")
    units = [tuple(value / norm for value in group) for group, norm in zip(groups, norms, strict=True)]
    summed = tuple(sum(unit[index] for unit in units) for index in range(len(template)))
    direction_norm = _joint_norm(summed)
    _require(direction_norm > three.JOINT_DEGENERACY_TOLERANCE, "accepted canonical joint vector is degenerate")
    rms = _joint_rms(summed)
    result = tuple((value / rms) * float(delta_sic) for value in summed)
    _require(all(bool(torch.isfinite(value).all().item()) for value in result), "scaled accepted canonical gradient is nonfinite")
    metrics = {
        "evaluator_joint_gradient_norms": dict(zip(("E_S", "E_I", "E_C"), norms, strict=True)),
        "normalized_sum_joint_norm": direction_norm,
        "normalized_sum_joint_rms": rms,
        "delta_SIC": float(delta_sic),
        "normalization_scope": "joint_encoder_generator_vector",
        "combination_rule": CANONICAL_COMBINATION_RULE,
        "scalar_canonical_loss_defined": False,
    }
    return result, metrics


def _calibrate_kappa_from_gradients(
    weighted_reconstruction_gradients: Sequence[Sequence[torch.Tensor]],
    canonical_gradients: Sequence[Sequence[torch.Tensor]],
    *,
    beta: float | None = None,
) -> tuple[float, list[dict[str, float]]]:
    _require(len(weighted_reconstruction_gradients) == 20 and len(canonical_gradients) == 20, "kappa calibration requires the exact twenty registered TRAIN batches")
    rows: list[dict[str, float]] = []
    for batch_index, (rec_gradient, canonical_gradient) in enumerate(zip(weighted_reconstruction_gradients, canonical_gradients, strict=True)):
        rec_norm = _joint_norm(rec_gradient)
        canonical_norm = _joint_norm(canonical_gradient)
        _require(rec_norm > 0.0 and canonical_norm > three.JOINT_DEGENERACY_TOLERANCE, f"kappa calibration batch {batch_index} has zero gradient support")
        ratio = rec_norm / canonical_norm
        _require(math.isfinite(ratio) and ratio > 0.0, f"kappa calibration batch {batch_index} ratio is invalid")
        rows.append({"batch_index": batch_index, "weighted_reconstruction_norm": rec_norm, "canonical_gradient_norm": canonical_norm, "ratio": ratio})
    result = statistics.median(row["ratio"] for row in rows)
    _require(math.isfinite(result) and result > 0.0, "median kappa is zero or nonfinite")
    if beta is not None:
        _require(math.isfinite(float(beta)) and float(beta) > 0.0, "fixed beta is not finite and positive")
    return result, rows


def _calibrate_kappa_from_norm_rows(rows: Sequence[Mapping[str, Any]], *, beta: float) -> tuple[float, list[dict[str, Any]]]:
    _require(len(rows) == 20, "kappa calibration requires the exact twenty registered TRAIN batches")
    _require(math.isfinite(float(beta)) and float(beta) > 0.0, "fixed beta is not finite and positive")
    normalized: list[dict[str, Any]] = []
    for index, source in enumerate(rows):
        rec_norm = float(source["weighted_reconstruction_norm"])
        canonical_norm = float(source["canonical_gradient_norm"])
        _require(rec_norm > 0.0 and canonical_norm > three.JOINT_DEGENERACY_TOLERANCE, f"kappa calibration batch {index} has zero gradient support")
        ratio = rec_norm / canonical_norm
        _require(math.isfinite(ratio) and ratio > 0.0, f"kappa calibration batch {index} ratio is invalid")
        normalized.append({**dict(source), "weighted_reconstruction_norm": rec_norm, "canonical_gradient_norm": canonical_norm, "ratio": ratio})
    result = statistics.median(row["ratio"] for row in normalized)
    _require(math.isfinite(result) and result > 0.0, "median kappa is zero or nonfinite")
    return result, normalized


def _add_gradient_vectors(
    bogt_gradient: Sequence[torch.Tensor], canonical_gradient: Sequence[torch.Tensor], *, kappa: float
) -> tuple[torch.Tensor, ...]:
    _require(len(bogt_gradient) == len(canonical_gradient) and len(bogt_gradient) > 0, "BOGT and canonical joint gradient layouts differ")
    _require(math.isfinite(kappa) and kappa > 0.0, "frozen kappa is invalid")
    result = tuple(left + float(kappa) * right for left, right in zip(bogt_gradient, canonical_gradient, strict=True))
    _require(all(bool(torch.isfinite(value).all().item()) for value in result), "combined parameter-gradient vector is nonfinite")
    return result


def _classify_result(
    *,
    generation: bool,
    safe: bool,
    optical: bool,
    s1_delta: float,
    s1_ci_lower: float,
    dev_delta: float,
    dev_ci_lower: float,
    local: bool,
) -> str:
    transfer = s1_delta >= PRACTICAL_TRANSFER_DELTA and s1_ci_lower > 0.0 and (dev_delta >= 0.0 or dev_ci_lower > 0.0)
    semantic_improvement = (s1_delta > 0.0 and s1_ci_lower > 0.0) or (dev_delta > 0.0 and dev_ci_lower > 0.0)
    if generation and safe and optical:
        return "BOGT_CANONICAL_GRADIENT_GENERATION_SUPPORTED"
    if semantic_improvement and (not safe or not optical):
        return "BOGT_CANONICAL_GAIN_WITH_RECONSTRUCTION_OR_OPTICAL_TRADEOFF"
    if transfer and safe and optical:
        return "BOGT_CANONICAL_GRADIENT_TRANSFER_SUPPORTED"
    if (local or (dev_delta > 0.0 and dev_ci_lower > 0.0)) and not transfer:
        return "BOGT_CANONICAL_GRADIENT_LOCAL_RESPONSE_WITHOUT_GENERATION_TRANSFER"
    if not semantic_improvement and not local:
        return "BOGT_CANONICAL_GRADIENT_NO_ADVANTAGE"
    return "M0_BOGT_CANONICAL_GRADIENT_MIXED"

def _parameter_tuple(consumer: Any) -> tuple[torch.nn.Parameter, torch.nn.Parameter]:
    named = dict(consumer.runtime.named_parameters())
    _require(tuple(named) == ("encoder.phase", "generator.phase"), "runtime trainable parameter surface changed")

def _gradient_buffer_snapshot(consumer: Any) -> dict[str, Any]:
    named = dict(consumer.runtime.named_parameters())
    _require(tuple(named) == ("encoder.phase", "generator.phase"), "gradient ledger trainable parameter surface changed")
    total_numel = sum(parameter.numel() for parameter in named.values())
    non_null: list[str] = []
    nonzero_parameters: list[str] = []
    per_parameter_norms: dict[str, float] = {}
    nonzero_elements = 0
    squared_norms: list[float] = []
    for name, parameter in named.items():
        gradient = parameter.grad
        if gradient is None:
            per_parameter_norms[name] = 0.0
            continue
        non_null.append(name)
        norm = float(gradient.detach().to(torch.float64).norm().cpu())
        per_parameter_norms[name] = norm
        squared_norms.append(norm * norm)
        count = int(torch.count_nonzero(gradient).item())
        nonzero_elements += count
        if count > 0:
            nonzero_parameters.append(name)
    aggregate_norm = math.sqrt(sum(squared_norms))
    return {
        "trainable_parameter_tensor_count": len(named),
        "trainable_parameter_numel": int(total_numel),
        "grad_is_not_none_count": len(non_null),
        "nonzero_grad_parameter_count": len(nonzero_parameters),
        "nonzero_grad_element_count": nonzero_elements,
        "aggregate_grad_l2_norm": aggregate_norm,
        "max_parameter_grad_l2_norm": max(per_parameter_norms.values(), default=0.0),
        "per_parameter_grad_l2_norm": per_parameter_norms,
        "first_offending_parameter_names": non_null[:8],
        "first_nonzero_parameter_names": nonzero_parameters[:8],
        "clean": not non_null,
    }


def _assert_gradients_clean(consumer: Any, *, phase: str) -> dict[str, Any]:
    snapshot = _gradient_buffer_snapshot(consumer)
    _require(snapshot["clean"], f"trainable .grad buffers are not clean at {phase}: {snapshot['first_offending_parameter_names']}")
    return snapshot


def _record_gradient_buffers(
    recorder: Any | None,
    consumer: Any,
    phase: str,
    *,
    batch_index: int | None = None,
    evaluator: str | None = None,
    expected_clean: bool | None = None,
) -> dict[str, Any]:
    snapshot = _gradient_buffer_snapshot(consumer)
    if recorder is not None and hasattr(recorder, "record_gradient_buffers"):
        recorder.record_gradient_buffers(
            phase,
            snapshot,
            batch_index=batch_index,
            evaluator=evaluator,
            expected_clean=expected_clean,
        )
    return snapshot


def _gradient_tuple(value: Sequence[torch.Tensor | None], parameters: Sequence[torch.Tensor]) -> tuple[torch.Tensor, ...]:
    result = tuple(torch.zeros_like(parameter) if gradient is None else gradient.detach() for parameter, gradient in zip(parameters, value, strict=True))
    _require(all(bool(torch.isfinite(item).all().item()) for item in result), "gradient vector contains nonfinite values")
    return result


def _canonical_losses_from_raw(
    raw: torch.Tensor,
    images: torch.Tensor,
    smooth: torch.nn.Module,
    independent: torch.nn.Module,
    ec: torch.nn.Module,
) -> dict[str, torch.Tensor]:
    raw_view = three._cmos_view(raw)
    target_view = three._cmos_view(images)
    s_features, _ = smooth(raw_view)
    i_features, _ = independent(raw_view)
    c_features, _ = ec(raw_view)
    with torch.no_grad():
        s_target, _ = smooth(target_view)
        i_target, _ = independent(target_view)
        c_target, _ = ec(target_view)
    return {
        "E_S": three._normalized_feature_loss(s_features, s_target, "L_S"),
        "E_I": three._normalized_feature_loss(i_features, i_target, "L_I"),
        "E_C": three._normalized_feature_loss(c_features, c_target, "L_C"),
    }


def _canonical_loss_for_evaluator(raw: torch.Tensor, images: torch.Tensor, evaluator: torch.nn.Module, name: str) -> torch.Tensor:
    raw_view = three._cmos_view(raw)
    target_view = three._cmos_view(images)
    generated_features, _ = evaluator(raw_view)
    with torch.no_grad():
        target_features, _ = evaluator(target_view)
    loss_name = {"E_S": "L_S", "E_I": "L_I", "E_C": "L_C"}[name]
    return three._normalized_feature_loss(generated_features, target_features, loss_name)

        "cuda": [value.clone() for value in torch.cuda.get_rng_state_all()] if torch.cuda.is_available() else [],
    }


def _normalize_canonical_component(gradient: Sequence[torch.Tensor], *, name: str) -> tuple[tuple[torch.Tensor, ...], float]:
    norm = _joint_norm(gradient)
    _require(norm > three.JOINT_DEGENERACY_TOLERANCE, f"{name} canonical gradient is zero or degenerate")
    return tuple((value.detach() / norm).to(device="cpu") for value in gradient), norm


def _combine_normalized_canonical_components(
    components: Mapping[str, Sequence[torch.Tensor]], *, delta_sic: float
) -> tuple[tuple[torch.Tensor, ...], dict[str, Any]]:
    _require(set(components) == {"E_S", "E_I", "E_C"}, "canonical combiner requires exactly E_S/E_I/E_C")
    ordered = [tuple(components[name]) for name in ("E_S", "E_I", "E_C")]
    _require(all(len(item) == len(ordered[0]) and item for item in ordered), "normalized canonical component layouts differ")
    summed = tuple(sum(item[index] for item in ordered) for index in range(len(ordered[0])))
    direction_norm = _joint_norm(summed)
    _require(direction_norm > three.JOINT_DEGENERACY_TOLERANCE, "accepted canonical joint vector is degenerate")
    rms = _joint_rms(summed)
    result = tuple(((value / rms) * float(delta_sic)).detach() for value in summed)
    _require(all(bool(torch.isfinite(value).all().item()) for value in result), "scaled accepted canonical gradient is nonfinite")
    return result, {
        "evaluator_joint_gradient_norms": {},
        "normalized_sum_joint_norm": direction_norm,
        "normalized_sum_joint_rms": rms,
        "delta_SIC": float(delta_sic),
        "normalization_scope": "joint_encoder_generator_vector",
        "combination_rule": CANONICAL_COMBINATION_RULE,
        "scalar_canonical_loss_defined": False,
    }


def _sequential_canonical_gradient(
    consumer: Any,
    batch: torch.Tensor,
    evaluators: Sequence[torch.nn.Module],
    *,
    recorder: Any | None = None,
    batch_index: int | None = None,
    phase_prefix: str = "canonical",
    keep_components: bool = False,
) -> tuple[tuple[torch.Tensor, ...], dict[str, Any], dict[str, float], dict[str, tuple[torch.Tensor, ...]]]:
    """Recompute each accepted evaluator at the same theta while retaining only CPU unit vectors."""
    _require(len(evaluators) == 3, "sequential canonical operator requires E_S/E_I/E_C")
    parameters = _parameter_tuple(consumer)
    device = torch.device(consumer.device)
    input_hash = tensor_sha256(batch.detach().cpu())
    rng_before = _rng_snapshot()
    names = ("E_S", "E_I", "E_C")
    unit_components: dict[str, tuple[torch.Tensor, ...]] = {}
    component_norms: dict[str, float] = {}
    loss_values: dict[str, float] = {}
    entry_gradient_state = _record_gradient_buffers(
        recorder,
        consumer,
        f"{phase_prefix}_entry_before_gradient_construction",
        batch_index=batch_index,
        expected_clean=True,
    )
    _require(entry_gradient_state["clean"], "trainable .grad buffers were non-clean before canonical gradient construction")

    for name, evaluator in zip(names, evaluators, strict=True):
        _require(all(parameter.grad is None and not parameter.requires_grad for parameter in evaluator.parameters()), f"{name} must remain frozen and gradient-free")
        evaluator.eval()
        if recorder is not None:
            recorder.sample(f"{phase_prefix}_{name}_before_forward", batch_index=batch_index, evaluator=name)
        evaluator.to(device)
        if recorder is not None:
            recorder.sample(f"{phase_prefix}_{name}_after_residency_activation", batch_index=batch_index, evaluator=name)
        residency_gradient_state = _record_gradient_buffers(
            recorder,
            consumer,
            f"{phase_prefix}_{name}_after_residency_activation",
            batch_index=batch_index,
            evaluator=name,
            expected_clean=True,
        )
        _require(residency_gradient_state["clean"], f"trainable .grad buffers became non-clean during {name} residency activation")

        breakdown = consumer.forward_loss(batch, data_role=bogt.TRAIN_ROLE)
        _require(hasattr(breakdown, "result") and breakdown.result is not None, "production BOGT loss breakdown lacks its forward result")
        model_forward_gradient_state = _record_gradient_buffers(
            recorder,
            consumer,
            f"{phase_prefix}_{name}_after_model_forward",
            batch_index=batch_index,
            evaluator=name,
            expected_clean=True,
        )
        _require(model_forward_gradient_state["clean"], f"trainable .grad buffers became non-clean during {name} model forward")
        raw = breakdown.result.raw_intensity
        loss = _canonical_loss_for_evaluator(raw, batch, evaluator, name)
        evaluator_forward_gradient_state = _record_gradient_buffers(
            recorder,
            consumer,
            f"{phase_prefix}_{name}_after_evaluator_forward_before_autograd",
            batch_index=batch_index,
            evaluator=name,
            expected_clean=True,
        )
        _require(evaluator_forward_gradient_state["clean"], f"trainable .grad buffers became non-clean during {name} evaluator forward")
        if recorder is not None:
            recorder.sample(f"{phase_prefix}_{name}_after_evaluator_forward", batch_index=batch_index, evaluator=name)
        gradients = torch.autograd.grad(loss, parameters, retain_graph=False, allow_unused=True)
        after_autograd_gradient_state = _record_gradient_buffers(
            recorder,
            consumer,
            f"{phase_prefix}_{name}_after_autograd_grad",
            batch_index=batch_index,
            evaluator=name,
            expected_clean=True,
        )
        _require(after_autograd_gradient_state["clean"], f"trainable .grad buffers became non-clean during {name} autograd.grad")
        gradient = _gradient_tuple(gradients, parameters)
        unit, norm = _normalize_canonical_component(gradient, name=name)
        loss_values[name] = float(loss.detach().cpu())
        component_norms[name] = norm
        unit_components[name] = unit
        if recorder is not None:
            recorder.sample(f"{phase_prefix}_{name}_after_autograd_grad", batch_index=batch_index, evaluator=name)

        del gradients, gradient, unit, norm, loss, raw, breakdown
        gc.collect()
        if recorder is not None:
            recorder.sample(f"{phase_prefix}_{name}_after_graph_release", batch_index=batch_index, evaluator=name)
        released_gradient_state = _record_gradient_buffers(
            recorder,
            consumer,
            f"{phase_prefix}_{name}_after_graph_release",
            batch_index=batch_index,
            evaluator=name,
            expected_clean=True,
        )
        _require(released_gradient_state["clean"], f"trainable .grad buffers became non-clean while releasing {name} graph")
        evaluator.to(torch.device("cpu"))
        if device.type == "cuda":
            torch.cuda.empty_cache()
        if recorder is not None:
            recorder.sample(f"{phase_prefix}_{name}_after_residency_transition", batch_index=batch_index, evaluator=name)
        transition_gradient_state = _record_gradient_buffers(
            recorder,
            consumer,
            f"{phase_prefix}_{name}_after_residency_transition",
            batch_index=batch_index,
            evaluator=name,
            expected_clean=True,
        )
        _require(transition_gradient_state["clean"], f"trainable .grad buffers became non-clean by {name} residency transition")
        _require(all(parameter.grad is None for parameter in evaluator.parameters()), f"{name} accumulated evaluator parameter .grad")

    _require(tensor_sha256(batch.detach().cpu()) == input_hash, "canonical phase mutated its fixed TRAIN batch")
    rng_after = _rng_snapshot()
    _require(_tree_equal(rng_before, rng_after), "canonical evaluator recomputation changed RNG state")
    combined, metrics = _combine_normalized_canonical_components(unit_components, delta_sic=three.DELTA_SIC)
    metrics["evaluator_joint_gradient_norms"] = component_norms
    return combined, metrics, loss_values, unit_components if keep_components else {}


def _reference_canonical_gradient(
    consumer: Any,
    batch: torch.Tensor,
    evaluators: Sequence[torch.nn.Module],
    *,
    recorder: Any | None = None,
    batch_index: int | None = None,
) -> tuple[tuple[torch.Tensor, ...], dict[str, Any], dict[str, float], dict[str, tuple[torch.Tensor, ...]]]:
    """One-batch reference matching the pre-repair shared-graph accepted implementation."""
    _require(len(evaluators) == 3, "canonical reference requires E_S/E_I/E_C")
    parameters = _parameter_tuple(consumer)
    for name, evaluator in zip(("E_S", "E_I", "E_C"), evaluators, strict=True):
        evaluator.eval()
        evaluator.to(consumer.device)
        if recorder is not None:
            recorder.sample("canonical_reference_evaluator_residency", batch_index=batch_index, evaluator=name)
    if recorder is not None:
        recorder.sample("canonical_reference_before_forward", batch_index=batch_index, evaluator="E_S/E_I/E_C")
    breakdown = consumer.forward_loss(batch, data_role=bogt.TRAIN_ROLE)
    raw = breakdown.result.raw_intensity
    losses = _canonical_losses_from_raw(raw, batch, evaluators[0], evaluators[1], evaluators[2])
    if recorder is not None:
        recorder.sample("canonical_reference_after_evaluator_forward", batch_index=batch_index, evaluator="E_S/E_I/E_C")
    gradients_by_name: dict[str, tuple[torch.Tensor, ...]] = {}
    loss_values: dict[str, float] = {}
    for index, name in enumerate(("E_S", "E_I", "E_C")):
        gradients = torch.autograd.grad(losses[name], parameters, retain_graph=index < 2, allow_unused=True)
        gradients_by_name[name] = _gradient_tuple(gradients, parameters)
        loss_values[name] = float(losses[name].detach().cpu())
        del gradients
    combined, metrics = _joint_canonical_gradient(gradients_by_name, delta_sic=three.DELTA_SIC)
    units = {
        name: tuple((value.detach() / metrics["evaluator_joint_gradient_norms"][name]).to(device="cpu") for value in gradients_by_name[name])
        for name in ("E_S", "E_I", "E_C")
    }
    combined_cpu = tuple(value.detach().cpu() for value in combined)
    del losses, gradients_by_name, raw, breakdown
    gc.collect()
    for name, evaluator in zip(("E_S", "E_I", "E_C"), evaluators, strict=True):
        evaluator.to(torch.device("cpu"))
        if recorder is not None:
            recorder.sample("canonical_reference_evaluator_released", batch_index=batch_index, evaluator=name)
    if torch.device(consumer.device).type == "cuda":
        torch.cuda.empty_cache()
    if recorder is not None:
        recorder.sample("canonical_reference_after_graph_release", batch_index=batch_index, evaluator="E_S/E_I/E_C")
    return combined_cpu, metrics, loss_values, units


def _verify_canonical_equivalence(
    consumer: Any,
    batch: torch.Tensor,
    evaluators: Sequence[torch.nn.Module],
    *,
    recorder: Any,
    batch_index: int,
    row_indices_sha256: str,
) -> dict[str, Any]:
    parameters = _parameter_tuple(consumer)
    parameter_hashes_before = {name: tensor_sha256(value.detach().cpu()) for name, value in consumer.runtime.named_parameters()}
    evaluator_hashes_before = [_module_state_sha256(model) for model in evaluators]
    batch_hash_before = tensor_sha256(batch.detach().cpu())
    rng_before = _rng_snapshot()

    reference, reference_metrics, reference_losses, reference_units = _reference_canonical_gradient(
        consumer,
        batch,
        evaluators,
        recorder=recorder,
        batch_index=batch_index,
    )
    rng_after_reference = _rng_snapshot()
    _require(_tree_equal(rng_before, rng_after_reference), "reference canonical operator changed RNG state")
    sequential, sequential_metrics, sequential_losses, sequential_units = _sequential_canonical_gradient(
        consumer,
        batch,
        evaluators,
        recorder=recorder,
        batch_index=batch_index,
        phase_prefix="equivalence_sequential",
        keep_components=True,
    )
    _require(torch.allclose(reference[0], sequential[0], rtol=CANONICAL_EQUIVALENCE_RTOL, atol=CANONICAL_EQUIVALENCE_ATOL), "sequential Encoder canonical gradient differs from accepted reference")
    _require(torch.allclose(reference[1], sequential[1], rtol=CANONICAL_EQUIVALENCE_RTOL, atol=CANONICAL_EQUIVALENCE_ATOL), "sequential Generator canonical gradient differs from accepted reference")
    component_comparisons: dict[str, dict[str, Any]] = {}
    for name in ("E_S", "E_I", "E_C"):
        _require(
            all(torch.allclose(left, right, rtol=CANONICAL_EQUIVALENCE_RTOL, atol=CANONICAL_EQUIVALENCE_ATOL) for left, right in zip(reference_units[name], sequential_units[name], strict=True)),
            f"sequential normalized {name} component differs from accepted reference",
        )
        ref_norm = float(reference_metrics["evaluator_joint_gradient_norms"][name])
        seq_norm = float(sequential_metrics["evaluator_joint_gradient_norms"][name])
        _require(math.isclose(ref_norm, seq_norm, rel_tol=CANONICAL_EQUIVALENCE_RTOL, abs_tol=CANONICAL_EQUIVALENCE_ATOL), f"sequential {name} norm differs from accepted reference")
        component_comparisons[name] = {
            "gradient_allclose": True,
            "reference_norm": ref_norm,
            "sequential_norm": seq_norm,
            "direction_cosine": _joint_cosine(reference_units[name], sequential_units[name]),
        }
    _require(math.isclose(canary_norm := _joint_norm(reference), _joint_norm(sequential), rel_tol=CANONICAL_EQUIVALENCE_RTOL, abs_tol=CANONICAL_EQUIVALENCE_ATOL), "sequential joint canonical norm differs from accepted reference")
    _require(math.isclose(reference_metrics["normalized_sum_joint_rms"], sequential_metrics["normalized_sum_joint_rms"], rel_tol=CANONICAL_EQUIVALENCE_RTOL, abs_tol=CANONICAL_EQUIVALENCE_ATOL), "sequential canonical joint RMS differs from accepted reference")
    direction_cosine = _joint_cosine(reference, sequential)
    _require(math.isclose(direction_cosine, 1.0, rel_tol=0.0, abs_tol=CANONICAL_EQUIVALENCE_RTOL), "sequential canonical direction differs from accepted reference")
    for name in ("E_S", "E_I", "E_C"):
        _require(math.isclose(reference_losses[name], sequential_losses[name], rel_tol=CANONICAL_EQUIVALENCE_RTOL, abs_tol=CANONICAL_EQUIVALENCE_ATOL), f"sequential {name} loss differs from accepted reference")
    parameter_hashes_after = {name: tensor_sha256(value.detach().cpu()) for name, value in consumer.runtime.named_parameters()}
    evaluator_hashes_after = [_module_state_sha256(model) for model in evaluators]
    _require(parameter_hashes_before == parameter_hashes_after, "canonical equivalence changed step600 trainable parameters")
    _require(evaluator_hashes_before == evaluator_hashes_after, "canonical equivalence changed frozen evaluator state")
    _require(batch_hash_before == tensor_sha256(batch.detach().cpu()), "canonical equivalence changed the fixed TRAIN input")
    _require(_tree_equal(rng_before, _rng_snapshot()), "canonical equivalence changed RNG state")
    _require(all(parameter.grad is None for parameter in parameters), "canonical equivalence accumulated trainable .grad")
    _require(_frozen_evaluators_clear(evaluators), "canonical equivalence accumulated evaluator .grad or unfroze an evaluator")
    group_norms = {
        "encoder": {"reference": _joint_norm((reference[0],)), "sequential": _joint_norm((sequential[0],))},
        "generator": {"reference": _joint_norm((reference[1],)), "sequential": _joint_norm((sequential[1],))},
    }
    del reference, sequential, reference_units, sequential_units
    gc.collect()
    if torch.device(consumer.device).type == "cuda":
        torch.cuda.empty_cache()
    recorder.sample("canonical_equivalence_verified", batch_index=batch_index, evaluator="E_S/E_I/E_C")
    return {
        "status": "PASS",
        "method": "accepted shared-graph reference versus sequential per-evaluator recomputation at identical theta and TRAIN rows",
        "batch_index": batch_index,
        "row_indices_sha256": row_indices_sha256,
        "rtol": CANONICAL_EQUIVALENCE_RTOL,
        "atol": CANONICAL_EQUIVALENCE_ATOL,
        "encoder_allclose": True,
        "generator_allclose": True,
        "encoder_norm": group_norms["encoder"],
        "generator_norm": group_norms["generator"],
        "joint_norm": canary_norm,
        "joint_rms": {"reference": reference_metrics["normalized_sum_joint_rms"], "sequential": sequential_metrics["normalized_sum_joint_rms"]},
        "direction_cosine": direction_cosine,
        "components": component_comparisons,
        "losses": {name: {"reference": reference_losses[name], "sequential": sequential_losses[name]} for name in ("E_S", "E_I", "E_C")},
        "optimizer_created_or_mutated": False,
        "parameters_unchanged": True,
        "evaluators_unchanged": True,
        "rng_unchanged": True,
        "trainable_grad_accumulation": False,
        "E_D_on_gpu": False,
    }


def _compute_gradient_vectors(
    consumer: Any,
    batch: torch.Tensor,
    *,
    smooth: torch.nn.Module,
    independent: torch.nn.Module,
    ec: torch.nn.Module,
    recorder: Any | None = None,
    batch_index: int | None = None,
    phase_prefix: str = "training",
) -> dict[str, Any]:
    parameters = _parameter_tuple(consumer)
    entry_gradient_state = _record_gradient_buffers(
        recorder,
        consumer,
        f"{phase_prefix}_before_BOGT_gradient_construction",
        batch_index=batch_index,
        expected_clean=True,
    )
    _require(entry_gradient_state["clean"], "trainable .grad buffers were non-clean before BOGT gradient construction")
    breakdown = consumer.forward_loss(batch, data_role=bogt.TRAIN_ROLE)
    _require(hasattr(breakdown, "result") and breakdown.result is not None, "production BOGT loss breakdown lacks its forward result")
    terms = (
        consumer.weights.alpha * breakdown.l_dist,
        consumer.weights.beta * breakdown.l_rec,
        consumer.weights.gamma * breakdown.l_op,
    )
    bogt_gradient = tuple(torch.zeros_like(parameter) for parameter in parameters)
    dist_generator_norm = 0.0
    for index, term in enumerate(terms):
        gradients = torch.autograd.grad(term, parameters, retain_graph=index < 2, allow_unused=True)
        component = _gradient_tuple(gradients, parameters)
        if index == 0:
            dist_generator_norm = _joint_norm((component[1],))
        bogt_gradient = tuple(total + value for total, value in zip(bogt_gradient, component, strict=True))
        del gradients, component
    loss_values = {
        "L_dist": float(breakdown.l_dist.detach().cpu()),
        "L_rec": float(breakdown.l_rec.detach().cpu()),
        "L_op": float(breakdown.l_op.detach().cpu()),
        "L_total": float(breakdown.l_total.detach().cpu()),
    }
    bogt_gradient_state = _record_gradient_buffers(
        recorder,
        consumer,
        f"{phase_prefix}_after_BOGT_autograd_grad",
        batch_index=batch_index,
        expected_clean=True,
    )
    _require(bogt_gradient_state["clean"], "BOGT autograd.grad left trainable .grad buffers populated")
    del terms, breakdown
    gc.collect()
    bogt_gradient = tuple(value.detach().cpu() for value in bogt_gradient)
    canonical_gradient, canonical_metrics, canonical_losses, _ = _sequential_canonical_gradient(
        consumer,
        batch,
        (smooth, independent, ec),
        recorder=recorder,
        batch_index=batch_index,
        phase_prefix=phase_prefix,
    )
    post_canonical_gradient_state = _record_gradient_buffers(
        recorder,
        consumer,
        f"{phase_prefix}_after_canonical_gradient_construction",
        batch_index=batch_index,
        expected_clean=True,
    )
    _require(post_canonical_gradient_state["clean"], "canonical gradient construction left trainable .grad buffers populated")
    loss_values.update({f"ell_{name[-1]}": value for name, value in canonical_losses.items()})
    _require(all(bool(torch.isfinite(value).all().item()) for value in bogt_gradient), "weighted BOGT gradient is nonfinite")
    _require(all(math.isfinite(value) for value in loss_values.values()), "one or more training scalar measurements are nonfinite")
    _require(dist_generator_norm == 0.0, "L_dist has a nonzero direct Generator gradient")
    return {
        "losses": loss_values,
        "G_BOGT": bogt_gradient,
        "G_canonical": canonical_gradient,
        "canonical_metrics": canonical_metrics,
        "L_dist_generator_gradient_norm": dist_generator_norm,
    }

def _optimizer_steps(checkpoint: Mapping[str, Any]) -> list[int]:
    states = checkpoint.get("optimizer_state", {}).get("state")
    _require(isinstance(states, Mapping) and len(states) == 2, "step600 optimizer state does not contain the two trainable parameter moments")
    result = []
    for state in states.values():
        step = state.get("step")
        result.append(int(step.item()) if torch.is_tensor(step) else int(step))
    return sorted(result)


def _prediction_vectors(record: Mapping[str, Any], *, rows: int, name: str) -> tuple[torch.Tensor, torch.Tensor]:
    labels = torch.as_tensor(record.get("true_labels"), dtype=torch.long, device="cpu")
    predictions = torch.as_tensor(record.get("prediction_labels"), dtype=torch.long, device="cpu")
    _require(labels.shape == predictions.shape == (rows,), f"{name} prediction vectors do not cover the fixed {rows} rows")
    _require(bool(((labels >= 0) & (labels < 10) & (predictions >= 0) & (predictions < 10)).all().item()), f"{name} contains an invalid class index")
    return labels, predictions


def _classification_metrics(labels: torch.Tensor, predictions: torch.Tensor, *, role: str) -> dict[str, Any]:
    _require(labels.shape == predictions.shape and labels.ndim == 1, "classification vectors differ in shape")
    confusion = torch.bincount(labels * 10 + predictions, minlength=100).reshape(10, 10)
    counts = confusion.sum(1)
    recalls = confusion.diag().to(torch.float64) / counts.clamp_min(1)
    return {
        "rows": int(labels.numel()),
        "data_role": role,
        "macro_recall": float(recalls.mean()),
        "per_class_recall": recalls.tolist(),
        "minimum_class_recall": float(recalls.min()),
        "confusion_matrix": confusion.tolist(),
        "prediction_histogram": confusion.sum(0).tolist(),
        "class_8_prediction_mass": float((predictions == 8).to(torch.float64).mean()),
        "true_labels": labels.tolist(),
        "prediction_labels": predictions.tolist(),
    }


def _validate_population_metrics(record: Mapping[str, Any], *, rows: int, name: str) -> dict[str, Any]:
    labels, predictions = _prediction_vectors(record, rows=rows, name=name)
    metrics = _classification_metrics(labels, predictions, role=str(record.get("data_role", "unknown")))
    expected_matrix = record.get("confusion_matrix")
    if expected_matrix is not None:
        _require(metrics["confusion_matrix"] == expected_matrix, f"{name} saved confusion matrix disagrees with row-level predictions")
    expected_macro = record.get("macro_recall")
    if expected_macro is not None:
        _require(math.isclose(metrics["macro_recall"], float(expected_macro), rel_tol=0.0, abs_tol=1.0e-12), f"{name} saved macro recall disagrees with row-level predictions")
    return metrics


def _validate_authorities(root: Path) -> dict[str, Any]:
    legacy, start = pure_bogt._fixed_authorities(root)
    _require(file_sha256(root / START_CHECKPOINT) == START_CHECKPOINT_SHA256, "accepted step600 checkpoint hash changed")
    _require(file_sha256(root / START_REPORT) == START_REPORT_SHA256, "accepted step600 report hash changed")
    _require(start.get("step") == START_STEP and start.get("sampler_state", {}).get("cursor") == START_STEP, "accepted step600 checkpoint cursor changed")
    _require(_optimizer_steps(start) == [START_STEP, START_STEP], "accepted step600 Adam moments are not at step600")
    _require(len(start.get("training_history", [])) == START_STEP, "accepted step600 history length changed")

    pure = readout._validate_authorities(root)
    endpoint = pure["step1200"]
    pure_report_path = root / readout.ATTEMPT_DIR / "report.json"
    pure_report = _read_json(pure_report_path)
    _require(pure_report.get("result") == "VALID_COMPLETED", "pure-BOGT step1200 readout is not a complete valid measurement")
    _require(pure_report.get("outer_gate") == readout.OUTER_GATE and pure_report.get("classification") == readout.CLASSIFICATION_D, "pure-BOGT comparator readout identity/classification changed")
    _require(file_sha256(root / readout.PARENT_STATUS_PATH) == pure["parent_status_hash"], "pure-BOGT parent INVALID_HARNESS record changed")
    _require(pure["parent_attempt"].get("attempt_02_state") == "INVALID_HARNESS", "pure-BOGT parent attempt state is not preserved as INVALID_HARNESS")
    _require(endpoint.get("parent_checkpoint_sha256") == START_CHECKPOINT_SHA256 and endpoint.get("step") == END_STEP, "pure-BOGT comparator checkpoint is not descended from accepted step600")
    _require(_optimizer_steps(endpoint) == [END_STEP, END_STEP], "pure-BOGT comparator Adam moments are not at step1200")
    _require(endpoint.get("data_identity") == start.get("data_identity"), "step600 and pure-BOGT comparator data identities differ")
    _require(all(torch.equal(a, b) for a, b in zip(start["sampler_state"]["permutations"], endpoint["sampler_state"]["permutations"], strict=True)), "pure-BOGT comparator TRAIN ordering differs from accepted step600")

    canonical_report, canonical_source = accepted_canonical._objective_authority(root)
    _require(canonical_source.get("source_manifest_id") == accepted_canonical.THREE_EVALUATOR_SOURCE_MANIFEST_ID, "accepted canonical evaluator source manifest changed")
    gradient_path = root / readout.ATTEMPT_DIR / "gradient_analysis.json"
    gradient_analysis = _read_json(gradient_path)
    batches = readout.build_gradient_batches(start["sampler_state"]["permutations"])
    batch_indices = gradient_analysis.get("batch_indices")
    _require(gradient_analysis.get("schema") == readout.GRADIENT_SCHEMA and gradient_analysis.get("E_D_in_gradient_graph") is False, "fixed-batch gradient authority or E_D firewall changed")
    _require(batch_indices == [batch.tolist() for batch in batches], "fixed twenty calibration batches do not match saved TRAIN cursors 1200..1219")
    _require(len(torch.unique(torch.cat(batches))) == 200, "fixed calibration rows overlap")
    _require(_optimizer_steps(start) == [START_STEP, START_STEP], "step600 Adam step identity changed during authority resolution")

    populations: dict[str, Any] = {}
    for step_name in ("step600", "step1200"):
        section = pure_report.get(step_name)
        _require(isinstance(section, Mapping), f"pure readout is missing {step_name} population evidence")
        for population, rows in (("DEV", DEV_ROWS), ("S1", S1_ROWS)):
            record = section.get(population, {}).get("E_D")
            _require(isinstance(record, Mapping), f"pure readout is missing {step_name} {population} E_D evidence")
            populations[f"{step_name}_{population}"] = _validate_population_metrics(record, rows=rows, name=f"pure {step_name} {population}")

    _require(abs(populations["step600_DEV"]["macro_recall"] - 0.1318) <= 0.001, "accepted step600 DEV E_D baseline did not reproduce")
    _require(abs(populations["step600_S1"]["macro_recall"] - 0.1355) <= 0.002, "accepted step600 S1 E_D baseline did not reproduce")
    row_review_path = root / ROOT_CAUSE_REVIEW_03_PATH
    row_review = _read_json(row_review_path)
    reviewed_trace = row_review["authoritative_train_order"]["trace"]
    trace_paths = {
        "A_authoritative": reviewed_trace.get("A_authoritative", []),
        "B_uninterrupted": reviewed_trace.get("B_uninterrupted", []),
        "C_save_restore": reviewed_trace.get("C_save_restore", []),
    }
    _require(all(len(records) == 8 for records in trace_paths.values()), "reviewed order trace does not cover steps601-608 on all paths")
    for index, global_step in enumerate(range(START_STEP + 1, START_STEP + 9)):
        order_position = global_step - 1
        expected_indices = torch.stack([permutation[order_position] for permutation in start["sampler_state"]["permutations"]])
        expected_ids = expected_indices.tolist()
        expected_hash = tensor_sha256(expected_indices)
        endpoint_record = endpoint["training_history"][global_step - 1]
        _require(endpoint_record.get("step") == global_step and endpoint_record.get("train_indices_sha256") == expected_hash, f"pure BOGT row authority is invalid at global step {global_step}")
        for path_name, records in trace_paths.items():
            record = records[index]
            _require(
                record.get("global_step") == global_step
                and record.get("cursor_before") == order_position
                and record.get("cursor_after") == global_step
                and record.get("order_position") == order_position
                and record.get("row_ids") == expected_ids
                and record.get("row_sha256") == expected_hash
                and record.get("endpoint_history_row_sha256") == endpoint_record.get("train_indices_sha256"),
                f"root-cause review row trace {path_name} differs from accepted order at global step {global_step}",
            )
    _require(
        all(trace_paths["A_authoritative"][i]["row_sha256"] == trace_paths["B_uninterrupted"][i]["row_sha256"] == trace_paths["C_save_restore"][i]["row_sha256"] for i in range(8)),
        "reviewed PATH A/B/C row identities are not equal",
    )
    return {
        "legacy_authorities": legacy,
        "step600": start,
        "step600_report": _read_json(root / START_REPORT),
        "pure": pure,
        "pure_report": pure_report,
        "gradient_analysis": gradient_analysis,
        "gradient_batches": batches,
        "canonical_report": canonical_report,
        "canonical_source": canonical_source,
        "populations": populations,
        "data_identity": start["data_identity"],
        "authority_hashes": {
            "step600_checkpoint": START_CHECKPOINT_SHA256,
            "step600_report": START_REPORT_SHA256,
            "pure_step1200_checkpoint": pure["endpoint_hash"],
            "pure_step1200_readout": file_sha256(pure_report_path),
            "fixed_gradient_batches": file_sha256(gradient_path),
            "canonical_evaluator_report": accepted_canonical.THREE_EVALUATOR_REPORT_SHA256,
            "canonical_evaluator_source": file_sha256(root / accepted_canonical.THREE_EVALUATOR_SOURCE),
            "E_S": robust.EXPECTED_SMOOTH_CHECKPOINT_SHA256,
            "E_I": robust.EXPECTED_INDEPENDENT_CHECKPOINT_SHA256,
            "E_C": robust.EXPECTED_EC_CHECKPOINT_SHA256,
            "E_D": three.ED_CHECKPOINT_SHA256,
            "row_order_root_cause_review": file_sha256(row_review_path),
        },
    }

def _load_training_evaluators(root: Path, device: torch.device) -> tuple[torch.nn.Module, torch.nn.Module, torch.nn.Module]:
    del device
    cpu = torch.device("cpu")
    smooth, _ = robust.successor._load_smooth_evaluator(root, cpu)
    independent, independent_payload = load_frozen_evaluator(root / robust.INDEPENDENT_CHECKPOINT, device=cpu, min_validation_accuracy=0.985)
    _require(file_sha256(root / robust.SMOOTH_CHECKPOINT) == robust.EXPECTED_SMOOTH_CHECKPOINT_SHA256, "accepted E_S checkpoint hash changed")
    _require(file_sha256(root / robust.INDEPENDENT_CHECKPOINT) == robust.EXPECTED_INDEPENDENT_CHECKPOINT_SHA256, "accepted E_I checkpoint hash changed")
    _require(independent_payload.get("architecture_kind") == "frozen_mnist_evaluator_v1", "accepted E_I architecture identity changed")
    payload = safe_torch_load(root / robust.EC_CHECKPOINT, expected_sha256=robust.EXPECTED_EC_CHECKPOINT_SHA256, map_location="cpu")
    _require(payload.get("semantic_authority_id") == third_confirmation.SEMANTIC_AUTHORITY_ID and payload.get("architecture_kind") == third_confirmation.ThirdConfirmationEvaluator.architecture_kind, "accepted E_C checkpoint identity changed")
    ec = third_confirmation.ThirdConfirmationEvaluator().to(cpu)
    ec.load_state_dict(payload["state_dict"], strict=True)
    ec.freeze()
    models = (smooth, independent, ec)
    for model in models:
        model.eval()
        for parameter in model.parameters():
            parameter.requires_grad_(False)
            parameter.grad = None
    _require(all(not parameter.requires_grad and parameter.grad is None for model in models for parameter in model.parameters()), "a canonical training evaluator is not frozen before backward")
    return models


def _validate_step600_runtime_contract(
    checkpoint: Mapping[str, Any],
    consumer: Any,
    *,
    expected_frozen_hash: str | None = None,
) -> dict[str, Any]:
    """Validate the accepted legacy checkpoint contract and reconstructed frozen runtime."""
    _require(checkpoint.get("schema") == STEP600_CHECKPOINT_SCHEMA, "accepted step600 checkpoint schema changed")
    run_identity = checkpoint.get("run_identity")
    _require(isinstance(run_identity, Mapping), "accepted step600 checkpoint has no run identity")
    config_identity = run_identity.get("config_identity")
    _require(isinstance(config_identity, Mapping), "accepted step600 checkpoint has no config identity")
    _require(config_identity.get("runtime") == STEP600_RUNTIME_CONTRACT, "accepted step600 frozen/trainable runtime contract changed")
    model_identity = run_identity.get("model_identity")
    _require(
        isinstance(model_identity, Mapping)
        and model_identity.get("runtime") == "LIGHTGEN_2D_CHEN_BOGT_COMPLEX_OLS_CONSUMER_V1"
        and model_identity.get("trainable") == ["encoder.phase", "generator.phase"]
        and model_identity.get("training_evaluator") == "FrozenMNISTEvaluator v1"
        and model_identity.get("independent_evaluator") == "SIC_INDEPENDENT_CONFIRMATION_EVALUATOR_D_V1",
        "accepted step600 model identity changed",
    )
    runtime_identity = consumer.runtime_identity()
    _require(
        runtime_identity.get("config", {}).get("propagation_grid_size") == bogt.PRODUCTION_PROPAGATION_GRID
        and runtime_identity.get("production_propagation_grid") == bogt.PRODUCTION_PROPAGATION_GRID,
        "reconstructed frozen runtime propagation grid differs from step600 contract",
    )
    _require(runtime_identity.get("transport") == "T=I_100 coefficient space", "reconstructed OLS transport identity changed")
    _require(
        torch.equal(
            consumer.runtime.ols.transport.detach().cpu(),
            torch.eye(100, dtype=torch.complex64),
        ),
        "reconstructed OLS transport is not T=I_100",
    )
    parameter_names = tuple(name for name, _ in consumer.runtime.named_parameters())
    _require(parameter_names == ("encoder.phase", "generator.phase"), "step600 runtime parameter groups changed")
    frozen_hash = consumer.frozen_state_sha256
    if expected_frozen_hash is not None:
        _require(frozen_hash == expected_frozen_hash, "step600 frozen optical runtime changed while loading trainable state")
    serialized_hash = checkpoint.get("frozen_runtime_state_sha256")
    if serialized_hash is not None:
        _require(serialized_hash == frozen_hash, "serialized step600 frozen runtime digest disagrees with reconstructed runtime")
    return {
        "frozen_runtime_state_sha256": frozen_hash,
        "frozen_runtime_digest_source": (
            "serialized_checkpoint_field" if serialized_hash is not None else "reconstructed_from_source_bound_runtime_contract"
        ),
        "serialized_frozen_runtime_digest_present": serialized_hash is not None,
        "runtime_contract": dict(STEP600_RUNTIME_CONTRACT),
        "runtime_identity": {
            "identity": runtime_identity.get("identity"),
            "module_sha256": runtime_identity.get("module_sha256"),
            "transport_sha256": runtime_identity.get("transport_sha256"),
            "propagation_grid_size": runtime_identity.get("config", {}).get("propagation_grid_size"),
        },
    }


def _load_step600_runtime_with_identity(
    root: Path,
    checkpoint: Mapping[str, Any],
    context: Any,
) -> tuple[Any, dict[str, Any]]:
    """Reconstruct the accepted step600 runtime and resolve its frozen identity once."""
    consumer = bogt.LightGen2DChenBOGTComplexOLSTrainingConsumer(
        device=torch.device("cuda:0"),
        propagation_grid_size=bogt.PRODUCTION_PROPAGATION_GRID,
    )
    references = bogt.FrozenBogtReferences.from_payload(checkpoint["references"])
    weights = bogt.BogtLossWeights.from_payload(checkpoint["weights"])
    external_references = parent._load_references(root / m0.FROZEN_REFERENCES, expected_data_identity=context.data_identity)
    external_weights = parent._load_weights(root / m0.FROZEN_WEIGHTS, expected_reference_identity=canonical_sha256(external_references.identity()))
    _require(references.identity() == external_references.identity(), "step600 checkpoint BOGT references differ from the frozen authority")
    _require(weights.identity() == external_weights.identity(), "step600 checkpoint BOGT weights differ from the frozen authority")
    consumer.bind_references(references, for_scientific=True)
    consumer.bind_weights(weights, for_scientific=True)
    frozen_before_trainable_load = consumer.frozen_state_sha256
    identity_before = _validate_step600_runtime_contract(
        checkpoint,
        consumer,
        expected_frozen_hash=frozen_before_trainable_load,
    )
    consumer.load_trainable_state_dict(checkpoint["trainable_parameter_state"])
    identity_after = _validate_step600_runtime_contract(
        checkpoint,
        consumer,
        expected_frozen_hash=frozen_before_trainable_load,
    )
    _require(
        identity_before["frozen_runtime_state_sha256"] == identity_after["frozen_runtime_state_sha256"]
        and identity_before["runtime_identity"] == identity_after["runtime_identity"],
        "step600 runtime identity changed while restoring accepted trainable parameters",
    )
    for parameter in consumer.runtime.parameters():
        parameter.requires_grad_(True)
        parameter.grad = None
    consumer.runtime.eval()
    _require(all(parameter.grad is None for parameter in consumer.runtime.parameters()), "step600 runtime identity resolution accumulated gradients")
    consumer._accepted_step600_runtime_identity = identity_after
    return consumer, identity_after


def _load_training_runtime(
    root: Path,
    checkpoint: Mapping[str, Any],
    context: Any,
    *,
    create_optimizer: bool = True,
) -> tuple[Any, torch.optim.Adam | None, tuple[torch.nn.Module, torch.nn.Module, torch.nn.Module]]:
    consumer, start_runtime_identity = _load_step600_runtime_with_identity(root, checkpoint, context)
    optimizer = _restore_step600_optimizer(consumer, checkpoint) if create_optimizer else None
    evaluators = _load_training_evaluators(root, torch.device("cpu"))
    _require(all(parameter.device.type == "cpu" for model in evaluators for parameter in model.parameters()), "canonical evaluators must begin phase-local on CPU")
    bogt.restore_rng_state(checkpoint["rng_state"])
    consumer._accepted_step600_runtime_identity = start_runtime_identity
    return consumer, optimizer, evaluators


def _restore_step600_optimizer(consumer: Any, checkpoint: Mapping[str, Any]) -> torch.optim.Adam:
    optimizer = consumer.create_optimizer()
    optimizer.load_state_dict(copy.deepcopy(checkpoint["optimizer_state"]))
    _require(consumer.optimizer_manifest(optimizer) == checkpoint["optimizer_configuration"], "restored step600 optimizer identity changed")
    _require(_optimizer_steps(checkpoint) == [START_STEP, START_STEP], "scientific Adam does not start at step600")
    return optimizer


def _trainable_state(consumer: Any) -> dict[str, torch.Tensor]:
    return {name: value.detach().cpu().clone() for name, value in consumer.trainable_state_dict().items()}


def _module_state_sha256(module: torch.nn.Module) -> str:
    return canonical_sha256({name: tensor_sha256(value.detach().cpu()) for name, value in module.state_dict().items()})


def _frozen_evaluators_clear(evaluators: Sequence[torch.nn.Module]) -> bool:
    return all(parameter.grad is None and not parameter.requires_grad for model in evaluators for parameter in model.parameters())


def _set_gradients(parameters: Sequence[torch.nn.Parameter], gradients: Sequence[torch.Tensor]) -> None:
    _require(len(parameters) == len(gradients), "gradient assignment parameter count changed")
    for parameter, gradient in zip(parameters, gradients, strict=True):
        _require(gradient.shape == parameter.shape and bool(torch.isfinite(gradient).all().item()), "gradient assignment has an invalid tensor")
        parameter.grad = gradient.detach().to(device=parameter.device, dtype=parameter.dtype).clone()


def _restore_disposable_preflight_state(
    consumer: Any,
    optimizer: torch.optim.Optimizer,
    sampler: ExtendedTrainSampler,
    *,
    trainable_state: Mapping[str, torch.Tensor],
    optimizer_state: Mapping[str, Any],
    sampler_state: Mapping[str, Any],
    rng_state: Mapping[str, Any],
    recorder: Any,
    phase_prefix: str,
) -> dict[str, Any]:
    """Restore a disposable counterfactual, including the separately stored .grad buffers."""
    consumer.load_trainable_state_dict(trainable_state)
    optimizer.load_state_dict(copy.deepcopy(optimizer_state))
    sampler.load_state_dict(sampler_state)
    bogt.restore_rng_state(rng_state)

    restored_parameters = _trainable_state(consumer)
    restored_optimizer = copy.deepcopy(optimizer.state_dict())
    restored_sampler = copy.deepcopy(sampler.state_dict())
    restored_rng = bogt._rng_state_payload()
    before_clear = _record_gradient_buffers(
        recorder,
        consumer,
        f"{phase_prefix}_after_model_optimizer_rng_cursor_restore_before_grad_clear",
        expected_clean=True,
    )

    # Loading parameter and optimizer state does not own Parameter.grad. Clear only
    # after the restored dirty state has been recorded, using the production Adam convention.
    optimizer.zero_grad(set_to_none=True)
    after_clear = _record_gradient_buffers(
        recorder,
        consumer,
        f"{phase_prefix}_after_optimizer_zero_grad",
        expected_clean=True,
    )
    _require(after_clear["clean"], f"{phase_prefix} did not restore clean trainable .grad buffers")
    _require(_tree_equal(restored_parameters, trainable_state), f"{phase_prefix} gradient clearing changed restored parameters")
    _require(_tree_equal(restored_optimizer, optimizer.state_dict()), f"{phase_prefix} gradient clearing changed restored optimizer state")
    _require(_tree_equal(restored_sampler, sampler.state_dict()), f"{phase_prefix} gradient clearing changed restored sampler cursor")
    _require(_tree_equal(restored_rng, bogt._rng_state_payload()), f"{phase_prefix} gradient clearing changed restored RNG state")
    return {
        "pre_clear_grad_is_not_none_count": int(before_clear["grad_is_not_none_count"]),
        "pre_clear_aggregate_grad_l2_norm": float(before_clear["aggregate_grad_l2_norm"]),
        "post_clear_grad_is_not_none_count": int(after_clear["grad_is_not_none_count"]),
        "parameters_unchanged_by_clear": True,
        "optimizer_state_unchanged_by_clear": True,
        "sampler_unchanged_by_clear": True,
        "rng_unchanged_by_clear": True,
        "clean_after_restore": True,
    }


def _joint_cosine(left: Sequence[torch.Tensor], right: Sequence[torch.Tensor]) -> float:
    left_norm = _joint_norm(left)
    right_norm = _joint_norm(right)
    _require(left_norm > 0.0 and right_norm > 0.0, "cannot compute cosine for a zero joint vector")
    dot = sum(float((a.detach().to(torch.float64) * b.detach().to(torch.float64)).sum().cpu()) for a, b in zip(left, right, strict=True))
    value = dot / (left_norm * right_norm)
    _require(math.isfinite(value), "joint gradient cosine is nonfinite")
    return value


def _apply_update_vectors(
    consumer: Any,
    optimizer: torch.optim.Adam,
    *,
    G_BOGT: Sequence[torch.Tensor],
    G_canonical: Sequence[torch.Tensor],
    kappa: float,
    recorder: Any | None = None,
    batch_index: int | None = None,
    phase_prefix: str = "optimizer_update",
) -> dict[str, Any]:
    parameters = _parameter_tuple(consumer)
    before_clear = _record_gradient_buffers(
        recorder,
        consumer,
        f"{phase_prefix}_before_optimizer_zero_grad",
        batch_index=batch_index,
        expected_clean=True,
    )
    _require(before_clear["clean"], f"trainable .grad buffers were non-clean before {phase_prefix}")
    G_total = _add_gradient_vectors(G_BOGT, G_canonical, kappa=kappa)
    _require(_joint_norm(G_BOGT) > 0.0 and _joint_norm(G_total) > 0.0, "BOGT or combined gradient is zero")
    optimizer.zero_grad(set_to_none=True)
    after_zero_grad = _record_gradient_buffers(
        recorder,
        consumer,
        f"{phase_prefix}_after_optimizer_zero_grad",
        batch_index=batch_index,
        expected_clean=True,
    )
    _require(after_zero_grad["clean"], f"optimizer zero_grad did not clear trainable buffers before {phase_prefix}")
    _set_gradients(parameters, G_total)
    assigned_gradient_state = _record_gradient_buffers(
        recorder,
        consumer,
        f"{phase_prefix}_after_G_total_assignment",
        batch_index=batch_index,
        expected_clean=False,
    )
    gradient_assignment_matches = all(
        torch.equal(parameter.grad, expected.detach().to(device=parameter.device, dtype=parameter.dtype))
        for parameter, expected in zip(parameters, G_total, strict=True)
    )
    _require(gradient_assignment_matches, f"assigned trainable .grad buffers differ from G_total for {phase_prefix}")
    _require(assigned_gradient_state["grad_is_not_none_count"] == len(parameters), f"G_total assignment did not populate every trainable .grad in {phase_prefix}")
    before = [parameter.detach().clone() for parameter in parameters]
    optimizer.step()
    after_step_gradient_state = _record_gradient_buffers(
        recorder,
        consumer,
        f"{phase_prefix}_after_optimizer_step",
        batch_index=batch_index,
        expected_clean=False,
    )
    _require(
        all(torch.equal(parameter.grad, expected.detach().to(device=parameter.device, dtype=parameter.dtype)) for parameter, expected in zip(parameters, G_total, strict=True)),
        f"optimizer step altered the current G_total gradient buffers in {phase_prefix}",
    )
    _require(after_step_gradient_state["grad_is_not_none_count"] == len(parameters), f"optimizer step lost a trainable gradient buffer in {phase_prefix}")
    group_updates = [float((parameter.detach() - previous).to(torch.float64).norm().cpu()) for parameter, previous in zip(parameters, before, strict=True)]
    _require(all(math.isfinite(value) and value > 0.0 for value in group_updates), "both Encoder and Generator must receive a real Adam update")
    return {
        "G_total": G_total,
        "group_update_norms": dict(zip(("encoder.phase", "generator.phase"), group_updates, strict=True)),
        "gradient_assignment_matches_G_total": gradient_assignment_matches,
        "G_BOGT_norm": _joint_norm(G_BOGT),
        "kappa_G_canonical_norm": _joint_norm(tuple(float(kappa) * value for value in G_canonical)),
        "cosine_G_BOGT_G_canonical": _joint_cosine(G_BOGT, G_canonical),
        "canonical_group_norms": {name: float(value.detach().to(torch.float64).norm().cpu()) for name, value in zip(("encoder.phase", "generator.phase"), G_canonical, strict=True)},
    }

    """Expose the exact accepted step600 per-class permutations through step1200."""

def _expected_train_record(training_history: Sequence[Mapping[str, Any]], next_position: int) -> Mapping[str, Any]:
    """Resolve the next batch from a cursor that points at the next unread permutation slot."""
    _require(isinstance(next_position, int) and 0 <= next_position < len(training_history), "next TRAIN position is outside the authoritative history")
    record = training_history[next_position]
    expected_step = next_position + 1
    _require(record.get("step") == expected_step, f"pure BOGT history index {next_position} does not identify global step {expected_step}")
    return record


def _train_order_identity(sampler: ExtendedTrainSampler) -> str:
    state = sampler.state_dict()
    return canonical_sha256({
        "schema": state["schema"],
        "seed": state["seed"],
        "class_count": state["class_count"],
        "permutation_sha256": [tensor_sha256(value) for value in state["permutations"]],
    })


def _rng_state_sha256() -> str:
    return canonical_sha256(bogt._rng_state_payload())


def _preflight_train_update(
    consumer: Any,
    optimizer: torch.optim.Adam,
    sampler: ExtendedTrainSampler,
    context: Any,
    evaluators: Sequence[torch.nn.Module],
    endpoint_history: Sequence[Mapping[str, Any]],
    accepted_permutations: Sequence[torch.Tensor],
    *,
    kappa: float,
    sampler_identity: str,
    path: str,
    recorder: Any,
    capture_gradient_vectors: bool = False,
) -> dict[str, Any]:
    cursor_before = sampler.cursor
    expected = _expected_train_record(endpoint_history, cursor_before)
    authoritative_indices = torch.stack([permutation[cursor_before] for permutation in accepted_permutations])
    authoritative_hash = tensor_sha256(authoritative_indices)
    _require(expected.get("train_indices_sha256") == authoritative_hash, f"pure BOGT history and accepted permutation disagree at global step {cursor_before + 1}")
    rng_before = _rng_state_sha256()
    result = _train_update(
        consumer,
        optimizer,
        sampler,
        context,
        evaluators,
        kappa=kappa,
        expected_row_hash=authoritative_hash,
        recorder=recorder,
        capture_gradient_vectors=capture_gradient_vectors,
    )
    rng_after = _rng_state_sha256()
    _require(result["global_step"] == cursor_before + 1 and sampler.cursor == cursor_before + 1, "preflight update cursor did not advance by one")
    _require(result["train_indices"] == authoritative_indices.tolist(), f"actual TRAIN row identities differ at global step {cursor_before + 1}")
    result.update({
        "path": path,
        "global_step_before": cursor_before,
        "global_step_after": sampler.cursor,
        "cursor_before": cursor_before,
        "cursor_after": sampler.cursor,
        "order_position": cursor_before,
        "permutation_pass_index": 0,
        "sampler_order_identity_sha256": sampler_identity,
        "authoritative_row_ids": authoritative_indices.tolist(),
        "endpoint_history_row_sha256": expected["train_indices_sha256"],
        "rng_state_sha256_before": rng_before,
        "rng_state_sha256_after": rng_after,
        "iterator_event": "direct indexed read from immutable saved per-class permutations; no Python iterator",
    })
    return result


def _train_update(
    consumer: Any,
    optimizer: torch.optim.Adam,
    sampler: ExtendedTrainSampler,
    context: Any,
    evaluators: Sequence[torch.nn.Module],
    *,
    kappa: float,
    expected_row_hash: str | None = None,
    recorder: Any | None = None,
    capture_gradient_vectors: bool = False,
) -> dict[str, Any]:
    before_gradient_state = _record_gradient_buffers(
        recorder,
        consumer,
        "train_update_entry_before_gradient_construction",
        batch_index=sampler.cursor,
        expected_clean=True,
    )
    _require(before_gradient_state["clean"], "train update entered with non-clean trainable .grad buffers")
    indices, labels = sampler.next_batch()
    _require(labels.tolist() == list(range(10)), "accepted class-balanced TRAIN labels changed")
    row_hash = tensor_sha256(indices)
    if expected_row_hash is not None:
        _require(row_hash == expected_row_hash, f"TRAIN sample order differs from pure BOGT at global step {sampler.cursor}")
    batch = context.images.index_select(0, indices).to(consumer.device)
    context.ledger.record(bogt.TRAIN_ROLE, int(indices.numel()))
    before_compute_gradient_state = _record_gradient_buffers(
        recorder,
        consumer,
        "train_update_before_current_step_gradient_construction",
        batch_index=sampler.cursor,
        expected_clean=True,
    )
    _require(before_compute_gradient_state["clean"], "train update lost clean .grad state before current-step gradient construction")
    gradients = _compute_gradient_vectors(
        consumer,
        batch,
        smooth=evaluators[0],
        independent=evaluators[1],
        ec=evaluators[2],
        recorder=recorder,
        batch_index=sampler.cursor,
        phase_prefix="training_canonical",
    )
    after_compute_gradient_state = _record_gradient_buffers(
        recorder,
        consumer,
        "train_update_after_current_step_gradient_construction",
        batch_index=sampler.cursor,
        expected_clean=True,
    )
    _require(after_compute_gradient_state["clean"], "autograd gradient-vector construction populated parameter .grad buffers")
    del batch
    applied = _apply_update_vectors(
        consumer,
        optimizer,
        G_BOGT=gradients["G_BOGT"],
        G_canonical=gradients["G_canonical"],
        kappa=kappa,
        recorder=recorder,
        batch_index=sampler.cursor,
        phase_prefix="train_update",
    )
    _require(_frozen_evaluators_clear(evaluators), "E_S/E_I/E_C acquired parameter gradients")
    _require(all(parameter.grad is None for parameter in context.independent_evaluator.parameters()), "independent E_D acquired a gradient during training")
    internal_steps = {int(item["step"].item()) for item in optimizer.state.values()}
    _require(internal_steps == {sampler.cursor}, "scientific Adam step does not follow the accepted global cursor")
    post_optimizer_gradient_state = _record_gradient_buffers(
        recorder,
        consumer,
        "train_update_after_optimizer_step_before_clear",
        batch_index=sampler.cursor,
        expected_clean=False,
    )
    _require(post_optimizer_gradient_state["grad_is_not_none_count"] == len(_parameter_tuple(consumer)), "optimizer step did not leave the expected current-step gradients")
    optimizer.zero_grad(set_to_none=True)
    cleared_gradient_state = _record_gradient_buffers(
        recorder,
        consumer,
        "train_update_after_optimizer_zero_grad",
        batch_index=sampler.cursor,
        expected_clean=True,
    )
    _require(cleared_gradient_state["clean"], "optimizer zero_grad failed to clear buffers after the completed update")
    canonical_norm = _joint_norm(gradients["G_canonical"])
    encoder_share = applied["canonical_group_norms"]["encoder.phase"] / canonical_norm
    generator_share = applied["canonical_group_norms"]["generator.phase"] / canonical_norm
    result = {
        "global_step": sampler.cursor,
        "completed_updates": sampler.cursor - START_STEP,
        "losses": gradients["losses"],
        "gradient_vectors": {
            "G_BOGT_norm": applied["G_BOGT_norm"],
            "G_canonical_norm": canonical_norm,
            "kappa_G_canonical_norm": applied["kappa_G_canonical_norm"],
            "cosine_G_BOGT_G_canonical": applied["cosine_G_BOGT_G_canonical"],
            "canonical_encoder_norm": applied["canonical_group_norms"]["encoder.phase"],
            "canonical_generator_norm": applied["canonical_group_norms"]["generator.phase"],
            "canonical_encoder_norm_share": encoder_share,
            "canonical_generator_norm_share": generator_share,
            "gradient_assignment_matches_G_total": applied["gradient_assignment_matches_G_total"],
            "gradient_buffers_clean_after_update": True,
        },
        "parameter_update_norms": applied["group_update_norms"],
        "train_indices": indices.tolist(),
        "train_indices_sha256": row_hash,
        "labels": labels.tolist(),
        "data_role": bogt.TRAIN_ROLE,
        "E_D_forward_count": 0,
        "scalar_canonical_loss_defined": False,
    }
    if capture_gradient_vectors:
        result["preflight_gradient_vectors"] = {
            "G_BOGT": tuple(value.detach().cpu().clone() for value in gradients["G_BOGT"]),
            "G_canonical": tuple(value.detach().cpu().clone() for value in gradients["G_canonical"]),
            "G_total": tuple(value.detach().cpu().clone() for value in applied["G_total"]),
        }
    return result


def _calibrate_kappa(
    consumer: Any,
    context: Any,
    evaluators: Sequence[torch.nn.Module],
    batches: Sequence[torch.Tensor],
    recorder: Any,
    *,
    started: float,
) -> dict[str, Any]:
    _require(len(batches) == 20, "kappa calibration is not the fixed twenty-batch protocol")
    ratio_rows: list[dict[str, Any]] = []
    loss_rows: list[dict[str, float]] = []
    for index, indices in enumerate(batches):
        labels = context.labels.index_select(0, indices)
        _require(labels.tolist() == list(range(10)), f"fixed calibration batch {index} is not class balanced")
        recorder.sample("calibration_batch_start", batch_index=index, progress={"target_batches": 20}, started=started)
        start_gradient_state = _record_gradient_buffers(
            recorder,
            consumer,
            "calibration_batch_start",
            batch_index=index,
            expected_clean=True,
        )
        _require(start_gradient_state["clean"], f"calibration batch {index} did not start with clean trainable .grad buffers")
        batch = context.images.index_select(0, indices).to(consumer.device)
        context.ledger.record(bogt.TRAIN_ROLE, int(indices.numel()))
        parameters = _parameter_tuple(consumer)
        breakdown = consumer.forward_loss(batch, data_role=bogt.TRAIN_ROLE)
        _require(hasattr(breakdown, "result") and breakdown.result is not None, "calibration BOGT forward lacks production raw output")
        loss_row = {
            "L_dist": float(breakdown.l_dist.detach().cpu()),
            "L_rec": float(breakdown.l_rec.detach().cpu()),
            "L_op": float(breakdown.l_op.detach().cpu()),
            "L_total": float(breakdown.l_total.detach().cpu()),
        }
        weighted_rec = float(consumer.weights.beta) * breakdown.l_rec
        gradients = torch.autograd.grad(weighted_rec, parameters, retain_graph=False, allow_unused=True)
        rec_gradient = _gradient_tuple(gradients, parameters)
        rec_norm = _joint_norm(rec_gradient)
        del gradients, rec_gradient, weighted_rec, breakdown
        gc.collect()
        rec_gradient_state = _record_gradient_buffers(
            recorder,
            consumer,
            "calibration_after_BOGT_reconstruction_gradient",
            batch_index=index,
            expected_clean=True,
        )
        _require(rec_gradient_state["clean"], f"reconstruction calibration batch {index} populated trainable .grad buffers")
        recorder.sample("calibration_after_BOGT_reconstruction_gradient", batch_index=index, evaluator=None, started=started)

        canonical_gradient, canonical_metrics, canonical_losses, _ = _sequential_canonical_gradient(
            consumer,
            batch,
            evaluators,
            recorder=recorder,
            batch_index=index,
            phase_prefix="calibration",
        )
        canonical_norm = _joint_norm(canonical_gradient)
        _require(canonical_norm > three.JOINT_DEGENERACY_TOLERANCE, f"canonical direction is zero on fixed calibration batch {index}")
        _require(_joint_norm((canonical_gradient[0],)) > 0.0 and _joint_norm((canonical_gradient[1],)) > 0.0, f"canonical direction does not reach both groups on batch {index}")
        canonical_gradient_state = _record_gradient_buffers(
            recorder,
            consumer,
            "calibration_after_canonical_gradient",
            batch_index=index,
            expected_clean=True,
        )
        _require(canonical_gradient_state["clean"], f"canonical calibration batch {index} populated trainable .grad buffers")
        ratio_rows.append({
            "batch_index": index,
            "weighted_reconstruction_norm": rec_norm,
            "canonical_gradient_norm": canonical_norm,
        })
        loss_row.update({f"ell_{name[-1]}": float(value) for name, value in canonical_losses.items()})
        loss_row.update({"batch_index": index, "row_indices_sha256": tensor_sha256(indices)})
        loss_rows.append(loss_row)
        del canonical_gradient, canonical_metrics, canonical_losses, batch
        gc.collect()
        if torch.device(consumer.device).type == "cuda":
            torch.cuda.empty_cache()
        batch_end_gradient_state = _record_gradient_buffers(
            recorder,
            consumer,
            "calibration_batch_end",
            batch_index=index,
            expected_clean=True,
        )
        _require(batch_end_gradient_state["clean"], f"calibration batch {index} ended with non-clean trainable .grad buffers")
        recorder.sample("calibration_batch_end", batch_index=index, progress={"completed_batches": index + 1, "target_batches": 20}, started=started)
    beta = float(consumer.weights.beta)
    kappa, ratios = _calibrate_kappa_from_norm_rows(ratio_rows, beta=beta)
    for batch, row in zip(batches, ratios, strict=True):
        row["row_indices_sha256"] = tensor_sha256(batch)
        row["beta"] = beta
    return {"kappa": kappa, "batch_rows": ratios, "baseline_losses": loss_rows, "beta": beta, "batch_start_cursor": readout.GRADIENT_BATCH_START, "batch_count": readout.GRADIENT_BATCH_COUNT, "calibration_data_role": bogt.TRAIN_ROLE}


def _kappa_receipt_payload(
    calibration: Mapping[str, Any],
    batches: Sequence[torch.Tensor],
    labels: torch.Tensor,
    source: Mapping[str, Any],
) -> dict[str, Any]:
    _require(len(batches) == 20 and len(calibration.get("batch_rows", [])) == 20, "kappa receipt requires the exact registered twenty batches")
    _require(calibration.get("batch_count") == 20 and calibration.get("calibration_data_role") == bogt.TRAIN_ROLE, "kappa receipt calibration role or count changed")
    _require(calibration.get("batch_start_cursor") == readout.GRADIENT_BATCH_START, "kappa receipt batch start cursor changed")
    rows: list[dict[str, Any]] = []
    for index, (indices, source_row) in enumerate(zip(batches, calibration["batch_rows"], strict=True)):
        expected_indices = indices.detach().cpu().to(torch.int64)
        batch_labels = labels.index_select(0, expected_indices.to(labels.device)).detach().cpu().to(torch.int64)
        row_hash = tensor_sha256(expected_indices)
        _require(batch_labels.tolist() == list(range(10)), f"kappa receipt batch {index} labels are not class-balanced")
        _require(source_row.get("batch_index") == index and source_row.get("row_indices_sha256") == row_hash, f"kappa receipt batch {index} identity differs from calibration")
        rec_norm = float(source_row["weighted_reconstruction_norm"])
        canonical_norm = float(source_row["canonical_gradient_norm"])
        ratio = float(source_row["ratio"])
        _require(all(math.isfinite(value) and value > 0.0 for value in (rec_norm, canonical_norm, ratio)), f"kappa receipt batch {index} contains an invalid gradient statistic")
        rows.append(
            {
                "batch_index": index,
                "row_indices": expected_indices.tolist(),
                "labels": batch_labels.tolist(),
                "row_indices_sha256": row_hash,
                "weighted_reconstruction_norm": rec_norm,
                "canonical_gradient_norm": canonical_norm,
                "ratio": ratio,
            }
        )
    kappa = float(calibration["kappa"])
    _require(math.isfinite(kappa) and kappa > 0.0, "kappa receipt final median is invalid")
    _require(kappa == statistics.median(row["ratio"] for row in rows), "kappa receipt median differs from the registered ratios")
    return {
        "schema": "m0_bogt_canonical_gradient_augmentation_kappa_receipt.v1",
        "task_id": TASK_ID,
        "outer_gate": OUTER_GATE,
        "authorization_id": AUTHORIZATION_ID,
        "source_manifest_id": source["source_manifest_id"],
        "source_semantics_id": source["execution_source_semantics_id"],
        "calibration_data_role": bogt.TRAIN_ROLE,
        "batch_start_cursor": int(calibration["batch_start_cursor"]),
        "batch_count": len(rows),
        "fixed_batches": rows,
        "final_median_kappa": kappa,
        "formula": "median(||beta*G_rec|| / ||G_canonical||) over the exact 20 fixed TRAIN batches",
    }


def _optimizer_steps_from_optimizer(optimizer: torch.optim.Optimizer) -> list[int]:
    steps: list[int] = []
    for state in optimizer.state.values():
        step = state.get("step")
        if step is not None:
            steps.append(int(step.item()) if torch.is_tensor(step) else int(step))
    return sorted(steps)

def _restore_recovery(
    payload: Mapping[str, Any],
    consumer: Any,
    optimizer: torch.optim.Adam,
    sampler: ExtendedTrainSampler,
    context: Any,
    source: Mapping[str, Any],
    identity: ResearchRunIdentity,
    *,
    kappa: float,
) -> list[dict[str, Any]]:
    _require(payload.get("schema") == CHECKPOINT_SCHEMA and payload.get("task_id") == TASK_ID and payload.get("outer_gate") == OUTER_GATE, "recovery task schema changed")
    _require(payload.get("authorization_id") == AUTHORIZATION_ID and payload.get("parent_checkpoint_sha256") == START_CHECKPOINT_SHA256, "recovery authorization or accepted parent changed")
    _require(payload.get("source_manifest_id") == identity.source_manifest_id and payload.get("source_manifest") == dict(source) and payload.get("run_identity") == identity.to_dict(), "recovery source/run identity changed")
    _require(payload.get("data_identity") == context.data_identity and payload.get("frozen_runtime_state_sha256") == consumer.frozen_state_sha256, "recovery data or frozen runtime identity changed")
    _require(math.isclose(float(payload.get("kappa", float("nan"))), kappa, rel_tol=0.0, abs_tol=0.0), "recovery kappa differs from the single fixed calibration")
    step = payload.get("step")
    history = payload.get("training_history")
    _require(isinstance(step, int) and START_STEP <= step <= END_STEP and payload.get("completed_updates") == step - START_STEP, "recovery global step/update count is invalid")
    _require(isinstance(history, list) and len(history) == step - START_STEP, "recovery history length differs from completed updates")
    _require(payload.get("sampler_state", {}).get("cursor") == step, "recovery sampler cursor differs from optimizer step")
    ledger = payload.get("read_ledger")
    _require(isinstance(ledger, Mapping), "recovery checkpoint lacks a data-role read ledger")
    required_roles = (bogt.TRAIN_ROLE, bogt.DEV_ROLE, bogt.PROTECTED_ROLE, bogt.TEST_ROLE, bogt.SEALED_ROLE, bogt.TRAIN_SMOKE_ROLE, bogt.SYNTHETIC_ROLE)
    restored_counts = {role: ledger.get(role, 0) for role in required_roles}
    _require(all(isinstance(count, int) and count >= 0 for count in restored_counts.values()), "recovery data-role read ledger is malformed")
    _require(all(restored_counts[role] == 0 for role in (bogt.PROTECTED_ROLE, bogt.TEST_ROLE, bogt.SEALED_ROLE)), "recovery read ledger records a forbidden data role")
    context.ledger.counts = restored_counts
    consumer.load_trainable_state_dict(payload["trainable_parameter_state"])
    optimizer.load_state_dict(payload["optimizer_state"])
    _require(consumer.optimizer_manifest(optimizer) == payload.get("optimizer_configuration"), "recovery Adam configuration changed")
    sampler.load_state_dict(payload["sampler_state"])
    bogt.restore_rng_state(payload["rng_state"])
    _require(_optimizer_steps_from_optimizer(optimizer) == [step, step], "recovery Adam internal steps do not match the saved global cursor")
    return [dict(record) for record in history]


def _tree_equal(left: Any, right: Any) -> bool:
    return pure_bogt._tree_equal(left, right)


def _tensor_equivalence_metrics(
    left: torch.Tensor,
    right: torch.Tensor,
    *,
    rtol: float,
    atol: float,
) -> dict[str, Any]:
    _require(isinstance(left, torch.Tensor) and isinstance(right, torch.Tensor), "recovery comparison requires tensor values")
    _require(left.shape == right.shape and left.dtype == right.dtype, "recovery comparison tensor shape or dtype changed")
    left_cpu = left.detach().cpu()
    right_cpu = right.detach().cpu()
    difference = left_cpu - right_cpu
    metric_dtype = torch.complex128 if difference.is_complex() else torch.float64
    left_metric = left_cpu.to(metric_dtype)
    right_metric = right_cpu.to(metric_dtype)
    difference_metric = difference.to(metric_dtype)
    left_norm = float(torch.linalg.vector_norm(left_metric))
    right_norm = float(torch.linalg.vector_norm(right_metric))
    difference_norm = float(torch.linalg.vector_norm(difference_metric))
    cosine = (
        float((left_metric.reshape(-1).conj() * right_metric.reshape(-1)).sum().real / (left_norm * right_norm))
        if left_norm > 0.0 and right_norm > 0.0
        else None
    )
    return {
        "exact": bool(torch.equal(left_cpu, right_cpu)),
        "within_tolerance": bool(torch.allclose(left_cpu, right_cpu, rtol=rtol, atol=atol, equal_nan=False)),
        "rtol": float(rtol),
        "atol": float(atol),
        "numel": left_cpu.numel(),
        "max_abs_diff": float(difference_metric.abs().max()) if difference_metric.numel() else 0.0,
        "l2_diff": difference_norm,
        "relative_l2_diff": difference_norm / max(left_norm, right_norm, 1.0e-30),
        "left_norm": left_norm,
        "right_norm": right_norm,
        "cosine": cosine,
    }


def _named_tensor_equivalence_metrics(
    left: Mapping[str, torch.Tensor],
    right: Mapping[str, torch.Tensor],
    *,
    rtol: float,
    atol: float,
) -> dict[str, Any]:
    if set(left) != set(right):
        return {"passed": False, "missing_from_right": sorted(set(left) - set(right)), "missing_from_left": sorted(set(right) - set(left)), "tensors": {}}
    tensors = {name: _tensor_equivalence_metrics(left[name], right[name], rtol=rtol, atol=atol) for name in sorted(left)}
    return {"passed": all(value["within_tolerance"] for value in tensors.values()), "tensors": tensors}


def _cpu_state(value: Any) -> Any:
    if isinstance(value, torch.Tensor):
        return value.detach().cpu().clone()
    if isinstance(value, Mapping):
        return {key: _cpu_state(item) for key, item in value.items()}
    if isinstance(value, tuple):
        return tuple(_cpu_state(item) for item in value)
    if isinstance(value, list):
        return [_cpu_state(item) for item in value]
    return copy.deepcopy(value)


def _semantic_optimizer_state(consumer: Any, optimizer: torch.optim.Optimizer) -> dict[str, Any]:
    names_by_parameter_id = {id(parameter): name for name, parameter in consumer.runtime.named_parameters()}
    groups = []
    for group in optimizer.param_groups:
        names = [names_by_parameter_id[id(parameter)] for parameter in group["params"]]
        configuration = {name: _cpu_state(value) for name, value in group.items() if name != "params"}
        groups.append({"parameter_names": names, "configuration": configuration})
    states = {
        names_by_parameter_id[id(parameter)]: _cpu_state(state)
        for parameter, state in optimizer.state.items()
    }
    return {"parameter_groups": groups, "states_by_parameter_name": states}


def _scalar_equivalence_metrics(left: Mapping[str, float], right: Mapping[str, float]) -> dict[str, Any]:
    if set(left) != set(right):
        return {"passed": False, "left_names": sorted(left), "right_names": sorted(right), "values": {}}
    values = {
        name: _tensor_equivalence_metrics(
            torch.tensor(float(left[name]), dtype=torch.float64),
            torch.tensor(float(right[name]), dtype=torch.float64),
            rtol=CANONICAL_EQUIVALENCE_RTOL,
            atol=CANONICAL_EQUIVALENCE_ATOL,
        )
        for name in sorted(left)
    }
    return {"passed": all(item["within_tolerance"] for item in values.values()), "values": values}


def _optimizer_recovery_equivalence(left: Mapping[str, Any], right: Mapping[str, Any]) -> dict[str, Any]:
    left_groups = left.get("parameter_groups")
    right_groups = right.get("parameter_groups")
    left_states = left.get("states_by_parameter_name")
    right_states = right.get("states_by_parameter_name")
    if not isinstance(left_groups, list) or not isinstance(right_groups, list) or not isinstance(left_states, Mapping) or not isinstance(right_states, Mapping):
        return {"passed": False, "reason": "malformed semantic optimizer state"}
    groups_equal = _tree_equal(left_groups, right_groups)
    if set(left_states) != set(right_states):
        return {"passed": False, "parameter_groups_exact": groups_equal, "state_parameter_names_equal": False, "states": {}}
    state_report: dict[str, Any] = {}
    state_passed = True
    for parameter_name in sorted(left_states):
        first = left_states[parameter_name]
        second = right_states[parameter_name]
        if not isinstance(first, Mapping) or not isinstance(second, Mapping) or set(first) != set(second):
            state_report[parameter_name] = {"passed": False, "slot_names_equal": False, "slots": {}}
            state_passed = False
            continue
        slots: dict[str, Any] = {}
        for slot_name in sorted(first):
            left_value, right_value = first[slot_name], second[slot_name]
            if slot_name == "step":
                exact = _tree_equal(left_value, right_value)
                slots[slot_name] = {"exact": exact, "within_tolerance": exact, "left": int(left_value.item()) if torch.is_tensor(left_value) else int(left_value), "right": int(right_value.item()) if torch.is_tensor(right_value) else int(right_value)}
            elif isinstance(left_value, torch.Tensor) and isinstance(right_value, torch.Tensor) and (left_value.is_floating_point() or left_value.is_complex()):
                slots[slot_name] = _tensor_equivalence_metrics(
                    left_value,
                    right_value,
                    rtol=accepted_canonical.RECOVERY_UPDATE_RTOL,
                    atol=accepted_canonical.RECOVERY_UPDATE_ATOL,
                )
            else:
                exact = _tree_equal(left_value, right_value)
                slots[slot_name] = {"exact": exact, "within_tolerance": exact}
            state_passed = state_passed and bool(slots[slot_name]["within_tolerance"])
        state_report[parameter_name] = {"passed": all(item["within_tolerance"] for item in slots.values()), "slot_names_equal": True, "slots": slots}
    return {
        "passed": groups_equal and state_passed,
        "parameter_groups_exact": groups_equal,
        "state_parameter_names_equal": True,
        "state_by_parameter_name": state_report,
        "floating_slot_tolerance": {"rtol": accepted_canonical.RECOVERY_UPDATE_RTOL, "atol": accepted_canonical.RECOVERY_UPDATE_ATOL},
    }


def _gradient_recovery_equivalence(left: Mapping[str, Sequence[torch.Tensor]], right: Mapping[str, Sequence[torch.Tensor]]) -> dict[str, Any]:
    if set(left) != set(right):
        return {"passed": False, "left_components": sorted(left), "right_components": sorted(right), "components": {}}
    components: dict[str, Any] = {}
    for component_name in sorted(left):
        first = left[component_name]
        second = right[component_name]
        if len(first) != 2 or len(second) != 2:
            components[component_name] = {"passed": False, "reason": "unexpected trainable parameter count"}
            continue
        metrics = _named_tensor_equivalence_metrics(
            dict(zip(("encoder.phase", "generator.phase"), first, strict=True)),
            dict(zip(("encoder.phase", "generator.phase"), second, strict=True)),
            rtol=CANONICAL_EQUIVALENCE_RTOL,
            atol=CANONICAL_EQUIVALENCE_ATOL,
        )
        components[component_name] = metrics
    return {
        "passed": all(item.get("passed", False) for item in components.values()),
        "components": components,
        "tolerance": {"rtol": CANONICAL_EQUIVALENCE_RTOL, "atol": CANONICAL_EQUIVALENCE_ATOL},
    }


def _current_process_gpu_usage(device: torch.device) -> tuple[int | None, str]:
    index = torch.cuda.current_device() if device.index is None else int(device.index)
    try:
        result = subprocess.run(
            ["nvidia-smi", "--query-compute-apps=pid,used_memory", "--format=csv,noheader,nounits", "-i", str(index)],
            check=True,
            capture_output=True,
            text=True,
            timeout=5,
        )
    except (OSError, subprocess.SubprocessError) as error:
        return None, f"unavailable:{type(error).__name__}"
    matches: list[int] = []
    for line in result.stdout.splitlines():
        fields = [field.strip() for field in line.split(",")]
        if len(fields) != 2 or fields[0] in {"", "N/A"} or fields[1] in {"", "N/A"}:
            continue
        try:
            if int(fields[0]) == os.getpid():
                matches.append(int(fields[1]) * 1024 * 1024)
        except ValueError:
            continue
    return (sum(matches), "observed") if matches else (None, "no_current_process_row")


def _capture_resource_metrics(device: torch.device) -> dict[str, Any]:
    _require(device.type == "cuda" and torch.cuda.is_available(), "CUDA resource telemetry is unavailable", CanaryError)
    torch.cuda.synchronize(device)
    try:
        global_memory = pure_bogt._device_memory(device)
    except Exception as error:
        raise CanaryError(f"global GPU memory telemetry is unavailable: {type(error).__name__}: {error}") from error
    process_bytes, process_status = _current_process_gpu_usage(device)
    return {
        **global_memory,
        "current_process_gpu_bytes": process_bytes,
        "current_process_gpu_status": process_status,
        "torch_allocated_bytes": torch.cuda.memory_allocated(device),
        "torch_reserved_bytes": torch.cuda.memory_reserved(device),
        "torch_peak_allocated_bytes": torch.cuda.max_memory_allocated(device),
        "torch_peak_reserved_bytes": torch.cuda.max_memory_reserved(device),
    }


class PhaseResourceRecorder:
    def __init__(
        self,
        device: torch.device,
        *,
        telemetry_path: Path | None = None,
        gradient_ledger_path: Path | None = None,
        stop_path: Path | None = None,
        hard_limit: float = MAX_DEVICE_MEMORY_FRACTION,
        allow_existing_telemetry: bool = False,
    ) -> None:
        self.device = device
        self.telemetry_path = telemetry_path
        self.gradient_ledger_path = gradient_ledger_path
        self.stop_path = stop_path
        self.hard_limit = float(hard_limit)
        self.started = time.perf_counter()
        self.entries: list[dict[str, Any]] = []
        self.gradient_entries: list[dict[str, Any]] = []
        self.previous_gradient_clean: bool | None = None
        self.first_nonclean_transition: dict[str, Any] | None = None
        self.first_unexpected_nonclean_transition: dict[str, Any] | None = None
        _require(0.0 < self.hard_limit <= MAX_DEVICE_MEMORY_FRACTION, "resource recorder hard limit exceeds the frozen global GPU boundary")
        _require(telemetry_path is None or allow_existing_telemetry or not telemetry_path.exists(), "preflight resource telemetry is write-once")
        _require(gradient_ledger_path is None or not gradient_ledger_path.exists(), "preflight gradient-buffer ledger is write-once")
        _require(stop_path is None or not stop_path.exists(), "preflight resource-stop evidence is write-once")

    def record_gradient_buffers(
        self,
        phase: str,
        snapshot: Mapping[str, Any],
        *,
        batch_index: int | None = None,
        evaluator: str | None = None,
        expected_clean: bool | None = None,
    ) -> dict[str, Any]:
        current_clean = bool(snapshot["clean"])
        transition = self.previous_gradient_clean is True and not current_clean
        record = {
            "timestamp_utc": datetime.now(timezone.utc).isoformat(),
            "phase": str(phase),
            "batch_index": batch_index,
            "evaluator": evaluator,
            "expected_clean": expected_clean,
            "transition_from_clean_to_nonclean": transition,
            "unexpected_nonclean_state": expected_clean is True and not current_clean,
            "gradient_buffers": dict(snapshot),
        }
        if transition and self.first_nonclean_transition is None:
            self.first_nonclean_transition = dict(record)
        if record["unexpected_nonclean_state"] and self.first_unexpected_nonclean_transition is None:
            self.first_unexpected_nonclean_transition = dict(record)
        self.previous_gradient_clean = current_clean
        self.gradient_entries.append(record)
        if self.gradient_ledger_path is not None:
            self.gradient_ledger_path.parent.mkdir(parents=True, exist_ok=True)
            mode = "a" if self.gradient_ledger_path.exists() else "x"
            with self.gradient_ledger_path.open(mode, encoding="utf-8", newline="\n") as stream:
                stream.write(json.dumps(record, sort_keys=True, allow_nan=False) + "\n")
                stream.flush()
                os.fsync(stream.fileno())
        return record

    def sample(
        self,
        phase: str,
        *,
        batch_index: int | None = None,
        evaluator: str | None = None,
        progress: Mapping[str, Any] | None = None,
        started: float | None = None,
    ) -> dict[str, Any]:
        record = {
            **_capture_resource_metrics(self.device),
            "timestamp_utc": datetime.now(timezone.utc).isoformat(),
            "elapsed_seconds": time.perf_counter() - self.started,
            "phase_elapsed_seconds": 0.0 if started is None else max(0.0, time.perf_counter() - started),
            "phase": str(phase),
            "allocator_peak_scope": "since_previous_resource_sample",
            "batch_index": batch_index,
            "evaluator": evaluator,
            "progress": dict(progress or {}),
            "global_and_allocator_metrics_are_nonadditive": True,
        }
        self.entries.append(record)
        if self.telemetry_path is not None:
            self.telemetry_path.parent.mkdir(parents=True, exist_ok=True)
            mode = "a" if self.telemetry_path.exists() else "x"
            with self.telemetry_path.open(mode, encoding="utf-8", newline="\n") as stream:
                stream.write(json.dumps(record, sort_keys=True, allow_nan=False) + "\n")
                stream.flush()
                os.fsync(stream.fileno())
        fraction = float(record["global_used_fraction"])
        if fraction >= self.hard_limit:
            message = f"{phase} reached hard global GPU limit {self.hard_limit:.0%}: {fraction:.4%}"
            stop = {
                "schema": "m0_bogt_canonical_gradient_augmentation_resource_stop.v1",
                "classification": "PREFLIGHT_BLOCKED(resource)",
                "message": message,
                "final_snapshot": record,
                "resource_phase_records": self.entries,
                "attempt_created": False,
                "scientific_optimizer_updates": 0,
            }
            if self.stop_path is not None:
                atomic_write_new_json(self.stop_path, stop, artifact_kind="M0 canonical-gradient resource stop evidence")
            error = ResourceBudgetExceeded(message)
            setattr(error, "resource_phase_records", self.entries)
            raise error
        if self.device.type == "cuda" and torch.cuda.is_available():
            torch.cuda.reset_peak_memory_stats(self.device)
        return record

    def stop_for_budget(self, phase: str, message: str, *, progress: Mapping[str, Any] | None = None) -> None:
        record = self.sample(f"{phase}_before_resource_stop", progress=progress)
        stop = {
            "schema": "m0_bogt_canonical_gradient_augmentation_resource_stop.v1",
            "classification": "PREFLIGHT_BLOCKED(resource)",
            "message": message,
            "final_snapshot": record,
            "resource_phase_records": self.entries,
            "attempt_created": False,
            "scientific_optimizer_updates": 0,
        }
        if self.stop_path is not None:
            atomic_write_new_json(self.stop_path, stop, artifact_kind="M0 canonical-gradient resource stop evidence")
        error = ResourceBudgetExceeded(message)
        setattr(error, "resource_phase_records", self.entries)
        raise error


def _resource_recorder(
    telemetry_path: Path,
    *,
    gradient_ledger_path: Path | None = None,
    stop_path: Path | None = None,
    allow_existing_telemetry: bool = False,
) -> PhaseResourceRecorder:
    return PhaseResourceRecorder(
        torch.device("cuda:0"),
        telemetry_path=telemetry_path,
        gradient_ledger_path=gradient_ledger_path,
        stop_path=stop_path,
        allow_existing_telemetry=allow_existing_telemetry,
    )


def _is_resource_budget_error(error: BaseException) -> bool:
    resource_types = (ResourceBudgetExceeded, readout.ResourceBudgetExceeded, pure_bogt.ResourceBudgetExceeded)
    return isinstance(error, resource_types)


def _check_time(started: float, limit: float, phase: str, *, recorder: PhaseResourceRecorder | None = None) -> None:
    elapsed = time.perf_counter() - started
    if elapsed > limit:
        message = f"{phase} exceeded its frozen wall-clock budget: {elapsed:.2f}s > {limit:.2f}s"
        if recorder is not None:
            recorder.stop_for_budget(phase, message, progress={"elapsed_seconds": elapsed, "limit_seconds": limit})
        raise ResourceBudgetExceeded(message)


def _smoke_lifecycle(identity: ResearchRunIdentity, source: Mapping[str, Any], receipt: PreflightReceipt) -> dict[str, Any]:
    with tempfile.TemporaryDirectory(prefix="m0-cgga-lifecycle-") as temporary:
        root = Path(temporary)
        control = ResearchRunControl(identity, AUTHORIZATION_ID)
        control.mark_preflight_ready(receipt)
        control.start_execution()
        attempt = root / "smoke_attempt"
        effective = control.open_scientific_attempt(attempt, SCIENTIFIC_ATTEMPT_ID, receipt)
        atomic_write_new_json(attempt / "source_manifest.json", dict(source), artifact_kind="disposable lifecycle source")
        heartbeat = HeartbeatWriter(attempt, effective.run_id, total_steps=UPDATE_COUNT)
        heartbeat.update(state=RUN_PREPARING, phase="STATISTICS", step=0, latest_losses={"global_step": START_STEP}, force=True)
        heartbeat.update(state=RUNNING, phase="TRAINING", step=3, latest_losses={"global_step": START_STEP + 3}, force=True)
        control.complete()
        heartbeat.update(state=RUN_COMPLETED, phase="COMPLETED", step=3, latest_losses={"disposable_smoke": True}, force=True)
        _require(control.state == RUN_COMPLETED and heartbeat.latest_status_path.is_file(), "production lifecycle smoke did not reach a valid terminal state")
        atomic_write_new_json(attempt / "smoke_report.json", {"source_manifest_id": identity.source_manifest_id, "preflight_ready": receipt.status == PREFLIGHT_READY, "state": RUN_COMPLETED}, artifact_kind="disposable lifecycle report")
        saved = _read_json(heartbeat.latest_status_path)
        _require(saved.get("state") == RUN_COMPLETED and saved.get("phase") == "COMPLETED" and saved.get("step") == 3, "heartbeat terminal serialization changed")
        return {"status": "PASS", "attempt_directory_disposable": True, "source_manifest_id": identity.source_manifest_id, "terminal_state": saved["state"], "terminal_phase": saved["phase"], "report_serialization": True}


def _preflight_requalification(
    root: Path,
    authorities: Mapping[str, Any],
    context: Any,
    source: Mapping[str, Any],
    identity: ResearchRunIdentity,
    *,
    task_started: float,
) -> tuple[dict[str, Any], PreflightReceipt]:
    """Recheck the changed scientific entry within the cumulative preflight budget."""
    _require(torch.cuda.is_available(), "CUDA is required for source-bound requalification", ResourceBudgetExceeded)
    _require(not (root / SCIENTIFIC_ATTEMPT_DIR).exists(), "attempt_02 must remain absent before READY")
    _validated_repair_history(root, allow_existing_attempt01=True)
    prior = _read_json(root / PREFLIGHT_07_PATH)
    _require(prior.get("status") == PREFLIGHT_READY and prior.get("input_authority_hashes") == authorities["authority_hashes"], "inherited preflight_07 authority is not identical")
    _require(prior.get("step600_dev_baseline_qualification", {}).get("status") == "PASS", "inherited full DEV baseline is not qualified")
    _require(prior.get("canonical_operator_equivalence", {}).get("encoder_allclose") is True and prior.get("canonical_operator_equivalence", {}).get("generator_allclose") is True, "inherited canonical operator equivalence is not qualified")
    recorder = _resource_recorder(
        root / PREFLIGHT_09_RESOURCE_TELEMETRY_PATH,
        gradient_ledger_path=root / PREFLIGHT_09_GRADIENT_BUFFER_LEDGER_PATH,
        stop_path=root / PREFLIGHT_09_RESOURCE_STOP_PATH,
    )
    consumer = optimizer = evaluators = None
    phase = "load_step600_disposable_runtime"
    try:
        consumer, optimizer, evaluators = _load_training_runtime(root, authorities["step600"], context)
        _require(_optimizer_steps_from_optimizer(optimizer) == [START_STEP, START_STEP], "disposable Adam did not restore accepted step600")
        _require(all(parameter.grad is None for parameter in context.independent_evaluator.parameters()), "E_D gradient firewall failed")
        recorder.sample(phase)
        _check_time(task_started, MAX_PREFLIGHT_SECONDS, phase, recorder=recorder)

        phase = "reproduce_fixed_20_batch_kappa"
        calibration = _calibrate_kappa(consumer, context, evaluators, authorities["gradient_batches"], recorder, started=time.perf_counter())
        kappa_receipt = _kappa_receipt_payload(calibration, authorities["gradient_batches"], context.labels, source)
        kappa_path = root / PREFLIGHT_09_KAPPA_RECEIPT_PATH
        atomic_write_new_json(kappa_path, kappa_receipt, artifact_kind="M0 source-bound fixed kappa receipt")
        _require(
            math.isclose(float(calibration["kappa"]), SCIENTIFIC_KAPPA, rel_tol=KAPPA_REPRODUCTION_RTOL, abs_tol=0.0),
            f"fixed-batch kappa differs from frozen value: measured={calibration['kappa']!r}, frozen={SCIENTIFIC_KAPPA!r}",
        )
        _check_time(task_started, MAX_PREFLIGHT_SECONDS, phase, recorder=recorder)

        phase = "bogt_only_negative_control_and_three_real_updates"
        sampler = ExtendedTrainSampler(context, authorities["step600"]["sampler_state"])
        parameters_before = _trainable_state(consumer)
        optimizer_before = copy.deepcopy(optimizer.state_dict())
        sampler_before = copy.deepcopy(sampler.state_dict())
        rng_before = bogt._rng_state_payload()
        first_indices = torch.stack([permutation[START_STEP] for permutation in authorities["step600"]["sampler_state"]["permutations"]])
        batch = context.images.index_select(0, first_indices).to(consumer.device)
        vectors = _compute_gradient_vectors(consumer, batch, smooth=evaluators[0], independent=evaluators[1], ec=evaluators[2], recorder=recorder, batch_index=START_STEP, phase_prefix="requalification_negative_control")
        _require(_joint_norm((vectors["G_canonical"][0],)) > 0.0 and _joint_norm((vectors["G_canonical"][1],)) > 0.0, "canonical gradient does not reach both trainable groups")
        _apply_update_vectors(consumer, optimizer, G_BOGT=vectors["G_BOGT"], G_canonical=vectors["G_canonical"], kappa=SCIENTIFIC_KAPPA, recorder=recorder, batch_index=START_STEP, phase_prefix="requalification_augmented")
        augmented = _trainable_state(consumer)
        _restore_disposable_preflight_state(consumer, optimizer, sampler, trainable_state=parameters_before, optimizer_state=optimizer_before, sampler_state=sampler_before, rng_state=rng_before, recorder=recorder, phase_prefix="requalification_augmented_restore")
        _set_gradients(_parameter_tuple(consumer), vectors["G_BOGT"])
        optimizer.step()
        bogt_only = _trainable_state(consumer)
        differences = {name: float((augmented[name] - bogt_only[name]).norm()) for name in augmented}
        _require(all(math.isfinite(value) and value > 0.0 for value in differences.values()), "canonical term does not change both real updates")
        _restore_disposable_preflight_state(consumer, optimizer, sampler, trainable_state=parameters_before, optimizer_state=optimizer_before, sampler_state=sampler_before, rng_state=rng_before, recorder=recorder, phase_prefix="requalification_bogt_restore")
        del batch, vectors, augmented, bogt_only
        history = []
        for global_step in range(START_STEP + 1, START_STEP + 4):
            expected = authorities["pure"]["step1200"]["training_history"][global_step - 1]
            history.append(_train_update(consumer, optimizer, sampler, context, evaluators, kappa=SCIENTIFIC_KAPPA, expected_row_hash=expected["train_indices_sha256"], recorder=recorder))
            _check_time(task_started, MAX_PREFLIGHT_SECONDS, phase, recorder=recorder)
        _require(sampler.cursor == START_STEP + 3 and _optimizer_steps_from_optimizer(optimizer) == [START_STEP + 3, START_STEP + 3], "three real disposable updates did not advance Adam/cursor exactly")
        _require(not _tree_equal(parameters_before, _trainable_state(consumer)), "disposable joint update did not change trainable parameters")

        phase = "production_recovery_next_step_equivalence"
        with tempfile.TemporaryDirectory(prefix="m0-cgga-requalification-") as temporary:
            manager = RecoveryCheckpointManager(temporary, identity, source_manifest_id=identity.source_manifest_id, authorization_id=AUTHORIZATION_ID, keep=2)
            payload = _checkpoint_payload(consumer, optimizer, sampler, history, context, source, identity, kappa=SCIENTIFIC_KAPPA, calibration=calibration, cumulative_elapsed_seconds=time.perf_counter() - task_started)
            saved = manager.save({"phase": "TRAINING", "scientific_state": payload}, step=sampler.cursor)
            expected = authorities["pure"]["step1200"]["training_history"][sampler.cursor]
            uninterrupted = _train_update(consumer, optimizer, sampler, context, evaluators, kappa=SCIENTIFIC_KAPPA, expected_row_hash=expected["train_indices_sha256"], recorder=recorder)
            uninterrupted_parameters = _trainable_state(consumer)
            uninterrupted_optimizer = _semantic_optimizer_state(consumer, optimizer)
            wrapper = manager.load(saved)
            restored_history = _restore_recovery(wrapper["payload"]["scientific_state"], consumer, optimizer, sampler, context, source, identity, kappa=SCIENTIFIC_KAPPA)
            _require(len(restored_history) == 3 and sampler.cursor == START_STEP + 3, "recovery did not restore the saved next-unconsumed batch")
            replayed = _train_update(consumer, optimizer, sampler, context, evaluators, kappa=SCIENTIFIC_KAPPA, expected_row_hash=expected["train_indices_sha256"], recorder=recorder)
            parameter_equivalence = _named_tensor_equivalence_metrics(uninterrupted_parameters, _trainable_state(consumer), rtol=accepted_canonical.RECOVERY_UPDATE_RTOL, atol=accepted_canonical.RECOVERY_UPDATE_ATOL)
            optimizer_equivalence = _optimizer_recovery_equivalence(uninterrupted_optimizer, _semantic_optimizer_state(consumer, optimizer))
            _require(uninterrupted["train_indices_sha256"] == replayed["train_indices_sha256"] and parameter_equivalence["passed"] and optimizer_equivalence["passed"], "production save/recovery changed the next real update")
        _check_time(task_started, MAX_PREFLIGHT_SECONDS, phase, recorder=recorder)

        phase = "frozen_observation_and_terminal_lifecycle"
        smoke_indices = authorities["gradient_batches"][0]
        smoke_batch = context.images.index_select(0, smoke_indices).to(consumer.device)
        with torch.no_grad():
            raw = consumer.forward_loss(smoke_batch, data_role=bogt.TRAIN_ROLE).result.raw_intensity.detach().cpu()
        _require(raw.grad_fn is None and not raw.requires_grad, "frozen observation retained autograd graph")
        ed = readout._measure_ed_on_raw(context.independent_evaluator, raw, context.labels.index_select(0, smoke_indices), role=bogt.TRAIN_ROLE, recorder=recorder, phase="requalification_E_D_forward_only", task_started=time.perf_counter())
        _require(ed.get("rows") == TRAIN_BATCH_SIZE and all(parameter.grad is None for parameter in context.independent_evaluator.parameters()), "E_D forward-only confirmation failed")
        _require(_frozen_evaluators_clear(evaluators), "canonical evaluator gradient firewall failed")
        recorder.sample(phase)
        peak = max(recorder.entries, key=lambda item: float(item["global_used_fraction"]))
        receipt = PreflightReceipt(
            source_semantics_id=str(source["execution_source_semantics_id"]),
            authorization_id=AUTHORIZATION_ID,
            checks={
                CORRECTNESS_INVARIANTS: {"status": "PASS", "inherited_step600_DEV_and_canonical_equivalence": file_sha256(root / PREFLIGHT_07_PATH), "three_real_updates": True, "negative_control": differences, "recovery_next_step": True, "E_D_gradient_count": 0},
                RESOURCE_ENVELOPE: {"status": "PASS", "max_global_used_fraction": float(peak["global_used_fraction"])},
                IMPLEMENTATION_SELECTION: {"status": "PASS", "operator": "accepted E_S/E_I/E_C vector combination over joint Encoder+Generator"},
            },
            selected_engine="cuda:0 production optical runtime",
            resource={"global_device_limit": MAX_DEVICE_MEMORY_FRACTION, "phase_records": recorder.entries},
            status=PREFLIGHT_READY,
        )
        lifecycle = _smoke_lifecycle(identity, source, receipt)
        elapsed = time.perf_counter() - task_started
        _require(elapsed <= MAX_PREFLIGHT_SECONDS, "cumulative preflight budget exceeded", ResourceBudgetExceeded)
        prior_projection = prior["projected_execution"]
        projected = elapsed + float(prior_projection["training_seconds"]) + float(prior_projection["resource_safe_finalization_seconds_from_pure_comparator"])
        _require(projected <= MAX_TOTAL_SECONDS, "projected scientific execution exceeds four-hour budget", ResourceBudgetExceeded)
        ledger_path = root / PREFLIGHT_09_GRADIENT_BUFFER_LEDGER_PATH
        return {
            "schema": PREFLIGHT_SCHEMA,
            "task_id": TASK_ID,
            "outer_gate": OUTER_GATE,
            "authorization_id": AUTHORIZATION_ID,
            "status": PREFLIGHT_READY,
            "source_semantics_id": source["execution_source_semantics_id"],
            "source_manifest_id": identity.source_manifest_id,
            "attempt_created": False,
            "input_authority_hashes": dict(authorities["authority_hashes"]),
            "step600_dev_baseline_qualification": {**prior["step600_dev_baseline_qualification"], "inherited_from_preflight_07_sha256": file_sha256(root / PREFLIGHT_07_PATH)},
            "canonical_operator_equivalence": prior["canonical_operator_equivalence"],
            "kappa_calibration": calibration,
            "kappa_calibration_receipt": {"path": PREFLIGHT_09_KAPPA_RECEIPT_PATH.as_posix(), "sha256": file_sha256(kappa_path), "batch_count": 20, "source_manifest_id": identity.source_manifest_id},
            "three_disposable_updates": history,
            "production_recovery_next_step": {"passed": True, "parameter_equivalence": parameter_equivalence, "optimizer_equivalence": optimizer_equivalence},
            "production_shaped_lifecycle": lifecycle,
            "gradient_buffer_ledger_path": PREFLIGHT_09_GRADIENT_BUFFER_LEDGER_PATH.as_posix(),
            "gradient_buffer_ledger_sha256": file_sha256(ledger_path),
            "resource_phase_records": recorder.entries,
            "resource_peak_global_fraction": float(peak["global_used_fraction"]),
            "projected_execution": {"preflight_seconds": elapsed, "training_seconds": prior_projection["training_seconds"], "finalization_seconds": prior_projection["resource_safe_finalization_seconds_from_pure_comparator"], "total_seconds": projected, "max_seconds": MAX_TOTAL_SECONDS},
            "receipt": receipt.to_dict(),
            "elapsed_seconds": elapsed,
        }, receipt
    except Exception as error:
        setattr(error, "preflight_phase", phase)
        setattr(error, "resource_phase_records", recorder.entries)
        raise
    finally:
        if evaluators is not None:
            for model in evaluators:
                model.to(torch.device("cpu"))
        del consumer, optimizer, evaluators
        gc.collect()
        if torch.cuda.is_available():
            torch.cuda.empty_cache()


def _preflight(
    root: Path,
    authorities: Mapping[str, Any],
    context: Any,
    source: Mapping[str, Any],
    identity: ResearchRunIdentity,
    *,
    task_started: float,
) -> tuple[dict[str, Any], PreflightReceipt]:
    device = torch.device("cuda:0")
    _require(torch.cuda.is_available(), "CUDA is required for the source-bound production preflight", ResourceBudgetExceeded)
    repair_history = _validated_repair_history(root, allow_existing_attempt01=True)
    _require(repair_history.get("attempt_01_creation_permitted") is False, "preflight must preserve historical attempt_01")
    _require(not (root / SCIENTIFIC_ATTEMPT_DIR).exists(), "attempt_02 must remain absent during preflight qualification")
    _require(context.data_identity == authorities["data_identity"], "TRAIN/DEV identity differs from accepted checkpoint")
    endpoint = authorities["pure"]["step1200"]
    for global_step in range(START_STEP + 1, END_STEP + 1):
        item = endpoint["training_history"][global_step - 1]
        expected_indices = torch.stack([permutation[global_step - 1] for permutation in authorities["step600"]["sampler_state"]["permutations"]])
        _require(item.get("step") == global_step and item.get("train_indices_sha256") == tensor_sha256(expected_indices), f"pure BOGT TRAIN order authority differs at global step {global_step}")

    recorder = _resource_recorder(
        root / PREFLIGHT_09_RESOURCE_TELEMETRY_PATH,
        gradient_ledger_path=root / PREFLIGHT_09_GRADIENT_BUFFER_LEDGER_PATH,
        stop_path=root / PREFLIGHT_09_RESOURCE_STOP_PATH,
    )
    phase_started = time.perf_counter()
    consumer: Any | None = None
    optimizer: torch.optim.Adam | None = None
    evaluators: tuple[torch.nn.Module, torch.nn.Module, torch.nn.Module] | None = None
    sampler: ExtendedTrainSampler | None = None
    equivalence: dict[str, Any] | None = None
    calibration: dict[str, Any] | None = None
    baseline_qualification: dict[str, Any] | None = None
    disposable_trace: list[dict[str, Any]] = []
    kappa_receipt_written = False
    disposable_updates_completed = 0
    try:
        phase = "production_step600_dev_baseline_entry_and_full_confirmation"
        dev_rows_before = int(context.ledger.to_dict().get(bogt.DEV_ROLE, 0))
        with tempfile.TemporaryDirectory(prefix="m0-cgga-step600-dev-baseline-") as temporary:
            baseline_attempt = Path(temporary) / "baseline_attempt"
            baseline_attempt.mkdir()
            baseline = _measure_full_dev(
                root,
                authorities["step600"],
                context,
                authorities,
                recorder,
                baseline_attempt,
                identity.source_manifest_id,
                step=START_STEP,
                task_started=task_started,
            )
        accepted_step600_dev = authorities["populations"]["step600_DEV"]
        _require(abs(float(baseline["E_D"]["macro_recall"]) - 0.1318) <= 0.001, "preflight step600 full DEV E_D baseline does not reproduce the accepted value")
        _require(abs(float(baseline["L_rec"]) - float(authorities["pure_report"]["step600"]["DEV"]["L_rec"])) <= 1.0e-6, "preflight step600 full DEV reconstruction does not reproduce the accepted readout")
        _require(baseline["E_D"]["true_labels"] == accepted_step600_dev["true_labels"], "preflight step600 DEV row order differs from the accepted baseline")
        runtime_identity = baseline["runtime_identity"]
        _require(
            runtime_identity.get("frozen_runtime_digest_source") == "reconstructed_from_source_bound_runtime_contract"
            and runtime_identity.get("serialized_frozen_runtime_digest_present") is False,
            "preflight did not exercise the accepted legacy step600 runtime-identity reconstruction path",
        )
        dev_rows_after = int(context.ledger.to_dict().get(bogt.DEV_ROLE, 0))
        _require(dev_rows_after - dev_rows_before == DEV_ROWS, "step600 DEV baseline qualification did not record exactly the fixed DEV population")
        _require(all(not parameter.requires_grad and parameter.grad is None for parameter in context.independent_evaluator.parameters()), "E_D was not frozen and gradient-free after baseline qualification")
        baseline_qualification = {
            "status": "PASS",
            "entry": "production _measure_full_dev(step=600)",
            "checkpoint_sha256": START_CHECKPOINT_SHA256,
            "fixed_DEV_rows": DEV_ROWS,
            "runtime_identity": runtime_identity,
            "E_D_macro_recall": float(baseline["E_D"]["macro_recall"]),
            "E_D_forward_rows": int(baseline["E_D"]["rows"]),
            "E_D_gradient_count": 0,
            "L_rec": float(baseline["L_rec"]),
            "DEV_rows_recorded_in_qualification_ledger": dev_rows_after - dev_rows_before,
            "scientific_optimizer_created": False,
            "scientific_parameter_updates": 0,
            "measurement_artifact_scope": "temporary qualification directory; no scientific measurement artifact",
        }
        recorder.sample("preflight_step600_dev_baseline_entry_complete", progress=baseline_qualification, started=phase_started)
        del baseline
        gc.collect()
        torch.cuda.empty_cache()
        _check_time(task_started, MAX_PREFLIGHT_SECONDS, phase, recorder=recorder)

        phase = "restore_accepted_step600_model_evaluators_without_optimizer"
        consumer, optimizer, evaluators = _load_training_runtime(root, authorities["step600"], context, create_optimizer=False)
        _require(_tree_equal(_trainable_state(consumer), {name: value.detach().cpu() for name, value in authorities["step600"]["trainable_parameter_state"].items()}), "preflight runtime parameters differ from accepted step600")
        _require(optimizer is None, "optimizer must not exist during canonical operator equivalence")
        _require(all(parameter.device.type == "cpu" for model in evaluators for parameter in model.parameters()), "training evaluators were not initially CPU-resident")
        _require(all(parameter.device.type == "cpu" for parameter in context.independent_evaluator.parameters()), "independent E_D must remain CPU-resident during calibration")
        sampler = ExtendedTrainSampler(context, authorities["step600"]["sampler_state"])
        recorder.sample("preflight_step600_runtime_loaded", started=phase_started)
        _record_gradient_buffers(recorder, consumer, "preflight_loaded_runtime_before_canonical_equivalence", expected_clean=True)

        phase = "verify_accepted_canonical_gradient_numerical_equivalence"
        _record_gradient_buffers(recorder, consumer, "canonical_equivalence_before", expected_clean=True)
        fixed_indices = authorities["gradient_batches"][0]
        fixed_labels = context.labels.index_select(0, fixed_indices)
        _require(fixed_labels.tolist() == list(range(10)), "canonical equivalence input is not the first registered class-balanced TRAIN batch")
        fixed_batch = context.images.index_select(0, fixed_indices).to(device)
        context.ledger.record(bogt.TRAIN_ROLE, int(fixed_indices.numel()))
        equivalence = _verify_canonical_equivalence(
            consumer,
            fixed_batch,
            evaluators,
            recorder=recorder,
            batch_index=0,
            row_indices_sha256=tensor_sha256(fixed_indices),
        )
        _record_gradient_buffers(recorder, consumer, "canonical_equivalence_after", expected_clean=True)
        del fixed_batch
        _require(optimizer is None, "canonical equivalence created an optimizer")
        _require(all(parameter.device.type == "cpu" for parameter in evaluators[0].parameters()), "canonical evaluator was not returned to CPU after equivalence")
        _require(all(parameter.device.type == "cpu" for parameter in context.independent_evaluator.parameters()), "E_D left CPU during canonical equivalence")
        optimizer = _restore_step600_optimizer(consumer, authorities["step600"])
        _require(_optimizer_steps_from_optimizer(optimizer) == [START_STEP, START_STEP], "preflight Adam did not restore step600 after equivalence")
        recorder.sample("preflight_step600_adam_restored_after_equivalence", started=phase_started)

        phase = "calibrate_kappa_on_registered_20_train_batches"
        _record_gradient_buffers(recorder, consumer, "kappa_calibration_before", expected_clean=True)
        calibration = _calibrate_kappa(consumer, context, evaluators, authorities["gradient_batches"], recorder, started=phase_started)
        _record_gradient_buffers(recorder, consumer, "kappa_calibration_after", expected_clean=True)
        kappa_receipt = _kappa_receipt_payload(calibration, authorities["gradient_batches"], context.labels, source)
        kappa_receipt_path = root / PREFLIGHT_09_KAPPA_RECEIPT_PATH
        atomic_write_new_json(kappa_receipt_path, kappa_receipt, artifact_kind="M0 canonical-gradient fixed-batch kappa receipt")
        kappa_receipt_written = True
        kappa_receipt_sha256 = file_sha256(kappa_receipt_path)
        _require(math.isclose(float(calibration["kappa"]), SCIENTIFIC_KAPPA, rel_tol=KAPPA_REPRODUCTION_RTOL, abs_tol=0.0), f"fixed-batch kappa differs from frozen value: measured={calibration['kappa']!r}, frozen={SCIENTIFIC_KAPPA!r}")
        _check_time(task_started, MAX_PREFLIGHT_SECONDS, phase, recorder=recorder)

        phase = "verify_joint_direction_and_bogt_negative_control"
        starting_parameters = _trainable_state(consumer)
        starting_optimizer = copy.deepcopy(optimizer.state_dict())
        starting_sampler = copy.deepcopy(sampler.state_dict())
        starting_rng = bogt._rng_state_payload()
        first_indices, first_labels = sampler.next_batch()
        _record_gradient_buffers(recorder, consumer, "negative_control_1_before_gradient_construction", batch_index=sampler.cursor, expected_clean=True)
        batch = context.images.index_select(0, first_indices).to(device)
        context.ledger.record(bogt.TRAIN_ROLE, int(first_indices.numel()))
        first_gradients = _compute_gradient_vectors(
            consumer,
            batch,
            smooth=evaluators[0],
            independent=evaluators[1],
            ec=evaluators[2],
            recorder=recorder,
            batch_index=sampler.cursor,
            phase_prefix="preflight_counterfactual_canonical",
        )
        _require(_joint_norm((first_gradients["G_canonical"][0],)) > 0.0 and _joint_norm((first_gradients["G_canonical"][1],)) > 0.0, "canonical joint vector does not reach both trainable groups")
        first_augmented = _apply_update_vectors(
            consumer,
            optimizer,
            G_BOGT=first_gradients["G_BOGT"],
            G_canonical=first_gradients["G_canonical"],
            kappa=calibration["kappa"],
            recorder=recorder,
            batch_index=sampler.cursor,
            phase_prefix="negative_control_1",
        )
        _record_gradient_buffers(recorder, consumer, "negative_control_1_after_optimizer_step", batch_index=sampler.cursor, expected_clean=False)
        augmented_state = _trainable_state(consumer)
        negative_control_1_restore = _restore_disposable_preflight_state(
            consumer,
            optimizer,
            sampler,
            trainable_state=starting_parameters,
            optimizer_state=starting_optimizer,
            sampler_state=starting_sampler,
            rng_state=starting_rng,
            recorder=recorder,
            phase_prefix="negative_control_1_restore",
        )
        _record_gradient_buffers(recorder, consumer, "negative_control_2_before_gradient_assignment", batch_index=sampler.cursor + 1, expected_clean=True)
        _set_gradients(_parameter_tuple(consumer), first_gradients["G_BOGT"])
        _record_gradient_buffers(recorder, consumer, "negative_control_2_after_BOGT_gradient_assignment", batch_index=sampler.cursor + 1, expected_clean=False)
        optimizer.step()
        _record_gradient_buffers(recorder, consumer, "negative_control_2_after_optimizer_step", batch_index=sampler.cursor + 1, expected_clean=False)
        bogt_only_state = _trainable_state(consumer)
        negative_control_differences = {name: float((augmented_state[name].to(torch.float64) - bogt_only_state[name].to(torch.float64)).norm()) for name in augmented_state}
        _require(all(math.isfinite(value) and value > 0.0 for value in negative_control_differences.values()), "canonical addition did not change both trainable parameter updates in the BOGT-only negative control")
        negative_control_2_restore = _restore_disposable_preflight_state(
            consumer,
            optimizer,
            sampler,
            trainable_state=starting_parameters,
            optimizer_state=starting_optimizer,
            sampler_state=starting_sampler,
            rng_state=starting_rng,
            recorder=recorder,
            phase_prefix="negative_control_2_restore",
        )
        _require(all(parameter.grad is None for parameter in context.independent_evaluator.parameters()), "E_D acquired a preflight gradient")
        _require(_frozen_evaluators_clear(evaluators), "canonical training evaluator parameter gradients are not empty")

        phase = "three_disposable_joint_updates_and_train_order_parity"
        _record_gradient_buffers(recorder, consumer, "normal_disposable_update_phase_entry", batch_index=sampler.cursor, expected_clean=True)
        endpoint_history = endpoint["training_history"]
        accepted_permutations = authorities["step600"]["sampler_state"]["permutations"]
        sampler_identity = _train_order_identity(sampler)
        for global_step in range(START_STEP + 1, START_STEP + 4):
            _require(sampler.cursor + 1 == global_step, "preflight update loop and next-unconsumed sampler cursor differ")
            update_started = time.perf_counter()
            trace = _preflight_train_update(
                consumer,
                optimizer,
                sampler,
                context,
                evaluators,
                endpoint_history,
                accepted_permutations,
                kappa=calibration["kappa"],
                sampler_identity=sampler_identity,
                path="shared_uninterrupted_and_restored_prefix",
                recorder=recorder,
            )
            trace["elapsed_seconds"] = time.perf_counter() - update_started
            disposable_trace.append(trace)
            disposable_updates_completed += 1
            _require(disposable_updates_completed == global_step - START_STEP, "preflight disposable update counter diverged from completed training calls")
            _require(trace["gradient_vectors"]["gradient_assignment_matches_G_total"] is True and trace["gradient_vectors"]["gradient_buffers_clean_after_update"] is True, "preflight update did not use current G_total and return to a clean gradient state")
            if disposable_updates_completed == 1:
                _require(_optimizer_steps_from_optimizer(optimizer) == [START_STEP + 1, START_STEP + 1], "first normal disposable update did not advance both Adam states exactly once")
            _check_time(task_started, MAX_PREFLIGHT_SECONDS, phase, recorder=recorder)
        _require(sampler.cursor == START_STEP + 3 and _optimizer_steps_from_optimizer(optimizer) == [START_STEP + 3, START_STEP + 3], "three disposable updates did not advance Adam/cursor by exactly three")
        recorder.sample("preflight_three_disposable_joint_updates", progress={"completed_updates": 3}, started=phase_started)

        phase = "production_recovery_next_update_equivalence"
        with tempfile.TemporaryDirectory(prefix="m0-cgga-recovery-") as temporary:
            manager = RecoveryCheckpointManager(temporary, identity, source_manifest_id=identity.source_manifest_id, authorization_id=AUTHORIZATION_ID, keep=2)
            payload = _checkpoint_payload(consumer, optimizer, sampler, disposable_trace, context, source, identity, kappa=calibration["kappa"], calibration=calibration, cumulative_elapsed_seconds=time.perf_counter() - task_started)
            step603_boundary = {
                "parameters": _trainable_state(consumer),
                "runtime_buffer_sha256": {name: tensor_sha256(value.detach().cpu()) for name, value in consumer.runtime.named_buffers()},
                "optimizer": _semantic_optimizer_state(consumer, optimizer),
                "optimizer_configuration": consumer.optimizer_manifest(optimizer),
                "rng": bogt._rng_state_payload(),
                "sampler": copy.deepcopy(sampler.state_dict()),
                "runtime_mode": consumer.runtime.training,
                "evaluator_modes": [model.training for model in evaluators],
                "E_D_mode": context.independent_evaluator.training,
                "runtime_identity": consumer.runtime_identity(),
            }
            checkpoint_path = manager.save({"phase": "TRAINING", "scientific_state": payload}, step=sampler.cursor)
            _require(payload["step"] == START_STEP + 3 and payload["sampler_state"]["cursor"] == START_STEP + 3, "production recovery checkpoint did not save the next-unconsumed position after step603")
            path_b_trace: list[dict[str, Any]] = []
            path_b_states: dict[int, dict[str, Any]] = {}
            for expected_step in range(START_STEP + 4, START_STEP + 9):
                _require(sampler.cursor + 1 == expected_step, "uninterrupted path cursor differs from the next global step")
                record = _preflight_train_update(
                    consumer,
                    optimizer,
                    sampler,
                    context,
                    evaluators,
                    endpoint_history,
                    accepted_permutations,
                    kappa=calibration["kappa"],
                    sampler_identity=sampler_identity,
                    path="B_uninterrupted",
                    recorder=recorder,
                    capture_gradient_vectors=True,
                )
                gradient_vectors = record.pop("preflight_gradient_vectors")
                path_b_trace.append(record)
                path_b_states[expected_step] = {
                    "parameters": _trainable_state(consumer),
                    "runtime_buffer_sha256": {name: tensor_sha256(value.detach().cpu()) for name, value in consumer.runtime.named_buffers()},
                    "optimizer": _semantic_optimizer_state(consumer, optimizer),
                    "rng": bogt._rng_state_payload(),
                    "sampler": copy.deepcopy(sampler.state_dict()),
                    "gradient_vectors": gradient_vectors,
                    "losses": dict(record["losses"]),
                    "runtime_mode": consumer.runtime.training,
                    "evaluator_modes": [model.training for model in evaluators],
                    "E_D_mode": context.independent_evaluator.training,
                }
                _check_time(task_started, MAX_PREFLIGHT_SECONDS, phase, recorder=recorder)
            path_b_final_sampler = copy.deepcopy(sampler.state_dict())
            wrapper = manager.load(checkpoint_path)
            restored = wrapper.get("payload", {}).get("scientific_state")
            _require(isinstance(restored, Mapping), "production recovery manager returned no scientific payload")
            disposable_trace = _restore_recovery(restored, consumer, optimizer, sampler, context, source, identity, kappa=calibration["kappa"])
            _require(sampler.cursor == START_STEP + 3 and _optimizer_steps_from_optimizer(optimizer) == [START_STEP + 3, START_STEP + 3], "production recovery did not restore step603 Adam and next-row cursor")
            restored_step603_boundary = {
                "parameters": _trainable_state(consumer),
                "runtime_buffer_sha256": {name: tensor_sha256(value.detach().cpu()) for name, value in consumer.runtime.named_buffers()},
                "optimizer": _semantic_optimizer_state(consumer, optimizer),
                "optimizer_configuration": consumer.optimizer_manifest(optimizer),
                "rng": bogt._rng_state_payload(),
                "sampler": copy.deepcopy(sampler.state_dict()),
                "runtime_mode": consumer.runtime.training,
                "evaluator_modes": [model.training for model in evaluators],
                "E_D_mode": context.independent_evaluator.training,
                "runtime_identity": consumer.runtime_identity(),
            }
            s0_restored_exact = {
                "parameters_exact": _tree_equal(step603_boundary["parameters"], restored_step603_boundary["parameters"]),
                "runtime_buffers_exact": step603_boundary["runtime_buffer_sha256"] == restored_step603_boundary["runtime_buffer_sha256"],
                "semantic_optimizer_slots_exact": _tree_equal(step603_boundary["optimizer"], restored_step603_boundary["optimizer"]),
                "optimizer_configuration_exact": _tree_equal(step603_boundary["optimizer_configuration"], restored_step603_boundary["optimizer_configuration"]),
                "rng_exact": _tree_equal(step603_boundary["rng"], restored_step603_boundary["rng"]),
                "sampler_exact": _tree_equal(step603_boundary["sampler"], restored_step603_boundary["sampler"]),
                "runtime_and_evaluator_modes_exact": step603_boundary["runtime_mode"] == restored_step603_boundary["runtime_mode"] and step603_boundary["evaluator_modes"] == restored_step603_boundary["evaluator_modes"] and step603_boundary["E_D_mode"] == restored_step603_boundary["E_D_mode"],
                "runtime_identity_exact": _tree_equal(step603_boundary["runtime_identity"], restored_step603_boundary["runtime_identity"]),
                "optimizer_parameter_order": [group["parameter_names"] for group in step603_boundary["optimizer"]["parameter_groups"]],
            }
            s0_restored_exact["passed"] = all(value for key, value in s0_restored_exact.items() if key != "optimizer_parameter_order")
            _require(s0_restored_exact["passed"], f"S0 production restore changed step603 state: {json.dumps(s0_restored_exact, sort_keys=True)}")
            path_c_trace: list[dict[str, Any]] = []
            recovery_step_comparisons: list[dict[str, Any]] = []
            for expected_step in range(START_STEP + 4, START_STEP + 9):
                _require(sampler.cursor + 1 == expected_step, "restored path cursor differs from the next global step")
                record = _preflight_train_update(
                    consumer,
                    optimizer,
                    sampler,
                    context,
                    evaluators,
                    endpoint_history,
                    accepted_permutations,
                    kappa=calibration["kappa"],
                    sampler_identity=sampler_identity,
                    path="C_save_restore",
                    recorder=recorder,
                    capture_gradient_vectors=True,
                )
                gradient_vectors = record.pop("preflight_gradient_vectors")
                path_c_trace.append(record)
                expected_state = path_b_states[expected_step]
                expected_parameters = expected_state["parameters"]
                actual_parameters = _trainable_state(consumer)
                parameter_comparison = _named_tensor_equivalence_metrics(
                    expected_parameters,
                    actual_parameters,
                    rtol=accepted_canonical.RECOVERY_UPDATE_RTOL,
                    atol=accepted_canonical.RECOVERY_UPDATE_ATOL,
                )
                loss_comparison = _scalar_equivalence_metrics(expected_state["losses"], record["losses"])
                optimizer_comparison = _optimizer_recovery_equivalence(expected_state["optimizer"], _semantic_optimizer_state(consumer, optimizer))
                gradient_comparison = _gradient_recovery_equivalence(expected_state["gradient_vectors"], gradient_vectors)
                comparison = {
                    "global_step": expected_step,
                    "train_rows_exact": record["train_indices"] == path_b_trace[expected_step - (START_STEP + 4)]["train_indices"],
                    "losses": loss_comparison,
                    "parameters": parameter_comparison,
                    "optimizer": optimizer_comparison,
                    "gradients": gradient_comparison,

def _stratified_unpaired_bootstrap(
    old_labels: torch.Tensor,
    old_predictions: torch.Tensor,
    new_labels: torch.Tensor,
    new_predictions: torch.Tensor,
) -> dict[str, Any]:
    _require(old_labels.shape == old_predictions.shape and new_labels.shape == new_predictions.shape, "unpaired S1 bootstrap vectors differ in shape")
    generator = torch.Generator(device="cpu").manual_seed(BOOTSTRAP_SEED)
    deltas = torch.zeros((BOOTSTRAP_REPLICATES,), dtype=torch.float64)
    class_counts: dict[str, dict[str, int]] = {}
    for digit in range(10):
        old_mask = old_labels == digit
        new_mask = new_labels == digit
        old_correct = (old_predictions[old_mask] == digit).to(torch.float64)
        new_correct = (new_predictions[new_mask] == digit).to(torch.float64)
        _require(old_correct.numel() > 0 and new_correct.numel() > 0, f"S1 bootstrap lacks class {digit}")
        _require(old_correct.numel() == 200 and new_correct.numel() == 200, f"S1 bootstrap class {digit} does not have exactly 200 independent rows")
        old_draw = torch.randint(old_correct.numel(), (BOOTSTRAP_REPLICATES, old_correct.numel()), generator=generator)
        new_draw = torch.randint(new_correct.numel(), (BOOTSTRAP_REPLICATES, new_correct.numel()), generator=generator)
        deltas += (new_correct[ new_draw ].mean(dim=1) - old_correct[old_draw].mean(dim=1)) / 10.0
        class_counts[str(digit)] = {"old": int(old_correct.numel()), "new": int(new_correct.numel())}
    interval = torch.quantile(deltas, torch.tensor([0.025, 0.975], dtype=torch.float64))
    old_macro = float(sum((old_predictions[old_labels == digit] == digit).to(torch.float64).mean() for digit in range(10)) / 10.0)
    new_macro = float(sum((new_predictions[new_labels == digit] == digit).to(torch.float64).mean() for digit in range(10)) / 10.0)
    return {
        "method": "independent class-stratified bootstrap difference; no row pairing asserted",
        "replicates": BOOTSTRAP_REPLICATES,
        "seed": BOOTSTRAP_SEED,
        "old_macro_recall": old_macro,
        "new_macro_recall": new_macro,
        "delta_macro_recall": new_macro - old_macro,
        "ci95": {"lower": float(interval[0]), "upper": float(interval[1])},
        "class_counts": class_counts,
    }

def _canonical_measurement_losses(
    consumer: Any,
    context: Any,
    evaluators: Sequence[torch.nn.Module],
    batches: Sequence[torch.Tensor],
    *,
    recorder: Any | None = None,
) -> list[dict[str, Any]]:
    consumer.runtime.eval()
    for model in evaluators:
        model.eval()
    rows: list[dict[str, Any]] = []
    with torch.no_grad():
        for index, indices in enumerate(batches):
            batch = context.images.index_select(0, indices).to(consumer.device)
            if recorder is not None:
                recorder.sample("canonical_measurement_batch_start", batch_index=index, started=None)
            breakdown = consumer.forward_loss(batch, data_role=bogt.TRAIN_ROLE)
            raw = breakdown.result.raw_intensity
            loss_values: dict[str, float] = {}
            for name, evaluator in zip(("E_S", "E_I", "E_C"), evaluators, strict=True):
                evaluator.to(consumer.device)
                loss = _canonical_loss_for_evaluator(raw, batch, evaluator, name)
                loss_values[f"ell_{name[-1]}"] = float(loss.detach().cpu())
                del loss
                evaluator.to(torch.device("cpu"))
                if torch.device(consumer.device).type == "cuda":
                    torch.cuda.empty_cache()
                if recorder is not None:
                    recorder.sample("canonical_measurement_evaluator_released", batch_index=index, evaluator=name)
            rows.append({"batch_index": index, "row_indices_sha256": tensor_sha256(indices), **loss_values})
            context.ledger.record(bogt.TRAIN_ROLE, int(indices.numel()))
            del batch, breakdown, raw
            gc.collect()
            if recorder is not None:
                recorder.sample("canonical_measurement_batch_end", batch_index=index)
    return rows

def _measure_full_dev(
    root: Path,

    context: Any,
    authorities: Mapping[str, Any],
    recorder: Any,
    attempt: Path,
    source_manifest_id: str,
    *,
    step: int,
    task_started: float,
) -> dict[str, Any]:
    if step == START_STEP:
        identity_consumer, runtime_identity = _load_step600_runtime_with_identity(root, checkpoint, context)
        trainable_before = _trainable_state(identity_consumer)

        checkpoint,

    payload = _checkpoint_payload(consumer, optimizer, sampler, history, context, source, identity, kappa=kappa, calibration=calibration, cumulative_elapsed_seconds=time.perf_counter() - task_started)
    payload.pop("cumulative_elapsed_seconds")
    path = attempt / f"scientific_step_{sampler.cursor:06d}.pt"
    _write_torch_or_compare(path, payload, f"scientific step{sampler.cursor} checkpoint")
    return path


def _read_recovery(
    manager: RecoveryCheckpointManager,
    consumer: Any,
    optimizer: torch.optim.Adam,
    sampler: ExtendedTrainSampler,
    context: Any,
    source: Mapping[str, Any],
    identity: ResearchRunIdentity,
    *,
    kappa: float,
) -> tuple[str, list[dict[str, Any]]]:
    latest = manager.latest()
    _require(latest is not None, "scientific attempt has no verified recovery checkpoint")
    wrapper = manager.load(latest)
    recovery = wrapper.get("payload", {})
    phase = recovery.get("phase")
    _require(phase in {"TRAINING", "FINALIZATION"}, "recovery phase is not registered")
    state = recovery.get("scientific_state")
    _require(isinstance(state, Mapping), "recovery checkpoint lacks scientific state")
    history = _restore_recovery(state, consumer, optimizer, sampler, context, source, identity, kappa=kappa)
    return str(phase), history

def _run_attempt(
    root: Path,
    authorities: Mapping[str, Any],
    context: Any,
    source: Mapping[str, Any],
    identity: ResearchRunIdentity,
    preflight: Mapping[str, Any],
    receipt: PreflightReceipt,
    *,
    resume: bool,
) -> dict[str, Any]:
    _require(receipt.status == PREFLIGHT_READY and receipt.source_semantics_id == source["execution_source_semantics_id"] and receipt.authorization_id == AUTHORIZATION_ID, "scientific execution lacks the current source-bound READY receipt")
    _require(context.data_identity == authorities["data_identity"], "scientific context data identity differs from step600")
    _require(not (root / SCIENTIFIC_ATTEMPT_DIR).exists() or resume, "attempt_02 is write-once and already exists")
    _require(preflight.get("status") == PREFLIGHT_READY and preflight.get("source_manifest_id") == source.get("source_manifest_id"), "READY preflight source identity changed")
    kappa = SCIENTIFIC_KAPPA
    calibration = preflight["kappa_calibration"]
    _require(math.isclose(float(calibration.get("kappa", float("nan"))), kappa, rel_tol=KAPPA_REPRODUCTION_RTOL, abs_tol=0.0) and calibration.get("batch_count") == 20, "fixed source-bound kappa differs from qualified value")

    attempt = root / SCIENTIFIC_ATTEMPT_DIR
    if resume:
        _require(attempt.is_dir(), "resume requires the existing attempt_02")
        _require(_read_json(attempt / "source_manifest.json") == dict(source), "attempt source manifest changed before resume")
        _require(_read_json(attempt / "preflight.json") == dict(preflight), "attempt READY receipt changed before resume")
        old = _read_json(attempt / "latest_status.json")
        _require(old.get("state") in {RUN_PREPARING, RUNNING, RUN_INTERRUPTED}, "only an interrupted or stale running attempt can resume")
        status_path = attempt / "status.json"
        if status_path.exists():
            _require(_read_json(status_path).get("state") == RUN_INTERRUPTED, "terminal attempt status cannot be resumed")
    else:
        _require(not attempt.exists(), "attempt_02 already exists; this authorization is write-once")

    control: ResearchRunControl | None = None
    consumer: Any | None = None
    optimizer: torch.optim.Adam | None = None
    evaluators: tuple[torch.nn.Module, torch.nn.Module, torch.nn.Module] | None = None
    sampler: ExtendedTrainSampler | None = None
    heartbeat: HeartbeatWriter | None = None
    recorder: Any | None = None
    manager: RecoveryCheckpointManager | None = None
    attempt_opened = False
    current_step = START_STEP
    history: list[dict[str, Any]] = []
    phase = "scientific_initialization"
    preflight_elapsed = float(preflight.get("elapsed_seconds", 0.0))
    _require(math.isfinite(preflight_elapsed) and 0.0 <= preflight_elapsed <= MAX_PREFLIGHT_SECONDS, "source-bound preflight elapsed time is invalid")
    execution_started = time.perf_counter()
    budget_origin = execution_started - preflight_elapsed
    try:
        if resume:
            control = ResearchRunControl(identity, AUTHORIZATION_ID, preflight_status=PREFLIGHT_READY, state=RUN_INTERRUPTED, receipt=receipt)
            effective = control.resume_scientific_attempt(attempt, SCIENTIFIC_ATTEMPT_ID, receipt)
        else:
            control = ResearchRunControl(identity, AUTHORIZATION_ID)
            control.mark_preflight_ready(receipt)
            control.start_execution()
            effective = control.open_scientific_attempt(attempt, SCIENTIFIC_ATTEMPT_ID, receipt)
        attempt_opened = True
        _process_instance_path(attempt)
        if not resume:
            atomic_write_new_json(attempt / "source_manifest.json", dict(source), artifact_kind="M0 canonical-gradient source manifest")
            atomic_write_new_json(attempt / "preflight.json", dict(preflight), artifact_kind="M0 source-bound READY receipt")
            atomic_write_new_json(
                attempt / "execution_manifest.json",
                {
                    "schema": "m0_bogt_canonical_gradient_augmentation_execution_manifest.v1",
                    "task_id": TASK_ID,
                    "outer_gate": OUTER_GATE,
                    "authorization": {"id": AUTHORIZATION_ID, "token": AUTHORIZATION_TOKEN, "scope": "one 600-update augmented trajectory from accepted step600"},
                    "starting_checkpoint": {"path": START_CHECKPOINT.as_posix(), "sha256": START_CHECKPOINT_SHA256, "step": START_STEP},
                    "pure_bogt_comparator": {"checkpoint_sha256": authorities["pure"]["endpoint_hash"], "parent_status": "INVALID_HARNESS", "readout": "VALID_COMPLETED"},
                    "source_manifest_id": identity.source_manifest_id,
                    "run_identity": effective.to_dict(),
                    "trainable": ["encoder.phase", "generator.phase"],
                    "frozen": ["W_OLS", "optical geometry", "BOGT references/weights", "E_S", "E_I", "E_C", "E_D"],
                    "optimizer": {"configuration": dict(authorities["step600"]["optimizer_configuration"]), "restored_step": START_STEP, "fresh_zero": False},
                    "canonical_operator": {"normalization": CANONICAL_NORMALIZATION_RULE, "combination": CANONICAL_COMBINATION_RULE, "delta_SIC": three.DELTA_SIC, "scalar_canonical_loss_defined": False},
                    "kappa": kappa,
                    "updates": UPDATE_COUNT,
                    "sampling": {"TRAIN": "accepted saved class-balanced permutations; exact pure BOGT row order", "S1": "same-class 16-NN fixed seed, 200 per class"},
                    "data_firewall": {"protected": 0, "official_test": 0, "sealed": 0},
                },
                artifact_kind="M0 canonical-gradient execution manifest",
            )
        existing_telemetry = attempt / "resource_telemetry.jsonl"
        recorder = _resource_recorder(
            existing_telemetry,
            stop_path=attempt / "resource_stop.json",
            allow_existing_telemetry=resume,
        )
        if resume and existing_telemetry.is_file():
            recorder.entries.extend(json.loads(line) for line in existing_telemetry.read_text(encoding="utf-8").splitlines() if line.strip())
        heartbeat = HeartbeatWriter(attempt, effective.run_id, total_steps=TRAIN_ROWS + S1_ROWS)
        if not resume:
            heartbeat.update(state=RUN_PREPARING, phase="STATISTICS", step=0, latest_losses={"step600_start": START_STEP}, force=True)
        phase = "step600_full_DEV_baseline_confirmation"
        _check_time(budget_origin, MAX_TOTAL_SECONDS, phase)
        dev0_path = attempt / f"step{START_STEP:06d}_dev_measurement.json"
        if dev0_path.exists():
            dev0 = _read_json(dev0_path)
        else:
            dev0 = _measure_full_dev(root, authorities["step600"], context, authorities, recorder, attempt, identity.source_manifest_id, step=START_STEP, task_started=time.perf_counter())
        _require(abs(float(dev0["E_D"]["macro_recall"]) - 0.1318) <= 0.001, "step600 full DEV E_D baseline does not reproduce the accepted value")
        accepted_step600_dev = authorities["populations"]["step600_DEV"]
        _require(abs(float(dev0["L_rec"]) - float(authorities["pure_report"]["step600"]["DEV"]["L_rec"])) <= 1.0e-6, "step600 full DEV reconstruction does not reproduce the accepted readout")
        _require(dev0["E_D"]["true_labels"] == accepted_step600_dev["true_labels"], "step600 full DEV row order differs from the accepted baseline")

        phase = "independent_scientific_runtime_reconstruction_from_accepted_step600"
        consumer, optimizer, evaluators = _load_training_runtime(root, authorities["step600"], context)
        start_assertions = _verify_checkpoint_start(consumer, optimizer, authorities["step600"])
        recorder.sample(phase)
        sampler = ExtendedTrainSampler(context, authorities["step600"]["sampler_state"])
        if resume:
            manager = RecoveryCheckpointManager(attempt, identity, source_manifest_id=identity.source_manifest_id, authorization_id=AUTHORIZATION_ID, keep=32)
            phase_name, history = _read_recovery(manager, consumer, optimizer, sampler, context, source, identity, kappa=kappa)
            latest = manager.latest()
            recovery_payload = manager.load(latest).get("payload", {}).get("scientific_state", {}) if latest is not None else {}
            previous_elapsed = float(recovery_payload.get("cumulative_elapsed_seconds", 0.0))
            _require(math.isfinite(previous_elapsed) and previous_elapsed >= preflight_elapsed, "resume cumulative execution time is invalid")
            budget_origin = time.perf_counter() - previous_elapsed
            current_step = sampler.cursor
            if phase_name == "FINALIZATION":
                _require(sampler.cursor == END_STEP and len(history) == UPDATE_COUNT, "finalization recovery is not at the fixed step1200 endpoint")
        else:
            _require(sampler.cursor == START_STEP and not history, "fresh attempt sampler did not start at step600")
            manager = RecoveryCheckpointManager(attempt, identity, source_manifest_id=identity.source_manifest_id, authorization_id=AUTHORIZATION_ID, keep=32)
            _save_recovery(manager, consumer, optimizer, sampler, history, context, source, identity, kappa=kappa, calibration=calibration, phase="TRAINING", task_started=budget_origin)
        initial_encoder = authorities["step600"]["trainable_parameter_state"]["encoder.phase"].detach().cpu().clone()
        initial_generator = authorities["step600"]["trainable_parameter_state"]["generator.phase"].detach().cpu().clone()
        initial_frozen_runtime = consumer.frozen_state_sha256
        initial_weights = consumer.weights.to_payload()
        initial_references = consumer.references.identity()
        initial_ed_state_sha = _module_state_sha256(context.independent_evaluator)
        evaluator_state_sha = {
            "E_S": _module_state_sha256(evaluators[0]),
            "E_I": _module_state_sha256(evaluators[1]),
            "E_C": _module_state_sha256(evaluators[2]),
        }
        current_step = sampler.cursor
        if resume and history and heartbeat is not None:
            heartbeat.update(state=RUNNING, phase="TRAINING", step=sampler.cursor - START_STEP, latest_losses=history[-1]["losses"], force=True)
        if not resume or (sampler.cursor < END_STEP and (not history or history[-1].get("global_step", 0) < END_STEP)):
            recovery_phase = phase_name if resume else None
            if resume:
                _require(recovery_phase == "TRAINING", "only training-phase recovery is allowed before finalization")
            heartbeat.update(state=RUNNING, phase="TRAINING", step=sampler.cursor - START_STEP, latest_losses={"global_step": sampler.cursor}, force=True)
            interrupt_factory = _interrupt_payload_factory(consumer, optimizer, sampler, history, context, source, identity, kappa=kappa, calibration=calibration, task_started=budget_origin)
            with InterruptController(heartbeat, manager, interrupt_factory, phase="TRAINING", total_steps=UPDATE_COUNT) as interrupt:
                while sampler.cursor < END_STEP:
                    _check_time(budget_origin, MAX_TOTAL_SECONDS, "600-update scientific training")
                    global_step = sampler.cursor + 1
                    expected = authorities["pure"]["step1200"]["training_history"][global_step - 1]
                    record = _train_update(consumer, optimizer, sampler, context, evaluators, kappa=kappa, expected_row_hash=expected["train_indices_sha256"], recorder=recorder)
                    history.append(record)
                    current_step = sampler.cursor
                    completed = sampler.cursor - START_STEP
                    if sampler.cursor in CHECKPOINT_STEPS:
                        _save_fixed_checkpoint(attempt, consumer, optimizer, sampler, history, context, source, identity, kappa=kappa, calibration=calibration, task_started=budget_origin)
                    if completed % RECOVERY_INTERVAL == 0 or sampler.cursor == END_STEP:
                        recovery_path = _save_recovery(manager, consumer, optimizer, sampler, history, context, source, identity, kappa=kappa, calibration=calibration, phase="TRAINING", task_started=budget_origin)
                        recorder.sample("scientific_augmented_training", progress={"global_step": sampler.cursor, "completed_updates": completed}, started=execution_started)
                        heartbeat.update(state=RUNNING, phase="TRAINING", step=completed, latest_losses=record["losses"], last_recovery_checkpoint=recovery_path.name, force=True)
                    else:
                        heartbeat.update(state=RUNNING, phase="TRAINING", step=completed, latest_losses=record["losses"], force=False)
                    _require(sampler.cursor == START_STEP + completed, "scientific training cursor/update counter diverged")
                _require(interrupt is not None, "training interrupt controller did not initialize")
            _require(sampler.cursor == END_STEP and len(history) == UPDATE_COUNT, "scientific arm did not execute exactly 600 updates")
            _require(_optimizer_steps_from_optimizer(optimizer) == [END_STEP, END_STEP], "scientific Adam did not finish at internal step1200")
            _require((attempt / f"scientific_step_{END_STEP:06d}.pt").is_file(), "step1200 fixed checkpoint was not persisted inside the update loop")
            _save_recovery(manager, consumer, optimizer, sampler, history, context, source, identity, kappa=kappa, calibration=calibration, phase="FINALIZATION", task_started=budget_origin)
        else:
            _require(sampler.cursor == END_STEP and _optimizer_steps_from_optimizer(optimizer) == [END_STEP, END_STEP], "resumed endpoint Adam/cursor is not at step1200")

        phase = "step1200_fixed_batch_canonical_loss_confirmation"
        endpoint_loss_path = attempt / "step1200_fixed_20_train_batch_evaluator_losses.json"
        if endpoint_loss_path.exists():
            endpoint_loss_rows = _read_json(endpoint_loss_path)["records"]
        else:
            endpoint_loss_rows = _canonical_measurement_losses(consumer, context, evaluators, authorities["gradient_batches"], recorder=recorder)
            _write_json_or_compare(endpoint_loss_path, {"data_role": bogt.TRAIN_ROLE, "batch_start_cursor": readout.GRADIENT_BATCH_START, "records": endpoint_loss_rows, "E_D_forward_count": 0}, "step1200 fixed-batch canonical loss measurements")
        start_loss_summary = _median_losses(calibration["baseline_losses"])
        endpoint_loss_summary = _median_losses(endpoint_loss_rows)
        local_response = {name: endpoint_loss_summary[name] < start_loss_summary[name] for name in ("ell_S", "ell_I", "ell_C")}
        local_response_supported = sum(local_response.values()) >= 2
        evaluator_state_sha_after_training = {
            "E_S": _module_state_sha256(evaluators[0]),
            "E_I": _module_state_sha256(evaluators[1]),
            "E_C": _module_state_sha256(evaluators[2]),
        }
        _require(evaluator_state_sha_after_training == evaluator_state_sha, "one or more canonical training evaluators changed during optimization")
        for model in evaluators:
            model.to(torch.device("cpu"))
        del evaluators
        evaluators = None
        gc.collect()
        torch.cuda.empty_cache()

        endpoint_path = attempt / f"scientific_step_{END_STEP:06d}.pt"
        endpoint_checkpoint = safe_torch_load(endpoint_path, map_location="cpu")
        endpoint_hash = file_sha256(endpoint_path)
        endpoint_references = bogt.FrozenBogtReferences.from_payload(endpoint_checkpoint["references"]).identity()
        source_manifest_id = identity.source_manifest_id
        phase = "full_train_ols_collection"
        _check_time(budget_origin, MAX_TOTAL_SECONDS, phase)
        ols_started = time.perf_counter()
        bank, ols_collection = readout._collect_train_ols(
            consumer,
            context,
            attempt,
            identity.source_manifest_id,
            endpoint_hash,
            recorder,
            heartbeat,
            ols_started,
        )
        context.ledger.record(bogt.TRAIN_ROLE, TRAIN_ROWS)

        phase = "fixed_s1_input_generation"
        old_coefficients = safe_torch_load(root / readout.STEP600_S1_OLS_PATH, map_location="cpu")
        generated_ols, s1_metadata, s1_arrays = readout._s1_inputs(bank, context, old_coefficients)
        requested = s1_arrays["requested"].detach().cpu().to(torch.long)
        _require(generated_ols.shape == (S1_ROWS, 100) and bool(torch.equal(torch.bincount(requested, minlength=10), torch.full((10,), 200))), "fixed S1 did not produce exactly 200 requests per class")
        coeff_payload = {
            "schema": "m0_bogt_canonical_gradient_augmented_s1_ols.v1",
            "z": generated_ols.detach().cpu().to(torch.complex64).contiguous(),
            "requested_labels": requested,
            "sample_indices": s1_arrays["sample_indices"].detach().cpu().long(),
            "seeds": s1_arrays["seeds"].detach().cpu().long(),
            "anchor_positions": s1_arrays["anchor_positions"].detach().cpu().long(),
            "neighbor_positions": s1_arrays["neighbor_positions"].detach().cpu().long(),
            "metadata": s1_metadata,
            "data_role": bogt.TRAIN_ROLE,
            "source_step": END_STEP,
            "source_train_ols_sha256": tensor_sha256(bank),
            "source_manifest_id": identity.source_manifest_id,
            "selected_checkpoint_sha256": endpoint_hash,
            "sampler_family": "S1_same_class_16NN",
            "optimizer_created": False,
            "updates": 0,
        }
        _write_torch_or_compare(attempt / "generated_s1_ols_coefficients.pt", coeff_payload, "augmented step1200 S1 OLS coefficients")
        phase = "fixed_s1_raw_cmos_generation"
        s1_started = time.perf_counter()
        raw_s1, powers_s1, s1_generation = readout._generate_s1_raw(consumer, generated_ols, requested, attempt, source_manifest_id, endpoint_hash, recorder, heartbeat, s1_started)
        sheet = readout._write_contact_sheet(raw_s1, requested, attempt / "s1_augmented_step1200_contact_sheet.png", common_vmax=1.0)
        raw_s1_validity = readout._combine_raw_validity([readout._raw_chunk_validity(raw_s1[start : start + readout.S1_RAW_CHUNK_ROWS]) for start in range(0, S1_ROWS, readout.S1_RAW_CHUNK_ROWS)])
        _require(s1_generation["optical_power_finite"] and bool(torch.isfinite(powers_s1).all().item()), "S1 optical power measurement is invalid")
        del bank, generated_ols, old_coefficients, powers_s1
        del consumer, optimizer, sampler
        consumer = optimizer = sampler = None
        gc.collect()
        torch.cuda.empty_cache()

        phase = "step1200_full_DEV_optical_and_E_D_confirmation"
        dev_started = time.perf_counter()
        dev1 = _measure_full_dev(root, endpoint_checkpoint, context, authorities, recorder, attempt, identity.source_manifest_id, step=END_STEP, task_started=dev_started)
        s1_ed = readout._measure_ed_on_raw(
            context.independent_evaluator,
            raw_s1,
            requested,
            role=bogt.TRAIN_ROLE,
            recorder=recorder,
            phase="scientific_augmented_step1200_S1_E_D_confirmation",
            task_started=time.perf_counter(),
        )
        context.ledger.record(bogt.TRAIN_ROLE, S1_ROWS)
        s1_ed_metrics = _classification_metrics(torch.as_tensor(s1_ed["true_labels"], dtype=torch.long), torch.as_tensor(s1_ed["prediction_labels"], dtype=torch.long), role=bogt.TRAIN_ROLE)
        s1_ed_metrics["evaluator_device"] = s1_ed.get("evaluator_device")
        _write_json_or_compare(attempt / "step1200_s1_ed_measurement.json", s1_ed_metrics, "step1200 S1 E_D measurement")
        sheet_record = sheet

        pure_dev_record = authorities["pure_report"]["step1200"]["DEV"]["E_D"]
        pure_s1_record = authorities["pure_report"]["step1200"]["S1"]["E_D"]
        step600_dev_record = dev0["E_D"]
        labels0, pred0 = _prediction_vectors(step600_dev_record, rows=DEV_ROWS, name="step600 DEV")
        labelsp, predp = _prediction_vectors(pure_dev_record, rows=DEV_ROWS, name="pure step1200 DEV")
        labels1, pred1 = _prediction_vectors(dev1["E_D"], rows=DEV_ROWS, name="augmented step1200 DEV")
        _require(torch.equal(labels0, labelsp) and torch.equal(labels0, labels1), "DEV row order differs between step600, pure and augmented endpoints")
        dev_vs_pure = accepted_canonical._paired_bootstrap(labels0, predp == labels0, pred1 == labels0, replicates=BOOTSTRAP_REPLICATES, seed=BOOTSTRAP_SEED)
        dev_vs_start = accepted_canonical._paired_bootstrap(labels0, pred0 == labels0, pred1 == labels0, replicates=BOOTSTRAP_REPLICATES, seed=BOOTSTRAP_SEED)
        dev_delta_pure = float(dev1["E_D"]["macro_recall"]) - float(authorities["populations"]["step1200_DEV"]["macro_recall"])

        pure_s1_metrics = authorities["populations"]["step1200_S1"]
        old_s1_labels, old_s1_pred = _prediction_vectors(pure_s1_record, rows=S1_ROWS, name="pure step1200 S1")
        new_s1_labels, new_s1_pred = _prediction_vectors(s1_ed_metrics, rows=S1_ROWS, name="augmented step1200 S1")
        _require(torch.equal(old_s1_labels, new_s1_labels) and torch.equal(new_s1_labels, requested), "S1 requested labels changed between fixed pure and augmented populations")
        s1_bootstrap = _stratified_unpaired_bootstrap(old_s1_labels, old_s1_pred, new_s1_labels, new_s1_pred)
        s1_delta_pure = float(s1_ed_metrics["macro_recall"]) - float(pure_s1_metrics["macro_recall"])
        dev_reconstruction_delta = float(dev1["L_rec"]) - float(dev0["L_rec"])
        step1200_raw_valid = bool(dev1["raw_cmos_validity"]["finite"] and dev1["raw_cmos_validity"]["all_nonzero"] and dev1["raw_cmos_validity"]["all_nonconstant"] and raw_s1_validity["finite"] and raw_s1_validity["all_nonzero"] and raw_s1_validity["all_nonconstant"])
        optical_valid = bool(dev1["optical_power_finite"] and s1_generation["optical_power_finite"])
        reconstruction_safe = math.isfinite(dev_reconstruction_delta) and dev_reconstruction_delta <= RECONSTRUCTION_TOLERANCE
        generation_condition = bool(float(s1_ed_metrics["macro_recall"]) >= GENERATION_MACRO_THRESHOLD and float(s1_ed_metrics["minimum_class_recall"]) >= GENERATION_CLASS_THRESHOLD)
        dev_ci_pure = dev_vs_pure["ci95"]
        s1_ci = s1_bootstrap["ci95"]
        dev_supported = dev_delta_pure >= 0.0 or float(dev_ci_pure[0]) > 0.0
        transfer_supported = bool(s1_delta_pure >= PRACTICAL_TRANSFER_DELTA and float(s1_ci["lower"]) > 0.0 and dev_supported and reconstruction_safe and optical_valid)
        classif = _classify_result(
            generation=generation_condition,
            safe=reconstruction_safe and step1200_raw_valid,
            optical=optical_valid,
            s1_delta=s1_delta_pure,
            s1_ci_lower=float(s1_ci["lower"]),
            dev_delta=dev_delta_pure,
            dev_ci_lower=float(dev_ci_pure[0]),
            local=local_response_supported,
        )
        exact_checks = {
            "exact_600_updates": len(history) == UPDATE_COUNT,
            "adam_internal_step_1200": _optimizer_steps(endpoint_checkpoint) == [END_STEP, END_STEP],
            "step600_to_augmented_train_order_parity": all(record["train_indices_sha256"] == authorities["pure"]["step1200"]["training_history"][START_STEP + index]["train_indices_sha256"] for index, record in enumerate(history)),
            "canonical_gradient_reaches_encoder_and_generator": all(record["gradient_vectors"]["canonical_encoder_norm"] > 0.0 and record["gradient_vectors"]["canonical_generator_norm"] > 0.0 for record in history),
            "E_D_not_in_training_backward": True,

def _attempt_file_inventory(attempt: Path) -> list[dict[str, str]]:

    return [

        for item in sorted(attempt.rglob("*"))
        if item.is_file()
    ]

def _read_ready_preflight(path: Path, source: Mapping[str, Any]) -> tuple[dict[str, Any], PreflightReceipt]:
    preflight = _read_json(path)
    _require(preflight.get("schema") == PREFLIGHT_SCHEMA and preflight.get("status") == PREFLIGHT_READY, "existing preflight is not READY")
    _require(preflight.get("authorization_id") == AUTHORIZATION_ID and preflight.get("outer_gate") == OUTER_GATE, "existing preflight authorization or Outer Gate changed")
    _require(preflight.get("source_manifest_id") == source.get("source_manifest_id") and preflight.get("source_semantics_id") == source.get("execution_source_semantics_id"), "existing preflight is not bound to the current source")
    _require(preflight.get("source_manifest") == dict(source) and preflight.get("attempt_created") is False, "existing preflight source manifest or attempt boundary changed")
    receipt = PreflightReceipt.from_mapping(preflight["receipt"])
    _require(receipt.status == PREFLIGHT_READY and receipt.authorization_id == AUTHORIZATION_ID and receipt.source_semantics_id == source["execution_source_semantics_id"], "READY receipt does not match the current task/source")
    baseline = preflight.get("step600_dev_baseline_qualification", {})
    _require(
        baseline.get("status") == "PASS"
        and baseline.get("entry") == "production _measure_full_dev(step=600)"
        and baseline.get("checkpoint_sha256") == START_CHECKPOINT_SHA256
        and baseline.get("fixed_DEV_rows") == DEV_ROWS
        and baseline.get("E_D_forward_rows") == DEV_ROWS
        and baseline.get("E_D_gradient_count") == 0
        and baseline.get("DEV_rows_recorded_in_qualification_ledger") == DEV_ROWS
        and baseline.get("scientific_optimizer_created") is False
        and baseline.get("scientific_parameter_updates") == 0
        and baseline.get("runtime_identity", {}).get("frozen_runtime_digest_source") == "reconstructed_from_source_bound_runtime_contract"
        and baseline.get("runtime_identity", {}).get("serialized_frozen_runtime_digest_present") is False,
        "READY preflight does not prove the accepted legacy step600 DEV baseline production entry",
    )
    kappa_reference = preflight.get("kappa_calibration_receipt", {})
    kappa_path = path.parent / PREFLIGHT_09_KAPPA_RECEIPT_PATH.name
    _require(
        kappa_reference.get("path") == PREFLIGHT_09_KAPPA_RECEIPT_PATH.as_posix()
        and kappa_path.is_file()
        and file_sha256(kappa_path) == kappa_reference.get("sha256"),
        "READY preflight kappa receipt is missing or changed",
    )
    kappa_receipt = _read_json(kappa_path)
    _require(
        kappa_receipt.get("schema") == "m0_bogt_canonical_gradient_augmentation_kappa_receipt.v1"
        and kappa_receipt.get("authorization_id") == AUTHORIZATION_ID
        and kappa_receipt.get("source_manifest_id") == source.get("source_manifest_id")
        and kappa_receipt.get("source_semantics_id") == source.get("execution_source_semantics_id")
        and kappa_receipt.get("batch_count") == 20
        and len(kappa_receipt.get("fixed_batches", [])) == 20,
        "READY preflight kappa receipt identity or fixed-batch count changed",
    )
    _require(math.isclose(float(kappa_receipt.get("final_median_kappa", float("nan"))), SCIENTIFIC_KAPPA, rel_tol=KAPPA_REPRODUCTION_RTOL, abs_tol=0.0), "READY preflight kappa differs from the frozen scientific value")
    ledger_reference = preflight.get("gradient_buffer_ledger_path")
    ledger_path = path.parent / PREFLIGHT_09_GRADIENT_BUFFER_LEDGER_PATH.name
    _require(
        ledger_reference == PREFLIGHT_09_GRADIENT_BUFFER_LEDGER_PATH.as_posix()
        and ledger_path.is_file()
        and file_sha256(ledger_path) == preflight.get("gradient_buffer_ledger_sha256"),
        "READY preflight gradient-buffer ledger is missing or changed",
    )
    return preflight, receipt


def _build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(description=__doc__)
    action = parser.add_mutually_exclusive_group(required=True)
    action.add_argument("--preflight", action="store_true", help="run the write-once source-bound preflight_09")
    action.add_argument("--run", action="store_true", help="open clean attempt_02 only after READY")
    action.add_argument("--resume", action="store_true", help="resume the same interrupted attempt_02")
    return parser


def main(argv: Sequence[str] | None = None) -> int:
    parser = _build_parser()
    args = parser.parse_args(argv)
    root = _root()
    task_started = time.perf_counter()
    if args.preflight:
        previous_seconds = sum(float(_read_json(root / path)["elapsed_seconds"]) for path in (PREFLIGHT_07_PATH, PREFLIGHT_08_PATH))
        _require(previous_seconds < MAX_PREFLIGHT_SECONDS, "cumulative preflight budget exhausted", ResourceBudgetExceeded)
        task_started -= previous_seconds
    _validate_output_root(root, resume=args.resume)
    authorities = _validate_authorities(root)
    source = _source_manifest(root, authorities)
    identity = _identity(source, authorities)
    preflight_relative = _select_preflight_path(root)
    preflight_path = root / preflight_relative
    attempt_path = root / SCIENTIFIC_ATTEMPT_DIR
    run_path = root / RUNS_DIR

    _require(preflight_relative == PREFLIGHT_09_PATH, "current authorization does not select source-bound preflight_09")
    _require((root / ATTEMPT_DIR).is_dir(), "historical attempt_01 must remain immutable")
    if args.preflight or args.run:
        _require(not attempt_path.exists(), "attempt_02 already exists; use --resume only for an interrupted run")
    dll_path = pure_bogt._prepare_cuda_dll_path()
    if torch.cuda.is_available():
        torch.cuda.set_device(0)

    if args.run or args.resume:
        _require(preflight_path.is_file(), "attempt_02 requires the new source-bound READY preflight_09")
        preflight, receipt = _read_ready_preflight(preflight_path, source)
        context = readout._load_cpu_context(root, authorities["data_identity"])
        _require(context.data_identity == authorities["data_identity"], "scientific TRAIN/DEV context identity differs from accepted step600")
        report = _run_attempt(root, authorities, context, source, identity, preflight, receipt, resume=args.resume)
        print(json.dumps({"state": RUN_COMPLETED, "classification": report["classification"], "attempt": SCIENTIFIC_ATTEMPT_DIR.as_posix(), "scientific_updates": report["training"]["scientific_updates"]}, ensure_ascii=False))
        return 0

    if preflight_path.exists():
        existing = _read_json(preflight_path)
        if existing.get("status") == PREFLIGHT_READY:
            _read_ready_preflight(preflight_path, source)
        print(json.dumps({"status": existing.get("status"), "preflight": preflight_relative.as_posix(), "already_written": True, "attempt_created": False}, ensure_ascii=False))
        return 0 if existing.get("status") == PREFLIGHT_READY else 2

    context = readout._load_cpu_context(root, authorities["data_identity"])
    _require(context.data_identity == authorities["data_identity"], "preflight TRAIN/DEV context identity differs from accepted step600")
    try:
        preflight, receipt = _preflight_requalification(root, authorities, context, source, identity, task_started=task_started)
        preflight["cuda_runtime_dll_path"] = dll_path
        preflight["source_manifest"] = dict(source)
        preflight["source_manifest_id"] = identity.source_manifest_id
    except Exception as error:
        resource_block = _is_resource_budget_error(error)
        phase_records = getattr(error, "resource_phase_records", [])
        classification = "PREFLIGHT_BLOCKED(resource)" if resource_block else RUN_INVALID_HARNESS
        failure = {
            "schema": PREFLIGHT_SCHEMA,
            "task_id": TASK_ID,
            "outer_gate": OUTER_GATE,
            "authorization_id": AUTHORIZATION_ID,
            "status": PREFLIGHT_BLOCKED if resource_block else RUN_INVALID_HARNESS,
            "classification": classification,
            "source_manifest": dict(source),
            "source_semantics_id": source["execution_source_semantics_id"],
            "source_manifest_id": identity.source_manifest_id,
            "attempt_created": False,
            "failure": {
                "phase": getattr(error, "preflight_phase", "source_bound_preflight"),
                "type": type(error).__name__,
                "message": str(error),
                "resource_phase_records": phase_records,
                "gradient_buffer_ledger_path": PREFLIGHT_09_GRADIENT_BUFFER_LEDGER_PATH.as_posix(),
                "gradient_buffer_ledger_sha256": file_sha256(root / PREFLIGHT_09_GRADIENT_BUFFER_LEDGER_PATH) if (root / PREFLIGHT_09_GRADIENT_BUFFER_LEDGER_PATH).is_file() else None,
                "gradient_buffer_entries": getattr(error, "gradient_buffer_entries", []),
                "first_nonclean_gradient_transition": getattr(error, "first_nonclean_gradient_transition", None),
                "first_unexpected_nonclean_gradient_transition": getattr(error, "first_unexpected_nonclean_gradient_transition", None),
                "disposable_updates_completed": getattr(error, "disposable_updates_completed", 0),
                "kappa_receipt_path": PREFLIGHT_09_KAPPA_RECEIPT_PATH.as_posix() if (root / PREFLIGHT_09_KAPPA_RECEIPT_PATH).is_file() else None,
                "kappa_receipt_sha256": file_sha256(root / PREFLIGHT_09_KAPPA_RECEIPT_PATH) if (root / PREFLIGHT_09_KAPPA_RECEIPT_PATH).is_file() else None,
                "resource_stop_evidence": PREFLIGHT_09_RESOURCE_STOP_PATH.as_posix() if resource_block else None,
                "traceback": traceback.format_exc(),
            },
            "elapsed_seconds": time.perf_counter() - task_started,
        }
        if resource_block:
            blocked_receipt = PreflightReceipt(
                source_semantics_id=str(source["execution_source_semantics_id"]),
                authorization_id=AUTHORIZATION_ID,
                checks={
                    CORRECTNESS_INVARIANTS: {"status": "BLOCKED", "reason": str(error)},
                    RESOURCE_ENVELOPE: {"status": "BLOCKED", "resource_phase_records": phase_records},
                    IMPLEMENTATION_SELECTION: {"status": "BLOCKED"},
                },
                selected_engine=None,
                resource={"global_device_limit": MAX_DEVICE_MEMORY_FRACTION, "phase_records": phase_records},
                failure_signature=f"resource:{getattr(error, 'preflight_phase', 'source_bound_preflight')}",
                status=PREFLIGHT_BLOCKED,
            )
            failure["receipt"] = blocked_receipt.to_dict()
        run_path.mkdir(parents=True, exist_ok=True)
        atomic_write_new_json(preflight_path, failure, artifact_kind="M0 canonical-gradient source-bound preflight result")
        print(json.dumps({"status": failure["status"], "classification": classification, "failure": failure["failure"], "preflight": preflight_relative.as_posix(), "attempt_created": False}, ensure_ascii=False))
        return 2

    run_path.mkdir(parents=True, exist_ok=True)
    preflight["preflight_record_id"] = preflight_relative.stem
    atomic_write_new_json(preflight_path, preflight, artifact_kind="M0 canonical-gradient source-bound READY preflight")
    print(json.dumps({"status": PREFLIGHT_READY, "source_manifest_id": identity.source_manifest_id, "preflight": preflight_relative.as_posix(), "projected_execution": preflight["projected_execution"], "resource_peak_global_fraction": preflight["resource_peak_global_fraction"], "attempt_created": False}, ensure_ascii=False))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
