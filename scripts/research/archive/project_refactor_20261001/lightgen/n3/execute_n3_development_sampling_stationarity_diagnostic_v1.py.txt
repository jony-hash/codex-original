class _InvalidHarnessError(RuntimeError):
    """A declared data, numerical, or artifact boundary failed."""


def _require(condition: bool, message: str) -> None:
    if not condition:
        raise ValueError(message)


def _invalid(condition: bool, message: str) -> None:
    if not condition:
        raise _InvalidHarnessError(message)


def _mapping(value: Any, name: str) -> Mapping[str, Any]:
    _require(isinstance(value, Mapping), f"{name} must be a mapping")
    return value


def _sequence(value: Any, name: str) -> list[Any]:
    _require(isinstance(value, list), f"{name} must be a list")
    return value


def _relative(repo_root: Path, path: Path) -> str:
    return path.resolve().relative_to(repo_root.resolve()).as_posix()


def _read_json_exact(
    repo_root: Path,
    path: Path,
    expected_sha256: str,
    name: str,
) -> dict[str, Any]:
    target = repo_root / path
    _require(target.is_file(), f"{name} is missing: {target}")
    _require(file_sha256(target) == expected_sha256, f"{name} hash changed: {target}")
    value = json.loads(target.read_text(encoding="utf-8"))
    return dict(_mapping(value, name))


def _git_state(repo_root: Path) -> dict[str, Any]:
    branch = subprocess.run(
        ["git", "-C", str(repo_root), "branch", "--show-current"],
        check=True,
        capture_output=True,
        text=True,
    ).stdout.strip()
    head = subprocess.run(
        ["git", "-C", str(repo_root), "rev-parse", "HEAD"],
        check=True,
        capture_output=True,
        text=True,
    ).stdout.strip()
    status = subprocess.run(
        ["git", "-C", str(repo_root), "status", "--porcelain"],
        check=True,
        capture_output=True,
        text=True,
    ).stdout.splitlines()
    return {
        "branch": branch,
        "head": head,
        "dirty_paths": [line[3:] if len(line) >= 4 else line for line in status],
        "worktree_clean": not status,
    }


def _verify_file_hash(repo_root: Path, path: str, expected_sha256: str, name: str) -> None:
    target = repo_root / Path(path)
    _require(target.is_file(), f"{name} is missing: {target}")
    _require(file_sha256(target) == expected_sha256, f"{name} hash changed: {target}")

    development = definition._validate_development_authority(repo_root)
    empirical = definition._validate_empirical_ceiling(repo_root)

    _require(fold_authority.get("identity") == empirical["fold_identity"], "contract fold identity changed")

def _seed(role: str, replicate: int) -> int:
    material = f"{RNG_NAMESPACE}|{role}|{replicate}".encode("utf-8")
    return int.from_bytes(hashlib.sha256(material).digest()[:8], "big") & ((1 << 63) - 1)


def _validate_tensor(value: torch.Tensor, name: str) -> None:
    _invalid(torch.is_tensor(value) and bool(torch.isfinite(value).all()), f"{name} is non-finite")


def _moment_metrics(fit_q: torch.Tensor, heldout_q: torch.Tensor) -> tuple[dict[str, float], dict[str, float]]:
    _invalid(fit_q.ndim == 2 and heldout_q.ndim == 2 and fit_q.shape[1] == DIMENSION and heldout_q.shape[1] == DIMENSION, "moment q shape is invalid")
    _invalid(fit_q.shape[0] >= 2 and heldout_q.shape[0] >= 2, "moment subset is too small")
    _validate_tensor(fit_q, "fit q")
    _validate_tensor(heldout_q, "heldout q")
    fit_radial = fit_q.norm(dim=1)
    heldout_radial = heldout_q.norm(dim=1)
    fit_std = fit_q.std(dim=0, unbiased=True)
    heldout_std = heldout_q.std(dim=0, unbiased=True)
    radial_mean_difference = (fit_radial.mean() - heldout_radial.mean()).abs()
    radial_std_difference = (fit_radial.std(unbiased=True) - heldout_radial.std(unbiased=True)).abs()
    metrics = {
        "coordinate_mean_l2": float((fit_q.mean(dim=0) - heldout_q.mean(dim=0)).norm()),
        "coordinate_std_l2": float((fit_std - heldout_std).norm()),
        "radial_moment_l2": float(torch.sqrt(radial_mean_difference.square() + radial_std_difference.square())),
    }
    radial_mean_difference = float(radial_mean_difference)
    radial_std_difference = float(radial_std_difference)
    details = {
        "fit_radial_mean": float(fit_radial.mean()),
        "heldout_radial_mean": float(heldout_radial.mean()),
        "radial_mean_abs_difference": radial_mean_difference,
        "fit_radial_std": float(fit_radial.std(unbiased=True)),
        "heldout_radial_std": float(heldout_radial.std(unbiased=True)),
        "radial_std_abs_difference": radial_std_difference,
    }
    return metrics, details


def _batched_moment_metrics(fit_q: torch.Tensor, heldout_q: torch.Tensor) -> dict[str, torch.Tensor]:
    _invalid(fit_q.ndim >= 4 and heldout_q.ndim == fit_q.ndim, "batched moment q shape is invalid")
    _invalid(fit_q.shape[:-2] == heldout_q.shape[:-2] and fit_q.shape[-1] == DIMENSION and heldout_q.shape[-1] == DIMENSION, "batched moment dimensions differ")
    _invalid(fit_q.shape[-2] >= 2 and heldout_q.shape[-2] >= 2, "batched moment subset is too small")
    _validate_tensor(fit_q, "batched fit q")
    _validate_tensor(heldout_q, "batched heldout q")
    fit_mean = fit_q.mean(dim=-2)
    heldout_mean = heldout_q.mean(dim=-2)
    fit_std = fit_q.std(dim=-2, unbiased=True)
    heldout_std = heldout_q.std(dim=-2, unbiased=True)
    fit_radial = fit_q.norm(dim=-1)
    heldout_radial = heldout_q.norm(dim=-1)
    radial_mean_difference = (fit_radial.mean(dim=-1) - heldout_radial.mean(dim=-1)).abs()
    radial_std_difference = (fit_radial.std(dim=-1, unbiased=True) - heldout_radial.std(dim=-1, unbiased=True)).abs()
    result = {
        "coordinate_mean_l2": (fit_mean - heldout_mean).norm(dim=-1),
        "coordinate_std_l2": (fit_std - heldout_std).norm(dim=-1),
        "radial_moment_l2": torch.sqrt(radial_mean_difference.square() + radial_std_difference.square()),
    }
    for name, value in result.items():
        _invalid(bool(torch.isfinite(value).all()) and bool(value.ge(0.0).all()), f"batched {name} is invalid")
    return result


def _rank_values(value: torch.Tensor) -> torch.Tensor:
    values = value.to(dtype=torch.float64).reshape(-1).tolist()
    ordered = sorted(range(len(values)), key=lambda index: (values[index], index))
    ranks = torch.empty(len(values), dtype=torch.float64)
    start = 0
    while start < len(ordered):
        end = start + 1
        while end < len(ordered) and values[ordered[end]] == values[ordered[start]]:
            end += 1
        rank = (float(start + 1) + float(end)) / 2.0
        for position in range(start, end):
            ranks[ordered[position]] = rank
        start = end
    return ranks


def _association(first: torch.Tensor, second: torch.Tensor) -> dict[str, float | None]:
    left = first.to(dtype=torch.float64).reshape(-1)
    right = second.to(dtype=torch.float64).reshape(-1)
    _invalid(left.numel() >= 2 and right.shape == left.shape, "position association inputs are invalid")
    left_centered = left - left.mean()
    right_centered = right - right.mean()
    denominator = left_centered.norm() * right_centered.norm()
    if float(denominator) == 0.0:
        return {"pearson": None, "spearman": None}
    pearson = float(left_centered.dot(right_centered) / denominator)
    rank_left = _rank_values(left)
    rank_right = _rank_values(right)
    rank_left_centered = rank_left - rank_left.mean()
    rank_right_centered = rank_right - rank_right.mean()
    rank_denominator = rank_left_centered.norm() * rank_right_centered.norm()
    spearman = None if float(rank_denominator) == 0.0 else float(rank_left_centered.dot(rank_right_centered) / rank_denominator)
    return {"pearson": pearson, "spearman": spearman}


def _source_subset_summary(
    q: torch.Tensor,
    source_indices: torch.Tensor,
    labels: torch.Tensor,
    positions: torch.Tensor,
    *,
    expected_rows_per_class: int,
) -> dict[str, Any]:
    records: list[dict[str, Any]] = []
    pooled_positions: list[torch.Tensor] = []
    pooled_residuals: list[torch.Tensor] = []
    for label in range(CLASS_COUNT):
        class_positions = positions[labels.index_select(0, positions) == label]
        ordered = class_positions[torch.argsort(source_indices.index_select(0, class_positions), stable=True)]
        _invalid(int(ordered.numel()) == expected_rows_per_class, f"source-position subset class {label} count changed")
        energy = q.index_select(0, ordered).square().sum(dim=1).to(dtype=torch.float64)
        normalized_position = torch.arange(expected_rows_per_class, dtype=torch.float64) / float(expected_rows_per_class - 1)
        association = _association(normalized_position, energy)
        half = expected_rows_per_class // 2
        first_mean = energy[:half].mean()
        last_mean = energy[half:].mean()
        records.append(
            {
                "label": label,
                "source_index_first": int(source_indices[ordered[0]]),
                "source_index_last": int(source_indices[ordered[-1]]),
                "q_energy_first_half_mean": float(first_mean),
                "q_energy_last_half_mean": float(last_mean),
                "q_energy_last_minus_first_half_mean": float(last_mean - first_mean),
                "q_energy_position_association": association,
            }
        )
        pooled_positions.append(normalized_position)
        pooled_residuals.append(energy - energy.mean())
    pooled = _association(torch.cat(pooled_positions), torch.cat(pooled_residuals))
    return {
        "count": int(positions.numel()),
        "rows_per_class": expected_rows_per_class,
        "ordering": "within each class, source_index ascending; subset position=0..n-1 divided by n-1",
        "per_class": records,
        "pooled_class_stratified_q_energy_association": pooled,
        "role": "descriptive only; not an aggregate classification input",
    }


def _build_reference_partitions(labels: torch.Tensor) -> tuple[torch.Tensor, torch.Tensor]:
    class_positions = torch.stack(
        [
            (labels == label).nonzero(as_tuple=False).reshape(-1)
            for label in range(CLASS_COUNT)
        ]
    )
    _invalid(tuple(class_positions.shape) == (CLASS_COUNT, ROWS_PER_CLASS), "reference class position shape changed")
    heldout_partitions: list[torch.Tensor] = []
    fit_partitions: list[torch.Tensor] = []
    for replicate in range(PARTITION_REFERENCE_REPLICATES):
        generator = torch.Generator(device="cpu").manual_seed(_seed("partition-reference", replicate))
        permutations = torch.stack(
            [torch.randperm(ROWS_PER_CLASS, generator=generator, device="cpu") for _ in range(CLASS_COUNT)]
        )
        permuted_positions = class_positions.gather(1, permutations)
        heldout_indices = torch.stack(
            [
                permuted_positions[:, fold * HELDOUT_ROWS_PER_CLASS_PER_FOLD : (fold + 1) * HELDOUT_ROWS_PER_CLASS_PER_FOLD].reshape(-1)
                for fold in range(FOLD_COUNT)
            ]
        )
        fit_indices = torch.stack(
            [
                torch.cat(
                    [
                        permuted_positions[:, : fold * HELDOUT_ROWS_PER_CLASS_PER_FOLD],
                        permuted_positions[:, (fold + 1) * HELDOUT_ROWS_PER_CLASS_PER_FOLD :],
                    ],
                    dim=1,
                ).reshape(-1)
                for fold in range(FOLD_COUNT)
            ]
        )
        _invalid(tuple(heldout_indices.shape) == (FOLD_COUNT, HELDOUT_COUNT_PER_FOLD), "reference held-out fold shape changed")
        _invalid(tuple(fit_indices.shape) == (FOLD_COUNT, FIT_COUNT_PER_FOLD), "reference fit fold shape changed")
        for fold in range(FOLD_COUNT):
            _invalid(torch.unique(heldout_indices[fold]).numel() == HELDOUT_COUNT_PER_FOLD, "reference fold contains duplicate held-out rows")
            _invalid(torch.unique(fit_indices[fold]).numel() == FIT_COUNT_PER_FOLD, "reference fold contains duplicate fit rows")
            _invalid(torch.unique(torch.cat([heldout_indices[fold], fit_indices[fold]])).numel() == POOL_COUNT, "reference fit/held-out rows overlap")
            _invalid(
                torch.equal(
                    torch.bincount(labels.index_select(0, heldout_indices[fold]), minlength=CLASS_COUNT),
                    torch.full((CLASS_COUNT,), HELDOUT_ROWS_PER_CLASS_PER_FOLD, dtype=torch.int64),
                ),
                "reference held-out class balance changed",
            )
            _invalid(
                torch.equal(
                    torch.bincount(labels.index_select(0, fit_indices[fold]), minlength=CLASS_COUNT),
                    torch.full((CLASS_COUNT,), FIT_ROWS_PER_CLASS_PER_FOLD, dtype=torch.int64),
                ),
                "reference fit class balance changed",
            )
        heldout_partitions.append(heldout_indices)
        fit_partitions.append(fit_indices)
    heldout_result = torch.stack(heldout_partitions).contiguous()
    fit_result = torch.stack(fit_partitions).contiguous()
    _invalid(tuple(heldout_result.shape) == (PARTITION_REFERENCE_REPLICATES, FOLD_COUNT, HELDOUT_COUNT_PER_FOLD), "reference held-out tensor shape changed")
    _invalid(tuple(fit_result.shape) == (PARTITION_REFERENCE_REPLICATES, FOLD_COUNT, FIT_COUNT_PER_FOLD), "reference fit tensor shape changed")
    return heldout_result, fit_result


def _reference_metric_distributions(
    q: torch.Tensor,
    labels: torch.Tensor,
) -> tuple[dict[str, torch.Tensor], dict[int, dict[str, torch.Tensor]], torch.Tensor, torch.Tensor]:
    heldout_partitions, fit_partitions = _build_reference_partitions(labels)
    global_values = {name: [] for name in MOMENT_METRICS}
    class_values = {label: {name: [] for name in MOMENT_METRICS} for label in range(CLASS_COUNT)}
    for start in range(0, PARTITION_REFERENCE_REPLICATES, PARTITION_REFERENCE_BATCH_SIZE):
        stop = min(start + PARTITION_REFERENCE_BATCH_SIZE, PARTITION_REFERENCE_REPLICATES)
        batch_heldout_indices = heldout_partitions[start:stop]
        batch_fit_indices = fit_partitions[start:stop]
        batch_size = stop - start
        heldout_sampled = q.index_select(0, batch_heldout_indices.reshape(-1)).reshape(
            batch_size,
            FOLD_COUNT,
            CLASS_COUNT,
            HELDOUT_ROWS_PER_CLASS_PER_FOLD,
            DIMENSION,
        )
        fit_sampled = q.index_select(0, batch_fit_indices.reshape(-1)).reshape(
            batch_size,
            FOLD_COUNT,
            CLASS_COUNT,
            FIT_ROWS_PER_CLASS_PER_FOLD,
            DIMENSION,
        )
        global_batch = _batched_moment_metrics(
            fit_sampled.reshape(batch_size, FOLD_COUNT, FIT_COUNT_PER_FOLD, DIMENSION),
            heldout_sampled.reshape(batch_size, FOLD_COUNT, HELDOUT_COUNT_PER_FOLD, DIMENSION),
        )
        for name in MOMENT_METRICS:
            global_values[name].append(global_batch[name].reshape(-1).detach().cpu())
        class_batch = _batched_moment_metrics(
            fit_sampled,
            heldout_sampled,
        )
        for label in range(CLASS_COUNT):
            for name in MOMENT_METRICS:
                class_values[label][name].append(class_batch[name][:, :, label].reshape(-1).detach().cpu())
    global_result = {name: torch.cat(values).contiguous() for name, values in global_values.items()}
    class_result = {
        label: {name: torch.cat(values).contiguous() for name, values in metrics.items()}
        for label, metrics in class_values.items()
    }
    for name, values in global_result.items():
        _invalid(values.numel() == PARTITION_REFERENCE_REPLICATES * FOLD_COUNT, f"global reference count changed: {name}")
        _invalid(bool(torch.isfinite(values).all()) and bool(values.ge(0.0).all()), f"global reference invalid: {name}")
    for label in range(CLASS_COUNT):
        for name, values in class_result[label].items():
            _invalid(values.numel() == PARTITION_REFERENCE_REPLICATES * FOLD_COUNT, f"class reference count changed: {label}/{name}")
            _invalid(bool(torch.isfinite(values).all()) and bool(values.ge(0.0).all()), f"class reference invalid: {label}/{name}")
    return global_result, class_result, heldout_partitions, fit_partitions


def _reference_summary(values: torch.Tensor, quantile: float) -> dict[str, Any]:
    sorted_values = torch.sort(values).values
    threshold = torch.quantile(values, quantile)
    return {
        "count": int(values.numel()),
        "min": float(values.min()),
        "mean": float(values.mean()),
        "p50": float(torch.quantile(values, 0.50)),
        "quantile": quantile,
        "threshold": float(threshold),
        "max": float(values.max()),
        "values_sha256": tensor_sha256(values),
        "sorted_values_sha256": tensor_sha256(sorted_values),
    }


def _percentile(values: torch.Tensor, observed: float) -> float:
    _invalid(values.ndim == 1 and values.numel() > 0, "reference percentile values are invalid")
    return float((values <= float(observed)).sum().item() / values.numel())


def _validate_metric_map(metrics: Mapping[str, Any], name: str) -> None:
    for metric in MOMENT_METRICS:
        value = metrics.get(metric)
        _invalid(isinstance(value, (float, int)), f"{name} metric is not scalar: {metric}")
        _invalid(bool(torch.isfinite(torch.tensor(float(value), dtype=torch.float64))), f"{name} metric is non-finite: {metric}")
        _invalid(float(value) >= 0.0, f"{name} metric is negative: {metric}")


def _fold_class_pattern(exceptional_labels_by_fold: list[list[int]]) -> str:
    flattened = [label for labels in exceptional_labels_by_fold for label in labels]
    if not flattened:
        return "NONE"
    unique_labels = set(flattened)
    if any(len(labels) >= 5 for labels in exceptional_labels_by_fold):
        return "BROAD"
    if len(unique_labels) <= 2:
        return "CONCENTRATED"
    return "MIXED"


def _evaluate_fold(
    q: torch.Tensor,
    source_indices: torch.Tensor,
    labels: torch.Tensor,
    fold_ids: torch.Tensor,
    fold: int,
    global_reference: Mapping[str, torch.Tensor],
    class_reference: Mapping[int, Mapping[str, torch.Tensor]],
    contract: Mapping[str, Any],
) -> dict[str, Any]:
    heldout_positions = (fold_ids == fold).nonzero(as_tuple=False).reshape(-1)
    fit_positions = (fold_ids != fold).nonzero(as_tuple=False).reshape(-1)
    _invalid(int(fit_positions.numel()) == FIT_COUNT_PER_FOLD, f"fit count changed at fold {fold}")
    _invalid(int(heldout_positions.numel()) == HELDOUT_COUNT_PER_FOLD, f"held-out count changed at fold {fold}")
    fit_labels = labels.index_select(0, fit_positions)
    heldout_labels = labels.index_select(0, heldout_positions)
    _invalid(torch.equal(torch.bincount(fit_labels, minlength=CLASS_COUNT), torch.full((CLASS_COUNT,), FIT_ROWS_PER_CLASS_PER_FOLD, dtype=torch.int64)), f"fit class balance changed at fold {fold}")
    _invalid(torch.equal(torch.bincount(heldout_labels, minlength=CLASS_COUNT), torch.full((CLASS_COUNT,), HELDOUT_ROWS_PER_CLASS_PER_FOLD, dtype=torch.int64)), f"held-out class balance changed at fold {fold}")
    fit_q = q.index_select(0, fit_positions).contiguous()
    heldout_q = q.index_select(0, heldout_positions).contiguous()
    moment_metrics, moment_details = _moment_metrics(fit_q, heldout_q)
    _validate_metric_map(moment_metrics, f"fold {fold} official moment")
    global_thresholds = {name: _reference_summary(global_reference[name], GLOBAL_REFERENCE_QUANTILE) for name in MOMENT_METRICS}
    global_percentiles = {name: _percentile(global_reference[name], moment_metrics[name]) for name in MOMENT_METRICS}
    global_checks = {
        name: moment_metrics[name] <= global_thresholds[name]["threshold"]
        for name in MOMENT_METRICS
    }
    class_results: list[dict[str, Any]] = []
    exceptional_labels: list[int] = []
    for label in range(CLASS_COUNT):
        class_fit_positions = fit_positions[labels.index_select(0, fit_positions) == label]
        class_heldout_positions = heldout_positions[labels.index_select(0, heldout_positions) == label]
        class_metrics, class_details = _moment_metrics(
            q.index_select(0, class_fit_positions).contiguous(),
            q.index_select(0, class_heldout_positions).contiguous(),
        )
        _validate_metric_map(class_metrics, f"fold {fold} class {label} moment")
        thresholds = {name: _reference_summary(class_reference[label][name], CLASS_REFERENCE_QUANTILE) for name in MOMENT_METRICS}
        percentiles = {name: _percentile(class_reference[label][name], class_metrics[name]) for name in MOMENT_METRICS}
        checks = {name: class_metrics[name] <= thresholds[name]["threshold"] for name in MOMENT_METRICS}
        exceptional = not all(checks.values())
        if exceptional:
            exceptional_labels.append(label)
        class_results.append(
            {
                "label": label,
                "fit_count": int(class_fit_positions.numel()),
                "heldout_count": int(class_heldout_positions.numel()),
                "moment_discrepancies": class_metrics,
                "moment_details": class_details,
                "reference_thresholds": thresholds,
                "reference_percentiles": percentiles,
                "reference_compatible": all(checks.values()),
                "reference_metric_checks": checks,
                "exceptional": exceptional,
            }
        )
    source_position_authority = _mapping(_mapping(contract.get("authorities"), "contract authorities").get("source_position"), "source-position authority")
    position = {
        "registered_authority": {
            "path": source_position_authority["path"],
            "sha256": source_position_authority["sha256"],
            "global_statistic": source_position_authority["global_statistic"],
            "class_conditioned_statistic": source_position_authority["class_conditioned_statistic"],
            "drift_supported": bool(
                source_position_authority["global_statistic"]["drift_supported"]
                or source_position_authority["class_conditioned_statistic"]["drift_supported"]
            ),
        },
        "fit_subset_raw_q_energy": _source_subset_summary(
            q, source_indices, labels, fit_positions, expected_rows_per_class=FIT_ROWS_PER_CLASS_PER_FOLD
        ),
        "heldout_subset_raw_q_energy": _source_subset_summary(
            q, source_indices, labels, heldout_positions, expected_rows_per_class=HELDOUT_ROWS_PER_CLASS_PER_FOLD
        ),
        "role": "registered full-pool source-position statistic drives the source-position input; fold subset raw-q summaries are descriptive only",
    }
    empirical_authority = _mapping(_mapping(contract.get("authorities"), "contract authorities").get("empirical_ceiling"), "empirical ceiling authority")
    empirical_pattern = _sequence(empirical_authority.get("fold_pattern"), "empirical ceiling fold pattern")
    empirical_link = bool(empirical_pattern[fold])
    return {
        "fold": fold,
        "operationally_valid": True,
        "fit_heldout_disjoint": True,
        "fit_source_indices_sha256": tensor_sha256(source_indices.index_select(0, fit_positions)),
        "heldout_source_indices_sha256": tensor_sha256(source_indices.index_select(0, heldout_positions)),
        "class_composition": {
            "fit_counts": [int(value) for value in torch.bincount(fit_labels, minlength=CLASS_COUNT).tolist()],
            "heldout_counts": [int(value) for value in torch.bincount(heldout_labels, minlength=CLASS_COUNT).tolist()],
            "exactly_balanced": True,
            "composition_difference_explains_result": False,
        },
        "moment_discrepancies": moment_metrics,
        "moment_details": moment_details,
        "reference_thresholds": global_thresholds,
        "reference_percentiles": global_percentiles,
        "partition_reference_compatible": all(global_checks.values()),
        "partition_reference_metric_checks": global_checks,
        "statistically_exceptional": not all(global_checks.values()),
        "class_conditioned": {
            "per_class": class_results,
            "exceptional_labels": exceptional_labels,
            "exceptional_label_count": len(exceptional_labels),
            "pattern": _fold_class_pattern([exceptional_labels]),
        },
        "position_diagnostics": position,
        "empirical_ceiling_linkage": {
            "official_fold": fold,
            "existing_compatibility_pass": empirical_link,
            "recomputed_or_redefined": False,
        },
    }


def _class_conditioned_aggregate(fold_results: list[Mapping[str, Any]]) -> dict[str, Any]:
    exceptional_labels_by_fold = [
        [int(value) for value in _mapping(item["class_conditioned"], f"fold {index} class result").get("exceptional_labels", [])]
        for index, item in enumerate(fold_results)
    ]
    label_fold_counts = {
        str(label): sum(label in labels for labels in exceptional_labels_by_fold)
        for label in range(CLASS_COUNT)
    }
    max_per_fold = max((len(labels) for labels in exceptional_labels_by_fold), default=0)
    repeated_labels = [int(label) for label, count in label_fold_counts.items() if count >= REPEATED_EXCEPTIONAL_CLASS_FOLD_COUNT]
    drift_supported = (
        max_per_fold >= MATERIAL_EXCEPTIONAL_CLASS_COUNT_IN_FOLD
        or bool(repeated_labels)
    )
    return {
        "exceptional_labels_by_fold": exceptional_labels_by_fold,
        "exceptional_label_count_by_fold": [len(labels) for labels in exceptional_labels_by_fold],
        "label_exceptional_fold_counts": label_fold_counts,
        "max_exceptional_labels_in_one_fold": max_per_fold,
        "repeated_exceptional_labels": repeated_labels,
        "pattern": _fold_class_pattern(exceptional_labels_by_fold),
        "composition_difference": False,
        "drift_supported": drift_supported,
        "drift_rule": "one official fold has at least two exceptional labels or one label is exceptional in at least two official folds",
    }


def _classification(
    exceptional_fold_count: int,
    source_position_drift: bool,
    class_conditioned_drift: bool,
) -> str:
    if (
        source_position_drift
        or class_conditioned_drift
        or exceptional_fold_count >= MATERIAL_EXCEPTIONAL_FOLD_COUNT
    ):
        return CLASSIFICATION_LIMITATION
    if exceptional_fold_count == 0:
        return CLASSIFICATION_SUPPORTED
    return CLASSIFICATION_MIXED

def _write_report(repo_root: Path, output_dir: Path, report: Mapping[str, Any]) -> Path:
    if output_dir.exists():
        raise FileExistsError(f"refusing to reuse write-once report output: {output_dir}")
    output_dir.parent.mkdir(parents=True, exist_ok=True)
    temporary = Path(tempfile.mkdtemp(prefix=f".{output_dir.name}.", dir=output_dir.parent))
    try:
        atomic_write_new_json(
            temporary / REPORT_NAME,
            report,
            artifact_kind="N3 development sampling stationarity report",
        )
        if output_dir.exists():
            raise FileExistsError(f"write-once output appeared during diagnostic: {output_dir}")
        os.replace(temporary, output_dir)
    finally:
        if temporary.exists():
            shutil.rmtree(temporary)
    return output_dir / REPORT_NAME


def execute(repo_root: Path, output_dir: Path = REPORT_OUTPUT_DIR) -> dict[str, Any]:
    git_before = _git_state(repo_root)
    _require(git_before["branch"] == EXPECTED_BRANCH, "current branch is not the N3 authority branch")
    _require(git_before["head"] == EXPECTED_STARTING_HEAD, "current HEAD changed from frozen authority base")
    unexpected_dirty = [path for path in git_before["dirty_paths"] if path not in SAFE_DIRTY_PATHS]
    _require(not unexpected_dirty, f"unrelated local changes invalidate provenance: {unexpected_dirty}")
    _require(not output_dir.exists(), f"write-once diagnostic output already exists: {output_dir}")

    contract = _validate_contract(repo_root)
    try:
        q, source_indices, labels, fold_ids, pool_identity, fold_identity = _validate_development_identity(repo_root, contract)
        with torch.no_grad():
            global_reference, class_reference, heldout_partitions, fit_partitions = _reference_metric_distributions(q, labels)
            fold_results = [
                _evaluate_fold(
                    q,
                    source_indices,
                    labels,
                    fold_ids,
                    fold,
                    global_reference,
                    class_reference,
                    contract,
                )
                for fold in range(FOLD_COUNT)
            ]
    except _InvalidHarnessError as error:
        report_path = _write_report(repo_root, output_dir, _blocked_report(repo_root, git_before, contract, error))
        return {
            "result": "BLOCKED",
            "classification": None,
            "contract": _relative(repo_root, repo_root / CONTRACT_PATH),
            "contract_sha256": EXPECTED_CONTRACT_SHA256,
            "report": _relative(repo_root, report_path),
            "report_sha256": file_sha256(report_path),
            "blocker": "INVALID_HARNESS",
            "sealed_confirmation_consumed": False,
            "training_authorized": False,
            "formal_promotion": False,
            "generation_readiness": "not_ready",
        }

    report = _valid_report(
        repo_root,
        git_before,
        contract,
        q,
        source_indices,
        labels,
        pool_identity,
        fold_identity,
        fold_results,
        global_reference,
        class_reference,
        heldout_partitions,
        fit_partitions,
    )
    report_path = _write_report(repo_root, output_dir, report)
    return {
        "result": "VALID_COMPLETED",
        "classification": report["classification"],
        "contract": _relative(repo_root, repo_root / CONTRACT_PATH),
        "contract_sha256": EXPECTED_CONTRACT_SHA256,
        "report": _relative(repo_root, report_path),
        "report_sha256": file_sha256(report_path),
        "exceptional_folds": report["aggregate"]["partition_reference_exceptional_fold_count"],
        "fold_count": FOLD_COUNT,
        "git_branch": git_before["branch"],
        "starting_head": git_before["head"],
        "ending_head": report["git"]["ending_head"],
        "sealed_confirmation_consumed": False,
        "training_authorized": False,
        "formal_promotion": False,
        "generation_readiness": "not_ready",
    }


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--repo-root", type=Path, default=Path(__file__).resolve().parents[4])
    parser.add_argument("--output-dir", type=Path, default=Path(__file__).resolve().parents[4] / REPORT_OUTPUT_DIR)
    args = parser.parse_args()
    repo_root = args.repo_root.resolve()
    output_dir = args.output_dir if args.output_dir.is_absolute() else repo_root / args.output_dir
    print(json.dumps(execute(repo_root, output_dir), indent=2, sort_keys=True))


if __name__ == "__main__":
    main()
