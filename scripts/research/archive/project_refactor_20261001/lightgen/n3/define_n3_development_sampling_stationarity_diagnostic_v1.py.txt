def _require(condition: bool, message: str) -> None:
    if not condition:
        raise ValueError(message)


def _mapping(value: Any, name: str) -> Mapping[str, Any]:
    _require(isinstance(value, Mapping), f"{name} must be a mapping")
    return value


def _sequence(value: Any, name: str) -> list[Any]:
    _require(isinstance(value, list), f"{name} must be a list")
    return value


def _relative(repo_root: Path, path: Path) -> str:
    return path.resolve().relative_to(repo_root.resolve()).as_posix()


def _read_json_exact(
    repo_root: Path,
    path: Path,
    expected_sha256: str,
    name: str,
) -> dict[str, Any]:
    target = repo_root / path
    _require(target.is_file(), f"{name} is missing: {target}")
    _require(file_sha256(target) == expected_sha256, f"{name} hash changed: {target}")
    value = json.loads(target.read_text(encoding="utf-8"))
    return dict(_mapping(value, name))


def _git_state(repo_root: Path) -> dict[str, Any]:
    branch = subprocess.run(
        ["git", "-C", str(repo_root), "branch", "--show-current"],
        check=True,
        capture_output=True,
        text=True,
    ).stdout.strip()
    head = subprocess.run(
        ["git", "-C", str(repo_root), "rev-parse", "HEAD"],
        check=True,
        capture_output=True,
        text=True,
    ).stdout.strip()
    status = subprocess.run(
        ["git", "-C", str(repo_root), "status", "--porcelain"],
        check=True,
        capture_output=True,
        text=True,
    ).stdout.splitlines()
    return {
        "branch": branch,
        "head": head,
        "dirty_paths": [line[3:] if len(line) >= 4 else line for line in status],
        "worktree_clean": not status,
    }


def _validate_current_state(repo_root: Path) -> dict[str, Any]:
    paths = (Path("docs/CURRENT_STATUS.md"), Path("docs/status/current_handoff.md"))
    markers = (
        "N3_FIT_HELDOUT_EMPIRICAL_CEILING_NOT_SUPPORTED",
        "NO_MINIMAL_INTERVENTION_SUPPORTED",
        "training_authorized=false",
        "formal_promotion=false",
        "generation_readiness=not_ready",
    )
    for path in paths:
        target = repo_root / path
        _require(target.is_file(), f"current state file is missing: {target}")
        text = target.read_text(encoding="utf-8")
        for marker in markers:
            _require(marker in text, f"current state marker is missing from {path}: {marker}")
    return {
        "paths": [_relative(repo_root, repo_root / path) for path in paths],
        "required_markers": list(markers),
        "state_conflict": False,
    }

def _validate_empirical_ceiling(repo_root: Path) -> dict[str, Any]:
    report = _read_json_exact(
        repo_root,
        EMPIRICAL_CEILING_REPORT,
        EMPIRICAL_CEILING_REPORT_SHA256,
        "empirical-fit ceiling report",
    )
    _require(report.get("schema") == "n3_empirical_fit_joint_bootstrap_ceiling_diagnostic_report.v1", "empirical ceiling schema changed")
    _require(report.get("status") == "valid_completed" and report.get("result") == "VALID_COMPLETED", "empirical ceiling is incomplete")
    _require(report.get("classification") == "N3_FIT_HELDOUT_EMPIRICAL_CEILING_NOT_SUPPORTED", "empirical ceiling classification changed")
    contract = _mapping(report.get("contract"), "empirical ceiling contract")
    _require(contract.get("sha256") == EMPIRICAL_CEILING_CONTRACT_SHA256, "empirical ceiling contract identity changed")
    lineage = _mapping(report.get("development_lineage"), "empirical ceiling development lineage")
    _require(lineage.get("fold_assignment_sha256") == FOLD_ASSIGNMENT_SHA256, "empirical ceiling fold assignment changed")
    _require(lineage.get("fit_rows_per_fold") == FIT_COUNT_PER_FOLD, "empirical ceiling fit size changed")
    _require(lineage.get("heldout_rows_per_fold") == HELDOUT_COUNT_PER_FOLD, "empirical ceiling held-out size changed")
    fold_identity = _mapping(lineage.get("fold_identity"), "empirical ceiling fold identity")
    _require(fold_identity.get("assignment_sha256") == FOLD_ASSIGNMENT_SHA256, "empirical ceiling fold identity changed")
    _require(fold_identity.get("fold_count") == FOLD_COUNT, "empirical ceiling fold count changed")
    folds = _sequence(fold_identity.get("folds"), "empirical ceiling fold records")
    _require(len(folds) == FOLD_COUNT, "empirical ceiling fold records are incomplete")
    for fold, item in enumerate(folds):
        record = _mapping(item, f"empirical ceiling fold {fold}")
        _require(record.get("fold") == fold, f"empirical ceiling fold order changed at {fold}")
        _require(record.get("fit_count") == FIT_COUNT_PER_FOLD, f"empirical ceiling fit count changed at {fold}")
        _require(record.get("heldout_count") == HELDOUT_COUNT_PER_FOLD, f"empirical ceiling held-out count changed at {fold}")
        _require(record.get("fit_heldout_disjoint") is True, f"empirical ceiling overlap at {fold}")
    empirical = _mapping(report.get("empirical_fit"), "empirical ceiling result")
    fold_results = _sequence(empirical.get("fold_results"), "empirical ceiling results")
    _require(len(fold_results) == FOLD_COUNT, "empirical ceiling results are incomplete")
    pattern: list[bool] = []
    for fold, item in enumerate(fold_results):
        record = _mapping(item, f"empirical ceiling result {fold}")
        _require(record.get("fold") == fold, f"empirical ceiling result order changed at {fold}")
        _require(record.get("operationally_valid") is True, f"empirical ceiling fold {fold} is not operational")
        pattern.append(bool(record.get("compatibility")))
    _require(pattern == [False, True, False, True, True], "empirical ceiling fold pattern changed")
    aggregate = _mapping(report.get("aggregate"), "empirical ceiling aggregate")
    _require(aggregate.get("compatible_fold_count") == 3, "empirical ceiling compatible count changed")
    _require(aggregate.get("all_five_folds_required") is True, "empirical ceiling fold criterion changed")
    return {
        "path": _relative(repo_root, repo_root / EMPIRICAL_CEILING_REPORT),
        "sha256": EMPIRICAL_CEILING_REPORT_SHA256,
        "schema": report["schema"],
        "contract_sha256": EMPIRICAL_CEILING_CONTRACT_SHA256,
        "classification": report["classification"],
        "fold_pattern": pattern,
        "compatible_fold_count": 3,
        "fold_identity": dict(fold_identity),
    }

    analysis = _mapping(report.get("source_index_partition_position_analysis"), "source-position analysis")
    _require(analysis.get("ordering") == POSITION_ORDERING, "source-position ordering convention changed")
    _require(analysis.get("membership_changed") is False, "source-position membership changed")
    _require(analysis.get("new_split_created") is False, "source-position created a new split")
    _require(analysis.get("source_index_sha256") == comparison.EXPECTED_SOURCE_INDEX_SHA256, "source-position source identity changed")
    class_counts = _mapping(analysis.get("r2_last_minus_first_sign_counts"), "source-position class half-difference counts")
    spearman_counts = _mapping(analysis.get("r2_position_spearman_sign_counts"), "source-position class Spearman counts")
    _require(
        class_counts.get("positive") == CLASS_COUNT
        and class_counts.get("negative") == 0
        and class_counts.get("zero") == 0,
        "source-position class half-difference convention changed",
    )
    _require(
        spearman_counts.get("positive") == CLASS_COUNT
        and spearman_counts.get("negative") == 0
        and spearman_counts.get("zero_or_undefined") == 0,
        "source-position class Spearman convention changed",
    )
    _require(analysis.get("systematic_source_position_trend_supported") is True, "registered source-position trend changed")
    pooled = _mapping(analysis.get("pooled_class_stratified"), "source-position pooled class-stratified analysis")
    pooled_r2 = _mapping(pooled.get("r2"), "source-position pooled r2 analysis")
    association = _mapping(pooled_r2.get("association_with_normalized_position"), "source-position pooled r2 association")
    global_spearman = association.get("spearman")
    _require(isinstance(global_spearman, (int, float)) and float(global_spearman) == float(global_spearman), "source-position global statistic is invalid")

        "path": _relative(repo_root, repo_root / SOURCE_POSITION_REPORT),

            "value": float(global_spearman),
            "absolute_threshold": POSITION_GLOBAL_ABS_SPEARMAN_THRESHOLD,
            "drift_supported": abs(float(global_spearman)) >= POSITION_GLOBAL_ABS_SPEARMAN_THRESHOLD,
        },
        "class_conditioned_statistic": {
            "name": "all_class_r2_half_difference_and_spearman_sign_coherence",
            "half_difference_sign_counts": dict(class_counts),
            "spearman_sign_counts": dict(spearman_counts),
            "required_class_count": POSITION_CLASS_TREND_REQUIRED_COUNT,
            "drift_supported": True,
        },
        "registered_systematic_trend": True,
        "membership_changed": False,
        "new_split_created": False,
    }


def validate_contract_payload(contract: Mapping[str, Any]) -> None:
    """Validate the frozen semantic surface without loading q values."""

    _require(contract.get("schema") == CONTRACT_SCHEMA, "contract schema changed")
    _require(contract.get("version") == 1, "contract version changed")
    _require(contract.get("task_id") == TASK_ID and contract.get("gate") == GATE, "contract task changed")
    _require(contract.get("classification") == READY_CLASSIFICATION, "contract readiness changed")
    _require(contract.get("claim_classification") == CLAIM_CLASSIFICATION, "contract claim changed")
    _require(contract.get("definition_only") is True, "definition boundary changed")

    diagnostic = _mapping(contract.get("diagnostic"), "diagnostic")
    for key, expected in {
        "name": "DEVELOPMENT_SAMPLING_STATIONARITY",
        "pool_scope": "frozen 1000-row N3 development bank only",
        "official_fold_observations_unchanged": True,
        "producer_family_evaluated": False,
        "producer_selection": False,
        "sealed_confirmation_input": False,
        "training": False,
    }.items():
        _require(diagnostic.get(key) == expected, f"diagnostic semantic changed: {key}")

    development = _mapping(contract.get("development_evaluation_contract"), "development evaluation contract")
    for key, expected in {
        "pool_count": POOL_COUNT,
        "dimension": DIMENSION,
        "class_count": CLASS_COUNT,
        "rows_per_class": ROWS_PER_CLASS,
        "fold_count": FOLD_COUNT,
        "fit_rows_per_fold": FIT_COUNT_PER_FOLD,
        "heldout_rows_per_fold": HELDOUT_COUNT_PER_FOLD,
        "fit_rows_per_class_per_fold": FIT_ROWS_PER_CLASS_PER_FOLD,
        "heldout_rows_per_class_per_fold": HELDOUT_ROWS_PER_CLASS_PER_FOLD,
        "fold_assignment_sha256": FOLD_ASSIGNMENT_SHA256,
        "labels_as_model_inputs": False,
        "official_fold_resplit": False,
        "official_fold_reselection": False,
        "cross_fold_pooling": False,
    }.items():
        _require(development.get(key) == expected, f"development evaluation changed: {key}")

    moment = _mapping(contract.get("moment_stability_contract"), "moment stability contract")
    _require(moment.get("metrics") == list(MOMENT_METRICS), "moment metrics changed")
    _require(moment.get("std_denominator") == "n-1 within each subset", "standard deviation denominator changed")
    _require(moment.get("radial_variable") == "row_l2_norm(q)", "radial variable changed")
    _require(moment.get("all_metrics_nonnegative_and_finite_required") is True, "moment validity boundary changed")

    position = _mapping(contract.get("source_position_contract"), "source-position contract")
    _require(position.get("ordering") == POSITION_ORDERING, "source-position ordering changed")
    _require(position.get("variable") == POSITION_VARIABLE, "source-position variable changed")
    _require(position.get("existing_authority_reused") is True, "source-position authority reuse changed")
    _require(position.get("new_selector_created") is False, "new source-position selector was created")
    _require(position.get("global_statistic", {}).get("absolute_threshold") == POSITION_GLOBAL_ABS_SPEARMAN_THRESHOLD, "source-position global threshold changed")
    _require(position.get("class_conditioned_statistic", {}).get("required_class_count") == CLASS_COUNT, "source-position class threshold changed")
    _require(position.get("fold_subset_diagnostics_role") == "descriptive only; not an aggregate classification input", "fold position role changed")

    partition = _mapping(contract.get("finite_partition_reference_contract"), "finite partition reference")
    for key, expected in {
        "replicates": PARTITION_REFERENCE_REPLICATES,
        "batch_size": PARTITION_REFERENCE_BATCH_SIZE,
        "global_quantile": GLOBAL_REFERENCE_QUANTILE,
        "class_quantile": CLASS_REFERENCE_QUANTILE,
        "randomized_partitions_are_reference_only": True,
        "official_folds_remain_observations": True,
        "same_class_counts": True,
        "data_dependent_tuning": False,
        "replacement_of_official_folds": False,
    }.items():
        _require(partition.get(key) == expected, f"partition reference changed: {key}")
    _require(
        partition.get("scheme")
        == "for each replicate and class, torch.randperm(100) with the derived CPU seed; consecutive blocks of 20 rows define pseudo-fold held-outs; fit is the complement",
        "partition reference scheme changed",
    )
    _require(
        partition.get("seed_derivation")

    _require(partition.get("generator") == "torch.Generator(device='cpu')", "partition reference generator changed")

    _require(class_stability.get("reference_quantile") == CLASS_REFERENCE_QUANTILE, "class reference quantile changed")

    _require(
        class_stability.get("drift_rule")

    _require(class_stability.get("composition_rule") == "all official and pseudo folds require exactly 20 rows per class; composition difference cannot explain a result", "class composition rule changed")

    classification = _mapping(contract.get("classification_contract"), "classification contract")
    _require(classification.get("supported") == CLASSIFICATION_SUPPORTED, "supported classification changed")
    _require(classification.get("limitation") == CLASSIFICATION_LIMITATION, "limitation classification changed")
    _require(classification.get("mixed") == CLASSIFICATION_MIXED, "mixed classification changed")
    _require(classification.get("material_exceptional_fold_count") == MATERIAL_EXCEPTIONAL_FOLD_COUNT, "material fold threshold changed")
    _require(
        classification.get("rule")
        == "LIMITATION if source-position drift or class-conditioned drift is supported or exceptional official folds >=2; SUPPORTED if no drift and exceptional official folds=0; otherwise MIXED_OR_UNRESOLVED",
        "classification rule changed",
    )

    governance = _mapping(contract.get("governance"), "contract governance")
    for key, expected in {

        "formal_promotion": False,
        "generation_readiness": "not_ready",
        "sealed_confirmation_consumed": False,
        "producer_family_evaluated": False,

    boundary = _mapping(contract.get("execution_boundary"), "execution boundary")

        "q_values_loaded_by_definition_consumer": False,

        "cuda": False,
        "training": False,
    }.items():
        _require(boundary.get(key) == expected, f"execution boundary changed: {key}")
    totality = _mapping(contract.get("contract_totality"), "contract totality")
    _require(totality.get("status") == "passed", "contract totality did not pass")

def _write_contract(repo_root: Path, output_dir: Path, payload: Mapping[str, Any]) -> Path:
    if output_dir.exists():
        raise FileExistsError(f"refusing to reuse write-once contract output: {output_dir}")
    output_dir.parent.mkdir(parents=True, exist_ok=True)
    temporary = Path(tempfile.mkdtemp(prefix=f".{output_dir.name}.", dir=output_dir.parent))
    try:
        atomic_write_new_json(
            temporary / CONTRACT_NAME,
            payload,
            artifact_kind="N3 development sampling stationarity contract",
        )
        if output_dir.exists():
            raise FileExistsError(f"write-once output appeared during contract definition: {output_dir}")
        os.replace(temporary, output_dir)
    finally:
        if temporary.exists():
            shutil.rmtree(temporary)
    return output_dir / CONTRACT_NAME


def execute(repo_root: Path, output_dir: Path = OUTPUT_DIR) -> dict[str, Any]:
    git = _git_state(repo_root)
    _require(git["branch"] == EXPECTED_BRANCH, "current branch is not the N3 authority branch")
    _require(git["head"] == EXPECTED_STARTING_HEAD, "starting committed HEAD changed")
    unexpected_dirty = [path for path in git["dirty_paths"] if path not in SAFE_DIRTY_PATHS]
    _require(not unexpected_dirty, f"unrelated local changes invalidate provenance: {unexpected_dirty}")
    _require(not output_dir.exists(), f"write-once contract output already exists: {output_dir}")

    state = _validate_current_state(repo_root)
    development = _validate_development_authority(repo_root)
    empirical = _validate_empirical_ceiling(repo_root)
    position = _validate_source_position_authority(repo_root)
    payload = _build_payload(repo_root, git, state, development, empirical, position)
    contract_path = _write_contract(repo_root, output_dir, payload)
    contract = _read_json_exact(repo_root, output_dir / CONTRACT_NAME, file_sha256(contract_path), "new stationarity contract")
    validate_contract_payload(contract)
    return {
        "result": "READY",
        "classification": READY_CLASSIFICATION,
        "claim_classification": CLAIM_CLASSIFICATION,
        "contract": _relative(repo_root, contract_path),
        "contract_sha256": file_sha256(contract_path),
        "git_branch": git["branch"],
        "git_starting_head": git["head"],
        "git_ending_head": _git_state(repo_root)["head"],
        "q_values_loaded": False,
        "scientific_execution": False,
        "sealed_confirmation_consumed": False,
        "training_authorized": False,
        "formal_promotion": False,
        "generation_readiness": "not_ready",
    }


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--repo-root", type=Path, default=Path(__file__).resolve().parents[4])
    parser.add_argument("--output-dir", type=Path, default=OUTPUT_DIR)
    args = parser.parse_args()
    repo_root = args.repo_root.resolve()
    output_dir = args.output_dir if args.output_dir.is_absolute() else repo_root / args.output_dir
    print(json.dumps(execute(repo_root, output_dir), indent=2, sort_keys=True))


if __name__ == "__main__":
    main()
