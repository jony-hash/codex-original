class ContractError(RuntimeError):
    """Raised when the N3 model-order contract cannot be frozen safely."""


def _require(condition: bool, message: str) -> None:
    if not condition:
        raise ContractError(message)


def _mapping(value: object, name: str) -> Mapping[str, Any]:
    _require(isinstance(value, Mapping), f"{name} must be a mapping")
    return value


def _path(root: Path, value: str | Path) -> Path:
    relative = Path(value)
    _require(not relative.is_absolute(), f"{value} must be repository-relative")
    return (root / relative).resolve()


def _relative(root: Path, path: Path) -> str:
    return path.resolve().relative_to(root.resolve()).as_posix()


def _artifact_ref(root: Path, relative: str | Path, expected: str, role: str) -> dict[str, str]:
    target = _path(root, relative)
    _require(target.is_file(), f"{role} is missing: {target}")
    observed = file_sha256(target).lower()
    _require(observed == expected.lower(), f"{role} hash changed")
    return {"path": _relative(root, target), "sha256": expected.lower(), "role": role}


def _validate_report(root: Path, relative: Path, expected_sha: str, expected_classification: str) -> dict[str, str]:
    reference = _artifact_ref(root, relative, expected_sha, "historical diagnostic report")
    payload = _mapping(json.loads((root / relative).read_text(encoding="utf-8")), "historical diagnostic report")
    _require(payload.get("classification") == expected_classification, f"historical diagnostic classification changed: {relative}")
    return reference

def _load_train_labels(root: Path) -> tuple[torch.Tensor, torch.Tensor]:
    source = _path(root, TRAINING_SOURCE)
    _require(file_sha256(source).lower() == TRAINING_SOURCE_SHA256, "canonical TRAIN source identity changed")
    payload = torch.load(source, map_location="cpu", weights_only=False)
    _require(isinstance(payload, (tuple, list)) and len(payload) == 2, "canonical TRAIN source schema changed")
    images, labels = payload
    _require(isinstance(images, torch.Tensor) and isinstance(labels, torch.Tensor), "canonical TRAIN tensors are missing")
    _require(tuple(images.shape) == (60_000, 28, 28), "canonical TRAIN image geometry changed")
    _require(labels.shape == (60_000,) and labels.dtype == torch.long, "canonical TRAIN label geometry changed")
    return images, labels


def _excluded_rows(labels: torch.Tensor, train_indices: torch.Tensor) -> dict[str, Any]:
    groups = {
        "TRAIN_A": TRAIN_A,
        "TRAIN_B": TRAIN_B,
        "MULTIVIEW_C1": MULTIVIEW_C1,
        "MULTIVIEW_C2": MULTIVIEW_C2,
        "MULTIVIEW_C3": MULTIVIEW_C3,
        "MULTIVIEW_C4": MULTIVIEW_C4,
    }
    all_rows = [row for values in groups.values() for row in values]
    _require(len(set(all_rows)) == len(all_rows), "historical diagnostic rows overlap")
    excluded = torch.tensor(sorted(all_rows), dtype=torch.long)
    _require(bool(torch.isin(excluded, train_indices).all().item()), "historical diagnostic row left canonical TRAIN")
    details: dict[str, Any] = {}
    for name, values in groups.items():
        indices = torch.tensor(values, dtype=torch.long)
        labels_for_rows = labels.index_select(0, indices)
        details[name] = {
            "indices": list(values),
            "count": len(values),
            "label_histogram": torch.bincount(labels_for_rows, minlength=CLASS_COUNT).tolist(),
            "role": "historical diagnostic exclusion; not an N3 candidate population",
        }
    return {"groups": details, "union": excluded.tolist(), "union_count": int(excluded.numel())}


def build_partitions(root: Path) -> dict[str, Any]:
    _images, labels = _load_train_labels(root)
    train_indices, development_indices = stratified_mnist_indices(
        labels, seed=SPLIT_SEED, validation_per_label=VALIDATION_PER_LABEL
    )
    train_indices = torch.sort(train_indices).values.contiguous()
    development_indices = torch.sort(development_indices).values.contiguous()
    _require(train_indices.numel() == 55_000 and development_indices.numel() == 5_000, "canonical split counts changed")
    _require(torch.unique(torch.cat((train_indices, development_indices))).numel() == 60_000, "canonical split overlaps")
    excluded = _excluded_rows(labels, train_indices)
    excluded_tensor = torch.tensor(excluded["union"], dtype=torch.long)
    available = train_indices[~torch.isin(train_indices, excluded_tensor)]
    roles: dict[str, dict[str, Any]] = {}
    per_class_total = FIT_ROWS_PER_CLASS + SELECT_ROWS_PER_CLASS + CONFIRM_ROWS_PER_CLASS
    for role, count in (
        ("N3_FIT", FIT_ROWS_PER_CLASS),
        ("N3_SELECT", SELECT_ROWS_PER_CLASS),
        ("N3_CONFIRM", CONFIRM_ROWS_PER_CLASS),
    ):
        by_class: dict[str, list[int]] = {}
        flat: list[int] = []
        for digit in range(CLASS_COUNT):
            class_rows = available[labels.index_select(0, available) == digit]
            _require(class_rows.numel() >= per_class_total, f"class {digit} lacks N3 support")
            start = {
                "N3_FIT": 0,
                "N3_SELECT": FIT_ROWS_PER_CLASS,
                "N3_CONFIRM": FIT_ROWS_PER_CLASS + SELECT_ROWS_PER_CLASS,
            }[role]
            selected = class_rows[start : start + count]
            _require(selected.numel() == count, f"{role} class {digit} count changed")
            values = [int(item) for item in selected.tolist()]
            by_class[str(digit)] = values
            flat.extend(values)
        tensor = torch.tensor(flat, dtype=torch.long)
        roles[role] = {
            "count": int(tensor.numel()),
            "rows_per_class": count,
            "classes": list(range(CLASS_COUNT)),
            "indices": flat,
            "indices_sha256": tensor_sha256(tensor),
            "label_histogram": torch.bincount(labels.index_select(0, tensor), minlength=CLASS_COUNT).tolist(),
            "indices_by_class": by_class,
            "ordering": "class-major, within-class canonical TRAIN row order",
            "data_role": "TRAIN",
        }
    tensors = [torch.tensor(roles[name]["indices"], dtype=torch.long) for name in ("N3_FIT", "N3_SELECT", "N3_CONFIRM")]
    _require(torch.unique(torch.cat(tensors)).numel() == sum(tensor.numel() for tensor in tensors), "N3 roles overlap")
    _require(not bool(torch.isin(torch.cat(tensors), excluded_tensor).any().item()), "N3 role consumed a historical diagnostic row")
    return {
        "source": {"path": TRAINING_SOURCE.as_posix(), "sha256": TRAINING_SOURCE_SHA256, "source_count": 60_000},
        "canonical_split": {
            "algorithm": "stratified_mnist_indices(labels, seed=1234, validation_per_label=500)",
            "seed": SPLIT_SEED,
            "train_count": int(train_indices.numel()),
            "development_count": int(development_indices.numel()),
            "train_indices_sha256": tensor_sha256(train_indices),
            "development_indices_sha256": tensor_sha256(development_indices),
            "development_used_by_n3": False,
        },
        "historical_exclusions": excluded,
        "roles": roles,
        "protected_test_sealed": {"protected": 0, "test": 0, "sealed": 0},
    }


def parameter_count(rank: int) -> dict[str, int]:
    _require(rank in RANK_DOMAIN, "rank is outside the frozen candidate domain")
    basis = PARAMETER_DIMENSION * rank
    coefficients = CLASS_COUNT * rank
    return {
        "rank": rank,
        "basis_B": basis,
        "class_coefficients_A": coefficients,
        "stored_trainable_correction_parameters": basis + coefficients,
        "effective_gauge_dof_if_B_is_orthonormal": rank * (PARAMETER_DIMENSION + CLASS_COUNT - rank),
        "shared_baseline_parameters": PARAMETER_DIMENSION,
        "shared_baseline_parameters_trainable": 0,
    }


def _validate_start_checkpoint(root: Path) -> dict[str, Any]:
    path = _path(root, START_CHECKPOINT)
    payload = _mapping(safe_torch_load(path, expected_sha256=START_CHECKPOINT_SHA256, map_location="cpu"), "accepted step9000 checkpoint")
    _require(payload.get("completed_updates") == START_STEP, "accepted checkpoint cursor changed")
    _require(payload.get("trainable_parameter_names") == ["generator.phase"], "accepted trainable boundary changed")
    generator = payload.get("generator_state")
    _require(isinstance(generator, torch.Tensor) and tuple(generator.shape) == PHASE_SHAPE and generator.dtype == torch.float32, "accepted generator geometry changed")
    _require(bool(torch.isfinite(generator).all().item()), "accepted generator state is nonfinite")
    return {
        "path": START_CHECKPOINT.as_posix(),
        "sha256": START_CHECKPOINT_SHA256,
        "schema": payload.get("schema"),
        "completed_updates": START_STEP,
        "generator_shape": list(PHASE_SHAPE),
        "generator_dtype": str(generator.dtype),
        "generator_tensor_sha256": tensor_sha256(generator),
        "role": "frozen shared baseline theta_shared for every candidate",
    }

def write_contract(root: Path, output_dir: str | Path = DEFAULT_OUTPUT_DIR) -> Path:
    root = root.resolve()
    target = (root / output_dir if not Path(output_dir).is_absolute() else Path(output_dir)).resolve()
    outputs_root = (root / "outputs").resolve()
    _require(outputs_root in target.parents and target not in {outputs_root, outputs_root / "runs"}, "contract output must be a distinct child under outputs")
    _require(not target.exists(), f"contract output already exists: {target}")
    payload = build_contract(root)
    target.mkdir(parents=True, exist_ok=False)
    path = target / CONTRACT_NAME
    atomic_write_new_json(path, payload, artifact_kind="N3 class-conditioned low-rank structural model-order contract")
    return path


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)

    parser.add_argument("--output-dir", type=Path, default=DEFAULT_OUTPUT_DIR)
    args = parser.parse_args()
    path = write_contract(args.repo_root, args.output_dir)
    print(json.dumps({"result": "READY", "classification": CLASSIFICATION, "outer_gate": OUTER_GATE, "contract": path.as_posix(), "scientific_candidate_training_executed": False, "rank_selected": False}, sort_keys=True))


if __name__ == "__main__":
    main()
