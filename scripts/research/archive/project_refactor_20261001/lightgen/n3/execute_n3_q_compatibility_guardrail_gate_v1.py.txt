class InvalidHarnessError(RuntimeError):
    """A frozen input, evaluator, or output-boundary invariant failed."""


def _invalid(condition: bool, message: str) -> None:
    if not condition:
        raise InvalidHarnessError(message)


def _mapping(value: Any, name: str) -> Mapping[str, Any]:
    _invalid(isinstance(value, Mapping), f"{name} must be a mapping")
    return value


def _relative(repo_root: Path, path: Path) -> str:
    return path.resolve().relative_to(repo_root.resolve()).as_posix()


def _git_state(repo_root: Path) -> dict[str, Any]:
    branch = subprocess.run(
        ["git", "-C", str(repo_root), "branch", "--show-current"],
        check=True,
        capture_output=True,
        text=True,
    ).stdout.strip()
    head = subprocess.run(
        ["git", "-C", str(repo_root), "rev-parse", "HEAD"],
        check=True,
        capture_output=True,
        text=True,
    ).stdout.strip()
    status = subprocess.run(
        ["git", "-C", str(repo_root), "status", "--porcelain"],
        check=True,
        capture_output=True,
        text=True,
    ).stdout.splitlines()
    return {
        "branch": branch,
        "head": head,
        "dirty_paths": [line[3:] if len(line) >= 4 else line for line in status],
        "worktree_clean": not status,
    }

def _validate_contract_authorities(repo_root: Path, contract: Mapping[str, Any]) -> dict[str, str]:
    authorities = _mapping(contract.get("authorities"), "contract authorities")
    snapshot: dict[str, str] = {}
    for name, value in authorities.items():
        item = _mapping(value, f"contract authority {name}")
        path = repo_root / Path(str(item["path"]))
        expected = str(item["sha256"])
        _invalid(path.is_file(), f"contract authority is missing: {path}")
        actual = file_sha256(path)
        _invalid(actual == expected, f"contract authority hash changed: {path}")
        snapshot[str(item["path"])] = actual
    return snapshot

def _validate_preflight(repo_root: Path, contract: Mapping[str, Any]) -> tuple[dict[str, Any], dict[str, str]]:
    git = _git_state(repo_root)
    _invalid(git["branch"] == EXPECTED_BRANCH, "current branch is not the calibration authority branch")
    _invalid(git["head"] == EXPECTED_STARTING_HEAD, "current HEAD is not the frozen authority base")
    authority_snapshot = _validate_contract_authorities(repo_root, contract)
    _validate_current_boundaries(repo_root, contract)
    _validate_prior_results(repo_root, contract)
    source, fold, rows, fold_ids, q_meta = v2_ceiling._validate_source_and_fold(repo_root)
    _invalid(q_meta.get("values_loaded") is False, "V2 q values were loaded during preflight")
    _invalid(len(rows) == POOL_COUNT, "V2 source row count changed")
    _invalid(fold["protocol"]["fold_count"] == FOLD_COUNT, "V2 fold count changed")
    _invalid(fold["protocol"]["fit_count_per_fold"] == FIT_COUNT_PER_FOLD, "V2 fit count changed")
    _invalid(fold["protocol"]["heldout_count_per_fold"] == HELDOUT_COUNT_PER_FOLD, "V2 heldout count changed")
    _validate_fold_partition(fold_ids)
    return {
        "git": git,
        "source": source,
        "fold": fold,
        "rows": rows,
        "fold_ids": fold_ids,
        "q_meta": q_meta,
    }, authority_snapshot


def _validate_fold_partition(fold_ids: torch.Tensor) -> None:
    values = fold_ids.to(dtype=torch.int64).reshape(-1)
    _invalid(tuple(values.shape) == (POOL_COUNT,), "fold assignment shape changed")
    _invalid(tuple(int(value) for value in values.unique(sorted=True).tolist()) == tuple(range(FOLD_COUNT)), "fold assignment domain changed")
    for fold in range(FOLD_COUNT):
        heldout = (values == fold).nonzero(as_tuple=False).reshape(-1)
        fitting = (values != fold).nonzero(as_tuple=False).reshape(-1)
        _invalid(heldout.numel() == HELDOUT_COUNT_PER_FOLD, f"fold {fold} heldout count changed")
        _invalid(fitting.numel() == FIT_COUNT_PER_FOLD, f"fold {fold} fit count changed")
        _invalid(not bool(torch.isin(fitting, heldout).any()), f"fold {fold} fit/heldout overlap")

def _seed(producer_id: str, fold: int, trial: int, role: str, bank: int) -> int:
    material = f"{RNG_NAMESPACE}|producer={producer_id}|fold={fold}|trial={trial}|role={role}|bank={bank}".encode("utf-8")
    return int.from_bytes(hashlib.sha256(material).digest()[:8], "big") & ((1 << 63) - 1)


def validate_seed_namespace(producer_id: str, fold: int, trial: int) -> list[int]:
    roles = [("pseudo-data", 0)]
    roles.extend(("data-vs-model", index) for index in range(DATA_VS_MODEL_REPLICATES))
    roles.extend(("model-vs-model-reference", index) for index in range(MODEL_VS_MODEL_REPLICATES))
    seeds = [_seed(producer_id, fold, trial, role, bank) for role, bank in roles]
    _invalid(len(seeds) == len(set(seeds)), "RNG namespace collision within a trial")
    return seeds

def _compact_trial(
    producer_id: str,
    fold: int,
    trial: int,
    pseudo_record: Mapping[str, Any],
    model_records: list[Mapping[str, Any]],
    reference_records: list[Mapping[str, Any]],
    evaluated: Mapping[str, Any],
) -> dict[str, Any]:
    components = {name: bool(evaluated["compatibility_components"][name]) for name in SCALAR_METRICS}
    _invalid(set(components) == set(SCALAR_METRICS), "trial metric component set changed")
    return {
        "producer_id": producer_id,
        "fold": fold,
        "trial": trial,
        "primary_pass": components[PRIMARY_METRIC],
        "current_full_gate_pass": all(components.values()),
        "compatibility_components": components,
        "failed_guardrails": [name for name in SCALAR_METRICS if not components[name]],
        "data_vs_model_mean": _scalar_metrics(_mapping(evaluated["data_vs_model_mean"], "DATA-vs-MODEL mean")),
        "model_vs_model_reference_q95": _scalar_metrics(_mapping(evaluated["model_vs_model_reference_q95"], "MODEL-vs-MODEL q95")),
        "per_coordinate_all_pass": bool(evaluated["per_coordinate_all_pass"]),
        "per_coordinate_vector_role": "descriptive only; not a scalar compatibility gate",
        "sampling_records": [
            _compact_sampling_record(pseudo_record),
            *[_compact_sampling_record(record) for record in model_records],
            *[_compact_sampling_record(record) for record in reference_records],
        ],
    }

def _fold_summary(producer_id: str, fold: int, fit_summary: Mapping[str, Any], trial_records: list[dict[str, Any]]) -> dict[str, Any]:
    _invalid(len(trial_records) == TRIAL_COUNT_PER_PRODUCER_FOLD, f"{producer_id} fold {fold} trial count is incomplete")
    false_failures = {
        name: sum(not bool(trial["compatibility_components"][name]) for trial in trial_records)
        for name in SCALAR_METRICS
    }
    return {
        "producer_id": producer_id,
        "fold": fold,
        "trials": len(trial_records),
        "current_full_gate_pass_count": sum(bool(trial["current_full_gate_pass"]) for trial in trial_records),
        "current_full_gate_pass_rate": sum(bool(trial["current_full_gate_pass"]) for trial in trial_records) / float(len(trial_records)),
        "primary_energy_distance_false_fail_count": false_failures[PRIMARY_METRIC],
        "primary_energy_distance_false_fail_rate": false_failures[PRIMARY_METRIC] / float(len(trial_records)),
        "guardrail_false_failures": {
            name: {
                "false_fail_count": false_failures[name],
                "false_fail_rate": false_failures[name] / float(len(trial_records)),
            }
            for name in SCALAR_METRICS
        },
        "fit": dict(fit_summary),
        "trial_records": trial_records,
    }


def _binomial_lower_cdf(k: int, n: int, probability: float) -> float:
    if k < 0:
        return 0.0
    if k >= n:
        return 1.0
    if probability <= 0.0:
        return 1.0
    if probability >= 1.0:
        return 0.0
    log_p = math.log(probability)
    log_one_minus_p = math.log1p(-probability)
    logs = [
        math.lgamma(n + 1.0)
        - math.lgamma(j + 1.0)
        - math.lgamma(n - j + 1.0)
        + j * log_p
        + (n - j) * log_one_minus_p
        for j in range(k + 1)
    ]
    maximum = max(logs)
    return math.exp(maximum) * sum(math.exp(value - maximum) for value in logs)


def clopper_pearson_upper(false_fail_count: int, trial_count: int) -> float:
    _invalid(0 <= false_fail_count <= trial_count and trial_count > 0, "binomial count is invalid")
    if false_fail_count == trial_count:
        return 1.0
    if false_fail_count == 0:
        return 1.0 - math.exp(math.log(ALPHA) / trial_count)
    lower = 0.0
    upper = 1.0
    for _ in range(80):
        midpoint = (lower + upper) / 2.0
        if _binomial_lower_cdf(false_fail_count, trial_count, midpoint) > ALPHA:
            lower = midpoint
        else:
            upper = midpoint
    _invalid(math.isfinite(upper) and 0.0 <= upper <= 1.0, "Clopper-Pearson upper bound is invalid")
    return upper


def _interval(false_fail_count: int, trial_count: int) -> dict[str, Any]:
    return {
        "method": "exact one-sided Clopper-Pearson binomial upper confidence bound",
        "confidence_level": CONFIDENCE_LEVEL,
        "alpha": ALPHA,
        "lower_bound": 0.0,
        "upper_bound": clopper_pearson_upper(false_fail_count, trial_count),
    }


def _cell_failure_counts(fold_summaries: list[Mapping[str, Any]], metric: str) -> list[dict[str, Any]]:
    cells = []
    for item in fold_summaries:
        count = int(item["guardrail_false_failures"][metric]["false_fail_count"])
        trials = int(item["trials"])
        cells.append(
            {
                "producer_id": item["producer_id"],
                "fold": int(item["fold"]),
                "false_fail_count": count,
                "trials": trials,
                "false_fail_rate": count / float(trials),
                "uncertainty": _interval(count, trials),
            }
        )
    return cells


def _guardrail_summary(metric: str, fold_summaries: list[Mapping[str, Any]]) -> dict[str, Any]:
    cells = _cell_failure_counts(fold_summaries, metric)
    pooled_count = sum(int(cell["false_fail_count"]) for cell in cells)
    pooled_trials = sum(int(cell["trials"]) for cell in cells)
    worst = max(cells, key=lambda cell: (float(cell["uncertainty"]["upper_bound"]), float(cell["false_fail_rate"]), -int(cell["fold"])))
    independent_eligible = metric != PRIMARY_METRIC and float(worst["uncertainty"]["upper_bound"]) <= FALSE_FAIL_UPPER_BOUND
    return {
        "metric": metric,
        "primary": metric == PRIMARY_METRIC,
        "producer_fold_cells": cells,
        "pooled": {
            "false_fail_count": pooled_count,
            "trials": pooled_trials,
            "false_fail_rate": pooled_count / float(pooled_trials),
            "uncertainty": _interval(pooled_count, pooled_trials),
        },
        "worst_cell": worst,
        "worst_cell_upper_bound": float(worst["uncertainty"]["upper_bound"]),
        "independently_hard_eligible": independent_eligible,
        "eligibility_rule": "worst producer-by-fold Clopper-Pearson upper bound <= 0.01; primary energy_distance is excluded from this selection",
        "final_v3_role": "HARD_PRIMARY_UNCHANGED" if metric == PRIMARY_METRIC else ("HARD" if independent_eligible else "DESCRIPTIVE_ONLY"),
    }


def _add_calibrated_gate_results(fold_summaries: list[dict[str, Any]], hard_guardrails: list[str]) -> None:
    for item in fold_summaries:
        calibrated_count = 0
        for trial in item["trial_records"]:
            calibrated = all(bool(trial["compatibility_components"][name]) for name in hard_guardrails)
            trial["calibrated_gate_pass"] = calibrated
            calibrated_count += calibrated
        item["calibrated_gate_hard_guardrails"] = list(hard_guardrails)
        item["calibrated_gate_pass_count"] = calibrated_count
        item["calibrated_gate_pass_rate"] = calibrated_count / float(item["trials"])


def _producer_reliability(producer_id: str, fold_summaries: list[Mapping[str, Any]], hard_guardrails: list[str]) -> dict[str, Any]:
    ordered = sorted((item for item in fold_summaries if item["producer_id"] == producer_id), key=lambda item: int(item["fold"]))
    _invalid(len(ordered) == FOLD_COUNT, f"{producer_id} five-fold records are incomplete")
    full_rates = [float(item["current_full_gate_pass_rate"]) for item in ordered]
    calibrated_rates = [float(item["calibrated_gate_pass_rate"]) for item in ordered]
    direct_full = []
    direct_calibrated = []
    for trial in range(TRIAL_COUNT_PER_PRODUCER_FOLD):
        full = all(bool(item["trial_records"][trial]["current_full_gate_pass"]) for item in ordered)
        calibrated = all(bool(item["trial_records"][trial]["calibrated_gate_pass"]) for item in ordered)
        direct_full.append(full)
        direct_calibrated.append(calibrated)
    return {
        "producer_id": producer_id,
        "folds": [
            {
                "fold": int(item["fold"]),
                "current_full_gate_pass_count": int(item["current_full_gate_pass_count"]),
                "current_full_gate_pass_rate": float(item["current_full_gate_pass_rate"]),
                "calibrated_gate_pass_count": int(item["calibrated_gate_pass_count"]),
                "calibrated_gate_pass_rate": float(item["calibrated_gate_pass_rate"]),
            }
            for item in ordered
        ],
        "paired_trial_indices": list(range(TRIAL_COUNT_PER_PRODUCER_FOLD)),
        "direct_current_full_gate": {
            "all_five_fold_pass_count": sum(direct_full),
            "trials": len(direct_full),
            "all_five_fold_pass_rate": sum(direct_full) / float(len(direct_full)),
        },
        "direct_calibrated_gate": {
            "hard_guardrails": list(hard_guardrails),
            "all_five_fold_pass_count": sum(direct_calibrated),
            "trials": len(direct_calibrated),
            "all_five_fold_pass_rate": sum(direct_calibrated) / float(len(direct_calibrated)),
            "target": FIVE_FOLD_RELIABILITY_TARGET,
            "meets_target": sum(direct_calibrated) / float(len(direct_calibrated)) >= FIVE_FOLD_RELIABILITY_TARGET,
        },
        "diagnostic_product_of_calibrated_fold_rates": math.prod(calibrated_rates),
        "diagnostic_product_not_used_for_acceptance": True,
        "current_full_gate_fold_rates": full_rates,
    }


def _fit_fold(
    producer_id: str,
    fold: int,
    q: torch.Tensor,
    source_indices: torch.Tensor,
    labels: torch.Tensor,
    fold_ids: torch.Tensor,
) -> tuple[dict[str, Any], list[dict[str, Any]], set[int]]:
    fit_positions = (fold_ids != fold).nonzero(as_tuple=False).reshape(-1)
    heldout_positions = (fold_ids == fold).nonzero(as_tuple=False).reshape(-1)
    _invalid(fit_positions.numel() == FIT_COUNT_PER_FOLD and heldout_positions.numel() == HELDOUT_COUNT_PER_FOLD, f"fold {fold} partition changed")
    fit_q = q.index_select(0, fit_positions).contiguous()
    fit_source_indices = source_indices.index_select(0, fit_positions).contiguous()
    fit_labels = labels.index_select(0, fit_positions).contiguous()
    _invalid(torch.equal(torch.bincount(fit_labels, minlength=10), torch.full((10,), 400, dtype=torch.int64)), f"fold {fold} fit class balance changed")
    print(json.dumps({"stage": TASK_ID, "producer": producer_id, "fold": fold, "status": "fit_started"}), flush=True)
    fit = _fit_producer(producer_id, fit_q)
    fit_summary = _fit_summary(producer_id, fold, fit_q, fit_positions, fit_source_indices, fit_labels, fit)
    trial_records: list[dict[str, Any]] = []
    all_seeds: set[int] = set()
    for trial in range(TRIAL_COUNT_PER_PRODUCER_FOLD):
        seeds = validate_seed_namespace(producer_id, fold, trial)
        _invalid(all_seeds.isdisjoint(seeds), f"RNG seed collision within {producer_id} fold {fold}")
        all_seeds.update(seeds)
        pseudo_data, pseudo_record = _sample_bank(producer_id, fit, fold, trial, "pseudo-data", 0, PSEUDO_DATA_SAMPLE_COUNT)
        model_banks: list[torch.Tensor] = []
        model_records: list[dict[str, Any]] = []
        for bank in range(DATA_VS_MODEL_REPLICATES):
            sampled, record = _sample_bank(producer_id, fit, fold, trial, "data-vs-model", bank, MODEL_SAMPLE_COUNT)
            model_banks.append(sampled)
            model_records.append(record)
        reference_banks: list[torch.Tensor] = []
        reference_records: list[dict[str, Any]] = []
        for bank in range(MODEL_VS_MODEL_REPLICATES):
            sampled, record = _sample_bank(producer_id, fit, fold, trial, "model-vs-model-reference", bank, MODEL_SAMPLE_COUNT)
            reference_banks.append(sampled)
            reference_records.append(record)
        evaluated = _evaluate_trial(pseudo_data, model_banks, reference_banks)
        trial_records.append(_compact_trial(producer_id, fold, trial, pseudo_record, model_records, reference_records, evaluated))
        if (trial + 1) % 20 == 0 or trial + 1 == TRIAL_COUNT_PER_PRODUCER_FOLD:
            print(json.dumps({"stage": TASK_ID, "producer": producer_id, "fold": fold, "trial": trial + 1, "trials": TRIAL_COUNT_PER_PRODUCER_FOLD, "current_full_gate_pass_so_far": sum(bool(item["current_full_gate_pass"]) for item in trial_records)}), flush=True)
    return _fold_summary(producer_id, fold, fit_summary, trial_records), trial_records, all_seeds


def _write_json(path: Path, payload: Mapping[str, Any]) -> None:
    atomic_write_new_json(path, payload, artifact_kind="N3 q compatibility guardrail calibration artifact")

    preflight, authority_before = _validate_preflight(repo_root, contract)
    _invalid(file_sha256(repo_root / CONTRACT_PATH) == contract_sha256, "calibration contract changed before q load")
    q, source_indices, labels, identities = _load_q_after_preflight(repo_root, preflight)
    _invalid(len(identities) == POOL_COUNT, "V2 stable source identities are incomplete")
    _invalid(file_sha256(repo_root / Path(str(_mapping(_mapping(contract["authorities"], "contract authorities")["q_artifact"], "q authority")["path"]))) == _mapping(_mapping(contract["authorities"], "contract authorities")["q_artifact"], "q authority")["sha256"], "q authority changed after load")

    # The existing V1 harness provides the registered producer equations and

    old_namespace = self_null.RNG_NAMESPACE

    fold_summaries: list[dict[str, Any]] = []
    seed_count = 0
    all_seeds: set[int] = set()
    created_at = dt.datetime.now(dt.timezone.utc).isoformat()
    try:
        with torch.no_grad():
            for producer_id in PRODUCER_IDS:
                for fold in range(FOLD_COUNT):
                    summary, _trials, seeds = _fit_fold(producer_id, fold, q, source_indices, labels, preflight["fold_ids"])
                    _invalid(all_seeds.isdisjoint(seeds), f"global RNG seed collision for {producer_id} fold {fold}")
                    all_seeds.update(seeds)
                    seed_count += len(seeds)
                    fold_summaries.append(summary)
                    print(json.dumps({"stage": TASK_ID, "producer": producer_id, "fold": fold, "status": "completed", "current_full_gate_pass_rate": summary["current_full_gate_pass_rate"]}), flush=True)
                    _validate_authority_snapshot(authority_before, _authority_snapshot(repo_root, contract))
    finally:
        self_null.RNG_NAMESPACE = old_namespace
    _validate_authority_snapshot(authority_before, _authority_snapshot(repo_root, contract))
    _invalid(len(fold_summaries) == len(PRODUCER_IDS) * FOLD_COUNT, "producer/fold coverage is incomplete")
    _invalid(seed_count == len(PRODUCER_IDS) * FOLD_COUNT * TRIAL_COUNT_PER_PRODUCER_FOLD * (1 + DATA_VS_MODEL_REPLICATES + MODEL_VS_MODEL_REPLICATES), "sampling seed count is incomplete")
