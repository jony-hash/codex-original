"""Freeze the deterministic five-fold mapping for the V2 development bank.

The fold authority is created after the source rows and q artifact are
published, but before any compatibility metric is evaluated.  It reads only
JSON metadata and file hashes; q tensor values are not loaded here.
"""

import argparse
import json
import os
import shutil
import tempfile
from pathlib import Path
from typing import Any, Mapping

import torch

from ldm_project.v2.evidence import atomic_write_new_json, canonical_sha256, file_sha256, tensor_sha256
from scripts.research.lightgen.n3 import define_n3_development_bank_v2 as definition


TASK_ID = "N3_DEVELOPMENT_BANK_V2_FIVE_FOLD_PROTOCOL"
SCHEMA = "n3_development_bank_v2_five_fold_protocol.v1"
EXPECTED_BRANCH = definition.EXPECTED_BRANCH
EXPECTED_STARTING_HEAD = definition.EXPECTED_STARTING_HEAD

SOURCE_MANIFEST = definition.OUTPUT_DIR / definition.ARTIFACT_NAME
SOURCE_MANIFEST_SHA256 = "4bc251af91b3a6cd0235fe113117443d871a564bd05634fa9e61b5ab0b36f812"
Q_MATERIALIZATION_DIR = Path(
    "outputs/20260909_v2/runs/"
    "n3_development_bank_v2_q_materialization_v1_attempt_01"
)
Q_MATERIALIZATION_MANIFEST = Q_MATERIALIZATION_DIR / "n3_development_bank_v2_q_materialization_manifest.v1.json"
Q_MATERIALIZATION_MANIFEST_SHA256 = "422d8c65b78bc459197ed46504399cb9c87ba8f7184eb2405e0faa85353995d8"
Q_ARTIFACT = Q_MATERIALIZATION_DIR / "n3_development_bank_v2_q.v1.pt"
Q_ARTIFACT_SHA256 = "fe48c8fdb9bb2bf732076fabdf475288baa6f7d628ca10039a83bbda54fcfcbf"
Q_TENSOR_SHA256 = "d660fd8dc449725edab081d00c8f553f151029cddef9210360a7a6fad4281209"

CLASS_COUNT = definition.CLASS_COUNT
ROWS_PER_CLASS = definition.ROWS_PER_CLASS
POOL_COUNT = definition.POOL_COUNT
FOLD_COUNT = 5
HELDOUT_PER_CLASS = ROWS_PER_CLASS // FOLD_COUNT
FIT_PER_CLASS = ROWS_PER_CLASS - HELDOUT_PER_CLASS
HELDOUT_COUNT = CLASS_COUNT * HELDOUT_PER_CLASS
FIT_COUNT = POOL_COUNT - HELDOUT_COUNT
ASSIGNMENT_ALGORITHM = "within_class_source_index_ascending_round_robin_v1"
POSITION_ORDERING = "selected rows sorted by source_index ascending; exact ties by stable source-row identity ascending"

OUTPUT_DIR = Path(
    "outputs/20260909_v2/runs/"
    "n3_development_bank_v2_five_fold_protocol_v1_attempt_01"
)
ARTIFACT_NAME = "n3_development_bank_v2_fold_manifest.v1.json"

SAFE_DIRTY_PATHS = set(definition.SAFE_DIRTY_PATHS) | {
    "scripts/research/lightgen/n3/define_n3_development_bank_v2_five_fold_protocol.py",
    "scripts/research/lightgen/n3/execute_n3_development_bank_v2_empirical_ceiling.py",
    "tests/v2/test_n3_development_bank_v2.py",
}


def _require(condition: bool, message: str) -> None:
    if not condition:
        raise ValueError(message)


def _mapping(value: Any, name: str) -> Mapping[str, Any]:
    _require(isinstance(value, Mapping), f"{name} must be a mapping")
    return value


def _relative(repo_root: Path, path: Path) -> str:
    return path.resolve().relative_to(repo_root.resolve()).as_posix()


def _git_state(repo_root: Path) -> dict[str, Any]:
    return definition._git_state(repo_root)


def _read_json_exact(repo_root: Path, path: Path, expected_sha256: str, name: str) -> dict[str, Any]:
    target = repo_root / path
    _require(target.is_file(), f"{name} is missing: {target}")
    _require(file_sha256(target) == expected_sha256, f"{name} hash changed: {target}")
    return dict(_mapping(json.loads(target.read_text(encoding="utf-8")), name))


def _load_source(repo_root: Path) -> tuple[dict[str, Any], list[dict[str, Any]]]:
    source = _read_json_exact(repo_root, SOURCE_MANIFEST, SOURCE_MANIFEST_SHA256, "V2 source manifest")
    definition.validate_source_manifest_payload(source)
    rows = source["rows"]
    return source, [dict(_mapping(row, "source row")) for row in rows]


def _load_q_metadata(repo_root: Path, source: Mapping[str, Any]) -> dict[str, Any]:
    manifest = _read_json_exact(
        repo_root,
        Q_MATERIALIZATION_MANIFEST,
        Q_MATERIALIZATION_MANIFEST_SHA256,
        "V2 q materialization manifest",
    )
    _require(manifest.get("schema") == "n3_development_bank_v2_q_materialization_manifest.v1", "q materialization schema changed")
    _require(manifest.get("status") == "VALID_COMPLETED", "q materialization is incomplete")
    source_ref = _mapping(manifest.get("source_manifest"), "q source reference")
    _require(source_ref.get("path") == _relative(repo_root, repo_root / SOURCE_MANIFEST), "q source path changed")
    _require(source_ref.get("sha256") == SOURCE_MANIFEST_SHA256, "q source hash changed")
    _require(source_ref.get("bank_semantic_sha256") == source["bank_semantic_sha256"], "q bank semantic identity changed")
    q_ref = _mapping(manifest.get("q_artifact"), "q artifact reference")
    _require(q_ref.get("path") == Q_ARTIFACT.name, "q artifact path changed")
    q_path = repo_root / Q_ARTIFACT
    _require(q_path.is_file() and file_sha256(q_path) == Q_ARTIFACT_SHA256, "q artifact file hash changed")
    _require(q_ref.get("sha256") == Q_ARTIFACT_SHA256, "q artifact reference hash changed")
    _require(q_ref.get("q_tensor_sha256") == Q_TENSOR_SHA256, "q tensor identity changed")
    _require(q_ref.get("shape") == [POOL_COUNT, 100] and q_ref.get("dtype") == "torch.float64", "q tensor shape or dtype changed")
    identity = _mapping(manifest.get("source_identity"), "q source identity")
    _require(identity.get("row_manifest_sha256") == source["identity"]["row_manifest_sha256"], "q row identity changed")
    _require(identity.get("source_indices_sha256") == source["identity"]["selected_source_indices_sha256"], "q source index identity changed")
    _require(identity.get("labels_sha256") == source["identity"]["selected_labels_sha256"], "q label identity changed")
    _require(manifest.get("sealed_confirmation_consumed") is False, "q materialization consumed sealed confirmation")
    return {
        "manifest_path": _relative(repo_root, repo_root / Q_MATERIALIZATION_MANIFEST),
        "manifest_sha256": Q_MATERIALIZATION_MANIFEST_SHA256,
        "artifact_path": _relative(repo_root, q_path),
        "artifact_sha256": Q_ARTIFACT_SHA256,
        "tensor_sha256": Q_TENSOR_SHA256,
        "shape": q_ref["shape"],
        "dtype": q_ref["dtype"],
        "values_loaded": False,
    }


def _assignment(rows: list[dict[str, Any]]) -> tuple[list[int], list[dict[str, Any]], dict[str, Any]]:
    _require(len(rows) == POOL_COUNT, "V2 source row count changed")
    fold_ids = [-1] * POOL_COUNT
    fold_rows = [dict(row) for row in rows]
    for label in range(CLASS_COUNT):
        positions = [position for position, row in enumerate(rows) if int(row["label"]) == label]
        _require(len(positions) == ROWS_PER_CLASS, f"class {label} row count changed")
        ordered = sorted(
            positions,
            key=lambda position: (
                int(rows[position]["source_index"]),
                str(rows[position]["stable_source_row_identity"]),
            ),
        )
        for rank, position in enumerate(ordered):
            _require(rows[position].get("selected_within_class_rank") == rank, f"class {label} source ordering changed")
            fold_ids[position] = rank % FOLD_COUNT
            fold_rows[position]["within_class_source_order_rank"] = rank
            fold_rows[position]["heldout_fold"] = rank % FOLD_COUNT
    _require(all(value >= 0 for value in fold_ids), "fold assignment is incomplete")
    assignment_payload = {
        "algorithm": ASSIGNMENT_ALGORITHM,
        "ordering": POSITION_ORDERING,
        "fold_count": FOLD_COUNT,
        "bank_row_position": list(range(POOL_COUNT)),
        "source_indices": [int(row["source_index"]) for row in rows],
        "labels": [int(row["label"]) for row in rows],
        "fold_id_by_bank_row": fold_ids,
    }
    assignment_sha256 = canonical_sha256(assignment_payload)
    return fold_ids, fold_rows, {
        "algorithm": ASSIGNMENT_ALGORITHM,
        "ordering": POSITION_ORDERING,
        "fold_count": FOLD_COUNT,
        "fold_id_by_bank_row": fold_ids,
        "assignment_sha256": assignment_sha256,
        "rng_used": False,
        "q_values_used": False,
    }


def _fold_summaries(fold_ids: list[int], rows: list[dict[str, Any]]) -> list[dict[str, Any]]:
    source_indices = torch.tensor([int(row["source_index"]) for row in rows], dtype=torch.int64)
    labels = torch.tensor([int(row["label"]) for row in rows], dtype=torch.int64)
    result: list[dict[str, Any]] = []
    for fold in range(FOLD_COUNT):
        held = torch.tensor([position for position, value in enumerate(fold_ids) if value == fold], dtype=torch.int64)
        fit = torch.tensor([position for position, value in enumerate(fold_ids) if value != fold], dtype=torch.int64)
        held_labels = labels.index_select(0, held)
        fit_labels = labels.index_select(0, fit)
        _require(held.numel() == HELDOUT_COUNT and fit.numel() == FIT_COUNT, f"fold {fold} count changed")
        _require(torch.equal(torch.bincount(held_labels, minlength=CLASS_COUNT), torch.full((CLASS_COUNT,), HELDOUT_PER_CLASS, dtype=torch.int64)), f"fold {fold} held-out balance changed")
        _require(torch.equal(torch.bincount(fit_labels, minlength=CLASS_COUNT), torch.full((CLASS_COUNT,), FIT_PER_CLASS, dtype=torch.int64)), f"fold {fold} fit balance changed")
        result.append({
            "fold": fold,
            "heldout_count": int(held.numel()),
            "fit_count": int(fit.numel()),
            "heldout_per_class": HELDOUT_PER_CLASS,
            "fit_per_class": FIT_PER_CLASS,
            "heldout_row_positions_sha256": tensor_sha256(held),
            "fit_row_positions_sha256": tensor_sha256(fit),
            "heldout_source_indices_sha256": tensor_sha256(source_indices.index_select(0, held)),
            "fit_source_indices_sha256": tensor_sha256(source_indices.index_select(0, fit)),
            "heldout_labels_sha256": tensor_sha256(held_labels),
            "fit_labels_sha256": tensor_sha256(fit_labels),
            "fit_heldout_disjoint": True,
            "full_bank_covered_once": True,
        })
    return result


def _build_payload(repo_root: Path, git: Mapping[str, Any], source: Mapping[str, Any], q_meta: Mapping[str, Any], rows: list[dict[str, Any]], fold_ids: list[int], fold_identity: Mapping[str, Any]) -> dict[str, Any]:
    indices = torch.tensor([int(row["source_index"]) for row in rows], dtype=torch.int64)
    labels = torch.tensor([int(row["label"]) for row in rows], dtype=torch.int64)
    fold_rows_sha256 = canonical_sha256(rows)
    return {
        "schema": SCHEMA,
        "version": 1,
        "task_id": TASK_ID,
        "status": "FROZEN_BEFORE_VALIDATION",
        "semantic_authority_id": definition.SEMANTIC_AUTHORITY_ID,
        "bank_semantic_sha256": source["bank_semantic_sha256"],
        "git": {
            "branch": git["branch"],
            "starting_committed_head": git["head"],
            "definition_script": _relative(repo_root, Path(__file__).resolve()),
            "definition_script_sha256": file_sha256(Path(__file__).resolve()),
            "dirty_paths": git["dirty_paths"],
            "worktree_clean": git["worktree_clean"],
        },
        "source_manifest": {
            "path": _relative(repo_root, repo_root / SOURCE_MANIFEST),
            "sha256": SOURCE_MANIFEST_SHA256,
            "row_manifest_sha256": source["identity"]["row_manifest_sha256"],
            "source_indices_sha256": tensor_sha256(indices),
            "labels_sha256": tensor_sha256(labels),
        },
        "q_artifact": dict(q_meta),
        "protocol": {
            "class_count": CLASS_COUNT,
            "rows_per_class": ROWS_PER_CLASS,
            "pool_count": POOL_COUNT,
            "fold_count": FOLD_COUNT,
            "heldout_count_per_fold": HELDOUT_COUNT,
            "fit_count_per_fold": FIT_COUNT,
            "heldout_per_class_per_fold": HELDOUT_PER_CLASS,
            "fit_per_class_per_fold": FIT_PER_CLASS,
            "algorithm": ASSIGNMENT_ALGORITHM,
            "ordering": POSITION_ORDERING,
            "heldout_rule": "within-class source-order rank mod 5",
            "row_to_fold_frozen_before_validation": True,
            "assignment_sha256": fold_identity["assignment_sha256"],
            "rng_used": False,
            "q_values_used": False,
        },
        "rows": rows,
        "fold_identity": {
            **dict(fold_identity),
            "fold_rows_sha256": fold_rows_sha256,
            "folds": _fold_summaries(fold_ids, rows),
        },
        "identity": {
            "fold_rows_sha256": fold_rows_sha256,
            "source_indices_sha256": tensor_sha256(indices),
            "labels_sha256": tensor_sha256(labels),
            "class_counts": torch.bincount(labels, minlength=CLASS_COUNT).tolist(),
        },
        "execution_boundary": {
            "q_values_loaded": False,
            "scientific_validation_executed": False,
            "producer_evaluated": False,
            "sealed_confirmation_consumed": False,
            "training_authorized": False,
            "formal_promotion": False,
            "generation_readiness": "not_ready",
        },
        "governance": {
            "development_only": True,
            "producer_selection": False,
            "training_authorized": False,
            "formal_promotion": False,
            "generation_readiness": "not_ready",
            "sealed_confirmation_consumed": False,
        },
        "artifact_policy": {
            "write_once": True,
            "immutable_after_publish": True,
            "source_manifest_modified": False,
            "q_artifact_modified": False,
            "old_1000_row_authority_modified": False,
            "sealed_reservation_modified": False,
        },
    }


def validate_fold_manifest_payload(payload: Mapping[str, Any]) -> None:
    _require(payload.get("schema") == SCHEMA, "fold manifest schema changed")
    _require(payload.get("version") == 1 and payload.get("task_id") == TASK_ID, "fold manifest identity changed")
    _require(payload.get("status") == "FROZEN_BEFORE_VALIDATION", "fold manifest was not frozen before validation")
    _require(payload.get("semantic_authority_id") == definition.SEMANTIC_AUTHORITY_ID, "fold semantic authority changed")
    protocol = _mapping(payload.get("protocol"), "fold protocol")
    for key, expected in {
        "class_count": CLASS_COUNT,
        "rows_per_class": ROWS_PER_CLASS,
        "pool_count": POOL_COUNT,
        "fold_count": FOLD_COUNT,
        "heldout_count_per_fold": HELDOUT_COUNT,
        "fit_count_per_fold": FIT_COUNT,
        "heldout_per_class_per_fold": HELDOUT_PER_CLASS,
        "fit_per_class_per_fold": FIT_PER_CLASS,
        "algorithm": ASSIGNMENT_ALGORITHM,
        "ordering": POSITION_ORDERING,
        "heldout_rule": "within-class source-order rank mod 5",
        "row_to_fold_frozen_before_validation": True,
        "rng_used": False,
        "q_values_used": False,
    }.items():
        _require(protocol.get(key) == expected, f"fold protocol changed: {key}")
    rows = payload.get("rows")
    _require(isinstance(rows, list) and len(rows) == POOL_COUNT, "fold row list is incomplete")
    fold_ids: list[int] = []
    source_indices: list[int] = []
    labels: list[int] = []
    for position, item in enumerate(rows):
        row = _mapping(item, f"fold row {position}")
        _require(row.get("bank_row_position") == position, "fold row position changed")
        _require(isinstance(row.get("source_index"), int) and isinstance(row.get("label"), int), "fold row identity is invalid")
        _require(row.get("heldout_fold") in range(FOLD_COUNT), "fold id is invalid")
        _require(row.get("within_class_source_order_rank") == row.get("selected_within_class_rank"), "source order rank changed")
        _require(row.get("stable_source_row_identity") == definition._stable_source_row_identity(int(row["source_index"]), int(row["label"])), "stable row identity changed")
        source_indices.append(int(row["source_index"]))
        labels.append(int(row["label"]))
        fold_ids.append(int(row["heldout_fold"]))
    source_tensor = torch.tensor(source_indices, dtype=torch.int64)
    label_tensor = torch.tensor(labels, dtype=torch.int64)
    _require(torch.unique(source_tensor).numel() == POOL_COUNT, "fold source rows are not unique")
    _require(torch.equal(torch.bincount(label_tensor, minlength=CLASS_COUNT), torch.full((CLASS_COUNT,), ROWS_PER_CLASS, dtype=torch.int64)), "fold class balance changed")
    assignment_payload = {
        "algorithm": ASSIGNMENT_ALGORITHM,
        "ordering": POSITION_ORDERING,
        "fold_count": FOLD_COUNT,
        "bank_row_position": list(range(POOL_COUNT)),
        "source_indices": source_indices,
        "labels": labels,
        "fold_id_by_bank_row": fold_ids,
    }
    fold_identity = _mapping(payload.get("fold_identity"), "fold identity")
    _require(fold_identity.get("assignment_sha256") == canonical_sha256(assignment_payload), "fold assignment hash changed")
    _require(fold_identity.get("fold_rows_sha256") == canonical_sha256(rows), "fold row hash changed")
    _require(protocol.get("assignment_sha256") == fold_identity.get("assignment_sha256"), "protocol assignment binding changed")
    for fold in range(FOLD_COUNT):
        held = [position for position, value in enumerate(fold_ids) if value == fold]
        fit = [position for position, value in enumerate(fold_ids) if value != fold]
        _require(len(held) == HELDOUT_COUNT and len(fit) == FIT_COUNT, f"fold {fold} count changed")
        _require([labels[position] for position in held].count(0) == HELDOUT_PER_CLASS, f"fold {fold} class balance changed")
        _require(torch.equal(torch.bincount(label_tensor.index_select(0, torch.tensor(held)), minlength=CLASS_COUNT), torch.full((CLASS_COUNT,), HELDOUT_PER_CLASS, dtype=torch.int64)), f"fold {fold} held-out balance changed")
        _require(torch.equal(torch.bincount(label_tensor.index_select(0, torch.tensor(fit)), minlength=CLASS_COUNT), torch.full((CLASS_COUNT,), FIT_PER_CLASS, dtype=torch.int64)), f"fold {fold} fit balance changed")
    boundary = _mapping(payload.get("execution_boundary"), "fold execution boundary")
    for key in ("q_values_loaded", "scientific_validation_executed", "producer_evaluated", "sealed_confirmation_consumed", "training_authorized", "formal_promotion"):
        _require(boundary.get(key) is False, f"fold execution boundary changed: {key}")
    _require(boundary.get("generation_readiness") == "not_ready", "fold generation readiness changed")


def _write_artifact(repo_root: Path, output_dir: Path, payload: Mapping[str, Any]) -> Path:
    if output_dir.exists():
        raise FileExistsError(f"refusing to reuse write-once output: {output_dir}")
    output_dir.parent.mkdir(parents=True, exist_ok=True)
    temporary = Path(tempfile.mkdtemp(prefix=f".{output_dir.name}.", dir=output_dir.parent))
    try:
        atomic_write_new_json(temporary / ARTIFACT_NAME, payload, artifact_kind="N3 V2 fold manifest")
        if output_dir.exists():
            raise FileExistsError(f"write-once output appeared during fold definition: {output_dir}")
        os.replace(temporary, output_dir)
    finally:
        if temporary.exists():
            shutil.rmtree(temporary)
    return output_dir / ARTIFACT_NAME


def execute(repo_root: Path, output_dir: Path = OUTPUT_DIR) -> dict[str, Any]:
    git = _git_state(repo_root)
    _require(git["branch"] == EXPECTED_BRANCH, "current branch is not the N3 authority branch")
    _require(git["head"] == EXPECTED_STARTING_HEAD, "starting committed HEAD changed")
    unexpected = [path for path in git["dirty_paths"] if path not in SAFE_DIRTY_PATHS]
    _require(not unexpected, f"unrelated local changes invalidate provenance: {unexpected}")
    _require(not output_dir.exists(), f"write-once output already exists: {output_dir}")
    source, rows = _load_source(repo_root)
    q_meta = _load_q_metadata(repo_root, source)
    fold_ids, fold_rows, fold_identity = _assignment(rows)
    payload = _build_payload(repo_root, git, source, q_meta, fold_rows, fold_ids, fold_identity)
    validate_fold_manifest_payload(payload)
    artifact_path = _write_artifact(repo_root, output_dir, payload)
    published = json.loads(artifact_path.read_text(encoding="utf-8"))
    validate_fold_manifest_payload(published)
    return {
        "result": "FROZEN_BEFORE_VALIDATION",
        "artifact": _relative(repo_root, artifact_path),
        "artifact_sha256": file_sha256(artifact_path),
        "source_manifest_sha256": SOURCE_MANIFEST_SHA256,
        "bank_semantic_sha256": source["bank_semantic_sha256"],
        "q_artifact_sha256": Q_ARTIFACT_SHA256,
        "q_tensor_sha256": Q_TENSOR_SHA256,
        "assignment_sha256": fold_identity["assignment_sha256"],
        "fold_count": FOLD_COUNT,
        "fit_count_per_fold": FIT_COUNT,
        "heldout_count_per_fold": HELDOUT_COUNT,
        "fit_per_class_per_fold": FIT_PER_CLASS,
        "heldout_per_class_per_fold": HELDOUT_PER_CLASS,
        "q_values_loaded": False,
        "sealed_confirmation_consumed": False,
        "producer_evaluated": False,
        "training_authorized": False,
        "formal_promotion": False,
        "generation_readiness": "not_ready",
    }


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--repo-root", type=Path, default=Path(__file__).resolve().parents[4])
    parser.add_argument("--output-dir", type=Path, default=OUTPUT_DIR)
    args = parser.parse_args()
    repo_root = args.repo_root.resolve()
    output_dir = args.output_dir if args.output_dir.is_absolute() else repo_root / args.output_dir
    print(json.dumps(execute(repo_root, output_dir), indent=2, sort_keys=True))


if __name__ == "__main__":
    main()
