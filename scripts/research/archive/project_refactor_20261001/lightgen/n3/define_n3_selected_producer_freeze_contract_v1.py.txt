class InvalidHarnessError(RuntimeError):
    """A frozen authority or output-boundary invariant failed."""


def _require(condition: bool, message: str) -> None:
    if not condition:
        raise InvalidHarnessError(message)


def _mapping(value: Any, name: str) -> Mapping[str, Any]:
    _require(isinstance(value, Mapping), f"{name} must be a mapping")
    return value


def _relative(repo_root: Path, path: Path) -> str:
    return path.resolve().relative_to(repo_root.resolve()).as_posix()


def _read_json_exact(
    repo_root: Path,
    path: Path,
    expected_sha256: str,
    name: str,
) -> dict[str, Any]:
    target = repo_root / path
    _require(target.is_file(), f"{name} is missing: {target}")
    _require(file_sha256(target) == expected_sha256, f"{name} hash changed: {target}")
    try:
        value = json.loads(target.read_text(encoding="utf-8"))
    except (OSError, UnicodeError, json.JSONDecodeError) as exc:
        raise InvalidHarnessError(f"{name} cannot be read as JSON: {target}") from exc
    return dict(_mapping(value, name))


def _ref(
    repo_root: Path,
    path: Path,
    expected_sha256: str,
    role: str,
    schema: str,
) -> dict[str, Any]:
    target = repo_root / path
    _require(target.is_file(), f"{role} is missing: {target}")
    _require(file_sha256(target) == expected_sha256, f"{role} hash changed: {target}")
    return {
        "path": _relative(repo_root, target),
        "sha256": expected_sha256,
        "role": role,
        "schema": schema,
    }


def _git_state(repo_root: Path) -> dict[str, Any]:
    branch = subprocess.run(
        ["git", "-C", str(repo_root), "branch", "--show-current"],
        check=True,
        capture_output=True,
        text=True,
    ).stdout.strip()
    head = subprocess.run(
        ["git", "-C", str(repo_root), "rev-parse", "HEAD"],
        check=True,
        capture_output=True,
        text=True,
    ).stdout.strip()
    status = subprocess.run(
        ["git", "-C", str(repo_root), "status", "--porcelain"],
        check=True,
        capture_output=True,
        text=True,
    ).stdout.splitlines()
    return {
        "branch": branch,
        "head": head,
        "dirty_paths": [line[3:] if len(line) >= 4 else line for line in status],
        "worktree_clean": not status,
    }


def midrank_plotting_positions(values: Sequence[float]) -> list[float]:
    """Apply the selected candidate's stable average-rank convention."""

    n = len(values)
    _require(n >= 2, "synthetic marginal table must contain at least two rows")
    ordered = sorted(range(n), key=lambda index: (float(values[index]), index))
    positions = [0.0] * n
    cursor = 0
    while cursor < n:
        end = cursor + 1
        value = float(values[ordered[cursor]])
        while end < n and float(values[ordered[end]]) == value:
            end += 1
        average_rank = ((cursor + 1) + end) / 2.0
        plotting_position = (average_rank - 0.5) / float(n)
        for offset in range(cursor, end):
            positions[ordered[offset]] = plotting_position
        cursor = end
    _require(all(0.0 < value < 1.0 for value in positions), "plotting positions are not interior")
    return positions


def piecewise_linear_empirical_quantile(
    sorted_values: Sequence[float], probability: float
) -> float:
    """Apply endpoint-tail and linear interpolation semantics on synthetic data."""

    n = len(sorted_values)
    _require(n >= 2, "synthetic quantile table must contain at least two rows")
    values = [float(value) for value in sorted_values]
    _require(values == sorted(values), "quantile table must be sorted")
    p = float(probability)
    first_position = (1.0 - 0.5) / float(n)
    last_position = (float(n) - 0.5) / float(n)
    if p <= first_position:
        return values[0]
    if p >= last_position:
        return values[-1]
    upper = next(
        index
        for index in range(1, n)
        if p <= (float(index + 1) - 0.5) / float(n)
    )
    lower = upper - 1
    lower_position = (float(lower + 1) - 0.5) / float(n)
    upper_position = (float(upper + 1) - 0.5) / float(n)
    fraction = (p - lower_position) / (upper_position - lower_position)
    return values[lower] + fraction * (values[upper] - values[lower])

def define_contract(repo_root: Path) -> dict[str, Any]:
    root = repo_root.resolve()
    git = _git_state(root)
    _require(git["branch"] == EXPECTED_BRANCH, f"unexpected branch: {git['branch']}")
    _require(git["head"] == EXPECTED_HEAD, f"unexpected HEAD: {git['head']}")
    _governance, governance_snapshot = _validate_current_governance(root)
    source, q_manifest, fold, ceiling = _validate_v2_authorities(root)
    replay, _lineage, _evaluator, _calibration = _validate_v3_selection(root)
    _registry, _comparison, candidate, _historical = _validate_n4_and_gaussian(root)
    authorities = _authority_bundle(root, governance_snapshot)
    payload = _contract_payload(
        root,
        git,
        authorities,
        _v2_authority_summary(source, q_manifest, fold, ceiling),
        candidate,
        replay,
        governance_snapshot,
    )
    return payload


def publish_contract(repo_root: Path, payload: Mapping[str, Any]) -> Path:
    target = repo_root / OUTPUT_DIR / CONTRACT_NAME
    if target.exists():
        raise FileExistsError(f"refusing to overwrite selected-producer contract: {target}")
    atomic_write_new_json(target, payload, artifact_kind="selected-producer freeze contract")
    return target


def _parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--repo-root", type=Path, default=Path(__file__).resolve().parents[4])
    parser.add_argument("--publish", action="store_true", help="publish the write-once contract after validation")
    return parser.parse_args()


def main() -> int:
    args = _parse_args()
    try:
        payload = define_contract(args.repo_root)
        target = None
        if args.publish:
            target = publish_contract(args.repo_root.resolve(), payload)
        output = {
            "result": "READY",
            "classification": READY_CLASSIFICATION,
            "semantic_authority": PRODUCER_STATE_SEMANTIC_ID,
            "contract_schema": CONTRACT_SCHEMA,
            "artifact_path": _relative(args.repo_root.resolve(), target) if target else None,
            "artifact_sha256": file_sha256(target) if target else None,
            "selected_development_producer_candidate": SELECTED_DEVELOPMENT_CANDIDATE,
            "frozen_producer_state": None,
            "training_authorized": False,
            "formal_promotion": False,
            "generation_readiness": "not_ready",
            "sealed_confirmation": "RESERVED_NOT_MATERIALIZED",
            "next_action": "EXECUTE_N3_SELECTED_PRODUCER_FULL_DEVELOPMENT_FIT_V1",
        }
        print(json.dumps(output, indent=2, sort_keys=True))
        return 0
    except (InvalidHarnessError, FileNotFoundError, FileExistsError) as exc:
        print(json.dumps({"result": "BLOCKED", "reason": str(exc)}, indent=2, sort_keys=True))
        return 2
