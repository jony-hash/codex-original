The source manifest is already frozen before this consumer starts.  This
consumer reads only the selected MNIST image rows through the registered
fixed step-5000 encoder/OLS prefix, stores q=abs(c/sqrt(s)), and never enters
the decoder, evaluator, optimizer, training, or sealed-confirmation path.
"""

def _require(condition: bool, message: str) -> None:
    if not condition:
        raise ValueError(message)


def _mapping(value: Any, name: str) -> Mapping[str, Any]:
    _require(isinstance(value, Mapping), f"{name} must be a mapping")
    return value


def _relative(repo_root: Path, path: Path) -> str:
    return path.resolve().relative_to(repo_root.resolve()).as_posix()


def _git_state(repo_root: Path) -> dict[str, Any]:
    return definition._git_state(repo_root)


def _load_source(repo_root: Path) -> tuple[dict[str, Any], torch.Tensor, torch.Tensor, list[str]]:
    path = repo_root / SOURCE_MANIFEST
    _require(path.is_file(), "frozen V2 source manifest is missing")
    _require(file_sha256(path) == SOURCE_MANIFEST_SHA256, "frozen V2 source manifest hash changed")
    payload = json.loads(path.read_text(encoding="utf-8"))
    definition.validate_source_manifest_payload(payload)
    rows = payload["rows"]
    indices = torch.tensor([int(row["source_index"]) for row in rows], dtype=torch.int64)
    labels = torch.tensor([int(row["label"]) for row in rows], dtype=torch.int64)
    identities = [str(row["stable_source_row_identity"]) for row in rows]
    _require(tensor_sha256(indices) == payload["identity"]["selected_source_indices_sha256"], "source index hash is stale")
    _require(tensor_sha256(labels) == payload["identity"]["selected_labels_sha256"], "source label hash is stale")
    _require(len(set(identities)) == POOL_COUNT, "source row identities are not unique")
    return dict(payload), indices, labels, identities


def _load_selected_images(repo_root: Path, indices: torch.Tensor) -> torch.Tensor:
    path = repo_root / RAW_IMAGES_PATH
    _require(file_sha256(path) == RAW_IMAGES_SHA256, "raw MNIST image artifact changed")
    with path.open("rb") as handle:
        header = handle.read(16)
        _require(len(header) == 16, "raw MNIST image header is incomplete")
        magic, count, height, width = struct.unpack(">IIII", header)
        _require((magic, count, height, width) == (2051, SOURCE_COUNT, 28, 28), "raw MNIST image header changed")
        images = torch.empty((indices.numel(), 1, 28, 28), dtype=torch.float32)
        for position, source_index in sorted(enumerate(indices.tolist()), key=lambda item: item[1]):
            handle.seek(16 + int(source_index) * 28 * 28)
            raw = handle.read(28 * 28)
            _require(len(raw) == 28 * 28, "selected MNIST image row is incomplete")
            images[position, 0] = torch.tensor(list(raw), dtype=torch.float32).reshape(28, 28).div_(255.0)
    _require(torch.isfinite(images).all() and images.ge(0.0).all() and images.le(1.0).all(), "selected MNIST images are invalid")
    return images.contiguous()


def _state_digest(model: torch.nn.Module) -> str:
    return canonical_sha256({name: tensor_sha256(value) for name, value in model.state_dict().items()})


def _load_model_and_scale(repo_root: Path) -> tuple[bogt.ScaledChenBOGTModel, torch.Tensor, dict[str, Any]]:
    spec_path = repo_root / MRA_CONFIG_PATH
    _require(file_sha256(spec_path) == MRA_CONFIG_SHA256, "registered MRA config changed")
    spec = mra._load_spec(spec_path)
    mra._assert_contract(spec)
    formal, formal_path = mra._load_formal(spec)
    _require(formal_path.resolve() == (repo_root / FORMAL_CONFIG_PATH).resolve(), "formal config path changed")
    _require(file_sha256(formal_path) == FORMAL_CONFIG_SHA256, "formal config changed")

    checkpoint_path = repo_root / CHECKPOINT_PATH
    _require(file_sha256(checkpoint_path) == CHECKPOINT_SHA256, "step-5000 checkpoint changed")
    checkpoint = safe_torch_load(checkpoint_path, expected_sha256=CHECKPOINT_SHA256, map_location="cpu")
    _require(isinstance(checkpoint, Mapping), "step-5000 checkpoint is not a mapping")
    _require(checkpoint.get("schema") == "scaled_chen_bogt_horizon_checkpoint.v1", "checkpoint schema changed")
    _require(int(checkpoint.get("global_step", -1)) == 5000 and isinstance(checkpoint.get("model_state"), Mapping), "checkpoint step or state changed")
    model = mra._load_scaled_model(formal, checkpoint, torch.device("cpu"))
    _require(model.training is False, "fixed q model is in training mode")
    _require(not any(parameter.requires_grad for parameter in model.parameters()), "fixed q model is not frozen")
    _require(bogt.mode_bank_sha256(model) == MODE_BANK_SHA256, "fixed OLS mode-bank identity changed")

    calibration_path = repo_root / CALIBRATION_PATH
    calibration = safe_torch_load(calibration_path, expected_sha256=CALIBRATION_SHA256, map_location="cpu")
    calibration = _mapping(calibration, "unit-aware calibration")
    _require(calibration.get("schema") == CALIBRATION_SCHEMA and calibration.get("contract_version") == CALIBRATION_CONTRACT, "calibration identity changed")
    _require(calibration.get("sample_count") == 1000 and calibration.get("port_count") == DIMENSION, "calibration dimensions changed")
    _require(calibration.get("dataset_manifest_hash") == CALIBRATION_DATASET_MANIFEST_SHA256, "calibration dataset identity changed")
    _require(calibration.get("calibration_indices_sha256") == CALIBRATION_INDICES_SHA256, "calibration index identity changed")
    scale = calibration.get("port_scale")
    raw_scale = calibration.get("raw_port_mean")
    _require(torch.is_tensor(scale) and torch.is_tensor(raw_scale), "calibration scale tensors are missing")
    _require(scale.shape == (DIMENSION,) and raw_scale.shape == (DIMENSION,), "calibration scale shape changed")
    _require(torch.equal(scale, raw_scale), "registered q scale is not the frozen raw port mean")
    _require(torch.is_floating_point(scale) and torch.isfinite(scale).all() and scale.gt(0.0).all(), "calibration scale is invalid")
    return model, scale.detach().cpu().contiguous(), {
        "checkpoint_path": _relative(repo_root, checkpoint_path),
        "checkpoint_sha256": CHECKPOINT_SHA256,
        "checkpoint_global_step": 5000,
        "mode_bank_sha256": MODE_BANK_SHA256,
        "calibration_path": _relative(repo_root, calibration_path),
        "calibration_sha256": CALIBRATION_SHA256,
        "calibration_contract": CALIBRATION_CONTRACT,
        "calibration_scale_sha256": tensor_sha256(scale.detach().cpu()),
        "model_state_digest_before": _state_digest(model),
    }


def _collect_q(model: bogt.ScaledChenBOGTModel, images: torch.Tensor, scale: torch.Tensor) -> torch.Tensor:
    sqrt_scale = torch.sqrt(scale).to(dtype=torch.float32)
    chunks: list[torch.Tensor] = []
    batch_count = (images.shape[0] + BATCH_SIZE - 1) // BATCH_SIZE
    with torch.inference_mode():
        for batch_number, start in enumerate(range(0, images.shape[0], BATCH_SIZE), start=1):
            stop = min(start + BATCH_SIZE, images.shape[0])
            rendered = base.render_intensity_batch(images[start:stop])
            coefficients = replay._encoded_complex_prefix(model, rendered.amplitude)
            normalized = coefficients / sqrt_scale[None, :].to(device=coefficients.device)
            q = normalized.abs().to(dtype=torch.float64).detach().cpu().contiguous()
            _require(tuple(q.shape) == (stop - start, DIMENSION), "q batch shape changed")
            _require(torch.isfinite(q).all() and q.ge(0.0).all(), "q batch is invalid")
            chunks.append(q)
            if batch_number == 1 or batch_number == batch_count or batch_number % 25 == 0:
                print(json.dumps({"stage": "q_materialization", "processed_rows": stop, "total_rows": images.shape[0]}), flush=True)
    q = torch.cat(chunks, dim=0).contiguous()
    _require(tuple(q.shape) == (images.shape[0], DIMENSION), "q tensor shape changed")
    _require(torch.isfinite(q).all() and q.ge(0.0).all(), "q tensor is invalid")
    return q

    _require(state_after == model_identity["model_state_digest_before"], "fixed model state changed during q materialization")
    return {
        "schema": SCHEMA,
        "version": 1,
        "task_id": TASK_ID,
        "status": "VALID_COMPLETED",
        "semantic_authority_id": definition.SEMANTIC_AUTHORITY_ID,
        "q_coordinate": "R2_AMPLITUDE",
        "q_definition": "q=abs(c/sqrt(s))=abs(R3_COMPLEX)",
        "source_tensor_path": "ScaledChenBOGTModel._coefficients(encoded)",
        "source_stage": "encoder_final_field_to_fixed_ols_modal_projection",
        "source_dtype": "torch.complex64",
        "q_dtype": "torch.float64",
        "q_shape": [POOL_COUNT, DIMENSION],
        "source_manifest_path": _relative(repo_root, repo_root / SOURCE_MANIFEST),
        "source_manifest_sha256": SOURCE_MANIFEST_SHA256,
        "bank_semantic_sha256": source["bank_semantic_sha256"],
        "source_row_manifest_sha256": source["identity"]["row_manifest_sha256"],
        "q_tensor_sha256": tensor_sha256(q),
        "source_indices_sha256": tensor_sha256(indices),
        "labels_sha256": tensor_sha256(labels),
        "source_row_identity_sha256": canonical_sha256(identities),

        "labels": labels,
        "source_row_identity": identities,
        "q": q,

        "execution": {

            "selected_source_rows_only": True,
            "full_source_image_tensor_loaded": False,
            "sealed_confirmation_rows_read": False,

def _write_output(repo_root: Path, output_dir: Path, payload: Mapping[str, Any]) -> dict[str, Any]:
    if output_dir.exists():
        raise FileExistsError(f"refusing to reuse write-once output: {output_dir}")
    output_dir.parent.mkdir(parents=True, exist_ok=True)
    temporary = Path(tempfile.mkdtemp(prefix=f".{output_dir.name}.", dir=output_dir.parent))
    try:
        q_path = temporary / Q_ARTIFACT_NAME
        atomic_torch_save_new(q_path, payload, artifact_kind="N3 V2 q artifact")
        q_sha256 = file_sha256(q_path)
        manifest = {
            "schema": MANIFEST_SCHEMA,
            "version": 1,
            "task_id": TASK_ID,
            "status": "VALID_COMPLETED",
            "source_manifest": {
                "path": payload["source_manifest_path"],
                "sha256": payload["source_manifest_sha256"],
                "bank_semantic_sha256": payload["bank_semantic_sha256"],
            },
            "q_artifact": {
                "path": Q_ARTIFACT_NAME,
                "sha256": q_sha256,
                "size_bytes": q_path.stat().st_size,
                "schema": SCHEMA,
                "q_tensor_sha256": payload["q_tensor_sha256"],
                "shape": payload["q_shape"],
                "dtype": payload["q_dtype"],
            },
            "source_identity": {
                "row_manifest_sha256": payload["source_row_manifest_sha256"],
                "source_indices_sha256": payload["source_indices_sha256"],
                "labels_sha256": payload["labels_sha256"],
                "source_row_identity_sha256": payload["source_row_identity_sha256"],
            },
            "frozen_ols_authority": payload["frozen_ols_authority"],
            "execution": payload["execution"],
            "governance": payload["governance"],
            "sealed_confirmation_consumed": False,
        }
        atomic_write_new_json(
            temporary / MATERIALIZATION_MANIFEST_NAME,
            manifest,
            artifact_kind="N3 V2 q materialization manifest",
        )
        if output_dir.exists():
            raise FileExistsError(f"write-once output appeared during q materialization: {output_dir}")
        os.replace(temporary, output_dir)
    finally:
        if temporary.exists():
            shutil.rmtree(temporary)
    return {
        "q_artifact": _relative(repo_root, output_dir / Q_ARTIFACT_NAME),
        "q_artifact_sha256": q_sha256,
        "q_manifest": _relative(repo_root, output_dir / MATERIALIZATION_MANIFEST_NAME),
        "q_manifest_sha256": file_sha256(output_dir / MATERIALIZATION_MANIFEST_NAME),
        "q_tensor_sha256": payload["q_tensor_sha256"],
    }


def execute(repo_root: Path, output_dir: Path = OUTPUT_DIR) -> dict[str, Any]:
    started = time.perf_counter()
    git = _git_state(repo_root)
    _require(git["branch"] == EXPECTED_BRANCH, "current branch is not the N3 authority branch")
    _require(git["head"] == EXPECTED_STARTING_HEAD, "starting committed HEAD changed")
    unexpected = [path for path in git["dirty_paths"] if path not in SAFE_DIRTY_PATHS]
    _require(not unexpected, f"unrelated local changes invalidate provenance: {unexpected}")
    _require(not output_dir.exists(), f"write-once output already exists: {output_dir}")
    source, indices, labels, identities = _load_source(repo_root)
    images = _load_selected_images(repo_root, indices)
    model, scale, model_identity = _load_model_and_scale(repo_root)
    q = _collect_q(model, images, scale)
    state_after = _state_digest(model)
    del images, model
    payload = _q_payload(
        repo_root,
        source,
        indices,
        labels,
        identities,
        q,
        model_identity,
        state_after,
        time.perf_counter() - started,
    )
    result = _write_output(repo_root, output_dir, payload)
    return {
        "result": "VALID_COMPLETED",
        "semantic_authority_id": definition.SEMANTIC_AUTHORITY_ID,
        "source_manifest": _relative(repo_root, repo_root / SOURCE_MANIFEST),
        "source_manifest_sha256": SOURCE_MANIFEST_SHA256,
        "bank_semantic_sha256": source["bank_semantic_sha256"],
        "selected_count": POOL_COUNT,
        "selected_per_class": definition.ROWS_PER_CLASS,
        "q_tensor_sha256": payload["q_tensor_sha256"],
        "model_state_unchanged": payload["execution"]["model_state_unchanged"],
        "sealed_confirmation_consumed": False,
        "producer_evaluated": False,
        "training_authorized": False,
        "formal_promotion": False,
        "generation_readiness": "not_ready",
        **result,
    }


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--repo-root", type=Path, default=Path(__file__).resolve().parents[4])
    parser.add_argument("--output-dir", type=Path, default=OUTPUT_DIR)
    args = parser.parse_args()
    repo_root = args.repo_root.resolve()
    output_dir = args.output_dir if args.output_dir.is_absolute() else repo_root / args.output_dir
    print(json.dumps(execute(repo_root, output_dir), indent=2, sort_keys=True))


if __name__ == "__main__":
    main()
