class _InvalidHarnessError(RuntimeError):
    def __init__(self, message: str, details: Mapping[str, Any]) -> None:
        super().__init__(message)
        self.details = dict(details)


def _require(condition: bool, message: str) -> None:
    if not condition:
        raise ValueError(message)


def _invalid(
    message: str,
    *,
    candidate: str,
    stage: str,
    details: Mapping[str, Any] | None = None,
) -> None:
    payload: dict[str, Any] = {
        "candidate": candidate,
        "stage": stage,
        "failure_semantics": "INVALID_HARNESS",
        "contract_action": "stop the candidate/fold before scientific compatibility or selection",
    }
    if details:
        payload.update(details)
    raise _InvalidHarnessError(message, payload)


def _mapping(value: Any, name: str) -> Mapping[str, Any]:
    _require(isinstance(value, Mapping), f"{name} must be a mapping")
    return value


def _sequence(value: Any, name: str) -> list[Any]:
    _require(isinstance(value, list), f"{name} must be a list")
    return value


def _relative(repo_root: Path, path: Path) -> str:
    return path.resolve().relative_to(repo_root.resolve()).as_posix()


def _read_json_exact(
    repo_root: Path,
    path: Path,
    expected_sha256: str,
    name: str,
) -> dict[str, Any]:
    target = repo_root / path
    _require(target.is_file(), f"{name} is missing: {target}")
    _require(file_sha256(target) == expected_sha256, f"{name} hash changed: {target}")
    value = json.loads(target.read_text(encoding="utf-8"))
    return dict(_mapping(value, name))


def _git_state(repo_root: Path) -> dict[str, Any]:
    branch = subprocess.run(
        ["git", "-C", str(repo_root), "branch", "--show-current"],
        check=True,
        capture_output=True,
        text=True,
    ).stdout.strip()
    head = subprocess.run(
        ["git", "-C", str(repo_root), "rev-parse", "HEAD"],
        check=True,
        capture_output=True,
        text=True,
    ).stdout.strip()
    status = subprocess.run(
        ["git", "-C", str(repo_root), "status", "--porcelain"],
        check=True,
        capture_output=True,
        text=True,
    ).stdout.splitlines()
    return {
        "branch": branch,
        "head": head,
        "dirty_paths": [line[3:] if len(line) >= 4 else line for line in status],
        "worktree_clean": not status,
    }


def _seed(candidate: str, role: str, fold: int, replicate: int) -> int:
    material = f"{RNG_NAMESPACE}|{candidate}|{role}|fold={fold}|replicate={replicate}".encode("utf-8")
    return int.from_bytes(hashlib.sha256(material).digest()[:8], "big") & ((1 << 63) - 1)


def _numeric_summary(value: torch.Tensor) -> dict[str, float]:
    view = value.detach().to(dtype=torch.float64).reshape(-1)
    _require(view.numel() > 0 and bool(torch.isfinite(view).all()), "numeric summary input is invalid")
    return {"min": float(view.min()), "mean": float(view.mean()), "max": float(view.max())}

def _validate_development_identity(
    repo_root: Path,
    baseline_contract: Mapping[str, Any],
) -> tuple[torch.Tensor, torch.Tensor, torch.Tensor, torch.Tensor, dict[str, Any]]:
    q, source_indices, labels, pool_identity = comparison._load_development_pool(repo_root)
    fold_ids, fold_identity = comparison._assign_folds(source_indices, labels)
    _require(tuple(q.shape) == (POOL_COUNT, DIMENSION), "development q shape changed")
    _require(bool(torch.isfinite(q).all()) and bool(q.ge(0.0).all()), "development q is invalid")
    _require(pool_identity["count"] == POOL_COUNT, "development pool count changed")
    _require(pool_identity["dimension"] == DIMENSION, "development pool dimension changed")
    _require(pool_identity["per_class"] == POOL_COUNT // CLASS_COUNT, "development class balance changed")

    development = _mapping(baseline_contract.get("development_identity"), "baseline development identity")
    _require(development.get("count") == POOL_COUNT, "baseline development count changed")
    _require(development.get("per_class") == POOL_COUNT // CLASS_COUNT, "baseline class balance changed")
    _require(development.get("source_indices_sha256") == pool_identity["source_indices_sha256"], "source index identity changed")
    _require(development.get("labels_sha256") == pool_identity["labels_sha256"], "label identity changed")

def _fit_candidate_m2(q_fit: torch.Tensor) -> dict[str, Any]:
    values = q_fit.to(dtype=torch.float64).contiguous()
    if values.ndim != 2 or tuple(values.shape) != (FIT_COUNT_PER_FOLD, DIMENSION):
        _invalid("M2 fit shape is invalid", candidate=CANDIDATE_M2, stage="fit")
    if not bool(torch.isfinite(values).all()) or not bool(values.ge(0.0).all()):
        _invalid("M2 fitting q is invalid", candidate=CANDIDATE_M2, stage="fit")

    frozen = baseline._fit_fold(values)
    transformed = torch.log1p(values)
    if not bool(torch.isfinite(transformed).all()):
        _invalid("M2 log1p transform is non-finite", candidate=CANDIDATE_M2, stage="fit")
    mu = transformed.mean(dim=0)
    centered = transformed - mu
    sigma = torch.sqrt(centered.square().sum(dim=0).div(float(FIT_COUNT_PER_FOLD - 1)))
    if not bool(torch.isfinite(mu).all()) or not bool(torch.isfinite(sigma).all()):
        _invalid("M2 fitted parameters are non-finite", candidate=CANDIDATE_M2, stage="fit")
    if not bool(sigma.gt(definition.SIGMA_DEGENERACY_THRESHOLD).all()):
        _invalid("M2 fitted scale is not strictly positive", candidate=CANDIDATE_M2, stage="fit")
    boundary_standard = -mu / sigma
    alpha = torch.special.ndtr(boundary_standard)
    if not bool(torch.isfinite(alpha).all()) or not bool(alpha.ge(0.0).all()) or not bool(alpha.lt(1.0).all()):
        _invalid("M2 truncation probability is invalid", candidate=CANDIDATE_M2, stage="fit")
    return {
        "frozen": frozen,
        "q_fit": values,
        "transformed": transformed,
        "mu": mu,
        "sigma": sigma,
        "alpha": alpha,
    }


def _fit_candidate_d(q_fit: torch.Tensor) -> dict[str, Any]:
    values = q_fit.to(dtype=torch.float64).contiguous()
    if values.ndim != 2 or tuple(values.shape) != (FIT_COUNT_PER_FOLD, DIMENSION):
        _invalid("D fit shape is invalid", candidate=CANDIDATE_D, stage="fit")
    if not bool(torch.isfinite(values).all()) or not bool(values.ge(0.0).all()):
        _invalid("D fitting q is invalid", candidate=CANDIDATE_D, stage="fit")
    frozen = baseline._fit_fold(values)
    _require(tuple(frozen["sorted_values"].shape) == (FIT_COUNT_PER_FOLD, DIMENSION), "D empirical marginal shape is invalid")
    _require(torch.equal(frozen["sorted_values"], torch.sort(values, dim=0).values), "D empirical marginal changed")
    dependence = _mapping(frozen.get("dependence"), "D dependence")
    _require(tuple(dependence["R_shrunk"].shape) == (DIMENSION, DIMENSION), "D dependence shape is invalid")
    return {"frozen": frozen}


def _finite_probability(
    value: torch.Tensor,
    *,
    candidate: str,
    stage: str,
    fold: int,
    role: str,
    replicate: int,
    name: str,
) -> None:
    finite = torch.isfinite(value)
    interior = value.gt(0.0) & value.lt(1.0)
    if not bool(finite.all()) or not bool(interior.all()):
        _invalid(
            f"{name} is not finite and strictly interior",
            candidate=candidate,
            stage=stage,
            details={
                "fold": fold,
                "role": role,
                "replicate": replicate,
                "invalid_count": int((~(finite & interior)).sum()),
                "nonfinite_count": int((~finite).sum()),
            },
        )


def _regularized_incomplete_beta_continued_fraction(
    a: float,
    b: float,
    x: torch.Tensor,
    *,
    fold: int,
    role: str,
    replicate: int,
) -> torch.Tensor:
    if x.numel() == 0 or not bool(torch.isfinite(x).all()) or not bool(x.gt(0.0).all()) or not bool(x.lt(1.0).all()):
        _invalid(
            "D incomplete-beta input is invalid",
            candidate=CANDIDATE_D,
            stage="sampling",
            details={"fold": fold, "role": role, "replicate": replicate},
        )
    dtype = x.dtype
    fpm = torch.tensor(1.0e-300, dtype=dtype)
    tolerance = 1.0e-14
    max_iterations = 256

    def continued_fraction(first_a: float, second_b: float, argument: torch.Tensor) -> torch.Tensor:
        qab = first_a + second_b
        qap = first_a + 1.0
        qam = first_a - 1.0
        c = torch.ones_like(argument)
        d = 1.0 - qab * argument / qap
        d = torch.where(d.abs().lt(fpm), torch.full_like(d, fpm), d)
        d = d.reciprocal()
        h = d.clone()
        active = torch.ones_like(argument, dtype=torch.bool)
        for iteration in range(1, max_iterations + 1):
            m = float(iteration)
            m2 = 2.0 * m
            aa = m * (second_b - m) * argument / ((qam + m2) * (first_a + m2))
            d1 = 1.0 + aa * d
            d1 = torch.where(d1.abs().lt(fpm), torch.full_like(d1, fpm), d1)
            c1 = 1.0 + aa / c
            c1 = torch.where(c1.abs().lt(fpm), torch.full_like(c1, fpm), c1)
            d1 = d1.reciprocal()
            h1 = h * d1 * c1

            aa = -(first_a + m) * (qab + m) * argument / ((first_a + m2) * (qap + m2))
            d2 = 1.0 + aa * d1
            d2 = torch.where(d2.abs().lt(fpm), torch.full_like(d2, fpm), d2)
            c2 = 1.0 + aa / c1
            c2 = torch.where(c2.abs().lt(fpm), torch.full_like(c2, fpm), c2)
            d2 = d2.reciprocal()
            delta = d2 * c2
            h2 = h1 * delta
            converged = (delta - 1.0).abs().le(tolerance)
            h = torch.where(active, h2, h)
            d = torch.where(active, d2, d)
            c = torch.where(active, c2, c)
            active = active & ~converged
        if bool(active.any()):
            _invalid(
                "D Student-t CDF continued fraction did not converge",
                candidate=CANDIDATE_D,
                stage="sampling",
                details={"fold": fold, "role": role, "replicate": replicate},
            )
        return h

    threshold = (a + 1.0) / (a + b + 2.0)
    lower_mask = x.lt(threshold)
    result = torch.empty_like(x)
    if bool(lower_mask.any()):
        lower_x = x[lower_mask]
        log_beta_term = (
            torch.special.gammaln(torch.tensor(a + b, dtype=dtype))
            - torch.special.gammaln(torch.tensor(a, dtype=dtype))
            - torch.special.gammaln(torch.tensor(b, dtype=dtype))
            + a * torch.log(lower_x)
            + b * torch.log1p(-lower_x)
        )
        result[lower_mask] = torch.exp(log_beta_term) * continued_fraction(a, b, lower_x) / a
    if bool((~lower_mask).any()):
        upper_x = x[~lower_mask]
        complement = 1.0 - upper_x
        log_beta_term = (
            torch.special.gammaln(torch.tensor(a + b, dtype=dtype))
            - torch.special.gammaln(torch.tensor(a, dtype=dtype))
            - torch.special.gammaln(torch.tensor(b, dtype=dtype))
            + a * torch.log(upper_x)
            + b * torch.log1p(-upper_x)
        )
        result[~lower_mask] = 1.0 - torch.exp(log_beta_term) * continued_fraction(b, a, complement) / b
    if not bool(torch.isfinite(result).all()):
        _invalid(
            "D Student-t CDF is non-finite",
            candidate=CANDIDATE_D,
            stage="sampling",
            details={"fold": fold, "role": role, "replicate": replicate},
        )
    return result


def _student_t5_cdf(value: torch.Tensor, *, fold: int, role: str, replicate: int) -> torch.Tensor:
    scores = value.to(dtype=torch.float64).contiguous()
    if not bool(torch.isfinite(scores).all()):
        _invalid(
            "D Student-t latent scores are non-finite",
            candidate=CANDIDATE_D,
            stage="sampling",
            details={"fold": fold, "role": role, "replicate": replicate},
        )
    x = 5.0 / (5.0 + scores.square())
    if not bool(torch.isfinite(x).all()) or not bool(x.gt(0.0).all()) or not bool(x.le(1.0).all()):
        _invalid(
            "D Student-t beta argument is invalid",
            candidate=CANDIDATE_D,
            stage="sampling",
            details={"fold": fold, "role": role, "replicate": replicate},
        )
    cdf = torch.empty_like(scores)
    one_mask = x.eq(1.0)
    cdf[one_mask] = 0.5
    interior = ~one_mask
    if bool(interior.any()):
        beta_value = _regularized_incomplete_beta_continued_fraction(
            2.5,
            0.5,
            x[interior],
            fold=fold,
            role=role,
            replicate=replicate,
        )
        signed_scores = scores[interior]
        cdf[interior] = torch.where(
            signed_scores.lt(0.0),
            0.5 * beta_value,
            1.0 - 0.5 * beta_value,
        )
    _finite_probability(
        cdf,
        candidate=CANDIDATE_D,
        stage="sampling",
        fold=fold,
        role=role,
        replicate=replicate,
        name="D Student-t CDF",
    )
    return cdf


def _sample_candidate_m2(
    fit: Mapping[str, Any],
    *,
    fold: int,
    role: str,
    replicate: int,
) -> tuple[torch.Tensor, dict[str, Any]]:
    seed = _seed(CANDIDATE_M2, role, fold, replicate)
    generator = torch.Generator(device="cpu").manual_seed(seed)
    frozen = _mapping(fit.get("frozen"), "M2 frozen fit")
    dependence = _mapping(frozen.get("dependence"), "M2 dependence")
    noise = torch.randn((GENERATED_BANK_SIZE, DIMENSION), generator=generator, dtype=torch.float64)
    z_new = noise.mm(dependence["cholesky"].transpose(0, 1))
    u_new = torch.special.ndtr(z_new)
    _finite_probability(
        u_new,
        candidate=CANDIDATE_M2,
        stage="sampling",
        fold=fold,
        role=role,
        replicate=replicate,
        name="M2 copula probability",
    )
    alpha = fit["alpha"]
    p_new = alpha + (1.0 - alpha) * u_new
    if not bool(torch.isfinite(p_new).all()) or not bool(p_new.ge(alpha).all()) or not bool(p_new.gt(0.0).all()) or not bool(p_new.lt(1.0).all()):
        _invalid(
            "M2 inverse-CDF probability is invalid",
            candidate=CANDIDATE_M2,
            stage="sampling",
            details={
                "fold": fold,
                "role": role,
                "replicate": replicate,
                "invalid_count": int((~(torch.isfinite(p_new) & p_new.ge(alpha) & p_new.gt(0.0) & p_new.lt(1.0))).sum()),
            },
        )
    latent = fit["mu"] + fit["sigma"] * torch.special.ndtri(p_new)
    if not bool(torch.isfinite(latent).all()):
        _invalid(
            "M2 inverse-CDF latent is non-finite",
            candidate=CANDIDATE_M2,
            stage="sampling",
            details={"fold": fold, "role": role, "replicate": replicate},
        )
    y_raw = latent
    if not bool(torch.isfinite(y_raw).all()) or not bool(y_raw.ge(-Y_ROUNDOFF_TOLERANCE).all()):
        _invalid(
            "M2 y output violates the fixed support boundary",
            candidate=CANDIDATE_M2,
            stage="sampling",
            details={
                "fold": fold,
                "role": role,
                "replicate": replicate,
                "support_boundary_failure_count": int((~(torch.isfinite(y_raw) & y_raw.ge(-Y_ROUNDOFF_TOLERANCE))).sum()),
            },
        )
    q_raw = torch.expm1(y_raw)
    finite_q = torch.isfinite(q_raw)
    negative_q = q_raw.lt(0.0)
    canonicalized = negative_q & q_raw.ge(-Q_ROUNDOFF_TOLERANCE) & y_raw.ge(-Y_ROUNDOFF_TOLERANCE)
    material_negative = negative_q & ~canonicalized
    invalid_q = ~finite_q | material_negative
    if bool(invalid_q.any()):
        finite_values = q_raw[finite_q]
        _invalid(
            "M2 q output is invalid",
            candidate=CANDIDATE_M2,
            stage="sampling",
            details={
                "fold": fold,
                "role": role,
                "replicate": replicate,
                "invalid_count": int(invalid_q.sum()),
                "nonfinite_count": int((~finite_q).sum()),
                "negative_count": int(negative_q.sum()),
                "materially_negative_count": int(material_negative.sum()),
                "canonicalization_count": int(canonicalized.sum()),
                "q_min": float(finite_values.min()) if finite_values.numel() else None,
                "q_max": float(finite_values.max()) if finite_values.numel() else None,
            },
        )
    q_new = q_raw.clone()
    q_new[canonicalized] = 0.0
    if not bool(torch.isfinite(q_new).all()) or not bool(q_new.ge(0.0).all()):
        _invalid(
            "M2 canonicalized q output violates support",
            candidate=CANDIDATE_M2,
            stage="sampling",
            details={"fold": fold, "role": role, "replicate": replicate},
        )
    return q_new, {
        "candidate": CANDIDATE_M2,
        "role": role,
        "fold": fold,
        "replicate": replicate,
        "seed": seed,
        "rng": "torch.Generator(device='cpu')",
        "bank_size": GENERATED_BANK_SIZE,
        "z_finite": True,
        "u_finite_strict_interior": True,
        "p_finite_strict_interior": True,
        "y_finite": True,
        "y_support_verified": True,
        "q_raw_finite": True,
        "q_finite": True,
        "q_nonnegative": True,
        "support_by_construction": True,
        "proposal_rejection": False,
        "positivity_acceptance_fraction": 1.0,
        "canonicalization_count": int(canonicalized.sum()),
        "materially_negative_count": 0,
        "noise_sha256": tensor_sha256(noise),
        "z_sha256": tensor_sha256(z_new),
        "u_sha256": tensor_sha256(u_new),
        "alpha_sha256": tensor_sha256(alpha),
        "p_sha256": tensor_sha256(p_new),
        "y_sha256": tensor_sha256(y_raw),
        "q_raw_sha256": tensor_sha256(q_raw),
        "q_sha256": tensor_sha256(q_new),
        "q_summary": _numeric_summary(q_new),
    }


def _sample_candidate_d(
    fit: Mapping[str, Any],
    *,
    fold: int,
    role: str,
    replicate: int,
) -> tuple[torch.Tensor, dict[str, Any]]:
    seed = _seed(CANDIDATE_D, role, fold, replicate)
    generator = torch.Generator(device="cpu").manual_seed(seed)
    frozen = _mapping(fit.get("frozen"), "D frozen fit")
    dependence = _mapping(frozen.get("dependence"), "D dependence")
    gaussian_noise = torch.randn((GENERATED_BANK_SIZE, DIMENSION), generator=generator, dtype=torch.float64)
    radial_noise = torch.randn((GENERATED_BANK_SIZE, STUDENT_T_DEGREES_OF_FREEDOM), generator=generator, dtype=torch.float64)
    gaussian = gaussian_noise.mm(dependence["cholesky"].transpose(0, 1))
    radial_statistic = radial_noise.square().sum(dim=1).div(float(STUDENT_T_DEGREES_OF_FREEDOM))
    if not bool(torch.isfinite(radial_statistic).all()) or not bool(radial_statistic.gt(0.0).all()):
        _invalid(
            "D radial scale is invalid",
            candidate=CANDIDATE_D,
            stage="sampling",
            details={"fold": fold, "role": role, "replicate": replicate},
        )
    t_new = gaussian / torch.sqrt(radial_statistic).unsqueeze(1)
    if not bool(torch.isfinite(t_new).all()):
        _invalid(
            "D Student-t latent is non-finite",
            candidate=CANDIDATE_D,
            stage="sampling",
            details={"fold": fold, "role": role, "replicate": replicate},
        )
    u_new = _student_t5_cdf(t_new, fold=fold, role=role, replicate=replicate)
    q_new = baseline._empirical_quantile_batch(frozen["sorted_values"], u_new)
    finite_q = torch.isfinite(q_new)
    nonnegative_q = q_new.ge(0.0)
    if not bool(finite_q.all()) or not bool(nonnegative_q.all()):
        _invalid(
            "D empirical-quantile output is invalid",
            candidate=CANDIDATE_D,
            stage="sampling",
            details={
                "fold": fold,
                "role": role,
                "replicate": replicate,
                "invalid_count": int((~(finite_q & nonnegative_q)).sum()),
                "nonfinite_count": int((~finite_q).sum()),
                "negative_count": int((~nonnegative_q).sum()),
            },
        )
    return q_new, {
        "candidate": CANDIDATE_D,
        "role": role,
        "fold": fold,
        "replicate": replicate,
        "seed": seed,
        "rng": "torch.Generator(device='cpu')",
        "bank_size": GENERATED_BANK_SIZE,
        "degrees_of_freedom": STUDENT_T_DEGREES_OF_FREEDOM,
        "nu_verified": STUDENT_T_DEGREES_OF_FREEDOM == 5,
        "gaussian_noise_sha256": tensor_sha256(gaussian_noise),
        "radial_noise_sha256": tensor_sha256(radial_noise),
        "radial_statistic_sha256": tensor_sha256(radial_statistic),
        "t_sha256": tensor_sha256(t_new),
        "u_sha256": tensor_sha256(u_new),
        "q_sha256": tensor_sha256(q_new),
        "z_finite": True,
        "u_finite_strict_interior": True,
        "q_finite": True,
        "q_nonnegative": True,
        "proposal_rejection": False,
        "positivity_acceptance_fraction": 1.0,
        "q_summary": _numeric_summary(q_new),
    }


def _pair_metrics(first: torch.Tensor, second: torch.Tensor) -> dict[str, Any]:
    left = first.to(dtype=torch.float64).contiguous()
    right = second.to(dtype=torch.float64).contiguous()
    result = dict(comparison._pair_metrics(left, right))
    per_coordinate = [
        comparison.validated_energy_distance(
            left[:, coordinate : coordinate + 1],
            right[:, coordinate : coordinate + 1],
        )
        for coordinate in range(DIMENSION)
    ]
    _require(all(math.isfinite(float(value)) for value in per_coordinate), "per-coordinate marginal ED is non-finite")
    result["marginal_energy_distance_mean"] = float(sum(per_coordinate) / DIMENSION)
    result["marginal_energy_distance_max"] = float(max(per_coordinate))
    result["per_coordinate_marginal_energy_distance"] = [float(value) for value in per_coordinate]
    _require(all(math.isfinite(float(result[name])) for name in SCALAR_METRICS), "pair metric is non-finite")
    return result


def _mean_metrics(metrics: list[Mapping[str, Any]]) -> dict[str, Any]:
    _require(metrics, "metric list is empty")
    result = {name: float(sum(float(item[name]) for item in metrics) / len(metrics)) for name in SCALAR_METRICS}
    vector = torch.tensor(
        [item["per_coordinate_marginal_energy_distance"] for item in metrics],
        dtype=torch.float64,
    ).mean(dim=0)
    _require(bool(torch.isfinite(vector).all()), "mean marginal vector is non-finite")
    result["per_coordinate_marginal_energy_distance"] = [float(value) for value in vector]
    return result


def _q95_metrics(metrics: list[Mapping[str, Any]]) -> dict[str, Any]:
    _require(metrics, "reference metric list is empty")
    result = {
        name: float(
            torch.quantile(
                torch.tensor([float(item[name]) for item in metrics], dtype=torch.float64),
                COMPATIBILITY_Q,
            )
        )
        for name in SCALAR_METRICS
    }
    vector = torch.quantile(
        torch.tensor([item["per_coordinate_marginal_energy_distance"] for item in metrics], dtype=torch.float64),
        COMPATIBILITY_Q,
        dim=0,
    )
    _require(bool(torch.isfinite(vector).all()), "q95 marginal vector is non-finite")
    result["per_coordinate_marginal_energy_distance"] = [float(value) for value in vector]
    return result


def _relative_improvement(reference: float, candidate: float) -> dict[str, Any]:
    _require(math.isfinite(reference) and math.isfinite(candidate), "relative-improvement input is non-finite")
    if reference == 0.0:
        return {"status": "UNDEFINED_ZERO_REFERENCE", "value": None}
    value = (reference - candidate) / reference
    _require(math.isfinite(value), "relative improvement is non-finite")
    return {"status": "DEFINED_DESCRIPTIVE_ONLY", "value": float(value)}


def _interpretation(compatible: Mapping[str, bool]) -> str:
    marginal = all(bool(compatible[name]) for name in MARGINAL_METRICS)
    joint = all(bool(compatible[name]) for name in JOINT_METRICS)
    if not marginal and not joint:
        return "MARGINAL_MISMATCH_REMAINS"
    if marginal and not joint:
        return "MARGINALS_IMPROVED_JOINT_DEPENDENCE_REMAINS"
    if marginal and joint:
        return "NO_RESIDUAL_FAILURE"
    return "MIXED"

        "R_emp_sha256": tensor_sha256(dependence["R_emp"]),
        "R_shrunk_sha256": tensor_sha256(dependence["R_shrunk"]),

    }
    if candidate == CANDIDATE_M2:

                "marginal_parameterization_verified": True,
                "support_boundary": "y>=0 and q>=0",
                "fixed_truncation_boundary_fitted_scalar_count": 0,

        )
    else:
        summary.update(
            {
                "degrees_of_freedom": STUDENT_T_DEGREES_OF_FREEDOM,
                "nu_verified": True,
                "empirical_marginal_unchanged": True,
            }
        )
    return summary


def _evaluate_fold(
    candidate: str,
    fit: Mapping[str, Any],
    q: torch.Tensor,
    fold_ids: torch.Tensor,
    fold: int,
    original_report: Mapping[str, Any],
) -> dict[str, Any]:
    heldout_positions = (fold_ids == fold).nonzero(as_tuple=False).reshape(-1)
    fit_positions = (fold_ids != fold).nonzero(as_tuple=False).reshape(-1)
    _require(int(heldout_positions.numel()) == HELDOUT_COUNT_PER_FOLD, "held-out count changed")
    _require(int(fit_positions.numel()) == FIT_COUNT_PER_FOLD, "fit count changed")
    heldout = q.index_select(0, heldout_positions).contiguous()

    sampler = _sample_candidate_m2 if candidate == CANDIDATE_M2 else _sample_candidate_d
    data_banks: list[torch.Tensor] = []
    data_sampling: list[dict[str, Any]] = []
    for replicate in range(DATA_VS_MODEL_REPLICATES):
        bank, sampling = sampler(fit, fold=fold, role="data-vs-model", replicate=replicate)
        data_banks.append(bank)
        data_sampling.append(sampling)
    model_banks: list[torch.Tensor] = []
    model_sampling: list[dict[str, Any]] = []
    for replicate in range(MODEL_VS_MODEL_REPLICATES):
        bank, sampling = sampler(fit, fold=fold, role="model-vs-model", replicate=replicate)
        model_banks.append(bank)
        model_sampling.append(sampling)

    data_metrics = [_pair_metrics(heldout, bank) for bank in data_banks]
    reference_pairs = [
        _pair_metrics(model_banks[left], model_banks[right])
        for left, right in MODEL_REFERENCE_PAIRS
    ]
    data_mean = _mean_metrics(data_metrics)
    reference_q95 = _q95_metrics(reference_pairs)
    compatibility_components = {
        name: float(data_mean[name]) <= float(reference_q95[name])
        for name in SCALAR_METRICS
    }
    per_coordinate_checks = [
        float(data_mean["per_coordinate_marginal_energy_distance"][coordinate])
        <= float(reference_q95["per_coordinate_marginal_energy_distance"][coordinate])
        for coordinate in range(DIMENSION)
    ]
    _require(all(isinstance(value, bool) for value in compatibility_components.values()), "compatibility mapping is invalid")
    marginal_compatible = all(compatibility_components[name] for name in MARGINAL_METRICS)
    joint_compatible = all(compatibility_components[name] for name in JOINT_METRICS)
    original_fold = _mapping(
        _sequence(original_report.get("fold_execution"), "original fold execution")[fold],
        f"original fold {fold}",
    )
    original_mean = _mapping(original_fold.get("data_vs_model_mean"), f"original fold {fold} mean")
    relative = {
        name: _relative_improvement(float(original_mean[name]), float(data_mean[name]))
        for name in SCALAR_METRICS
    }
    support_summary = {
        "finite": all(bool(item["q_finite"]) for item in data_sampling + model_sampling),
        "nonnegative": all(bool(item["q_nonnegative"]) for item in data_sampling + model_sampling),
        "materially_negative_count": int(sum(int(item.get("materially_negative_count", 0)) for item in data_sampling + model_sampling)),
        "roundoff_canonicalization_count": int(sum(int(item.get("canonicalization_count", 0)) for item in data_sampling + model_sampling)),
    }
    return {
        "candidate": candidate,
        "fold": fold,
        "operationally_valid": True,
        "fit_heldout_disjoint": True,
        "fit_summary": _fit_summary(candidate, fit, fold),
        "data_sampling": data_sampling,
        "model_sampling": model_sampling,
        "sampling_all_finite": support_summary["finite"],
        "sampling_all_nonnegative": support_summary["nonnegative"],
        "support_summary": support_summary,
        "data_vs_model_replicates": data_metrics,
        "data_vs_model_mean": data_mean,
        "model_vs_model_reference_pairs": reference_pairs,
        "model_vs_model_reference_q95": reference_q95,
        "compatibility_rule": "fixed-replicate mean DATA_VS_MODEL <= q95 MODEL_VS_MODEL for q-space primary plus covariance/norm and aggregate marginal guardrails",
        "compatibility_components": compatibility_components,
        "compatibility": all(compatibility_components.values()),
        "per_coordinate_marginal_q95_checks": per_coordinate_checks,
        "per_coordinate_vector_role": "descriptive-only; not a separate compatibility gate",
        "marginal_compatible": marginal_compatible,
        "joint_compatible": joint_compatible,
        "marginal_vs_joint_interpretation": _interpretation(compatibility_components),
        "relative_improvement_vs_original_copula": {
            "descriptive_only": True,
            "selection_uses_this": False,
            "metrics": relative,
        },
    }


def _candidate_result(
    candidate: str,
    fits: list[Mapping[str, Any]],
    fold_results: list[Mapping[str, Any]],
) -> dict[str, Any]:
    _require(len(fits) == FOLD_COUNT and len(fold_results) == FOLD_COUNT, f"{candidate} fold count is incomplete")
    _require(all(bool(item["operationally_valid"]) for item in fold_results), f"{candidate} execution is not operationally valid")
    compatible_fold_count = sum(bool(item["compatibility"]) for item in fold_results)
    return {
        "candidate": candidate,
        "family": CANDIDATE_M2_FAMILY if candidate == CANDIDATE_M2 else CANDIDATE_D_FAMILY,
        "fit_validity": [dict(_mapping(item["fit_summary"], f"{candidate} fit summary")) for item in fold_results],
        "fold_results": [dict(item) for item in fold_results],
        "compatible_fold_count": compatible_fold_count,
        "fold_count": FOLD_COUNT,
        "development_supported": compatible_fold_count == FOLD_COUNT,
        "support_rule": "all five folds must be operationally valid and complete-compatible",
    }


def _write_report(repo_root: Path, output_dir: Path, report: Mapping[str, Any]) -> Path:
    if output_dir.exists():
        raise FileExistsError(f"refusing to reuse write-once output: {output_dir}")
    output_dir.parent.mkdir(parents=True, exist_ok=True)
    temporary = Path(tempfile.mkdtemp(prefix=f".{output_dir.name}.", dir=output_dir.parent))
    try:
        atomic_write_new_json(
            temporary / REPORT_NAME,
            report,
            artifact_kind="N3 oracle-guided minimal intervention comparison V2 report",
        )
        if output_dir.exists():
            raise FileExistsError(f"write-once output appeared during comparison: {output_dir}")
        os.replace(temporary, output_dir)
    finally:
        if temporary.exists():
            shutil.rmtree(temporary)
    return output_dir / REPORT_NAME


def _build_blocked_report(
    repo_root: Path,
    output_dir: Path,
    git_before: Mapping[str, Any],
    frozen_contract: Mapping[str, Any],

def execute(repo_root: Path, output_dir: Path = OUTPUT_DIR) -> dict[str, Any]:
    git_before = _git_state(repo_root)
    _require(git_before["branch"] == EXPECTED_BRANCH, "current branch is not the N3 authority branch")
    _require(git_before["head"] == EXPECTED_STARTING_HEAD, "current HEAD changed from frozen V2 contract base")
    unexpected_dirty = [path for path in git_before["dirty_paths"] if path not in SAFE_DIRTY_PATHS]
    _require(not unexpected_dirty, f"unrelated local changes invalidate provenance: {unexpected_dirty}")
    _require(not output_dir.exists(), f"V2 comparison output already exists: {output_dir}")

    validated = _validate_contract(repo_root)
    frozen_contract = validated["contract"]
    original_report = validated["original_report"]
    baseline_contract = validated["baseline_contract"]

    q, source_indices, labels, fold_ids, development_identity = _validate_development_identity(
        repo_root,
        baseline_contract,
    )

    try:
        with torch.no_grad():
            candidate_fits: dict[str, list[dict[str, Any]]] = {CANDIDATE_M2: [], CANDIDATE_D: []}
            for fold in range(FOLD_COUNT):
                fit_positions = (fold_ids != fold).nonzero(as_tuple=False).reshape(-1)
                q_fit = q.index_select(0, fit_positions).contiguous()
                candidate_fits[CANDIDATE_M2].append(_fit_candidate_m2(q_fit))
                candidate_fits[CANDIDATE_D].append(_fit_candidate_d(q_fit))

            candidate_fold_results: dict[str, list[dict[str, Any]]] = {CANDIDATE_M2: [], CANDIDATE_D: []}
            for candidate in (CANDIDATE_M2, CANDIDATE_D):
                for fold in range(FOLD_COUNT):
                    candidate_fold_results[candidate].append(
                        _evaluate_fold(
                            candidate,
                            candidate_fits[candidate][fold],
                            q,
                            fold_ids,
                            fold,
                            original_report,
                        )
                    )
    except _InvalidHarnessError as error:
        report_path = _build_blocked_report(
            repo_root,
            output_dir,
            git_before,
            frozen_contract,
            baseline_contract,
            original_report,
            q,
            source_indices,
            labels,
            development_identity,
            error,
        )
        return {
            "result": "BLOCKED",
            "classification": None,
            "selected_candidate": None,
            "report": _relative(repo_root, report_path),
            "report_sha256": file_sha256(report_path),
            "blocker": "INVALID_HARNESS",
            "sealed_confirmation_consumed": False,
            "training_authorized": False,
            "formal_promotion": False,
            "generation_readiness": "not_ready",
        }

    candidate_m2_result = _candidate_result(
        CANDIDATE_M2,
        candidate_fits[CANDIDATE_M2],
        candidate_fold_results[CANDIDATE_M2],
    )
    candidate_d_result = _candidate_result(
        CANDIDATE_D,
        candidate_fits[CANDIDATE_D],
        candidate_fold_results[CANDIDATE_D],
    )
    report = _valid_report(
        repo_root,
        git_before,
        frozen_contract,
        q,
        source_indices,
        labels,
        development_identity,
        candidate_m2_result,
        candidate_d_result,
    )
    report_path = _write_report(repo_root, output_dir, report)
    return {
        "result": "VALID_COMPLETED",
        "classification": report["classification"],
        "selected_candidate": report["selection"]["selected_candidate"],
        "report": _relative(repo_root, report_path),
        "report_sha256": file_sha256(report_path),
        "m2_compatible_folds": candidate_m2_result["compatible_fold_count"],
        "d_compatible_folds": candidate_d_result["compatible_fold_count"],
        "git_branch": git_before["branch"],
        "starting_head": git_before["head"],
        "ending_head": report["git"]["ending_head"],
        "sealed_confirmation_consumed": False,
        "training_authorized": False,
        "formal_promotion": False,
        "generation_readiness": "not_ready",
    }


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--repo-root", type=Path, default=Path(__file__).resolve().parents[4])
    parser.add_argument("--output-dir", type=Path, default=OUTPUT_DIR)
    args = parser.parse_args()
    repo_root = args.repo_root.resolve()
    output_dir = args.output_dir if args.output_dir.is_absolute() else repo_root / args.output_dir
    print(json.dumps(execute(repo_root, output_dir), indent=2, sort_keys=True))


if __name__ == "__main__":
    main()
