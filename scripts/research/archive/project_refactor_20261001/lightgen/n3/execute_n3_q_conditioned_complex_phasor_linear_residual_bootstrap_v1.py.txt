class InvalidHarnessError(RuntimeError):
    """A complete measurement cannot be trusted because the harness is invalid."""


class BlockedExecution(RuntimeError):
    """A required frozen authority cannot be reproduced without substitution."""


def _require(condition: bool, message: str, *, blocked: bool = False) -> None:
    if condition:
        return
    error = BlockedExecution if blocked else InvalidHarnessError
    raise error(message)


def _mapping(value: Any, name: str) -> Mapping[str, Any]:
    _require(isinstance(value, Mapping), f"{name} must be a mapping")
    return value


def _relative(repo_root: Path, path: Path) -> str:
    return path.resolve().relative_to(repo_root.resolve()).as_posix()


def _git_state(repo_root: Path) -> dict[str, Any]:
    def _run(*args: str) -> str:
        result = subprocess.run(
            ["git", "-C", str(repo_root), *args],
            check=True,
            capture_output=True,
            text=True,
        )
        return result.stdout.strip()

    status_result = subprocess.run(
        ["git", "-C", str(repo_root), "status", "--porcelain"],
        check=True,
        capture_output=True,
        text=True,
    )
    status = status_result.stdout.splitlines()
    return {
        "branch": _run("branch", "--show-current"),
        "head": _run("rev-parse", "HEAD"),
        "dirty_paths": [line[3:] if len(line) >= 4 else line for line in status],
        "worktree_clean": not status,
    }


def _read_json_exact(repo_root: Path, path: Path, expected_sha256: str, name: str) -> dict[str, Any]:
    target = repo_root / path
    _require(target.is_file(), f"{name} is missing: {target}", blocked=True)
    _require(file_sha256(target) == expected_sha256, f"{name} hash changed: {target}", blocked=True)
    try:
        value = json.loads(target.read_text(encoding="utf-8"))
    except (OSError, UnicodeError, json.JSONDecodeError) as exc:
        raise InvalidHarnessError(f"{name} cannot be read as JSON: {target}") from exc
    return dict(_mapping(value, name))


def _ref(repo_root: Path, path: Path, *, role: str, expected_sha256: str | None = None) -> dict[str, Any]:
    target = repo_root / path
    _require(target.is_file(), f"{role} is missing: {target}", blocked=True)
    actual = file_sha256(target)
    _require(expected_sha256 is None or actual == expected_sha256, f"{role} hash changed: {target}", blocked=True)
    return {"path": _relative(repo_root, target), "sha256": actual, "role": role}

def _validate_current_governance(repo_root: Path, contract: Mapping[str, Any]) -> dict[str, Any]:
    try:
        current = contract_definition._validate_authorities(repo_root)["current"]
    except contract_definition.InvalidHarnessError as exc:
        raise BlockedExecution(str(exc)) from exc
    state_ref = _mapping(current.get("governance"), "current governance reference")
    contract_governance = _mapping(contract.get("governance"), "contract governance")
    _require(
        state_ref.get("sha256") == contract_governance.get("state_sha256_before")
        and contract_governance.get("state_sha256_must_remain_unchanged") is True,
        "current governance does not match the frozen contract",
        blocked=True,
    )
    for path in (CURRENT_STATUS_PATH, CURRENT_HANDOFF_PATH):
        text = (repo_root / path).read_text(encoding="utf-8")
        _require(TASK_ID in text, f"{path} does not name the execution action", blocked=True)
        _require("training_authorized=false" in text and "generation_readiness=not_ready" in text, f"{path} opened downstream governance", blocked=True)
    return {
        "path": state_ref["path"],
        "sha256_before": state_ref["sha256"],
        "active_objective": current["active_objective"],
        "hard_blocker": current["hard_blocker"],
        "authorization_tier": current["authorization_tier"],
        "selected_producer": "EMPIRICAL_GAUSSIAN_COPULA",
        "n3_amplitude_producer_confirmation": "PASSED",
        "sealed_consumed": True,
        "sealed_reused": False,
        "training_authorized": False,
        "formal_promotion": False,
        "generation_readiness": "not_ready",
    }

        "selected_producer_and_prior_confirmation": runtime["prior"],

        "previous_minimal_q_conditioned_phase_donor": runtime["minimal_phase_execution"],

    for key, value in expected.items():
        _require(dict(authorities[key]) == dict(value), f"frozen authority differs from runtime: {key}", blocked=True)
    frozen_current = _mapping(authorities["current_state"], "frozen current state")
    runtime_current = _mapping(runtime["current"], "runtime current state")
    _require(
        frozen_current.get("active_objective") == runtime_current.get("active_objective")
        and frozen_current.get("hard_blocker") == runtime_current.get("hard_blocker")
        and frozen_current.get("authorization_tier") == runtime_current.get("authorization_tier")
        and _mapping(frozen_current.get("governance"), "frozen governance").get("sha256")
        == _mapping(runtime_current.get("governance"), "runtime governance").get("sha256"),
        "frozen current governance differs from runtime",
        blocked=True,
    )
    definition_ref = _mapping(authorities["definition_implementation"], "definition implementation")
    definition_path = repo_root / Path(str(definition_ref["path"]))
    _require(file_sha256(definition_path) == definition_ref.get("sha256"), "contract definition implementation changed", blocked=True)
    metric_authority = _mapping(authorities["metric_and_evaluator"], "metric authority")
    for key in ("energy_distance", "evaluator", "aggregation"):
        ref = _mapping(metric_authority.get(key), f"metric authority {key}")
        _require(file_sha256(repo_root / Path(str(ref["path"]))) == ref.get("sha256"), f"metric implementation changed: {key}", blocked=True)
    return expected

def _derive_seed(role: str, fold: int, trial: int | None, bank: int | None) -> int:
    allowed = {
        "CANDIDATE_Q_SAMPLE",
        "CANDIDATE_COMPLETE_RESIDUAL_SAMPLE",
        "CANDIDATE_MODEL_BANK",
        "CANDIDATE_PSEUDO_DATA",
    }
    _require(role in allowed, "unknown frozen candidate RNG role")
    _require(0 <= fold < FOLD_COUNT, "seed fold is invalid")
    trial_token = "none" if trial is None else str(trial)
    bank_token = "none" if bank is None else str(bank)
    material = f"{RNG_NAMESPACE}|role={role}|fold={fold}|trial={trial_token}|bank={bank_token}".encode("utf-8")
    return int.from_bytes(hashlib.sha256(material).digest()[:8], "big") & MAX_SIGNED_63


def _generator(seed: int) -> torch.Generator:
    _require(0 <= seed <= MAX_SIGNED_63, "seed is outside signed 63-bit range")
    return torch.Generator(device="cpu").manual_seed(seed)


def _sample_indices(row_count: int, sample_count: int, seed: int) -> torch.Tensor:
    _require(row_count > 0 and sample_count > 0, "sampling dimensions are invalid")
    return torch.randint(row_count, (sample_count,), generator=_generator(seed), dtype=torch.int64).contiguous()

def _bounded_diagnostic_sample(value: torch.Tensor, max_count: int = 256) -> torch.Tensor:
    view = value.detach().to(device="cpu", dtype=torch.float64).reshape(-1).contiguous()
    _require(view.numel() > 0 and torch.isfinite(view).all().item(), "diagnostic sample input is invalid")
    count = min(max_count, int(view.numel()))
    if count == view.numel():
        return view
    positions = torch.linspace(0, view.numel() - 1, count, dtype=torch.float64).round().to(dtype=torch.int64)
    return view.index_select(0, positions).contiguous()


def _fit_complex_phasor_model(x_fit: torch.Tensor, phase_fit: torch.Tensor, fold: int) -> dict[str, Any]:
    x = x_fit.detach().to(device="cpu", dtype=torch.float64).contiguous()
    phase = phase_fit.detach().to(device="cpu", dtype=torch.float64).contiguous()
    _require(x.shape == (FIT_COUNT, DIMENSION), f"fold {fold} Gaussianized q fit shape changed", blocked=True)
    _require(phase.shape == (FIT_COUNT, DIMENSION), f"fold {fold} phase fit shape changed", blocked=True)
    _require(torch.isfinite(x).all().item() and torch.isfinite(phase).all().item(), f"fold {fold} phase-OLS input is nonfinite")
    target = torch.complex(phase.cos(), phase.sin()).to(dtype=torch.complex128).contiguous()
    _require(torch.isfinite(target.real).all().item() and torch.isfinite(target.imag).all().item(), f"fold {fold} phasor target is nonfinite")
    design = torch.cat((torch.ones((FIT_COUNT, 1), dtype=torch.float64), x), dim=1).contiguous()
    _require(design.shape == (FIT_COUNT, DESIGN_FEATURES), f"fold {fold} design shape changed", blocked=True)
    left, singular_values, right_transpose = torch.linalg.svd(design, full_matrices=False)
    _require(
        torch.isfinite(left).all().item()
        and torch.isfinite(singular_values).all().item()
        and torch.isfinite(right_transpose).all().item(),
        f"fold {fold} OLS SVD is nonfinite",
    )
    _require(singular_values.shape == (DESIGN_FEATURES,) and singular_values[0] > 0.0, f"fold {fold} OLS singular spectrum is invalid")
    cutoff = OLS_RCOND * singular_values[0]
    rank = int((singular_values > cutoff).sum().item())
    if rank != DESIGN_FEATURES:
        raise InvalidHarnessError(f"fold {fold} INVALID_HARNESS: frozen OLS design rank {rank} < {DESIGN_FEATURES}")
    projected_target = left.transpose(0, 1).to(torch.complex128).matmul(target)
    coefficients = right_transpose.transpose(0, 1).to(torch.complex128).matmul(
        projected_target / singular_values.to(torch.complex128).unsqueeze(-1)
    ).contiguous()
    prediction = design.to(torch.complex128).matmul(coefficients).contiguous()
    residual = (target - prediction).contiguous()
    _require(torch.isfinite(coefficients.real).all().item() and torch.isfinite(coefficients.imag).all().item(), f"fold {fold} OLS coefficients are nonfinite")
    _require(torch.isfinite(residual.real).all().item() and torch.isfinite(residual.imag).all().item(), f"fold {fold} OLS residuals are nonfinite")
    singular_summary = _summary(singular_values)
    smallest = float(singular_values[-1])
    largest = float(singular_values[0])
    _require(smallest > 0.0, f"fold {fold} OLS smallest singular value is nonpositive")
    return {
        "x": x,
        "target": target,
        "design": design,
        "coefficients": coefficients,
        "prediction": prediction,
        "residual": residual,
        "rank": rank,
        "singular_values": singular_values,
        "fit_summary": {
            "effective_rank": rank,
            "required_rank": DESIGN_FEATURES,
            "rcond": OLS_RCOND,
            "singular_values": singular_summary,
            "largest_singular_value": largest,
            "smallest_singular_value": smallest,
            "condition_number": largest / smallest,
            "design_tensor_sha256": tensor_sha256(design),
            "phasor_target_tensor_sha256": tensor_sha256(target),
            "coefficient_tensor_sha256": tensor_sha256(coefficients),
            "residual_tensor_sha256": tensor_sha256(residual),
            "residual_norm": _summary(residual.abs().norm(dim=1)),
            "phase_phasor_prediction_error": _summary((target - prediction).abs().norm(dim=1)),
            "conditional_mean_magnitude": _summary(prediction.abs().norm(dim=1)),
            "numerical_validity": True,
        },
    }


def _compose_candidate(
    q_fit_model: Mapping[str, Any],
    phase_model: Mapping[str, Any],
    sample_count: int,
    fold: int,
    trial: int | None,
    bank: int | None,
    role: str,
) -> tuple[torch.Tensor, torch.Tensor, torch.Tensor, torch.Tensor, dict[str, Any]]:
    q_seed = _derive_seed("CANDIDATE_Q_SAMPLE", fold, trial, bank)
    residual_seed = _derive_seed("CANDIDATE_COMPLETE_RESIDUAL_SAMPLE", fold, trial, bank)
    q_new = prior_execution._sample_q_from_fold(q_fit_model, sample_count, q_seed)
    x_new = _gaussianized_q_coordinates_batch(q_new, q_fit_model["sorted_values"])
    design_new = torch.cat((torch.ones((sample_count, 1), dtype=torch.float64), x_new), dim=1).contiguous()
    coefficients = phase_model["coefficients"]
    residual_pool = phase_model["residual"]
    mean_new = design_new.to(torch.complex128).matmul(coefficients).contiguous()
    residual_indices = _sample_indices(FIT_COUNT, sample_count, residual_seed)
    residual_new = residual_pool.index_select(0, residual_indices).contiguous()
    preprojection = (mean_new + residual_new).contiguous()
    magnitudes = preprojection.abs()
    valid = (
        torch.isfinite(preprojection.real)
        & torch.isfinite(preprojection.imag)
        & torch.isfinite(magnitudes)
        & magnitudes.gt(PHASOR_PROJECTION_EPSILON)
    )
    invalid_coordinate_count = int((~valid).sum().item())
    invalid_sample_count = int((~valid.all(dim=1)).sum().item())
    _require(
        invalid_coordinate_count == 0,
        f"fold {fold} INVALID_HARNESS: {invalid_coordinate_count} phasor coordinates in {invalid_sample_count} samples violate projection epsilon",
    )
    unit_phasor = (preprojection / magnitudes).to(dtype=torch.complex128).contiguous()
    z_new = (q_new.to(torch.complex128) * unit_phasor).contiguous()
    _require(torch.isfinite(z_new.real).all().item() and torch.isfinite(z_new.imag).all().item(), f"fold {fold} generated complex state is nonfinite")
    _require(torch.all(z_new.abs() > 0).item(), f"fold {fold} generated amplitude contains exact zero")
    canonical = complex_definition.canonicalize_existing_global_phase(z_new).contiguous()
    _require(torch.isfinite(canonical.real).all().item() and torch.isfinite(canonical.imag).all().item(), f"fold {fold} canonical complex state is nonfinite")
    seed_record = {
        "role": role,
        "q_sample_role": "CANDIDATE_Q_SAMPLE",
        "residual_sample_role": "CANDIDATE_COMPLETE_RESIDUAL_SAMPLE",
        "q_seed": q_seed,
        "residual_seed": residual_seed,
        "fold": fold,
        "trial": trial,
        "bank": bank,
        "sample_count": sample_count,
        "residual_indices_sha256": tensor_sha256(residual_indices),
        "preprojection_magnitude": _summary(magnitudes),
        "projection_invalid_coordinate_count": invalid_coordinate_count,
        "projection_invalid_sample_count": invalid_sample_count,
        "complete_residual_vectors": True,
        "heldout_residuals": False,
        "no_retry": True,
        "best_of_n": False,
    }
    return canonical, q_new, residual_indices, magnitudes, seed_record

def _residual_statistics(model_indices: list[torch.Tensor], calibration_indices: list[torch.Tensor]) -> dict[str, Any]:
    all_indices = torch.cat([item.reshape(-1) for item in [*model_indices, *calibration_indices]]).to(dtype=torch.int64)
    _require(all_indices.numel() > 0 and all_indices.ge(0).all().item() and all_indices.lt(FIT_COUNT).all().item(), "residual sampling indices are invalid")
    counts = torch.bincount(all_indices, minlength=FIT_COUNT).to(dtype=torch.int64)
    used = counts.gt(0)
    total = int(all_indices.numel())
    unique = int(used.sum().item())
    shares = counts[used].to(torch.float64).div(float(total))
    sorted_counts, _ = counts.sort(descending=True)
    return {
        "sampling_unit": "complete 100-D residual vector",
        "generated_rows": total,
        "unique_residual_vectors": unique,
        "unique_fraction": float(unique / total),
        "reuse_rate": float(1.0 - unique / total),
        "maximum_residual_share": float(shares.max()),
        "top_10_residual_share": float(sorted_counts[:10].sum().item() / total),
        "herfindahl_index": float(shares.square().sum()),
        "model_bank_index_tensor_sha256": tensor_sha256(torch.stack(model_indices)),
        "calibration_index_tensor_sha256": tensor_sha256(torch.stack(calibration_indices)),
    }


def _run_fold(
    fold: int,
    q: torch.Tensor,
    z: torch.Tensor,
    phase: torch.Tensor,
    identities: list[str],
    fit_positions: torch.Tensor,
    heldout_positions: torch.Tensor,
) -> tuple[dict[str, Any], dict[str, torch.Tensor]]:
    _require(fit_positions.numel() == FIT_COUNT and heldout_positions.numel() == HELDOUT_COUNT, f"fold {fold} sizes changed", blocked=True)
    _require(not bool(torch.isin(fit_positions, heldout_positions).any()), f"fold {fold} fit/heldout overlap", blocked=True)
    fit_q = q.index_select(0, fit_positions).contiguous()
    heldout_z = z.index_select(0, heldout_positions).contiguous()
    fit_phase = phase.index_select(0, fit_positions).contiguous()
    _require(len({identities[int(position)] for position in fit_positions.tolist()}) == FIT_COUNT, f"fold {fold} identities are not unique", blocked=True)
    q_fit_model = prior_execution._fit_candidate_fold(fit_q)
    phase_model = _fit_complex_phasor_model(q_fit_model["z"], fit_phase, fold)

    model_banks: list[torch.Tensor] = []
    model_embeddings: list[torch.Tensor] = []
    model_residual_indices: list[torch.Tensor] = []
    model_preprojection: list[torch.Tensor] = []
    model_seed_records: list[dict[str, Any]] = []
    for bank in range(MODEL_BANK_COUNT):
        generated, _, residual_indices, magnitudes, seed_record = _compose_candidate(
            q_fit_model, phase_model, MODEL_BANK_SIZE, fold, None, bank, "CANDIDATE_MODEL_BANK"
        )
        model_banks.append(generated)
        model_embeddings.append(complex_definition.complex_real_embedding(generated))
        model_residual_indices.append(residual_indices)
        model_preprojection.append(_bounded_diagnostic_sample(magnitudes))
        model_seed_records.append({"bank": bank, **seed_record})

    calibration_values: list[float] = []
    calibration_residual_indices: list[torch.Tensor] = []
    calibration_preprojection: list[torch.Tensor] = []
    calibration_seed_records: list[dict[str, Any]] = []
    started = time.perf_counter()
    for trial in range(CALIBRATION_TRIAL_COUNT):
        pseudo, _, residual_indices, magnitudes, seed_record = _compose_candidate(
            q_fit_model, phase_model, HELDOUT_COUNT, fold, trial, None, "CANDIDATE_PSEUDO_DATA"
        )
        pseudo_embedding = complex_definition.complex_real_embedding(pseudo)
        value, _ = _primary_statistic(pseudo_embedding, model_embeddings)
        calibration_values.append(value)
        calibration_residual_indices.append(residual_indices)
        calibration_preprojection.append(_bounded_diagnostic_sample(magnitudes))
        calibration_seed_records.append({"trial": trial, **seed_record})
        if trial == 0 or trial + 1 == CALIBRATION_TRIAL_COUNT or (trial + 1) % 100 == 0:
            print(json.dumps({"stage": "candidate_model_self_calibration", "fold": fold, "completed_trials": trial + 1, "total_trials": CALIBRATION_TRIAL_COUNT, "elapsed_seconds": round(time.perf_counter() - started, 3)}), flush=True)

    calibration_tensor = torch.tensor(calibration_values, dtype=torch.float64).contiguous()
    _require(calibration_tensor.shape == (CALIBRATION_TRIAL_COUNT,) and torch.isfinite(calibration_tensor).all().item() and calibration_tensor.ge(0.0).all().item(), f"fold {fold} calibration is invalid")
    heldout_embedding = complex_definition.complex_real_embedding(heldout_z)
    observed_t, per_bank_observed = _primary_statistic(heldout_embedding, model_embeddings)
    exceedance = int(calibration_tensor.ge(observed_t).sum().item())
    p_mc = float((1 + exceedance) / (CALIBRATION_TRIAL_COUNT + 1))
    _require(0.0 <= p_mc <= 1.0 and torch.isfinite(torch.tensor(p_mc, dtype=torch.float64)).item(), f"fold {fold} p_MC is invalid")
    descriptive = _aggregate_diagnostics([_descriptive_pair_metrics(heldout_z, bank) for bank in model_banks])
    residual_stats = _residual_statistics(model_residual_indices, calibration_residual_indices)
    preprojection_all = torch.cat([*model_preprojection, *calibration_preprojection]).reshape(-1)
    fold_result = {
        "fold": fold,
        "operationally_valid": True,
        "fit_rows": FIT_COUNT,
        "fit_rows_per_class": FIT_PER_CLASS,
        "heldout_rows": HELDOUT_COUNT,
        "heldout_rows_per_class": HELDOUT_PER_CLASS,
        "fit_row_positions_sha256": tensor_sha256(fit_positions),
        "heldout_row_positions_sha256": tensor_sha256(heldout_positions),
        "fit_heldout_disjoint": True,
        "fit_source_scope": "fitting rows only",
        "heldout_source_scope": "heldout complex rows only for final T_candidate",
        "q_producer_fit": prior_execution._fit_summary_for_v2(q_fit_model, fold),
        "phase_model_fit": phase_model["fit_summary"],
        "model_self_calibration": {
            "performed_before_heldout_statistic": True,
            "B": CALIBRATION_TRIAL_COUNT,
            "pseudo_data_rows": HELDOUT_COUNT,
            "model_bank_count": MODEL_BANK_COUNT,
            "model_bank_size": MODEL_BANK_SIZE,
            "calibration_tensor_sha256": tensor_sha256(calibration_tensor),
            "trial_indices_inclusive": [0, CALIBRATION_TRIAL_COUNT - 1],
            "model_bank_sampling": model_seed_records,
            "calibration_sampling": calibration_seed_records,
            "heldout_used_for_sampling": False,
            "calibration_validity": {
                "valid": True,
                "complete_trial_count": CALIBRATION_TRIAL_COUNT,
                "finite_nonnegative": True,
                "same_candidate_for_pseudo_and_model": True,
                "no_retry": True,
                "no_early_stop": True,
            },
        },
        "primary": {
            "name": PRIMARY_METRIC,
            "role": "HARD_PRIMARY",
            "T_candidate": observed_t,
            "T_observed": observed_t,
            "per_model_bank_T_observed": per_bank_observed,
            "mc_exceedance": exceedance,
            "exceedance": exceedance,
            "p_MC": p_mc,
            "alpha": PER_FOLD_ALPHA,
            "pass": p_mc >= PER_FOLD_ALPHA,
            "inequality": "p_MC>=0.01",
        },
        "residual_statistics": residual_stats,
        "projection_diagnostics": {
            "projection_invalid_coordinate_count": 0,
            "projection_invalid_sample_count": 0,
            "preprojection_magnitude": _summary(preprojection_all),
            "preprojection_summary_scope": "deterministic evenly spaced bounded diagnostic sample from each generated batch; descriptive-only",
            "preprojection_summary_max_values_per_generated_batch": 256,
        },
        "diagnostics": descriptive,
        "generated_model_bank_tensor_sha256": tensor_sha256(torch.stack(model_banks)),
        "calibration_tensor_sha256": tensor_sha256(calibration_tensor),
    }
    tensors = {
        "model_banks": torch.stack(model_banks).contiguous(),
        "calibration": calibration_tensor,
        "model_residual_indices": torch.stack(model_residual_indices).contiguous(),
        "calibration_residual_indices": torch.stack(calibration_residual_indices).contiguous(),
        "phase_coefficients": phase_model["coefficients"].contiguous(),
    }
    return fold_result, tensors

def execute(repo_root: Path, output_dir: Path = OUTPUT_DIR) -> dict[str, Any]:
    repo_root = repo_root.resolve()
    output_dir = output_dir.resolve() if output_dir.is_absolute() else (repo_root / output_dir).resolve()
    _require(not output_dir.exists(), f"refusing to reuse write-once output: {output_dir}", blocked=True)
    git_before = _git_state(repo_root)
    _require(git_before["branch"] == EXPECTED_BRANCH, "current branch is not the authority branch", blocked=True)
    _require(git_before["head"] == EXPECTED_STARTING_HEAD, "current committed HEAD is not the frozen authority base", blocked=True)
    contract, authorities = _validate_contract(repo_root)

    parser.add_argument("--output-dir", type=Path, default=OUTPUT_DIR)
    args = parser.parse_args()
    try:
        result = execute(args.repo_root, args.output_dir)
    except BlockedExecution as exc:
        print(json.dumps({"result": "BLOCKED", "classification": None, "reason": str(exc)}, indent=2, sort_keys=True))
        return 3
    except InvalidHarnessError as exc:
        print(json.dumps({"result": "INVALID", "classification": "INVALID_HARNESS", "reason": str(exc)}, indent=2, sort_keys=True))
        return 2

    return 0
