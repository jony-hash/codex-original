class ContractDefinitionBlocked(RuntimeError):
    """The requested family cannot be totally specified without a new choice."""


def _require(condition: bool, message: str) -> None:
    if not condition:
        raise ValueError(message)


def _mapping(value: Any, name: str) -> Mapping[str, Any]:
    _require(isinstance(value, Mapping), f"{name} must be a mapping")
    return value


def _relative(repo_root: Path, path: Path) -> str:
    return path.resolve().relative_to(repo_root.resolve()).as_posix()


def _read_json_exact(
    repo_root: Path,
    path: Path,
    expected_sha256: str,
    name: str,
) -> dict[str, Any]:
    target = repo_root / path
    _require(target.is_file(), f"{name} is missing: {target}")
    _require(file_sha256(target) == expected_sha256, f"{name} hash changed: {target}")
    return dict(_mapping(json.loads(target.read_text(encoding="utf-8")), name))


def _git_state(repo_root: Path) -> dict[str, Any]:
    branch = subprocess.run(
        ["git", "-C", str(repo_root), "branch", "--show-current"],
        check=True,
        capture_output=True,
        text=True,
    ).stdout.strip()
    head = subprocess.run(
        ["git", "-C", str(repo_root), "rev-parse", "HEAD"],
        check=True,
        capture_output=True,
        text=True,
    ).stdout.strip()
    status = subprocess.run(
        ["git", "-C", str(repo_root), "status", "--porcelain"],
        check=True,
        capture_output=True,
        text=True,
    ).stdout.splitlines()
    return {
        "branch": branch,
        "head": head,
        "dirty_paths": [line[3:] if len(line) >= 4 else line for line in status],
        "worktree_clean": not status,
    }


def _authority_ref(repo_root: Path, path: Path, sha256: str, role: str) -> dict[str, Any]:
    target = repo_root / path
    _require(target.is_file(), f"{role} is missing: {target}")
    _require(file_sha256(target) == sha256, f"{role} hash changed: {target}")
    return {"path": _relative(repo_root, target), "sha256": sha256, "role": role}

def golden_section_maximize(
    objective: Callable[[float], float],
    lower: float,
    upper: float,
    *,
    abs_tolerance: float = OPTIMIZER_ABS_TOLERANCE,
    rel_tolerance: float = OPTIMIZER_REL_TOLERANCE,
    max_iterations: int = OPTIMIZER_MAX_ITERATIONS,
) -> dict[str, Any]:
    """Reference the frozen deterministic bounded 1-D maximizer on a scalar domain."""

    _require(math.isfinite(lower) and math.isfinite(upper) and lower < upper, "optimizer domain is invalid")
    _require(abs_tolerance > 0.0 and rel_tolerance > 0.0 and max_iterations > 0, "optimizer controls are invalid")

    def evaluate(x: float) -> float:
        value = float(objective(float(x)))
        if not math.isfinite(value):
            raise ValueError("optimizer objective is non-finite")
        return value

    golden = (math.sqrt(5.0) - 1.0) / 2.0
    left = float(lower)
    right = float(upper)
    endpoint_values = {left: evaluate(left), right: evaluate(right)}
    c = right - golden * (right - left)
    d = left + golden * (right - left)
    fc = evaluate(c)
    fd = evaluate(d)
    evaluations = 4
    iterations = 0

    def tolerance(a: float, b: float) -> float:
        return max(abs_tolerance, rel_tolerance * max(1.0, abs(a), abs(b)))

    def right_is_better(value_right: float, value_left: float, x_right: float, x_left: float) -> bool:
        tie = OBJECTIVE_TIE_TOLERANCE * max(1.0, abs(value_right), abs(value_left))
        if value_right > value_left + tie:
            return True
        if abs(value_right - value_left) <= tie:
            return x_right > x_left
        return False

    while right - left > tolerance(left, right):
        if iterations >= max_iterations:
            raise ValueError("bounded optimizer did not converge within the frozen iteration cap")
        iterations += 1
        if right_is_better(fd, fc, d, c):
            left = c
            c = d
            fc = fd
            d = left + golden * (right - left)
            fd = evaluate(d)
        else:
            right = d
            d = c
            fd = fc
            c = right - golden * (right - left)
            fc = evaluate(c)
        evaluations += 1

    midpoint = (left + right) / 2.0
    candidates = [(left, evaluate(left)), (midpoint, evaluate(midpoint)), (right, evaluate(right))]
    evaluations += 3
    candidates.extend(endpoint_values.items())
    best_x, best_value = candidates[0]
    for candidate_x, candidate_value in candidates[1:]:
        if right_is_better(candidate_value, best_value, float(candidate_x), float(best_x)):
            best_x, best_value = float(candidate_x), float(candidate_value)
    return {
        "x": float(best_x),
        "objective": float(best_value),
        "iterations": iterations,
        "evaluations": evaluations,
        "boundary_hit": bool(best_x == lower or best_x == upper),
    }


def _run_contract_totality_checks() -> dict[str, Any]:
    if not (math.isfinite(NU_LOWER) and math.isfinite(NU_UPPER) and 2.0 < NU_LOWER < NU_UPPER):
        raise ContractDefinitionBlocked("Student-t nu domain is not finite, bounded, and covariance-safe")
    if not (math.isfinite(ETA_LOWER) and math.isfinite(ETA_UPPER) and ETA_LOWER < ETA_UPPER):
        raise ContractDefinitionBlocked("log(nu-2) optimizer domain is not finite and ordered")
    if T_CDF_MAX_ITERATIONS <= 0 or T_QUANTILE_MAX_ITERATIONS <= 0:
        raise ContractDefinitionBlocked("Student-t CDF/quantile iteration bounds are not fixed")
    if not (T_CDF_TOLERANCE > 0.0 and T_QUANTILE_ABS_TOLERANCE > 0.0 and T_QUANTILE_REL_TOLERANCE > 0.0):
        raise ContractDefinitionBlocked("Student-t numerical tolerances are not fixed")

    def synthetic_objective(x: float) -> float:
        return -((x - 0.25) ** 2)

    first = golden_section_maximize(synthetic_objective, -1.0, 1.0)
    second = golden_section_maximize(synthetic_objective, -1.0, 1.0)
    _require(first == second, "bounded optimizer is not deterministic on the synthetic contract check")
    _require(abs(first["x"] - 0.25) < 1.0e-5, "bounded optimizer failed the synthetic interior maximum check")
    tie = golden_section_maximize(lambda _x: 1.0, -1.0, 1.0)
    _require(tie["boundary_hit"] is True and tie["x"] == 1.0, "optimizer tie/boundary rule is not deterministic")

    return {
        "status": "passed",
        "future_runtime_math_choices": [],
        "validated_domains": {
            "nu_lower": NU_LOWER,
            "nu_upper": NU_UPPER,

            "eta_upper": ETA_UPPER,
            "cdf_iterations": T_CDF_MAX_ITERATIONS,
            "quantile_iterations": T_QUANTILE_MAX_ITERATIONS,
            "optimizer_iterations": OPTIMIZER_MAX_ITERATIONS,
        },
        "synthetic_optimizer_checks": {
            "interior_maximum_deterministic": True,
            "boundary_tie_prefers_larger_nu": True,
            "no_development_q_used": True,

            "one_new_fitted_scalar_per_fold": True,

            "sampler_defined": True,

            "failure_rules_defined": True,
        },
    }

def _write_contract(repo_root: Path, output_dir: Path, payload: Mapping[str, Any]) -> Path:
    if output_dir.exists():
        raise FileExistsError(f"refusing to reuse write-once contract output: {output_dir}")
    output_dir.parent.mkdir(parents=True, exist_ok=True)
    temporary = Path(tempfile.mkdtemp(prefix=f".{output_dir.name}.", dir=output_dir.parent))
    try:
        atomic_write_new_json(
            temporary / CONTRACT_NAME,
            payload,
            artifact_kind="N3 fixed-R Student-t copula contract",
        )
        if output_dir.exists():
            raise FileExistsError(f"contract output appeared during definition: {output_dir}")
        os.replace(temporary, output_dir)
    finally:
        if temporary.exists():
            shutil.rmtree(temporary)
    return output_dir / CONTRACT_NAME


def execute(repo_root: Path, output_dir: Path = OUTPUT_DIR) -> dict[str, Any]:
    git = _git_state(repo_root)
    _require(git["branch"] == EXPECTED_BRANCH, "current branch is not the N3 authority branch")
    _require(git["head"] == EXPECTED_STARTING_HEAD, "starting committed HEAD changed")
    _require(not output_dir.exists(), f"write-once contract output already exists: {output_dir}")

    v2 = _validate_v2_authority(repo_root)
    n4 = _validate_n4_authority(repo_root)
    governance = _validate_current_governance(repo_root)
    totality = _run_contract_totality_checks()
    payload = _build_payload(repo_root, git, v2, n4, governance, totality)
    contract_path = _write_contract(repo_root, output_dir, payload)
    ending_git = _git_state(repo_root)
    return {
        "result": "READY",
        "classification": READY_CLASSIFICATION,
        "claim_classification": CLAIM_CLASSIFICATION,
        "semantic_authority": TASK_ID,
        "family_id": FAMILY_ID,
        "contract": _relative(repo_root, contract_path),
        "contract_sha256": file_sha256(contract_path),
        "git_branch": git["branch"],
        "git_starting_head": git["head"],
        "git_ending_head": ending_git["head"],
        "new_candidate_results_observed": False,
        "q_tensor_deserialized": False,
        "heldout_evaluation_executed": False,
        "sealed_confirmation_consumed": False,
        "selected_producer": None,
        "training_authorized": False,
        "formal_promotion": False,
        "generation_readiness": "not_ready",
    }


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--repo-root", type=Path, default=Path(__file__).resolve().parents[4])
    parser.add_argument("--output-dir", type=Path, default=OUTPUT_DIR)
    args = parser.parse_args()
    repo_root = args.repo_root.resolve()
    output_dir = args.output_dir if args.output_dir.is_absolute() else repo_root / args.output_dir
    try:
        result = execute(repo_root, output_dir)
    except ContractDefinitionBlocked as exc:
        result = {
            "result": "BLOCKED",
            "classification": None,
            "semantic_authority": TASK_ID,
            "blocker": str(exc),
            "q_tensor_deserialized": False,
            "heldout_evaluation_executed": False,
            "sealed_confirmation_consumed": False,
        }
    print(json.dumps(result, indent=2, sort_keys=True))


if __name__ == "__main__":
    main()
