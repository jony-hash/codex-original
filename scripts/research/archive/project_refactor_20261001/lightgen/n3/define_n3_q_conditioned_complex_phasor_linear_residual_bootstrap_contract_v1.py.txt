def _git_state(repo_root: Path) -> dict[str, Any]:
    def _run(*args: str) -> str:
        result = subprocess.run(
            ["git", "-C", str(repo_root), *args],
            check=True,
            capture_output=True,
            text=True,
        )
        return result.stdout.strip()

    status_result = subprocess.run(
        ["git", "-C", str(repo_root), "status", "--porcelain"],
        check=True,
        capture_output=True,
        text=True,
    )
    status = status_result.stdout.splitlines()
    return {
        "branch": _run("branch", "--show-current"),
        "head": _run("rev-parse", "HEAD"),
        "dirty_paths": [line[3:] if len(line) >= 4 else line for line in status],
        "worktree_clean": not status,
    }

def _validate_float_matrix(value: Any, name: str) -> torch.Tensor:
    _require(torch.is_tensor(value), f"{name} must be a tensor")
    _require(value.ndim == 2 and value.shape[1] == DIMENSION, f"{name} must have shape [N,{DIMENSION}]")
    view = value.detach().to(device="cpu", dtype=torch.float64).contiguous()
    _require(torch.isfinite(view).all().item(), f"{name} is nonfinite")
    return view


def _validate_complex_matrix(value: Any, name: str) -> torch.Tensor:
    _require(torch.is_tensor(value), f"{name} must be a tensor")
    _require(value.ndim == 2 and value.shape[1] == DIMENSION, f"{name} must have shape [N,{DIMENSION}]")
    view = value.detach().to(device="cpu", dtype=torch.complex128).contiguous()
    _require(torch.isfinite(view.real).all().item() and torch.isfinite(view.imag).all().item(), f"{name} is nonfinite")
    return view


def _fit_complex_phasor_linear_ols(x: torch.Tensor, target: torch.Tensor) -> tuple[torch.Tensor, torch.Tensor, int]:
    """Frozen future fitting rule, exercised only with synthetic probes here."""

    x_view = _validate_float_matrix(x, "Gaussianized q coordinates")
    target_view = _validate_complex_matrix(target, "complex unit-phasor target")
    _require(x_view.shape[0] == target_view.shape[0] and x_view.shape[0] >= DESIGN_FEATURES, "fit rows are insufficient")
    ones = torch.ones((x_view.shape[0], 1), dtype=torch.float64, device="cpu")
    design = torch.cat((ones, x_view), dim=1).contiguous()
    _require(design.shape[1] == DESIGN_FEATURES, "conditional design width changed")

    left, singular_values, right_transpose = torch.linalg.svd(design, full_matrices=False)
    _require(
        torch.isfinite(left).all().item()
        and torch.isfinite(singular_values).all().item()
        and torch.isfinite(right_transpose).all().item(),
        "OLS SVD returned nonfinite values",
    )
    _require(singular_values.numel() == DESIGN_FEATURES and singular_values[0] > 0.0, "OLS SVD has no positive leading singular value")
    cutoff = OLS_RCOND * singular_values[0]
    rank = int((singular_values > cutoff).sum().item())
    _require(rank == DESIGN_FEATURES, "HARD_STOP: frozen OLS design is rank-deficient")

    projected_target = left.conj().transpose(-2, -1).to(torch.complex128).matmul(target_view)
    coefficients = right_transpose.conj().transpose(-2, -1).to(torch.complex128).matmul(
        projected_target / singular_values.to(torch.complex128).unsqueeze(-1)
    )
    prediction = design.to(torch.complex128).matmul(coefficients)
    _require(torch.isfinite(coefficients.real).all().item() and torch.isfinite(coefficients.imag).all().item(), "OLS coefficients are nonfinite")
    _require(torch.isfinite(prediction.real).all().item() and torch.isfinite(prediction.imag).all().item(), "OLS prediction is nonfinite")
    return coefficients.contiguous(), prediction.contiguous(), rank


def _project_unit_phasors(value: torch.Tensor) -> torch.Tensor:
    value_view = _validate_complex_matrix(value, "pre-projection complex phasors")
    magnitude = value_view.abs()
    _require(
        torch.isfinite(magnitude).all().item() and bool((magnitude > PHASOR_PROJECTION_EPSILON).all()),
        "INVALID_HARNESS: generated phasor coordinate is nonfinite or below projection epsilon",
    )
    projected = value_view / magnitude
    _require(
        torch.isfinite(projected.real).all().item() and torch.isfinite(projected.imag).all().item(),
        "unit-phasor projection returned nonfinite values",
    )
    return projected.contiguous()


def derive_seed(role: str, fold: int, trial: int | None, bank: int | None) -> int:
    _require(role in {
        "CANDIDATE_Q_SAMPLE",
        "CANDIDATE_COMPLETE_RESIDUAL_SAMPLE",
        "CANDIDATE_MODEL_BANK",
        "CANDIDATE_PSEUDO_DATA",
    }, "unknown candidate RNG role")
    _require(0 <= fold < FOLD_COUNT, "seed fold is invalid")
    token_trial = "none" if trial is None else str(trial)
    token_bank = "none" if bank is None else str(bank)
    material = f"{RNG_NAMESPACE}|role={role}|fold={fold}|trial={token_trial}|bank={token_bank}"
    return int.from_bytes(hashlib.sha256(material.encode("utf-8")).digest()[:8], "big") & MAX_SIGNED_63


def _contract_totality_checks() -> dict[str, Any]:
    _require(DIMENSION == 100 and DESIGN_FEATURES == 101 and FOLD_COUNT == 5, "contract dimensions changed")
    _require(FIT_ROWS_PER_FOLD == 4000 and HELDOUT_ROWS_PER_FOLD == 1000, "fold row counts changed")
    _require(MODEL_BANK_COUNT == 4 and MODEL_BANK_SIZE == 200, "model-bank dimensions changed")
    _require(CALIBRATION_TRIAL_COUNT == 999, "calibration trials changed")
    _require(FAMILY_ALPHA == 0.05 and PER_FOLD_ALPHA == 0.01, "alpha values changed")
    _require(OLS_RCOND == 1.0e-12 and PHASOR_PROJECTION_EPSILON == 1.0e-12, "numerical thresholds changed")

    sorted_values = torch.linspace(0.1, 1.0, DIMENSION, dtype=torch.float64).reshape(1, -1).repeat(4, 1)
    sorted_values[1, 5] = sorted_values[0, 5]
    sorted_values[2, 5] = sorted_values[0, 5]
    sorted_values[:, 6:] = torch.maximum(sorted_values[:, 6:], sorted_values[:, 5:6] + 0.1)
    query = sorted_values[0:1].clone()
    query[0, 0] = 0.0
    query[0, 1] = 2.0
    query[0, 2] = (sorted_values[0, 2] + sorted_values[3, 2]) / 2.0
    coordinates = prior_contract._gaussianized_q_coordinates(query, sorted_values)
    _require(coordinates.shape == (1, DIMENSION) and coordinates.dtype == torch.float64, "Gaussianized q probe changed")

    x_probe = torch.zeros((DESIGN_FEATURES, DIMENSION), dtype=torch.float64)
    x_probe[1:, :] = torch.eye(DIMENSION, dtype=torch.float64)
    target_probe = torch.ones((DESIGN_FEATURES, DIMENSION), dtype=torch.complex128)
    coefficients, prediction, rank = _fit_complex_phasor_linear_ols(x_probe, target_probe)
    _require(coefficients.shape == (DESIGN_FEATURES, DIMENSION), "linear coefficient shape changed")
    _require(prediction.shape == target_probe.shape and rank == DESIGN_FEATURES, "linear OLS probe changed")

    unit_probe = _project_unit_phasors(torch.ones((1, DIMENSION), dtype=torch.complex128))
    _require(unit_probe.shape == (1, DIMENSION), "unit-phasor projection shape changed")
    try:
        _project_unit_phasors(torch.zeros((1, DIMENSION), dtype=torch.complex128))
    except InvalidHarnessError:
        zero_projection_rejection = True
    else:
        zero_projection_rejection = False
    _require(zero_projection_rejection, "zero phasor projection is not fail-closed")
    try:
        _fit_complex_phasor_linear_ols(
            torch.zeros((DESIGN_FEATURES, DIMENSION), dtype=torch.float64),
            target_probe,
        )
    except InvalidHarnessError:
        rank_deficiency_rejection = True
    else:
        rank_deficiency_rejection = False
    _require(rank_deficiency_rejection, "rank-deficient OLS design is not a hard stop")

    seed_values = {
        derive_seed(role, fold, trial, bank)
        for fold in range(FOLD_COUNT)
        for role, trial, bank in (
            ("CANDIDATE_Q_SAMPLE", None, 0),
            ("CANDIDATE_COMPLETE_RESIDUAL_SAMPLE", 0, None),
            ("CANDIDATE_MODEL_BANK", None, 0),
            ("CANDIDATE_PSEUDO_DATA", 0, None),
        )
    }
    _require(len(seed_values) == FOLD_COUNT * 4, "candidate RNG namespaces are not disjoint")
    _require(all(0 <= seed <= MAX_SIGNED_63 for seed in seed_values), "candidate RNG seed is outside signed 63-bit range")

    return {
        "status": "passed",
        "candidate_id": CANDIDATE_ID,
        "conditional_mean_probe": "passed",
        "solver": "CPU float64 thin SVD pseudoinverse with explicit rcond",
        "rank_deficiency_probe": "passed",
        "zero_or_below_epsilon_projection_probe": "passed",
        "q_values_loaded": False,
        "complex_ols_values_loaded": False,
        "sealed_values_loaded": False,
        "scientific_evaluation_executed": False,
        "training_executed": False,
        "cuda_used": False,
        "governance_mutated": False,
    }

def execute(repo_root: Path, output_dir: Path = OUTPUT_DIR) -> dict[str, Any]:
    _require(not output_dir.exists(), f"refusing to reuse write-once contract output: {output_dir}")
    git = _git_state(repo_root)
    _require(git["branch"] == EXPECTED_BRANCH, "current branch is not the authority branch")
    _require(git["head"] == EXPECTED_STARTING_HEAD, "current committed HEAD is not the frozen authority base")
    authorities = _validate_authorities(repo_root)
    totality = _contract_totality_checks()
    governance_before = file_sha256(repo_root / GOVERNANCE_STATE_PATH)
    _require(governance_before == authorities["current"]["governance"]["sha256"], "governance changed during preflight")
    payload = _build_payload(repo_root, git, authorities, totality)

    output_dir.parent.mkdir(parents=True, exist_ok=True)
    temporary = Path(tempfile.mkdtemp(prefix=f".{output_dir.name}.", dir=output_dir.parent))
    try:
        atomic_write_new_json(
            temporary / CONTRACT_NAME,
            payload,
            artifact_kind="N3 higher-expressivity q-conditioned phase contract",
        )
        if output_dir.exists():
            raise FileExistsError(f"contract output appeared during definition: {output_dir}")
        os.replace(temporary, output_dir)
    finally:
        if temporary.exists():
            shutil.rmtree(temporary)

    git_after = _git_state(repo_root)
    governance_after = file_sha256(repo_root / GOVERNANCE_STATE_PATH)
    _require(git_after["branch"] == git["branch"] and git_after["head"] == git["head"], "branch or committed HEAD changed")
    _require(governance_after == governance_before, "governance changed during definition")
    contract_path = output_dir / CONTRACT_NAME
    return {
        "status": "valid_completed",
        "result": "READY",
        "classification": READY_CLASSIFICATION,
        "semantic_authority": "N3_Q_CONDITIONED_COMPLEX_PHASOR_LINEAR_RESIDUAL_BOOTSTRAP_CONTRACT_V1",
        "candidate_id": CANDIDATE_ID,
        "contract": _relative(repo_root, contract_path),
        "contract_sha256": file_sha256(contract_path),
        "previous_control": "FULL_COMPLEX_ROW_EMPIRICAL_BOOTSTRAP_CONTROL=5/5 reused",
        "minimal_phase_donor": "Q_CONDITIONED_NEAREST_PHASE_DONOR_V1=0/5",
        "q_values_loaded": False,
        "complex_ols_values_loaded": False,
        "sealed_values_loaded": False,
        "scientific_evaluation_executed": False,
        "training_authorized": False,
        "formal_promotion": False,
        "generation_readiness": "not_ready",
        "governance_unchanged": True,
        "next_action": NEXT_ACTION,
    }


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--repo-root", type=Path, default=Path(__file__).resolve().parents[4])
    parser.add_argument("--output-dir", type=Path, default=OUTPUT_DIR)
    args = parser.parse_args()
    repo_root = args.repo_root.resolve()
    output_dir = args.output_dir if args.output_dir.is_absolute() else repo_root / args.output_dir
    try:
        print(json.dumps(execute(repo_root, output_dir), indent=2, sort_keys=True))
        return 0
    except (InvalidHarnessError, prior_contract.InvalidHarnessError) as exc:
        print(json.dumps({"status": "invalid", "result": "INVALID_HARNESS", "reason": str(exc)}, indent=2, sort_keys=True))
        return 2


if __name__ == "__main__":
    raise SystemExit(main())
