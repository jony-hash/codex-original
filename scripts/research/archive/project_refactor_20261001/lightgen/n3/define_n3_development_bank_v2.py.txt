def _require(condition: bool, message: str) -> None:
    if not condition:
        raise ValueError(message)


def _mapping(value: Any, name: str) -> Mapping[str, Any]:
    _require(isinstance(value, Mapping), f"{name} must be a mapping")
    return value


def _relative(repo_root: Path, path: Path) -> str:
    return path.resolve().relative_to(repo_root.resolve()).as_posix()


def _read_json_exact(repo_root: Path, path: Path, expected_sha256: str, name: str) -> dict[str, Any]:
    target = repo_root / path
    _require(target.is_file(), f"{name} is missing: {target}")
    _require(file_sha256(target) == expected_sha256, f"{name} hash changed: {target}")
    value = json.loads(target.read_text(encoding="utf-8"))
    return dict(_mapping(value, name))


def _git_state(repo_root: Path) -> dict[str, Any]:
    branch = subprocess.run(
        ["git", "-C", str(repo_root), "branch", "--show-current"],
        check=True,
        capture_output=True,
        text=True,
    ).stdout.strip()
    head = subprocess.run(
        ["git", "-C", str(repo_root), "rev-parse", "HEAD"],
        check=True,
        capture_output=True,
        text=True,
    ).stdout.strip()
    status = subprocess.run(
        ["git", "-C", str(repo_root), "status", "--porcelain"],
        check=True,
        capture_output=True,
        text=True,
    ).stdout.splitlines()
    return {
        "branch": branch,
        "head": head,
        "dirty_paths": [line[3:] if len(line) >= 4 else line for line in status],
        "worktree_clean": not status,
    }


def _load_source_labels(repo_root: Path) -> torch.Tensor:
    processed = repo_root / SOURCE_LABELS_PATH
    raw_path = repo_root / SOURCE_LABELS_RAW_PATH
    _require(file_sha256(processed) == SOURCE_LABELS_FILE_SHA256, "processed MNIST source artifact changed")
    _require(file_sha256(raw_path) == SOURCE_LABELS_RAW_SHA256, "raw MNIST label artifact changed")
    raw = raw_path.read_bytes()
    _require(len(raw) >= 8, "raw MNIST label header is incomplete")
    magic, count = struct.unpack(">II", raw[:8])
    _require(magic == 2049 and count == SOURCE_COUNT, "raw MNIST label header changed")
    labels = torch.tensor(list(raw[8:]), dtype=torch.int64).contiguous()
    _require(labels.shape == (SOURCE_COUNT,), "MNIST source label count changed")
    _require(tensor_sha256(labels) == SOURCE_LABELS_TENSOR_SHA256, "MNIST source label identity changed")
    _require(bool(torch.equal(torch.unique(labels), torch.arange(CLASS_COUNT))), "MNIST labels are not 0..9")
    return labels


def _int_tensor(values: Any, name: str) -> torch.Tensor:
    tensor = torch.tensor(values, dtype=torch.int64).contiguous()
    _require(tensor.ndim == 1, f"{name} must be one-dimensional")
    _require(torch.unique(tensor).numel() == tensor.numel(), f"{name} contains duplicate identities")
    _require(bool(tensor.ge(0).all()) and bool(tensor.lt(SOURCE_COUNT).all()), f"{name} is outside mnist_train")
    return tensor


def _label_slice(labels: torch.Tensor, indices: torch.Tensor, name: str, expected_index_sha256: str, expected_label_sha256: str) -> dict[str, Any]:
    observed_labels = labels.index_select(0, indices).contiguous()
    _require(tensor_sha256(indices) == expected_index_sha256, f"{name} index authority changed")
    _require(tensor_sha256(observed_labels) == expected_label_sha256, f"{name} label authority changed")
    return {
        "dataset": "mnist_train",
        "count": int(indices.numel()),
        "index_sha256": tensor_sha256(indices),
        "label_sha256": tensor_sha256(observed_labels),
        "class_counts": torch.bincount(observed_labels, minlength=CLASS_COUNT).tolist(),
    }


def _stable_source_row_identity(source_index: int, label: int) -> str:
    material = f"mnist_train|{source_index}|{label}|SOURCE_ROW_IDENTITY_V1".encode("utf-8")
    return hashlib.sha256(material).hexdigest()

def _select_rows(labels: torch.Tensor, excluded_sets: Mapping[str, set[int]]) -> tuple[list[dict[str, Any]], dict[str, Any]]:
    excluded = set().union(*excluded_sets.values())
    rows: list[dict[str, Any]] = []
    coverage: list[dict[str, Any]] = []
    labels_list = [int(value) for value in labels.tolist()]
    for label in range(CLASS_COUNT):
        eligible = [index for index, value in enumerate(labels_list) if value == label and index not in excluded]
        _require(len(eligible) >= ROWS_PER_CLASS, f"class {label} lacks 500 eligible source rows")
        selected_ranks = [((j * (len(eligible) - 1)) // (ROWS_PER_CLASS - 1)) for j in range(ROWS_PER_CLASS)]
        _require(len(set(selected_ranks)) == ROWS_PER_CLASS, f"class {label} spacing produced duplicate ranks")
        selected = [eligible[rank] for rank in selected_ranks]
        _require(selected == sorted(selected), f"class {label} selected rows are not source-index ordered")
        class_rows = []
        for selected_rank, (eligible_rank, source_index) in enumerate(zip(selected_ranks, selected, strict=True)):
            row = {
                "bank_row_position": len(rows),
                "source_index": source_index,
                "label": label,
                "stable_source_row_identity": _stable_source_row_identity(source_index, label),
                "eligible_source_rank": eligible_rank,
                "selected_within_class_rank": selected_rank,
            }
            rows.append(row)
            class_rows.append(row)
        coverage.append({
            "label": label,
            "eligible_count": len(eligible),
            "eligible_source_index_min": eligible[0],
            "eligible_source_index_max": eligible[-1],
            "selected_source_index_min": selected[0],
            "selected_source_index_max": selected[-1],
            "selected_eligible_rank_min": selected_ranks[0],
            "selected_eligible_rank_max": selected_ranks[-1],
            "selected_eligible_rank_sha256": tensor_sha256(torch.tensor(selected_ranks, dtype=torch.int64)),
            "selected_count": len(class_rows),
            "full_eligible_range_endpoints_selected": selected[0] == eligible[0] and selected[-1] == eligible[-1],
            "selection_rule": SELECTION_RULE,
        })
    return rows, {"per_class": coverage, "total_eligible_count": sum(item["eligible_count"] for item in coverage)}


def _semantic_payload(
    rows: list[dict[str, Any]],

    exclusion_payload = {
        name: {
            "count": value["count"],
            "index_sha256": value["index_sha256"],
        }
        for name, value in (
            ("N3_DEVELOPMENT_BANK_V1", source["old_n3"]),
            ("S2_VALIDATION", source["s2_validation"]),
            ("S1_CONFIRMATION", source["s1_confirmation"]),
            ("N3_SEALED_CONFIRMATION_RESERVATION", source["sealed_confirmation_reservation"]),
        )
    }
    return {
        "semantic_authority_id": SEMANTIC_AUTHORITY_ID,
        "dataset": "mnist_train",
        "source_index_range": ELIGIBLE_RANGE,
        "source_labels_tensor_sha256": SOURCE_LABELS_TENSOR_SHA256,
        "rows_per_class": ROWS_PER_CLASS,
        "pool_count": POOL_COUNT,
        "selection_rule": SELECTION_RULE,
        "source_row_identity_formula": SOURCE_ROW_IDENTITY_FORMULA,

def validate_source_manifest_payload(payload: Mapping[str, Any]) -> None:
    _require(payload.get("schema") == SCHEMA, "source manifest schema changed")
    _require(payload.get("version") == 1 and payload.get("task_id") == TASK_ID, "source manifest identity changed")
    _require(payload.get("status") == "FROZEN_SOURCE_ROWS", "source manifest is not frozen")
    _require(payload.get("semantic_authority_id") == SEMANTIC_AUTHORITY_ID, "semantic authority changed")
    source = _mapping(payload.get("source_domain"), "source domain")
    _require(source.get("dataset") == "mnist_train" and source.get("source_index_range") == ELIGIBLE_RANGE, "source domain changed")
    selection = _mapping(payload.get("selection"), "selection")
    _require(selection.get("formula") == SELECTION_RULE and selection.get("q_values_used") is False, "selection rule or q boundary changed")
    _require(selection.get("ols_values_used") is False and selection.get("image_values_used") is False, "selection observed forbidden values")
    rows = payload.get("rows")
    _require(isinstance(rows, list) and len(rows) == POOL_COUNT, "source row manifest is incomplete")
    indices: list[int] = []
    labels: list[int] = []
    identities: list[str] = []
    for position, item in enumerate(rows):
        row = _mapping(item, f"source row {position}")
        _require(row.get("bank_row_position") == position, "source row order changed")
        source_index = row.get("source_index")
        label = row.get("label")
        _require(isinstance(source_index, int) and 0 <= source_index < SOURCE_COUNT, "source index is invalid")
        _require(isinstance(label, int) and 0 <= label < CLASS_COUNT, "source label is invalid")
        _require(row.get("stable_source_row_identity") == _stable_source_row_identity(source_index, label), "stable row identity changed")
        indices.append(source_index)
        labels.append(label)
        identities.append(str(row["stable_source_row_identity"]))
    index_tensor = torch.tensor(indices, dtype=torch.int64)
    label_tensor = torch.tensor(labels, dtype=torch.int64)
    _require(torch.unique(index_tensor).numel() == POOL_COUNT, "source rows are not unique")
    _require(torch.equal(torch.bincount(label_tensor, minlength=CLASS_COUNT), torch.full((CLASS_COUNT,), ROWS_PER_CLASS, dtype=torch.int64)), "source rows are not class balanced")
    identity = _mapping(payload.get("identity"), "source identity")
    _require(identity.get("selected_source_indices_sha256") == tensor_sha256(index_tensor), "selected source hash changed")
    _require(identity.get("selected_labels_sha256") == tensor_sha256(label_tensor), "selected label hash changed")
    _require(identity.get("row_manifest_sha256") == canonical_sha256(rows), "row manifest hash changed")
    _require(identity.get("class_counts") == [ROWS_PER_CLASS] * CLASS_COUNT, "class counts changed")
    _require(identity.get("unique_source_rows") is True, "unique source-row invariant changed")
    _require(len(set(identities)) == POOL_COUNT, "stable source identities are not unique")
    exclusions = _mapping(payload.get("exclusions"), "exclusions")
    overlaps = _mapping(exclusions.get("v2_overlap_counts"), "V2 overlap counts")
    _require(dict(overlaps) == {name: 0 for name in ("N3_DEVELOPMENT_BANK_V1", "S2_VALIDATION", "S1_CONFIRMATION", "N3_SEALED_CONFIRMATION_RESERVATION")}, "source overlap gate changed")
    boundary = _mapping(payload.get("materialization_boundary"), "materialization boundary")
    for key in ("definition_only", "q_values_loaded", "q_values_materialized", "image_values_loaded", "sealed_confirmation_values_loaded", "sealed_confirmation_consumed", "producer_evaluated", "producer_selection", "training_authorized", "formal_promotion"):
        _require(boundary.get(key) is False if key != "definition_only" else boundary.get(key) is True, f"source boundary changed: {key}")
    _require(boundary.get("generation_readiness") == "not_ready", "generation readiness changed")


def _write_artifact(repo_root: Path, output_dir: Path, payload: Mapping[str, Any]) -> Path:
    if output_dir.exists():
        raise FileExistsError(f"refusing to reuse write-once output: {output_dir}")
    output_dir.parent.mkdir(parents=True, exist_ok=True)
    temporary = Path(tempfile.mkdtemp(prefix=f".{output_dir.name}.", dir=output_dir.parent))
    try:
        atomic_write_new_json(temporary / ARTIFACT_NAME, payload, artifact_kind="N3 V2 source manifest")
        if output_dir.exists():
            raise FileExistsError(f"write-once output appeared during definition: {output_dir}")
        os.replace(temporary, output_dir)
    finally:
        if temporary.exists():
            shutil.rmtree(temporary)
    return output_dir / ARTIFACT_NAME


def execute(repo_root: Path, output_dir: Path = OUTPUT_DIR) -> dict[str, Any]:
    git = _git_state(repo_root)
    _require(git["branch"] == EXPECTED_BRANCH, "current branch is not the N3 authority branch")
    _require(git["head"] == EXPECTED_STARTING_HEAD, "starting committed HEAD changed")
    unexpected = [path for path in git["dirty_paths"] if path not in SAFE_DIRTY_PATHS]
    _require(not unexpected, f"unrelated local changes invalidate provenance: {unexpected}")
    _require(not output_dir.exists(), f"write-once output already exists: {output_dir}")
    labels = _load_source_labels(repo_root)
    source = _authority_inputs(repo_root, labels)
    rows, coverage = _select_rows(labels, source["excluded_sets"])
    payload = _build_payload(repo_root, git, labels, source, rows, coverage)
    validate_source_manifest_payload(payload)
    artifact_path = _write_artifact(repo_root, output_dir, payload)
    published = json.loads(artifact_path.read_text(encoding="utf-8"))
    validate_source_manifest_payload(published)
    return {
        "result": "FROZEN_SOURCE_ROWS",
        "semantic_authority_id": SEMANTIC_AUTHORITY_ID,
        "artifact": _relative(repo_root, artifact_path),
        "artifact_sha256": file_sha256(artifact_path),
        "bank_semantic_sha256": published["bank_semantic_sha256"],
        "row_manifest_sha256": published["identity"]["row_manifest_sha256"],
        "selected_source_indices_sha256": published["identity"]["selected_source_indices_sha256"],
        "selected_labels_sha256": published["identity"]["selected_labels_sha256"],
        "eligible_count": published["source_domain"]["eligible_count"],
        "eligible_count_by_class": [item["eligible_count"] for item in published["selection"]["coverage"]["per_class"]],
        "selected_count": POOL_COUNT,
        "selected_per_class": ROWS_PER_CLASS,
        "excluded_source_union": published["exclusions"]["union"],
        "v2_overlap_counts": published["exclusions"]["v2_overlap_counts"],
        "q_values_used": False,
        "image_values_used": False,
        "sealed_confirmation_consumed": False,
        "producer_selection": False,
        "training_authorized": False,
        "formal_promotion": False,
        "generation_readiness": "not_ready",
    }


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--repo-root", type=Path, default=Path(__file__).resolve().parents[4])
    parser.add_argument("--output-dir", type=Path, default=OUTPUT_DIR)
    args = parser.parse_args()
    repo_root = args.repo_root.resolve()
    output_dir = args.output_dir if args.output_dir.is_absolute() else repo_root / args.output_dir
    print(json.dumps(execute(repo_root, output_dir), indent=2, sort_keys=True))


if __name__ == "__main__":
    main()
