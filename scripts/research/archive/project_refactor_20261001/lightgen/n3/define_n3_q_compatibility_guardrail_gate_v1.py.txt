def _require(condition: bool, message: str) -> None:
    if not condition:
        raise ValueError(message)


def _mapping(value: Any, name: str) -> Mapping[str, Any]:
    _require(isinstance(value, Mapping), f"{name} must be a mapping")
    return value


def _relative(repo_root: Path, path: Path) -> str:
    return path.resolve().relative_to(repo_root.resolve()).as_posix()


def _git_state(repo_root: Path) -> dict[str, Any]:
    branch = subprocess.run(
        ["git", "-C", str(repo_root), "branch", "--show-current"],
        check=True,
        capture_output=True,
        text=True,
    ).stdout.strip()
    head = subprocess.run(
        ["git", "-C", str(repo_root), "rev-parse", "HEAD"],
        check=True,
        capture_output=True,
        text=True,
    ).stdout.strip()
    status = subprocess.run(
        ["git", "-C", str(repo_root), "status", "--porcelain"],
        check=True,
        capture_output=True,
        text=True,
    ).stdout.splitlines()
    return {
        "branch": branch,
        "head": head,
        "dirty_paths": [line[3:] if len(line) >= 4 else line for line in status],
        "worktree_clean": not status,
    }


def _read_json(repo_root: Path, path: Path, expected_sha256: str, name: str) -> dict[str, Any]:
    target = repo_root / path
    _require(target.is_file(), f"{name} is missing: {target}")
    _require(file_sha256(target) == expected_sha256, f"{name} hash changed: {target}")
    value = json.loads(target.read_text(encoding="utf-8"))
    return dict(_mapping(value, name))


def _authority_ref(repo_root: Path, path: Path, role: str, expected_sha256: str | None = None) -> dict[str, Any]:
    target = repo_root / path
    _require(target.is_file(), f"authority is missing: {target}")
    actual = file_sha256(target)
    if expected_sha256 is not None:
        _require(actual == expected_sha256, f"authority hash changed: {target}")
    return {"path": _relative(repo_root, target), "sha256": actual, "role": role}

def _run_contract_totality_checks() -> dict[str, Any]:
    _require(len(PRODUCER_IDS) == 2 and len(set(PRODUCER_IDS)) == 2, "producer set is not exactly two and unique")
    _require(TRIAL_COUNT_PER_PRODUCER_FOLD == 200, "trial count is not the permitted CPU fallback")
    _require(PSEUDO_DATA_SAMPLE_COUNT == 1000 and MODEL_SAMPLE_COUNT == 200, "sample sizes changed")
    _require(DATA_VS_MODEL_REPLICATES == 4 and MODEL_VS_MODEL_REPLICATES == 8, "replicate counts changed")
    _require(MODEL_REFERENCE_PAIRS == ((0, 1), (2, 3), (4, 5), (6, 7)), "reference pairs changed")
    _require(len(SCALAR_METRICS) == 8 and SCALAR_METRICS[0] == PRIMARY_METRIC, "metric vector changed")
    _require(tuple(NON_PRIMARY_METRICS) == SCALAR_METRICS[1:], "primary/non-primary partition changed")
    _require(0.0 < COMPATIBILITY_Q < 1.0, "compatibility quantile is invalid")
    _require(0.0 < ALPHA < 1.0 and CONFIDENCE_LEVEL == 0.95, "interval confidence is invalid")
    _require(FALSE_FAIL_UPPER_BOUND == 0.01, "false-fail bound changed")
    _require(FIVE_FOLD_RELIABILITY_TARGET == 0.80, "five-fold reliability target changed")
    _require(all("|" not in value for value in PRODUCER_IDS), "producer namespace is ambiguous")
    return {
        "producer_count": len(PRODUCER_IDS),
        "fold_count": FOLD_COUNT,
        "trial_count_per_producer_fold": TRIAL_COUNT_PER_PRODUCER_FOLD,
        "total_trials": len(PRODUCER_IDS) * FOLD_COUNT * TRIAL_COUNT_PER_PRODUCER_FOLD,
        "metric_count": len(SCALAR_METRICS),
        "non_primary_guardrail_count": len(NON_PRIMARY_METRICS),
        "contract_totality": "passed",
    }

def execute(repo_root: Path, output_dir: Path = OUTPUT_DIR) -> dict[str, Any]:
    if output_dir.exists():
        raise FileExistsError(f"refusing to reuse write-once calibration contract: {output_dir}")
    git = _git_state(repo_root)
    _require(git["branch"] == EXPECTED_BRANCH, "current branch is not the authority branch")
    _require(git["head"] == EXPECTED_STARTING_HEAD, "current HEAD is not the frozen authority base")
    _run_contract_totality_checks()
    current = _validate_current_state(repo_root)
    prior = _validate_prior_evidence(repo_root)
    v2 = _validate_v2_authority(repo_root)
    payload = _build_payload(repo_root, git, current, prior, v2, _run_contract_totality_checks())
    output_dir.parent.mkdir(parents=True, exist_ok=True)
    temporary = Path(tempfile.mkdtemp(prefix=f".{output_dir.name}.", dir=output_dir.parent))
    try:
        atomic_write_new_json(temporary / CONTRACT_NAME, payload, artifact_kind="N3 q compatibility guardrail calibration contract")
        if output_dir.exists():
            raise FileExistsError(f"calibration contract appeared during definition: {output_dir}")
        os.replace(temporary, output_dir)
    finally:
        if temporary.exists():
            shutil.rmtree(temporary)
    path = output_dir / CONTRACT_NAME
    return {
        "status": "valid_completed",
        "result": "READY",
        "classification": READY_CLASSIFICATION,
        "contract": _relative(repo_root, path),
        "contract_sha256": file_sha256(path),
        "trial_count_per_producer_fold": TRIAL_COUNT_PER_PRODUCER_FOLD,
        "total_trials": len(PRODUCER_IDS) * FOLD_COUNT * TRIAL_COUNT_PER_PRODUCER_FOLD,
        "selected_producer": None,
        "training_authorized": False,
        "formal_promotion": False,
        "generation_readiness": "not_ready",
        "sealed_confirmation_consumed": False,
    }


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--output-dir", type=Path, default=OUTPUT_DIR)
    args = parser.parse_args()
    repo_root = Path(__file__).resolve().parents[4]
    output_dir = args.output_dir if args.output_dir.is_absolute() else repo_root / args.output_dir
    print(json.dumps(execute(repo_root, output_dir), indent=2, sort_keys=True))


if __name__ == "__main__":
    main()
