class ProtocolError(RuntimeError):
    """Raised when the execution protocol cannot be uniquely frozen."""


def _require(condition: bool, message: str) -> None:
    if not condition:
        raise ProtocolError(message)


def _mapping(value: object, name: str) -> Mapping[str, Any]:
    _require(isinstance(value, Mapping), f"{name} must be a mapping")
    return value


def _repo_path(root: Path, value: str | Path) -> Path:
    relative = Path(value)
    _require(not relative.is_absolute(), f"{value} must be repository-relative")
    return (root / relative).resolve()


def _relative(root: Path, path: Path) -> str:
    return path.resolve().relative_to(root.resolve()).as_posix()


def _artifact_ref(root: Path, relative: str | Path, expected: str, role: str) -> dict[str, str]:
    target = _repo_path(root, relative)
    _require(target.is_file(), f"{role} is missing: {target}")
    observed = file_sha256(target).lower()
    _require(observed == expected.lower(), f"{role} hash changed: {target}")
    return {"path": _relative(root, target), "sha256": expected.lower(), "role": role}


def _load_json(root: Path, relative: Path, expected_sha: str, name: str) -> dict[str, Any]:
    _artifact_ref(root, relative, expected_sha, name)
    payload = json.loads(_repo_path(root, relative).read_text(encoding="utf-8"))
    return dict(_mapping(payload, name))

def write_protocol(root: Path, output_dir: str | Path = DEFAULT_OUTPUT_DIR) -> Path:
    root = root.resolve()
    target = (root / output_dir if not Path(output_dir).is_absolute() else Path(output_dir)).resolve()
    outputs_root = (root / "outputs").resolve()
    _require(outputs_root in target.parents and target not in {outputs_root, outputs_root / "runs"}, "protocol output must be a distinct child under outputs")
    _require(not target.exists(), f"protocol output already exists: {target}")
    payload = build_protocol(root)
    target.mkdir(parents=True, exist_ok=False)
    path = target / PROTOCOL_NAME
    atomic_write_new_json(path, payload, artifact_kind="N3 low-rank fit protocol")
    return path


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--repo-root", type=Path, default=Path(__file__).resolve().parents[4])
    parser.add_argument("--output-dir", type=Path, default=DEFAULT_OUTPUT_DIR)
    args = parser.parse_args()
    path = write_protocol(args.repo_root, args.output_dir)
    print(json.dumps({"result": "READY", "classification": CLASSIFICATION, "outer_gate": OUTER_GATE, "protocol": path.as_posix(), "candidate_fit_executed": False, "rank_selection_executed": False, "cuda_executed": False}, sort_keys=True))


if __name__ == "__main__":
    main()
