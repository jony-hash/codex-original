class InvalidHarnessError(RuntimeError):
    """A malformed or changed authority makes the measurement invalid."""


class BlockedExecution(RuntimeError):
    """A required frozen authority cannot be reproduced without substitution."""


def _require(condition: bool, message: str, *, blocked: bool = False) -> None:
    if condition:
        return
    error = BlockedExecution if blocked else InvalidHarnessError
    raise error(message)


def _mapping(value: Any, name: str) -> Mapping[str, Any]:
    _require(isinstance(value, Mapping), f"{name} must be a mapping")
    return value


def _relative(repo_root: Path, path: Path) -> str:
    return path.resolve().relative_to(repo_root.resolve()).as_posix()


def _git_state(repo_root: Path) -> dict[str, Any]:
    def _run(*args: str) -> str:
        result = subprocess.run(
            ["git", "-C", str(repo_root), *args],
            check=True,
            capture_output=True,
            text=True,
        )
        return result.stdout.strip()

    status_result = subprocess.run(
        ["git", "-C", str(repo_root), "status", "--porcelain"],
        check=True,
        capture_output=True,
        text=True,
    )
    status = status_result.stdout.splitlines()
    return {
        "branch": _run("branch", "--show-current"),
        "head": _run("rev-parse", "HEAD"),
        "dirty_paths": [line[3:] if len(line) >= 4 else line for line in status],
        "worktree_clean": not status,
    }


def _read_json_exact(
    repo_root: Path,
    path: Path,
    expected_sha256: str,
    name: str,
) -> dict[str, Any]:
    target = repo_root / path
    _require(target.is_file(), f"{name} is missing: {target}", blocked=True)
    _require(file_sha256(target) == expected_sha256, f"{name} hash changed: {target}", blocked=True)
    try:
        value = json.loads(target.read_text(encoding="utf-8"))
    except (OSError, UnicodeError, json.JSONDecodeError) as exc:
        raise InvalidHarnessError(f"{name} cannot be read as JSON: {target}") from exc
    return dict(_mapping(value, name))


def _ref(
    repo_root: Path,
    path: Path,
    *,
    role: str,
    expected_sha256: str | None = None,
) -> dict[str, Any]:
    target = repo_root / path
    _require(target.is_file(), f"{role} is missing: {target}", blocked=True)
    actual = file_sha256(target)
    _require(expected_sha256 is None or actual == expected_sha256, f"{role} hash changed: {target}", blocked=True)
    return {"path": _relative(repo_root, target), "sha256": actual, "role": role}

def _validate_current_governance(repo_root: Path, contract: Mapping[str, Any]) -> dict[str, Any]:
    try:
        current = contract_definition._validate_current_state(repo_root)
    except contract_definition.InvalidHarnessError as exc:
        raise BlockedExecution(str(exc)) from exc
    state_ref = _mapping(current.get("governance"), "current governance reference")
    contract_governance = _mapping(contract.get("governance"), "contract governance")
    _require(
        state_ref.get("sha256") == contract_governance.get("state_sha256_before")
        and contract_governance.get("state_sha256_must_remain_unchanged") is True,
        "current governance does not match the frozen contract",
        blocked=True,
    )
    for path in (contract_definition.CURRENT_STATUS_PATH, contract_definition.CURRENT_HANDOFF_PATH):
        text = (repo_root / path).read_text(encoding="utf-8")
        _require(TASK_ID in text, f"{path} does not name the execution action", blocked=True)
        _require("training_authorized=false" in text and "generation_readiness=not_ready" in text, f"{path} opened downstream governance", blocked=True)
    return {
        "path": state_ref["path"],
        "sha256_before": state_ref["sha256"],
        "active_objective": current["active_objective"],
        "hard_blocker": current["hard_blocker"],
        "authorization_tier": current["authorization_tier"],
        "selected_producer": "EMPIRICAL_GAUSSIAN_COPULA",
        "n3_amplitude_producer_confirmation": "PASSED",
        "sealed_consumed": True,
        "sealed_reused": False,
        "training_authorized": False,
        "formal_promotion": False,
        "generation_readiness": "not_ready",
    }


def _validate_frozen_authorities(
    repo_root: Path,
    contract: Mapping[str, Any],
    authorities: Mapping[str, Any],
) -> dict[str, Any]:
    try:
        runtime = contract_definition._validate_authorities(repo_root)
    except contract_definition.InvalidHarnessError as exc:
        raise BlockedExecution(str(exc)) from exc
    expected_map = {
        "current_state": runtime["current"],
        "v2_development": runtime["v2"],
        "project_complex_ols": runtime["complex_ols"],
        "selected_producer_and_prior_confirmation": runtime["prior"],
        "metric_and_evaluator": runtime["metrics"],
    }
    for key, expected in expected_map.items():
        if key == "current_state":
            frozen_current = _mapping(authorities[key], "frozen current state")
            runtime_current = _mapping(expected, "runtime current state")
            _require(
                frozen_current.get("active_objective") == runtime_current.get("active_objective")
                and frozen_current.get("hard_blocker") == runtime_current.get("hard_blocker")
                and frozen_current.get("authorization_tier") == runtime_current.get("authorization_tier")
                and _mapping(frozen_current.get("governance"), "frozen governance snapshot").get("sha256")
                == _mapping(runtime_current.get("governance"), "runtime governance snapshot").get("sha256"),
                "frozen current governance differs from runtime",
                blocked=True,
            )
        else:
            _require(dict(authorities[key]) == dict(expected), f"frozen authority differs from runtime: {key}", blocked=True)
    definition_ref = _mapping(authorities["definition_implementation"], "definition implementation reference")
    definition_path = repo_root / Path(str(definition_ref["path"]))
    _require(file_sha256(definition_path) == definition_ref.get("sha256"), "contract definition implementation changed", blocked=True)
    metric_authority = _mapping(authorities["metric_and_evaluator"], "metric authority")
    for key in ("energy_distance", "evaluator", "aggregation"):
        ref = _mapping(metric_authority.get(key), f"metric authority {key}")
        path = repo_root / Path(str(ref["path"]))
        _require(file_sha256(path) == ref.get("sha256"), f"metric authority implementation changed: {key}", blocked=True)
    return expected_map

        "immutable V2 exact complex materialization",
    )
    try:
        z = prior_execution._validate_materialization_payload(payload, q, source_indices, labels, identities, fold_ids)
    except (prior_execution.InvalidHarnessError, prior_execution.BlockedExecution) as exc:
        raise BlockedExecution(str(exc)) from exc
    _require(torch.equal(z.abs().to(dtype=torch.float64), q), "q-to-complex correspondence is not bitwise exact", blocked=True)
    _require(payload.get("governance", {}).get("sealed_values_reused") is False, "sealed values were reused by materialization", blocked=True)
    materialization_ref = _ref(
        repo_root,
        PREVIOUS_MATERIALIZATION_PATH,
        role="immutable V2 exact complex OLS materialization reused as input",
        expected_sha256=PREVIOUS_MATERIALIZATION_SHA256,
    )
    manifest_ref = _ref(
        repo_root,
        PREVIOUS_MATERIALIZATION_MANIFEST_PATH,
        role="immutable V2 exact complex OLS materialization manifest",
        expected_sha256=PREVIOUS_MATERIALIZATION_MANIFEST_SHA256,
    )
    source_ref = {
        "path": _relative(repo_root, repo_root / prior_definition.V2_SOURCE_MANIFEST_PATH),
        "sha256": prior_definition.V2_SOURCE_MANIFEST_SHA256,
    }
    return q, z, fold_ids, identities, source_indices, {
        "source": source,
        "labels": labels,
        "v2_runtime": v2_runtime,
        "materialization": materialization_ref,
        "materialization_manifest": manifest_ref,
        "source_ref": source_ref,
        "materialization_tensor_sha256": tensor_sha256(z),
    }, payload


def _derive_seed(role: str, fold: int, trial: int | None, bank: int | None) -> int:
    _require(role in {"CANDIDATE_Q_SAMPLE", "CANDIDATE_MODEL_BANK", "CANDIDATE_PSEUDO_DATA"}, "unknown frozen RNG role")
    _require(0 <= fold < FOLD_COUNT, "seed fold is invalid")
    trial_token = "none" if trial is None else str(trial)
    bank_token = "none" if bank is None else str(bank)
    material = f"{RNG_NAMESPACE}|role={role}|fold={fold}|trial={trial_token}|bank={bank_token}".encode("utf-8")
    return int.from_bytes(hashlib.sha256(material).digest()[:8], "big") & MAX_SIGNED_63


def _gaussianized_q_coordinates_batch(query: torch.Tensor, sorted_values: torch.Tensor) -> torch.Tensor:
    """Vectorized form of the contract's exact fit-only empirical-CDF rule."""

    query_value = query.detach().to(device="cpu", dtype=torch.float64).contiguous()
    sorted_value = sorted_values.detach().to(device="cpu", dtype=torch.float64).contiguous()
    _require(query_value.ndim == 2 and query_value.shape[1] == DIMENSION, "query q shape is invalid")
    _require(sorted_value.ndim == 2 and sorted_value.shape[1] == DIMENSION and sorted_value.shape[0] >= 2, "sorted fit q shape is invalid")
    _require(torch.isfinite(query_value).all().item() and query_value.ge(0.0).all().item(), "query q is invalid")
    _require(torch.isfinite(sorted_value).all().item() and sorted_value.ge(0.0).all().item(), "sorted fit q is invalid")
    _require(torch.all(sorted_value[1:] >= sorted_value[:-1]).item(), "sorted fit q is not monotone")
    n_fit = int(sorted_value.shape[0])
    positions = (torch.arange(n_fit, dtype=torch.float64) + 0.5) / float(n_fit)
    result = torch.empty_like(query_value)
    for coordinate in range(DIMENSION):
        column = sorted_value[:, coordinate].contiguous()
        values = query_value[:, coordinate]
        left = torch.searchsorted(column, values, right=False)
        right = torch.searchsorted(column, values, right=True)
        safe_left = left.clamp(min=0, max=n_fit - 1)
        safe_right = (right - 1).clamp(min=0, max=n_fit - 1)
        equal = (left < n_fit) & (right > left) & (column.index_select(0, safe_left) == values)
        equal_u = (positions.index_select(0, safe_left) + positions.index_select(0, safe_right)).mul(0.5)
        lower_bound = values.le(column[0])
        upper_bound = values.ge(column[-1])
        upper = torch.searchsorted(column, values, right=False).clamp(min=1, max=n_fit - 1)
        lower = upper - 1
        lower_value = column.index_select(0, lower)
        upper_value = column.index_select(0, upper)
        lower_position = positions.index_select(0, lower)
        upper_position = positions.index_select(0, upper)
        fraction = (values - lower_value) / (upper_value - lower_value)
        interpolated = lower_position + fraction * (upper_position - lower_position)
        u = torch.where(equal, equal_u, torch.where(lower_bound, positions[0], torch.where(upper_bound, positions[-1], interpolated)))
        result[:, coordinate] = torch.special.ndtri(u)
    _require(torch.isfinite(result).all().item(), "Gaussianized q coordinates are non-finite")
    return result.contiguous()


def _summary(value: torch.Tensor) -> dict[str, float]:
    view = value.detach().to(device="cpu", dtype=torch.float64).reshape(-1)
    _require(view.numel() > 0 and torch.isfinite(view).all().item(), "summary input is invalid")
    return {
        "count": int(view.numel()),
        "min": float(view.min()),
        "p01": float(torch.quantile(view, 0.01)),
        "p50": float(torch.quantile(view, 0.50)),
        "mean": float(view.mean()),
        "p95": float(torch.quantile(view, 0.95)),
        "p99": float(torch.quantile(view, 0.99)),
        "max": float(view.max()),
    }


def _nearest_donors(
    query_y: torch.Tensor,
    donor_y: torch.Tensor,
    donor_ids: list[str],
) -> tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
    """Return original fit-table indices, exact-distance summaries, and raw-q distances.

    The squared-distance matrix gives the exact Euclidean argmin without a
    square-root pass for every donor.  Donors are evaluated in bytewise stable
    identity order, so an exact minimum tie selects the smallest identity.  A
    second norm calculation is made for rows whose squared distances tie, which
    preserves the contract's exact float64 distance-tie rule.
    """

    query_value = query_y.detach().to(device="cpu", dtype=torch.float64).contiguous()
    donor_value = donor_y.detach().to(device="cpu", dtype=torch.float64).contiguous()
    _require(query_value.ndim == 2 and query_value.shape[1] == DIMENSION, "query donor coordinates are invalid")
    _require(donor_value.shape == (FIT_COUNT, DIMENSION), "donor coordinates are invalid")
    _require(len(donor_ids) == FIT_COUNT, "donor identities are incomplete")
    _require(torch.isfinite(query_value).all().item() and torch.isfinite(donor_value).all().item(), "donor coordinates are non-finite")
    identity_order = sorted(range(FIT_COUNT), key=lambda index: donor_ids[index].encode("utf-8"))
    order_tensor = torch.tensor(identity_order, dtype=torch.int64)
    ordered_y = donor_value.index_select(0, order_tensor).contiguous()
    donor_norm = ordered_y.square().sum(dim=1).unsqueeze(0)
    query_norm = query_value.square().sum(dim=1, keepdim=True)
    distances_squared = (query_norm + donor_norm - 2.0 * query_value.mm(ordered_y.transpose(0, 1))).clamp_min(0.0)
    _require(torch.isfinite(distances_squared).all().item(), "nearest-donor squared distances are non-finite")
    minimum_squared, ordered_indices = distances_squared.min(dim=1)
    tie_counts = distances_squared.eq(minimum_squared.unsqueeze(1)).sum(dim=1)
    tie_rows = torch.nonzero(tie_counts.gt(1), as_tuple=False).reshape(-1)
    if tie_rows.numel():
        exact_distances = torch.linalg.vector_norm(
            query_value.index_select(0, tie_rows).unsqueeze(1) - ordered_y.unsqueeze(0),
            dim=2,
        )
        exact_minimum, exact_indices = exact_distances.min(dim=1)
        minimum_squared[tie_rows] = exact_minimum.square()
        ordered_indices[tie_rows] = exact_indices
    selected_original = order_tensor.index_select(0, ordered_indices).contiguous()
    distances = torch.sqrt(minimum_squared).contiguous()
    _require(torch.isfinite(distances).all().item() and distances.ge(0.0).all().item(), "nearest-donor distances are invalid")
    return selected_original, distances, tie_counts


def _nearest_donors_with_q(
    query_y: torch.Tensor,
    query_q: torch.Tensor,
    donor_y: torch.Tensor,
    donor_q: torch.Tensor,
    donor_ids: list[str],
) -> tuple[torch.Tensor, torch.Tensor, torch.Tensor, torch.Tensor]:
    _require(donor_q.shape == (FIT_COUNT, DIMENSION), "donor q table is invalid")
    selected, distances, tie_counts = _nearest_donors(query_y, donor_y, donor_ids)
    selected_q = donor_q.index_select(0, selected)
    raw_q_distance = torch.linalg.vector_norm(query_q - selected_q, dim=1).contiguous()
    _require(torch.isfinite(raw_q_distance).all().item() and raw_q_distance.ge(0.0).all().item(), "raw q nearest distances are invalid")
    return selected, distances, raw_q_distance, tie_counts


def _compose_candidate(
    fit: Mapping[str, Any],
    donor_y: torch.Tensor,
    donor_q: torch.Tensor,
    donor_phase: torch.Tensor,
    donor_ids: list[str],
    sample_count: int,
    fold: int,
    trial: int | None,
    bank: int | None,
    role: str,
) -> tuple[torch.Tensor, torch.Tensor, torch.Tensor, torch.Tensor, torch.Tensor, dict[str, Any]]:
    seed = _derive_seed("CANDIDATE_Q_SAMPLE", fold, trial, bank)
    q_new = prior_execution._sample_q_from_fold(fit, sample_count, seed)
    y_new = _gaussianized_q_coordinates_batch(q_new, fit["sorted_values"])
    donor_indices, nearest_distance, raw_q_distance, tie_counts = _nearest_donors_with_q(
        y_new,
        q_new,
        donor_y,
        donor_q,
        donor_ids,
    )
    phase_new = donor_phase.index_select(0, donor_indices).contiguous()
    generated = prior_definition.compose_complex_from_amplitude_phase(q_new, phase_new)
    _require(generated.dtype == torch.complex128, "candidate complex dtype changed")
    _require(torch.isfinite(generated.real).all().item() and torch.isfinite(generated.imag).all().item(), "candidate complex rows are non-finite")
    _require(torch.all(generated.abs() > 0).item(), "candidate complex rows contain exact-zero amplitude")
    seed_record = {
        "role": role,
        "q_sample_role": "CANDIDATE_Q_SAMPLE",
        "seed": seed,
        "fold": fold,
        "trial": trial,
        "bank": bank,
        "sample_count": sample_count,
        "donor_indices_sha256": tensor_sha256(donor_indices),
        "nearest_distance_summary": _summary(nearest_distance),
        "raw_q_distance_summary": _summary(raw_q_distance),
        "exact_distance_tie_row_count": int(tie_counts.gt(1).sum()),
        "complete_phase_vectors": True,
        "heldout_phase_donors": False,
        "cross_fold_phase_donors": False,
        "labels_as_inputs": False,
        "no_retry": True,
        "best_of_n": False,
    }
    return generated, q_new, donor_indices, nearest_distance, raw_q_distance, seed_record


def _energy_distance(first: torch.Tensor, second: torch.Tensor) -> float:
    left = first.detach().to(device="cpu", dtype=torch.float64).contiguous()
    right = second.detach().to(device="cpu", dtype=torch.float64).contiguous()
    _require(left.ndim == 2 and right.ndim == 2 and left.shape[1] == right.shape[1], "energy-distance inputs are invalid")
    value = metric_consumer._energy_distance(left, right)
    _require(isinstance(value, float) and torch.isfinite(torch.tensor(value, dtype=torch.float64)).item() and value >= 0.0, "energy distance is invalid")
    return value


def _primary_statistic(data_embedding: torch.Tensor, model_embeddings: list[torch.Tensor]) -> tuple[float, list[float]]:
    values = [_energy_distance(data_embedding, bank) for bank in model_embeddings]
    _require(len(values) == MODEL_BANK_COUNT, "primary model-bank count changed")
    result = float(sum(values) / MODEL_BANK_COUNT)
    _require(torch.isfinite(torch.tensor(result, dtype=torch.float64)).item() and result >= 0.0, "primary statistic is invalid")
    return result, values


def _covariance(value: torch.Tensor) -> torch.Tensor:
    view = value.to(dtype=torch.float64).contiguous()
    centered = view - view.mean(dim=0, keepdim=True)
    result = centered.transpose(0, 1).mm(centered).div(float(view.shape[0] - 1))
    _require(torch.isfinite(result).all().item(), "covariance is non-finite")
    return result


def _phase_unit(canonical: torch.Tensor) -> torch.Tensor:
    phase = torch.atan2(canonical.imag, canonical.real).to(dtype=torch.float64)
    return torch.cat((phase.cos(), phase.sin()), dim=1).contiguous()


def _descriptive_pair_metrics(heldout_z: torch.Tensor, bank_z: torch.Tensor) -> dict[str, Any]:
    heldout_canonical = prior_definition.canonicalize_existing_global_phase(heldout_z)
    bank_canonical = prior_definition.canonicalize_existing_global_phase(bank_z)
    heldout_embed = torch.cat((heldout_canonical.real, heldout_canonical.imag), dim=1).to(dtype=torch.float64).contiguous()
    bank_embed = torch.cat((bank_canonical.real, bank_canonical.imag), dim=1).to(dtype=torch.float64).contiguous()
    heldout_q = heldout_canonical.abs().to(dtype=torch.float64)
    bank_q = bank_canonical.abs().to(dtype=torch.float64)
    heldout_phase = _phase_unit(heldout_canonical)
    bank_phase = _phase_unit(bank_canonical)
    heldout_cov = _covariance(heldout_embed)
    bank_cov = _covariance(bank_embed)
    bank_cov_energy = bank_cov.square().sum()
    _require(torch.isfinite(bank_cov_energy).item() and bank_cov_energy > 0, "descriptive covariance reference is zero")
    heldout_q_centered = heldout_q - heldout_q.mean(dim=0, keepdim=True)
    bank_q_centered = bank_q - bank_q.mean(dim=0, keepdim=True)
    heldout_phase_centered = heldout_phase - heldout_phase.mean(dim=0, keepdim=True)
    bank_phase_centered = bank_phase - bank_phase.mean(dim=0, keepdim=True)
    heldout_cross = heldout_q_centered.transpose(0, 1).mm(heldout_phase_centered).div(float(heldout_q.shape[0] - 1))
    bank_cross = bank_q_centered.transpose(0, 1).mm(bank_phase_centered).div(float(bank_q.shape[0] - 1))
    per_coordinate = [
        _energy_distance(heldout_q[:, coordinate : coordinate + 1], bank_q[:, coordinate : coordinate + 1])
        for coordinate in range(DIMENSION)
    ]
    scalar = {
        "amplitude_q_discrepancy": _energy_distance(heldout_q, bank_q),
        "phase_circular_discrepancy": _energy_distance(heldout_phase, bank_phase),
        "covariance_relative_frobenius": float((heldout_cov - bank_cov).square().sum() / bank_cov_energy),
        "norm_energy_distance": _energy_distance(heldout_embed.norm(dim=1, keepdim=True), bank_embed.norm(dim=1, keepdim=True)),
        "squared_norm_energy_distance": _energy_distance(heldout_embed.square().sum(dim=1, keepdim=True), bank_embed.square().sum(dim=1, keepdim=True)),
        "marginal_q_mean_l2": float((heldout_q.mean(dim=0) - bank_q.mean(dim=0)).norm()),
        "marginal_q_std_l2": float((heldout_q.std(dim=0, unbiased=True) - bank_q.std(dim=0, unbiased=True)).norm()),
        "marginal_q_energy_distance_mean": float(sum(per_coordinate) / DIMENSION),
        "marginal_q_energy_distance_max": float(max(per_coordinate)),
    }
    _require(all(torch.isfinite(torch.tensor(value, dtype=torch.float64)).item() and value >= 0.0 for value in scalar.values()), "descriptive scalar is invalid")
    return {
        **scalar,
        "amplitude_phase_dependence": {
            "heldout_q_phase_cross_covariance_frobenius": float(heldout_cross.norm()),
            "model_q_phase_cross_covariance_frobenius": float(bank_cross.norm()),
            "absolute_frobenius_delta": float((heldout_cross - bank_cross).norm()),
            "descriptive_only": True,
        },
        "per_coordinate_q_energy_distance": [float(value) for value in per_coordinate],
        "descriptive_only": True,
    }


def _aggregate_diagnostics(items: list[Mapping[str, Any]]) -> dict[str, Any]:
    _require(len(items) == MODEL_BANK_COUNT, "descriptive model-bank count changed")
    names = (
        "amplitude_q_discrepancy",
        "phase_circular_discrepancy",
        "covariance_relative_frobenius",
        "norm_energy_distance",
        "squared_norm_energy_distance",
        "marginal_q_mean_l2",
        "marginal_q_std_l2",
        "marginal_q_energy_distance_mean",
        "marginal_q_energy_distance_max",
    )
    aggregate = {name: float(sum(float(item[name]) for item in items) / len(items)) for name in names}
    aggregate["per_coordinate_q_energy_distance"] = [
        float(sum(float(item["per_coordinate_q_energy_distance"][coordinate]) for item in items) / len(items))
        for coordinate in range(DIMENSION)
    ]
    dependence = {
        key: float(sum(float(item["amplitude_phase_dependence"][key]) for item in items) / len(items))
        for key in ("heldout_q_phase_cross_covariance_frobenius", "model_q_phase_cross_covariance_frobenius", "absolute_frobenius_delta")
    }
    return {
        "descriptive_only": True,
        "aggregate_mean_over_model_banks": aggregate,
        "amplitude_phase_dependence": {**dependence, "descriptive_only": True},
        "per_model_bank": items,
        "cannot_change_decision": True,
    }


def _donor_statistics(
    model_indices: list[torch.Tensor],
    calibration_indices: list[torch.Tensor],
    model_distances: list[torch.Tensor],
    calibration_distances: list[torch.Tensor],
    model_raw_q_distances: list[torch.Tensor],
    calibration_raw_q_distances: list[torch.Tensor],
) -> dict[str, Any]:
    model_index_tensor = torch.cat([item.reshape(-1) for item in model_indices]).to(dtype=torch.int64)
    calibration_index_tensor = torch.cat([item.reshape(-1) for item in calibration_indices]).to(dtype=torch.int64)
    all_indices = torch.cat((model_index_tensor, calibration_index_tensor))
    counts = torch.bincount(all_indices, minlength=FIT_COUNT).to(dtype=torch.int64)
    total = int(all_indices.numel())
    unique = int(counts.gt(0).sum())
    probabilities = counts[counts.gt(0)].to(dtype=torch.float64).div(float(total))
    _require(total > 0 and unique > 0, "donor statistics are empty")
    model_distance = torch.cat([item.reshape(-1) for item in model_distances])
    calibration_distance = torch.cat([item.reshape(-1) for item in calibration_distances])
    model_raw = torch.cat([item.reshape(-1) for item in model_raw_q_distances])
    calibration_raw = torch.cat([item.reshape(-1) for item in calibration_raw_q_distances])
    top_counts, _ = counts.sort(descending=True)
    return {
        "definitions": {
            "reuse_rate": "1 - unique_selected_donors / generated_rows",
            "concentration": "Herfindahl sum of squared selected-donor shares; top_10_share is the share of the ten most-used donors",
            "distance": "ordinary Euclidean distance in the frozen Gaussianized-q coordinates",
            "raw_q_distance": "descriptive ordinary Euclidean distance in raw q coordinates",
        },
        "generated_rows": total,
        "unique_donors": unique,
        "unique_donor_fraction": float(unique / total),
        "reuse_rate": float(1.0 - unique / total),
        "maximum_donor_share": float(probabilities.max()),
        "top_10_share": float(top_counts[:10].sum().item() / total),
        "herfindahl_index": float(probabilities.square().sum()),
        "nearest_donor_distance": _summary(torch.cat((model_distance, calibration_distance))),
        "nearest_donor_distance_model_banks": _summary(model_distance),
        "nearest_donor_distance_pseudo_calibration": _summary(calibration_distance),
        "raw_q_nearest_distance": _summary(torch.cat((model_raw, calibration_raw))),
        "raw_q_nearest_distance_model_banks": _summary(model_raw),
        "raw_q_nearest_distance_pseudo_calibration": _summary(calibration_raw),
    }


def _run_fold(
    fold: int,
    q: torch.Tensor,
    z: torch.Tensor,
    phase: torch.Tensor,
    identities: list[str],
    fit_positions: torch.Tensor,
    heldout_positions: torch.Tensor,
) -> tuple[dict[str, Any], dict[str, torch.Tensor]]:
    _require(fit_positions.numel() == FIT_COUNT and heldout_positions.numel() == HELDOUT_COUNT, f"fold {fold} sizes changed", blocked=True)
    _require(not bool(torch.isin(fit_positions, heldout_positions).any()), f"fold {fold} fit/heldout overlap", blocked=True)
    fit_q = q.index_select(0, fit_positions).contiguous()
    heldout_z = z.index_select(0, heldout_positions).contiguous()
    fit_phase = phase.index_select(0, fit_positions).contiguous()
    donor_ids = [identities[int(position)] for position in fit_positions.tolist()]
    _require(len(set(donor_ids)) == FIT_COUNT, f"fold {fold} donor identities are not unique", blocked=True)
    fit_model = prior_execution._fit_candidate_fold(fit_q)
    donor_y = fit_model["z"].detach().to(dtype=torch.float64).contiguous()
    donor_q = fit_q.detach().to(dtype=torch.float64).contiguous()
    donor_phase = fit_phase.detach().to(dtype=torch.float64).contiguous()
    _require(donor_y.shape == (FIT_COUNT, DIMENSION), f"fold {fold} donor Gaussianized q shape changed", blocked=True)
    _require(donor_phase.shape == (FIT_COUNT, DIMENSION), f"fold {fold} donor phase shape changed", blocked=True)

    model_banks: list[torch.Tensor] = []
    model_embeddings: list[torch.Tensor] = []
    model_indices: list[torch.Tensor] = []
    model_distances: list[torch.Tensor] = []
    model_raw_distances: list[torch.Tensor] = []
    model_seed_records: list[dict[str, Any]] = []
    for bank in range(MODEL_BANK_COUNT):
        generated, _, donor_index, distance, raw_distance, seed_record = _compose_candidate(
            fit_model,
            donor_y,
            donor_q,
            donor_phase,
            donor_ids,
            MODEL_BANK_SIZE,
            fold,
            None,
            bank,
            "CANDIDATE_MODEL_BANK",
        )
        model_banks.append(generated.contiguous())
        model_embeddings.append(prior_definition.complex_real_embedding(generated))
        model_indices.append(donor_index)
        model_distances.append(distance)
        model_raw_distances.append(raw_distance)
        model_seed_records.append({"bank": bank, **seed_record})

    calibration_values: list[float] = []
    calibration_indices: list[torch.Tensor] = []
    calibration_distances: list[torch.Tensor] = []
    calibration_raw_distances: list[torch.Tensor] = []
    calibration_seed_records: list[dict[str, Any]] = []
    started = time.perf_counter()
    for trial in range(CALIBRATION_TRIAL_COUNT):
        pseudo, _, donor_index, distance, raw_distance, seed_record = _compose_candidate(
            fit_model,
            donor_y,
            donor_q,
            donor_phase,
            donor_ids,
            HELDOUT_COUNT,
            fold,
            trial,
            None,
            "CANDIDATE_PSEUDO_DATA",
        )
        pseudo_embedding = prior_definition.complex_real_embedding(pseudo)
        value, _ = _primary_statistic(pseudo_embedding, model_embeddings)
        calibration_values.append(value)
        calibration_indices.append(donor_index)
        calibration_distances.append(distance)
        calibration_raw_distances.append(raw_distance)
        calibration_seed_records.append({"trial": trial, **seed_record})
        if trial == 0 or trial + 1 == CALIBRATION_TRIAL_COUNT or (trial + 1) % 100 == 0:
            print(
                json.dumps(
                    {
                        "stage": "candidate_model_self_calibration",
                        "fold": fold,
                        "completed_trials": trial + 1,
                        "total_trials": CALIBRATION_TRIAL_COUNT,
                        "elapsed_seconds": round(time.perf_counter() - started, 3),
                    }
                ),
                flush=True,
            )

    calibration_tensor = torch.tensor(calibration_values, dtype=torch.float64).contiguous()
    _require(
        calibration_tensor.shape == (CALIBRATION_TRIAL_COUNT,)

def _aggregate_primary(folds: list[Mapping[str, Any]]) -> dict[str, Any]:
    _require(len(folds) == FOLD_COUNT, "candidate fold count changed")
    pattern = [bool(item["primary"]["pass"]) for item in folds]
    return {
        "fold_count": FOLD_COUNT,
        "fold_pass_pattern": pattern,
        "passed_fold_count": int(sum(pattern)),
        "strict_five_of_five": all(pattern),
        "primary_metric": PRIMARY_METRIC,
        "per_fold_alpha": PER_FOLD_ALPHA,
        "family_alpha": FAMILY_ALPHA,
    }

def _artifact_ref(repo_root: Path, output_dir: Path, temporary: Path, path: Path, role: str, schema: str) -> dict[str, Any]:
    logical = output_dir / path.relative_to(temporary)
    return {
        "path": _relative(repo_root, logical),
        "sha256": file_sha256(path),
        "size_bytes": path.stat().st_size,
        "role": role,
        "schema": schema,
    }


def _assert_output_tree(temporary: Path) -> None:
    expected = {CANDIDATE_ARTIFACT_NAME, LINEAGE_NAME, REPORT_NAME}
    actual = {path.relative_to(temporary).as_posix() for path in temporary.rglob("*") if path.is_file()}
    _require(actual == expected, f"execution output tree is not exact: {sorted(actual)}")

def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--repo-root", type=Path, default=Path(__file__).resolve().parents[4])
    parser.add_argument("--output-dir", type=Path, default=OUTPUT_DIR)
    args = parser.parse_args()
    try:
        result = execute(args.repo_root, args.output_dir)
    except BlockedExecution as exc:
        print(json.dumps({"result": "BLOCKED", "classification": None, "reason": str(exc)}, indent=2, sort_keys=True))
        return 3
    except InvalidHarnessError as exc:
        print(json.dumps({"result": "INVALID", "classification": "INVALID_HARNESS", "reason": str(exc)}, indent=2, sort_keys=True))
        return 2
    print(json.dumps(result, indent=2, sort_keys=True))
    return 0
