def _require(condition: bool, message: str) -> None:
    if not condition:
        raise ValueError(message)


def _mapping(value: Any, name: str) -> Mapping[str, Any]:
    _require(isinstance(value, Mapping), f"{name} must be a mapping")
    return value


def _relative(repo_root: Path, path: Path) -> str:
    return path.resolve().relative_to(repo_root.resolve()).as_posix()


def _read_json_exact(path: Path, expected_sha256: str, label: str) -> dict[str, Any]:
    _require(path.is_file(), f"{label} is missing: {path}")
    _require(file_sha256(path) == expected_sha256, f"{label} hash changed: {path}")
    return dict(_mapping(json.loads(path.read_text(encoding="utf-8")), label))


def _git_state(repo_root: Path) -> dict[str, Any]:
    branch = subprocess.run(
        ["git", "-C", str(repo_root), "branch", "--show-current"],
        check=True,
        capture_output=True,
        text=True,
    ).stdout.strip()
    head = subprocess.run(
        ["git", "-C", str(repo_root), "rev-parse", "HEAD"],
        check=True,
        capture_output=True,
        text=True,
    ).stdout.strip()
    status = subprocess.run(
        ["git", "-C", str(repo_root), "status", "--porcelain"],
        check=True,
        capture_output=True,
        text=True,
    ).stdout.splitlines()
    dirty_paths = [line[3:] if len(line) >= 4 else line for line in status]
    return {"branch": branch, "head": head, "dirty_paths": dirty_paths}

def _load_fit_bank(repo_root: Path) -> torch.Tensor:
    payload = safe_torch_load(
        repo_root / FIT_BANK,
        expected_sha256=EXPECTED_FIT_BANK_SHA256,
    )
    bank = _mapping(payload, "fit bank")
    _require(bank.get("schema") == "ols_producer_bank.v1", "fit bank schema changed")
    _require(bank.get("role") == "OLS_PRODUCER_FIT_BANK_V1", "fit bank role changed")
    _require(bank.get("producer_lineage") == "OLS_LATENT_PRODUCER_V1", "fit bank lineage changed")
    _require(bank.get("source_coordinate") == "PROJECT_COMPLEX_OLS_V1 / R3_COMPLEX", "fit bank coordinate changed")
    _require(bank.get("count") == FIT_COUNT and bank.get("N") == DIMENSION, "fit bank shape changed")
    _require(bank.get("dtype") == "torch.complex64" and bank.get("shape") == [FIT_COUNT, DIMENSION], "fit bank dtype changed")
    partition = _mapping(bank.get("partition"), "fit bank partition")
    _require(partition.get("semantic_identity_sha256") == EXPECTED_PARTITION_IDENTITY_SHA256, "fit bank partition changed")
    r3 = bank.get("r3_complex")
    indices = bank.get("source_indices")
    positions = bank.get("source_row_positions")
    labels = bank.get("labels")
    _require(torch.is_tensor(r3), "fit bank R3 tensor is missing")
    _require(torch.is_tensor(indices), "fit bank indices are missing")
    _require(torch.is_tensor(positions), "fit bank positions are missing")
    _require(torch.is_tensor(labels), "fit bank labels are missing")
    _require(tuple(r3.shape) == (FIT_COUNT, DIMENSION) and r3.dtype == torch.complex64, "fit bank R3 shape/dtype changed")
    _require(tuple(indices.shape) == (FIT_COUNT,) and indices.dtype == torch.int64, "fit bank indices changed")
    _require(tuple(positions.shape) == (FIT_COUNT,) and positions.dtype == torch.int64, "fit bank positions changed")
    _require(tuple(labels.shape) == (FIT_COUNT,) and labels.dtype == torch.int64, "fit bank labels changed")
    _require(bool(torch.isfinite(r3).all()), "fit bank R3 is non-finite")
    _require(tensor_sha256(r3) == EXPECTED_FIT_R3_SHA256, "fit bank R3 identity changed")
    _require(tensor_sha256(indices) == EXPECTED_FIT_SOURCE_INDEX_SHA256, "fit bank source-index identity changed")
    _require(tensor_sha256(positions) == EXPECTED_FIT_ROW_POSITION_SHA256, "fit bank row-position identity changed")
    _require(tensor_sha256(labels) == EXPECTED_PARTITION_LABEL_SHA256, "fit bank label identity changed")
    _require(torch.equal(torch.bincount(labels, minlength=CLASS_COUNT), torch.full((CLASS_COUNT,), ROWS_PER_CLASS, dtype=torch.int64)), "fit bank class balance changed")
    return r3.detach().cpu().contiguous()


def _fit_parameters(
    fit_r3: torch.Tensor,
    expected_mu_sha256: str,
) -> dict[str, Any]:
    q = fit_r3.abs().to(dtype=torch.float64).contiguous()
    _require(tuple(q.shape) == (FIT_COUNT, DIMENSION), "fit q shape changed")
    _require(bool(torch.isfinite(q).all()), "fit q is non-finite")
    _require(bool(q.ge(0.0).all()), "fit q is negative")
    mu_fit = q.mean(dim=0)
    centered = q - mu_fit
    _require(bool(torch.isfinite(mu_fit).all()), "fit mean is non-finite")
    _require(bool(mu_fit.ge(0.0).all()), "fit mean is negative")
    _require(bool(torch.isfinite(centered).all()), "centered fit matrix is non-finite")
    _require(tensor_sha256(mu_fit) == expected_mu_sha256, "fit mean disagrees with model-order parent")

    previous_deterministic = torch.are_deterministic_algorithms_enabled()
    previous_threads = torch.get_num_threads()
    torch.use_deterministic_algorithms(True)
    torch.set_num_threads(1)
    try:
        left, singular_values, right_transposed = torch.linalg.svd(centered, full_matrices=True)
    finally:
        torch.set_num_threads(previous_threads)
        torch.use_deterministic_algorithms(previous_deterministic)

    _require(tuple(left.shape) == (FIT_COUNT, FIT_COUNT), "SVD left shape changed")
    _require(tuple(singular_values.shape) == (DIMENSION,), "SVD spectrum shape changed")
    _require(tuple(right_transposed.shape) == (DIMENSION, DIMENSION), "SVD right shape changed")
    _require(bool(torch.isfinite(left).all()), "SVD left basis is non-finite")
    _require(bool(torch.isfinite(singular_values).all()), "SVD spectrum is non-finite")
    _require(bool(torch.isfinite(right_transposed).all()), "SVD right basis is non-finite")
    _require(bool(singular_values.ge(0.0).all()), "SVD spectrum is negative")

    right_basis = right_transposed.transpose(0, 1).contiguous()
    W_99 = right_basis[:, :RANK].contiguous()
    gram = W_99.transpose(0, 1).mm(W_99)
    identity = torch.eye(RANK, dtype=torch.float64)
    _require(
        bool(torch.allclose(gram, identity, rtol=ORTHOGONALITY_RTOL, atol=ORTHOGONALITY_ATOL)),
        "W_99 is not orthonormal",
    )
    lambda_values = singular_values[:RANK].square().div(float(SAMPLE_COVARIANCE_DENOMINATOR))
    _require(tuple(W_99.shape) == (DIMENSION, RANK), "W_99 shape changed")
    _require(tuple(lambda_values.shape) == (RANK,), "lambda shape changed")
    _require(bool(torch.isfinite(lambda_values).all()), "lambda is non-finite")
    _require(bool(lambda_values.gt(0.0).all()), "lambda is not strictly positive")

    scores = centered.mm(W_99)
    reconstruction = mu_fit + scores.mm(W_99.transpose(0, 1))
    _require(bool(torch.isfinite(scores).all()), "fit scores are non-finite")
    _require(bool(torch.isfinite(reconstruction).all()), "fit reconstruction is non-finite")

    return {
        "mu_fit": mu_fit,
        "W_99": W_99,
        "lambda_1_to_99": lambda_values,
        "singular_values": singular_values,
        "scores": scores,
    }


def _write_artifacts(
    repo_root: Path,
    output_dir: Path,
    state: Mapping[str, Any],
    manifest: Mapping[str, Any],
) -> tuple[Path, Path]:
    if output_dir.exists():
        raise FileExistsError(f"refusing to reuse write-once output: {output_dir}")
    output_dir.parent.mkdir(parents=True, exist_ok=True)
    temporary = Path(tempfile.mkdtemp(prefix=f".{output_dir.name}.", dir=output_dir.parent))
    try:
        state_path = temporary / STATE_NAME
        manifest_path = temporary / MANIFEST_NAME
        atomic_torch_save_new(state_path, state, artifact_kind="N3 producer state")
        state_sha256 = file_sha256(state_path)
        manifest_payload = dict(manifest)
        manifest_payload["artifact"] = {
            "path": _relative(repo_root, output_dir / STATE_NAME),
            "sha256": state_sha256,
            "schema": STATE_SCHEMA,
        }
        atomic_write_new_json(manifest_path, manifest_payload, artifact_kind="N3 producer manifest")
        _require(sorted(path.name for path in temporary.iterdir()) == sorted((MANIFEST_NAME, STATE_NAME)), "producer output layout changed")
        if output_dir.exists():
            raise FileExistsError(f"write-once output appeared during execution: {output_dir}")
        os.replace(temporary, output_dir)
    finally:
        if temporary.exists():
            shutil.rmtree(temporary)
    return output_dir / STATE_NAME, output_dir / MANIFEST_NAME
