def test_canonical_pixel_loss_balances_foreground_and_background() -> None:
    target = torch.tensor([[[[0.0, 1.0], [0.0, 1.0]]]])
    output = torch.tensor([[[[1.0, 0.0], [0.0, 1.0]]]], requires_grad=True)

    loss = rescue._canonical_pixel_loss(output, target, foreground_threshold=0.1)
    loss.backward()

    assert float(loss) == pytest.approx(0.5)
    assert output.grad is not None and torch.isfinite(output.grad).all()
    assert torch.count_nonzero(output.grad).item() > 0


def test_photometric_affine_is_fitted_from_paired_training_views() -> None:
    target = torch.linspace(0.0, 1.0, 64, dtype=torch.float32).reshape(2, 1, 4, 8)
    raw = (target * 0.5 + 0.2).clamp(0.0, 1.0)

    gain, offset, record = rescue._fit_photometric_affine(raw, target)
    corrected = rescue._apply_photometric_affine(raw, gain, offset)

    assert gain == pytest.approx(2.0, abs=1e-6)
    assert offset == pytest.approx(-0.4, abs=1e-6)
    assert record["fit_role"] == "TRAIN"
    assert torch.allclose(corrected, target, atol=1e-6, rtol=1e-6)


def test_training_photometric_forward_uses_no_grad_not_inference_mode() -> None:
    class Runtime:
        training = True

        def eval(self) -> None:
            self.training = False

        def train(self, mode: bool) -> None:
            self.training = mode

        def forward_from_image(self, batch: torch.Tensor) -> SimpleNamespace:
            assert not torch.is_inference_mode_enabled()
            assert not torch.is_grad_enabled()
            assert not torch.is_inference(batch)
            raw = torch.nn.functional.interpolate(batch, size=(200, 200), mode="nearest")
            return SimpleNamespace(raw_intensity=raw)

    class Ledger:
        reads: list[tuple[str, int]]

        def __init__(self) -> None:
            self.reads = []

        def record(self, role: str, rows: int) -> None:
            self.reads.append((role, rows))

    labels = torch.arange(10).repeat_interleave(2)
    context = SimpleNamespace(
        labels=labels,
        train_indices=torch.arange(20),
        images=torch.rand(20, 1, 28, 28),
        ledger=Ledger(),
    )
    consumer = SimpleNamespace(runtime=Runtime(), device=torch.device("cpu"))

    _, _, fit = rescue._fit_train_photometric_affine(consumer, context, rows_per_class=2)

    assert fit["rows"] == 20
    assert fit["fit_role"] == "TRAIN"
    assert consumer.runtime.training
    assert sum(rows for _, rows in context.ledger.reads) == 20


def test_repaired_attempt_is_new_and_binds_the_failed_attempt(tmp_path: Path) -> None:
    assert rescue.ATTEMPT_ID.endswith(":attempt_02")
    assert rescue.ATTEMPT_DIR.name == "attempt_02"
    assert rescue.PRIOR_INVALID_ATTEMPT_DIR.name == "attempt_01"
    prior_attempt = tmp_path / rescue.PRIOR_INVALID_ATTEMPT_DIR
    prior_attempt.mkdir(parents=True)
    (prior_attempt / "status.json").write_text(json.dumps({
        "state": rescue.RUN_INVALID_HARNESS,
        "completed_step": rescue.START_STEP + 1,
        "scientific_updates": 1,
    }), encoding="utf-8")
    (prior_attempt / "failure_report.json").write_text(json.dumps({
        "state": rescue.RUN_INVALID_HARNESS,
        "scientific_updates": 1,
        "automatic_recovery_permitted": False,
        "error": "RuntimeError: Inference tensors cannot be saved for backward",
    }), encoding="utf-8")

    failed = rescue._prior_invalid_attempt_identity(tmp_path)
    assert failed["status"] == rescue.RUN_INVALID_HARNESS
    assert failed["scientific_updates"] == 1
    assert failed["new_attempt_start_step"] == rescue.START_STEP
    assert failed["restart_policy"].startswith("fresh-load-from-accepted-step600")

    torch.save({}, prior_attempt / "scientific_step_000700.pt")
    with pytest.raises(rescue.ActiveCanaryError, match="contains a checkpoint"):
        rescue._prior_invalid_attempt_identity(tmp_path)

def test_original_generation_target_requires_all_rows_and_each_class() -> None:
    metrics = {"rows": 2_000, "evaluated_rows": 2_000, "macro_recall": 0.60, "per_class_recall": [0.1] * 10}
    assert rescue._meets_original_generation_target(metrics)
    metrics["per_class_recall"] = [0.0] + [0.7] * 9
    assert not rescue._meets_original_generation_target(metrics)
    metrics["per_class_recall"] = [0.1] * 10
    metrics["evaluated_rows"] = 1_999
    assert not rescue._meets_original_generation_target(metrics)

    all_rows = {
        "rows": 2_000,
        "evaluated_rows": 2_000,
        "macro_recall": 0.60,
        "per_class_recall": [0.1] * 10,
    }
    assert rescue._resolve_calibration_branch(all_rows, all_rows)
    assert not rescue._resolve_calibration_branch(all_rows, metrics)


def test_generator_composition_keeps_canonical_direction_and_off_control() -> None:
    q_canonical = torch.tensor([1.0, -1.0, 0.5, -0.5])
    g_rec = -2.0 * q_canonical + torch.tensor([0.0, 0.0, 1.0, 1.0])
    g_op = 0.25 * q_canonical

    active = rescue._compose_generator_gradient(q_canonical, g_rec, g_op, 0.5)
    disabled = rescue._compose_generator_gradient(
        q_canonical, g_rec, g_op, 0.0, semantic_enabled=False,
    )

    assert rescue._dot(active["g_total"], active["g_sem"]) > 0.0
    assert torch.count_nonzero(disabled["g_sem"]).item() == 0
    assert torch.equal(disabled["g_total"], g_rec + g_op)


def test_gate_requires_both_recall_gains_and_class8_reduction() -> None:
    passed = {
        "DEV_recall_gain": True,
        "DEV_CI": True,
        "S1_recall_gain": True,
        "S1_CI": True,
        "DEV_class8_false_positive_reduction": True,
        "S1_class8_false_positive_reduction": True,
        "output_and_resource_validity": True,
    }
    assert rescue._resolve_gate(passed) == (
        rescue.RUN_COMPLETED,
        rescue.CLASSIFICATION_SUPPORTED,
        True,
    )

    passed["S1_class8_false_positive_reduction"] = False
    assert rescue._resolve_gate(passed) == (
        rescue.RUN_VALID_FAILED,
        rescue.RUN_VALID_FAILED,
        False,
    )

def test_terminal_status_writer_and_single_control_transition(tmp_path: Path) -> None:
    attempt = tmp_path / "attempt_01"
    attempt.mkdir()

    rescue._write_status(attempt, state=rescue.RUNNING, phase="TRAINING", step=700)
    rescue._write_status(
        attempt,
        state=rescue.RUN_VALID_FAILED,
        phase=rescue.RUN_VALID_FAILED,
        step=rescue.END_STEP,
        outer_gate_passed=False,
    )
    status = json.loads((attempt / "status.json").read_text(encoding="utf-8"))
    assert status["state"] == rescue.RUN_VALID_FAILED

    module = ast.parse(Path(rescue.__file__).read_text(encoding="utf-8"))
    execute = next(
        node for node in module.body
        if isinstance(node, ast.FunctionDef) and node.name == "_execute"
    )
    transitions = [
        node for node in ast.walk(execute)
        if isinstance(node, ast.Call)
        and isinstance(node.func, ast.Attribute)
        and isinstance(node.func.value, ast.Name)
        and node.func.value.id == "control"
        and node.func.attr in {"complete", "valid_failed"}
    ]
    candidate_branch = next(
        node for node in ast.walk(execute)
        if isinstance(node, ast.If)
        and isinstance(node.test, ast.Name)
        and node.test.id == "calibration_candidate"
    )
    candidate_complete = [
        node for node in ast.walk(candidate_branch)
        if isinstance(node, ast.Call)
        and isinstance(node.func, ast.Attribute)
        and isinstance(node.func.value, ast.Name)
        and node.func.value.id == "control"
        and node.func.attr == "complete"
    ]
    assert len(candidate_complete) == 1
    assert any(isinstance(node, ast.Return) for node in ast.walk(candidate_branch))
    assert sum(node.func.attr == "complete" for node in transitions) == 2
    assert sum(node.func.attr == "valid_failed" for node in transitions) == 1

    report_writes = [
        node for node in ast.walk(execute)
        if isinstance(node, ast.Call)
        and isinstance(node.func, ast.Name)
        and node.func.id == "atomic_write_new_json"
        and node.args
        and isinstance(node.args[0], ast.BinOp)
        and isinstance(node.args[0].right, ast.Constant)
        and node.args[0].right.value == "report.json"
    ]
    assert len(report_writes) == 2
    normal_report_write = max(report_writes, key=lambda node: node.lineno)
    normal_transitions = [
        node for node in transitions if node.lineno > normal_report_write.lineno
    ]
    assert {node.func.attr for node in normal_transitions} == {"complete", "valid_failed"}
