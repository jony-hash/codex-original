def test_v2_execution_heartbeat_phases_follow_run_control_contract() -> None:
    tree = ast.parse(Path(v2.__file__).read_text(encoding="utf-8"))
    phases = []
    for node in ast.walk(tree):
        if not isinstance(node, ast.Call) or not isinstance(node.func, ast.Attribute):
            continue
        if not isinstance(node.func.value, ast.Name) or node.func.value.id != "heartbeat" or node.func.attr != "update":
            continue
        phase = next((keyword.value for keyword in node.keywords if keyword.arg == "phase"), None)
        if isinstance(phase, ast.Constant) and isinstance(phase.value, str):
            phases.append(phase.value)
        else:
            assert isinstance(phase, ast.Name) and phase.id == "interrupted_phase"

    assert phases
    assert set(phases) <= PHASES

def _legacy_v1_update_counts() -> list[int]:
    counts = [0] * v2.FIT_ROWS
    for global_step in range(v1.UPDATE_COUNT_PER_ARM):
        chunk = global_step % v2.CHUNK_COUNT
        start = chunk * v2.BATCH_SIZE
        for row in range(start, start + v2.BATCH_SIZE):
            counts[row] += 1
    return counts

def _hold_attempt_lock(root: Path, acquired, release) -> None:
    with v2._attempt_execution_lock(root):
        acquired.set()
        release.wait(timeout=10)


def test_execution_lock_rejects_live_process_and_releases_after_process_exit(tmp_path: Path) -> None:
    context = multiprocessing.get_context("spawn")
    acquired = context.Event()
    release = context.Event()
    process = context.Process(target=_hold_attempt_lock, args=(tmp_path, acquired, release))
    process.start()
    try:
        assert acquired.wait(timeout=10)
        with pytest.raises(v2.RenderabilityV2Error, match="another process holds"):
            with v2._attempt_execution_lock(tmp_path):
                pytest.fail("a competing process acquired the execution lock")
    finally:
        release.set()
        process.join(timeout=10)
        if process.is_alive():
            process.terminate()
            process.join(timeout=5)
    assert process.exitcode == 0
    with v2._attempt_execution_lock(tmp_path):
        pass

def test_finite_difference_uses_perturbed_ols_parameter_not_gradient(monkeypatch: pytest.MonkeyPatch) -> None:
    class _ScaledRuntime(nn.Module):
        def __init__(self) -> None:
            super().__init__()
            self.encoder = nn.Module()
            self.generator = nn.Module()
            self.encoder.register_parameter("phase", nn.Parameter(torch.tensor(0.2)))
            self.generator.register_parameter("phase", nn.Parameter(torch.tensor(0.1)))

        def forward_from_ols(self, coefficients: torch.Tensor) -> SimpleNamespace:
            level = 1000.0 * coefficients.real[:, 0] + 600.0 * coefficients.imag[:, 0] + self.generator.phase
            raw = level.reshape(-1, 1, 1, 1).expand(-1, 1, 200, 200)
            return SimpleNamespace(raw_intensity=raw)

    class _ScaledConsumer:
        def __init__(self) -> None:
            self.runtime = _ScaledRuntime()

        def load_trainable_state_dict(self, state: dict[str, torch.Tensor]) -> None:
            self.runtime.load_state_dict(state)

    captured: dict[str, tuple[int, str, float, tuple[float, ...]]] = {}

    def capture_finite_difference(
        parameter: torch.Tensor,
        index: int,
        component: str,
        analytic: float,
        epsilon_candidates: tuple[float, ...],
        objective: object,
        *,
        name: str,
    ) -> dict[str, float]:
        del objective
        captured[name] = (index, component, analytic, epsilon_candidates)
        return {"epsilon": epsilon_candidates[0]}

    monkeypatch.setattr(v1, "_finite_difference_component", capture_finite_difference)
    coefficients = torch.full((v2.BATCH_SIZE, 100), complex(3.0e-5, 1.6e-5), dtype=torch.complex64)
    v2._finite_difference_checks(_ScaledConsumer(), coefficients, _target_batch(v2.BATCH_SIZE), torch.device("cpu"))

    for component in ("real", "imag"):
        index, observed_component, analytic, epsilons = captured[f"OLS {component}"]
        assert observed_component == component
        selected = coefficients.reshape(-1)[index]
        parameter_coordinate = selected.real if component == "real" else selected.imag
        expected = v2._ols_finite_difference_epsilons(float(parameter_coordinate))
        assert epsilons == pytest.approx(expected)
        assert abs(analytic) > 100.0 * abs(float(parameter_coordinate))

def test_independent_ols_row_optimizers_resume_at_the_same_next_batch(tmp_path: Path) -> None:
    def make_state() -> tuple[_FakeConsumer, list[nn.Parameter], list[torch.optim.Optimizer]]:
        consumer = _FakeConsumer()
        for parameter in consumer.runtime.parameters():
            parameter.requires_grad_(False)
        rows = [nn.Parameter(torch.tensor([0.25 + 0.0j], dtype=torch.complex64)) for _ in range(v2.FIT_ROWS)]
        optimizers = [v1._adam([row]) for row in rows]
        return consumer, rows, optimizers

    targets = _target_batch(v2.BATCH_SIZE)
    rows_index = range(v2.BATCH_SIZE)
    consumer, rows, optimizers = make_state()
    for _ in range(3):
        v2._train_ols_rows_step(consumer, optimizers, rows, targets, rows_index)
    expected_rows = [row.detach().clone() for row in rows]
    expected_states = [optimizer.state_dict() for optimizer in optimizers]

    consumer, rows, optimizers = make_state()
    v2._train_ols_rows_step(consumer, optimizers, rows, targets, rows_index)
    payload = {
        "phase": "OLS_INPUT_ONLY",
        "cursor": 1,
        "trainable_parameter_state": {name: value.detach() for name, value in consumer.runtime.named_parameters()},
        "complex_ols_rows": [row.detach() for row in rows],
        "optimizer_states": [optimizer.state_dict() for optimizer in optimizers],
        "rng_state": {"torch_cpu": torch.get_rng_state()},
    }
    source_id = "independent-ols-row-source"
    v2._save_recovery_checkpoint(tmp_path / "ols_recovery", payload, source_manifest_id=source_id)
    restored = v2._load_recovery_checkpoint(tmp_path / "ols_recovery", source_manifest_id=source_id)
    for row, optimizer, saved_row, saved_optimizer in zip(rows, optimizers, restored["complex_ols_rows"], restored["optimizer_states"], strict=True):
        row.data.copy_(saved_row)
        optimizer.load_state_dict(saved_optimizer)
    for _ in range(2):
        v2._train_ols_rows_step(consumer, optimizers, rows, targets, rows_index)

    assert all(torch.allclose(row.detach(), expected, rtol=0, atol=1e-7) for row, expected in zip(rows, expected_rows, strict=True))
    assert all(v1._tree_equal(optimizer.state_dict(), expected, atol=1e-7) for optimizer, expected in zip(optimizers, expected_states, strict=True))

        "rng_state": {"torch_cpu": torch.get_rng_state()},
    }
    source_id = "v2-source-test"
    pointer = v2._save_recovery_checkpoint(tmp_path / "recovery", payload, source_manifest_id=source_id)
    restored = v2._load_recovery_checkpoint(tmp_path / "recovery", source_manifest_id=source_id)

    generator.data.copy_(restored["generator_phase"])
    resumed_optimizer.load_state_dict(restored["optimizer_state"])
    v2._train_generator_step(consumer, resumed_optimizer, z, targets, indices)
    assert torch.allclose(generator.detach(), expected_parameter, rtol=0, atol=1e-7)
    assert v1._tree_equal(resumed_optimizer.state_dict(), expected_optimizer, atol=1e-7)
    assert pointer["sha256"]
    with pytest.raises(v2.RenderabilityV2Error, match="pointer is not bound"):
        v2._load_recovery_checkpoint(tmp_path / "recovery", source_manifest_id="tampered-source")


def _passing_metrics() -> dict[str, object]:
    return {
        "pixel": {"relative_reduction": {"foreground": 0.50, "background": 0.50}},
        "E_D": {"macro_recall": 0.50, "per_class_recall": [0.10] * 10},
    }


def _failing_metrics() -> dict[str, object]:
    return {
        "pixel": {"relative_reduction": {"foreground": 0.49, "background": 0.50}},
        "E_D": {"macro_recall": 0.50, "per_class_recall": [0.10] * 10},
    }


def test_four_classification_boundaries_use_the_fixed_gate() -> None:
    passing = _passing_metrics()
    failing = _failing_metrics()
    assert v1._classify_arms(passing, passing) == "BOTH_RENDERABILITY_UPPER_BOUNDS_REACHED"
    assert v1._classify_arms(passing, failing) == "FIXED_OLS_TO_GENERATOR_MAPPING_REACHED"
    assert v1._classify_arms(failing, passing) == "GENERATOR_REGION_REACHED_ONLY_BY_OLS_OPTIMIZATION"
    assert v1._classify_arms(failing, failing) == "NEITHER_BOUNDED_UPPER_BOUND_REACHED"
