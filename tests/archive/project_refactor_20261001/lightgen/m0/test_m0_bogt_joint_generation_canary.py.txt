def _sampler_fixture() -> tuple[SimpleNamespace, dict[str, object]]:
    labels = torch.arange(10).repeat_interleave(1_300)
    train_indices = torch.arange(labels.numel())
    context = SimpleNamespace(labels=labels, train_indices=train_indices)
    permutations = [torch.arange(digit * 1_300, (digit + 1) * 1_300) for digit in range(10)]
    saved: dict[str, object] = {
        "schema": canary.bogt.SAMPLER_SCHEMA,
        "seed": 1234,
        "steps": canary.START_STEP,
        "cursor": canary.START_STEP,
        "class_count": 10,
        "rows_per_class": [1_300] * 10,
        "permutations": permutations,
    }
    return context, saved


def test_sampler_extends_stored_permutations_without_reshuffling() -> None:
    context, saved = _sampler_fixture()
    sampler = canary.ExtendedTrainSampler(context, saved)

    expected = torch.tensor([saved["permutations"][digit][600] for digit in range(10)])
    first_batch, labels = sampler.next_batch()

    assert torch.equal(first_batch, expected)
    assert torch.equal(labels, torch.arange(10))
    assert sampler.cursor == 601
    assert sampler.state_dict()["steps"] == 1_200
    assert torch.equal(sampler.state_dict()["permutations"][0], saved["permutations"][0])


def test_sampler_rejects_changed_training_population() -> None:
    context, saved = _sampler_fixture()
    saved["permutations"] = list(saved["permutations"])
    saved["permutations"][4] = saved["permutations"][4][:-1]

    with pytest.raises(canary.CanaryError, match="permutation shape changed"):
        canary.ExtendedTrainSampler(context, saved)


def test_recovery_sampler_rejects_cursor_or_schedule_drift() -> None:
    context, saved = _sampler_fixture()
    sampler = canary.ExtendedTrainSampler(context, saved)
    state = sampler.state_dict()
    state["cursor"] = 1_201

    with pytest.raises(canary.CanaryError, match="cursor is invalid"):
        sampler.load_state_dict(state)

def test_measure_phase_selects_cuda_device_before_resetting_peak_stats(monkeypatch) -> None:
    events = []
    monkeypatch.setattr(canary.torch.cuda, "set_device", lambda device: events.append(("set", device)))
    monkeypatch.setattr(canary.torch.cuda, "reset_peak_memory_stats", lambda device: events.append(("reset", device)))
    monkeypatch.setattr(canary, "_resource", lambda device: {"global_used_fraction": 0.1})
    device = torch.device("cuda:0")

    value, metrics = canary._measure_phase(device, "test_phase", lambda: events.append(("call", None)) or "measured")

    assert value == "measured"
    assert [event[0] for event in events] == ["set", "reset", "call"]
    assert metrics["phase"] == "test_phase"

def test_attempt_heartbeat_uses_only_shared_lifecycle_phases(tmp_path) -> None:
    heartbeat = canary.HeartbeatWriter(tmp_path / "valid", "test-run", total_steps=600, interval_seconds=0.0)
    heartbeat.update(state=canary.RUN_PREPARING, phase="STATISTICS", step=0, force=True)
    heartbeat.update(state=canary.RUNNING, phase="TRAINING", step=0, force=True)
    heartbeat.update(state=canary.RUNNING, phase="POST_TRAIN_COLLECTION", step=600, force=True)
    heartbeat.update(state=canary.RUNNING, phase="FINALIZING", step=600, force=True)
    heartbeat.update(state=canary.RUN_VALID_FAILED, phase=canary.RUN_VALID_FAILED, step=600, force=True)

    invalid = canary.HeartbeatWriter(tmp_path / "invalid", "test-run-invalid", total_steps=600, interval_seconds=0.0)
    with pytest.raises(ValueError, match="heartbeat phase is invalid"):
        invalid.update(state=canary.RUN_PREPARING, phase="STEP600_BASELINE", step=0, force=True)


def test_each_original_loss_term_changes_the_combined_gradient() -> None:
    encoder = torch.nn.Parameter(torch.tensor([1.0, -0.5], dtype=torch.float64))
    generator = torch.nn.Parameter(torch.tensor([0.25, 2.0], dtype=torch.float64))
    terms = {
        "alpha_L_dist": (encoder.square().sum() * 0.5),
        "beta_L_rec": (generator.square().sum() * 0.3),
        "gamma_L_op": ((encoder.sum() + generator.sum()).square() * 0.1),
    }

    result = canary._term_gradient_controls(terms, [encoder, generator], ["encoder.phase", "generator.phase"])

    assert set(result["term_gradient_norms"]) == set(terms)
    assert all(value > 0 for value in result["disabled_term_gradient_differences"].values())
    assert all(value > 0 for value in result["full_gradient_norms"].values())


def test_gate_requires_all_original_generation_conditions() -> None:
    dev0 = {"L_rec": 0.05}
    dev1 = {"L_rec": 0.054}
    generated = {"E_D": {"evaluated": True, "macro_recall": 0.50, "per_class_recall": [0.10] * 10}, "optical_power_finite": True}
    raw = {"finite": True, "all_nonzero": True, "all_nonconstant": True, "count": 2_000}

    classification, conditions = canary._gate(dev0, dev1, generated, raw)

    assert classification == canary.CLASSIFICATION_SUPPORTED
    assert all(conditions.values())

    generated["E_D"]["macro_recall"] = 0.499
    classification, conditions = canary._gate(dev0, dev1, generated, raw)

    assert classification == "VALID_FAILED"
    assert not conditions["S1_ED_macro_recall_at_least_0_50"]


def test_gate_rejects_background_reconstruction_tradeoff_and_invalid_cmos() -> None:
    dev0 = {"L_rec": 0.05}
    dev1 = {"L_rec": 0.0551}
    generated = {"E_D": {"evaluated": True, "macro_recall": 0.9, "per_class_recall": [0.8] * 10}, "optical_power_finite": True}
    raw = {"finite": True, "all_nonzero": True, "all_nonconstant": False, "count": 2_000}

    classification, conditions = canary._gate(dev0, dev1, generated, raw)

    assert classification == "VALID_FAILED"
    assert not conditions["DEV_reconstruction_delta_at_most_0_005"]
    assert not conditions["all_2000_raw_CMOS_finite_nonzero_nonconstant"]
