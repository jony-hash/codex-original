def _metrics(*, improved: bool) -> dict[str, Any]:
    base = 0.5 if improved else 0.4
    return {name: {"macro_recall": base} for name in ("E_S", "E_I", "E_C")}


def _fit_losses(*, improved: bool, partial: bool = False) -> dict[str, float]:
    values = {name: 0.5 if improved else 1.0 for name in ("E_S", "E_I", "E_C")}
    if partial:
        values["E_S"] = 0.5
        values["E_C"] = 1.1
    return values

def test_preregistered_classification_cases_are_disjoint() -> None:
    fit0 = {"training_evaluator_feature_losses": _fit_losses(improved=False)}
    fit_improved = {"training_evaluator_feature_losses": _fit_losses(improved=True)}
    fit_partial = {"training_evaluator_feature_losses": _fit_losses(improved=False, partial=True)}
    obs0 = _metrics(improved=False)
    obs_improved = _metrics(improved=True)
    positive = {"observed_macro_recall_delta": 0.06, "ci95": [0.01, 0.10]}
    null = {"observed_macro_recall_delta": 0.0, "ci95": [-0.01, 0.01]}
    negative = {"observed_macro_recall_delta": -0.01, "ci95": [-0.02, 0.0]}
    safe = {"passed": True}
    unsafe = {"passed": False}

    assert canary._classify(fit0, fit_improved, obs0, obs0, positive, safe, True) == canary.CASE_A
    assert canary._classify(fit0, fit0, obs0, obs_improved, null, safe, True) == canary.CASE_B
    assert canary._classify(fit0, fit_improved, obs0, obs0, negative, safe, True) == canary.CASE_C
    assert canary._classify(fit0, fit0, obs0, obs0, null, safe, True) == canary.CASE_D
    assert canary._classify(fit0, fit_improved, obs0, obs0, positive, unsafe, True) == canary.CASE_E
    assert canary._classify(fit0, fit_partial, obs0, obs0, null, safe, True) == canary.CASE_F

def test_recovery_requires_exact_checkpoint_restore_and_tolerates_float32_replay_roundoff() -> None:
    expected = {"phase": torch.tensor([0.25, 1.0]), "adam": {"step": 4.0}, "trace": [{"loss": 0.5}]}
    replayed = {"phase": torch.tensor([0.25 + 2.9802322e-8, 1.0]), "adam": {"step": 4.0}, "trace": [{"loss": 0.5 + 2.0e-8}]}
    assert canary._tree_equal(expected, expected)
    assert canary._tree_close(
        replayed,
        expected,
        rtol=canary.RECOVERY_UPDATE_RTOL,
        atol=canary.RECOVERY_UPDATE_ATOL,
    )
    outside = {"phase": torch.tensor([0.25 + 2.0e-6, 1.0]), "adam": {"step": 4.0}, "trace": [{"loss": 0.5}]}
    assert not canary._tree_close(
        outside,
        expected,
        rtol=canary.RECOVERY_UPDATE_RTOL,
        atol=canary.RECOVERY_UPDATE_ATOL,
    )
    assert not canary._tree_close(
        {"phase": torch.ones(2)},
        {"phase": "wrong type"},
        rtol=canary.RECOVERY_UPDATE_RTOL,
        atol=canary.RECOVERY_UPDATE_ATOL,
    )
    assert not canary._tree_equal({"phase": torch.ones(2)}, {"phase": "wrong type"})

    for label, prediction in enumerate(rows):
        confusion[label // 10][prediction] += 1

        canary._prediction_histogram([10])

def test_fixed_population_authority_rejects_tampered_row_binding(monkeypatch: pytest.MonkeyPatch) -> None:
    root = canary._root()
    monkeypatch.setattr(canary, "_root", lambda: root)
    original_history = canary._preflight_repair_history
    monkeypatch.setattr(canary, "_preflight_repair_history", lambda path, resume: original_history(path, resume=False))
    inputs = canary._load_inputs(root)
    assert inputs["fit_indices"].numel() == 200
    assert inputs["observation_indices"].numel() == 100
    assert not bool(torch.isin(inputs["fit_indices"], inputs["observation_indices"]).any())

    assert config["run_preflight_record_path"] == canary.CLEAN_ATTEMPT_02_PREFLIGHT_PATH.as_posix()
    assert config["updates"] == 300
    del inputs

    original_loader = canary.safe_torch_load
    target_path = (root / canary.V2_TENSORS).resolve()

    def tampered_loader(path: Path, **kwargs: Any) -> Any:
        payload = original_loader(path, **kwargs)
        if Path(path).resolve() == target_path:
            payload = dict(payload)
            payload["fit_indices"] = payload["fit_indices"].clone()
            payload["fit_indices"][0] = (payload["fit_indices"][0] + 1) % 60_000
        return payload

    monkeypatch.setattr(canary, "safe_torch_load", tampered_loader)

def test_recovery_restores_adam_and_rejects_changed_frozen_encoder(monkeypatch: pytest.MonkeyPatch) -> None:
    class Runtime(nn.Module):
        def __init__(self) -> None:
            super().__init__()
            self.encoder = nn.Module()
            self.generator = nn.Module()
            self.encoder.register_parameter("phase", nn.Parameter(torch.tensor([3.0, 4.0])))
            self.generator.register_parameter("phase", nn.Parameter(torch.tensor([0.0, 0.0])))

    class Consumer:
        def __init__(self) -> None:
            self.runtime = Runtime()
            self.device = torch.device("cpu")

    monkeypatch.setattr(canary, "_restore_rng", lambda _state: None)
    consumer = Consumer()
    params = dict(consumer.runtime.named_parameters())
    optimizer = torch.optim.Adam([params["generator.phase"]], lr=1.0e-3)
    params["generator.phase"].grad = torch.tensor([1.0, -1.0])
    optimizer.step()
    config = {
        "fixed_population": {
            "fit_indices_sha256": "fit",
            "observation_indices_sha256": "observation",
            "fit_targets_sha256": "targets",
            "fit_ols_sha256": "fit-ols",
            "observation_ols_sha256": "observation-ols",
        },
        "starting_checkpoint": {"sha256": "step600"},
    }
    payload = {
        "schema": canary.CHECKPOINT_SCHEMA,
        "data_binding": canary._data_binding(config),
        "cursor": 1,
        "generator_phase": params["generator.phase"].detach().cpu().clone(),
        "encoder_phase": params["encoder.phase"].detach().cpu().clone(),
        "optimizer_state": copy.deepcopy(optimizer.state_dict()),
        "trace": [{"step": 1}],
        "measurement_phase_snapshots": {"0": torch.tensor([0.0, 0.0])},
        "rng_state": {},
    }
    expected_generator = params["generator.phase"].detach().cpu().clone()
    expected_encoder = params["encoder.phase"].detach().cpu().clone()
    params["generator.phase"].data.zero_()
    cursor, trace, phase_snapshots = canary._restore_recovery(consumer, optimizer, payload, config, expected_encoder)
    assert cursor == 1
    assert trace == [{"step": 1}]
    assert torch.equal(phase_snapshots["0"], torch.tensor([0.0, 0.0]))
    assert torch.equal(params["generator.phase"].detach().cpu(), expected_generator)
    assert float(optimizer.state[params["generator.phase"]]["step"].item()) == 1.0

    invalid = dict(payload)
    invalid["encoder_phase"] = expected_encoder + 1.0
    with pytest.raises(canary.CanaryError, match="accepted frozen state"):
        canary._restore_recovery(consumer, optimizer, invalid, config, expected_encoder)
    missing_snapshot = dict(payload)
    missing_snapshot["measurement_phase_snapshots"] = {}
    with pytest.raises(canary.CanaryError, match="snapshot set differs"):
        canary._restore_recovery(consumer, optimizer, missing_snapshot, config, expected_encoder)

def test_scientific_attempt_gets_fresh_zero_adam_after_preflight_smoke() -> None:
    class Runtime(nn.Module):
        def __init__(self) -> None:
            super().__init__()
            self.encoder = nn.Module()
            self.encoder.register_parameter("phase", nn.Parameter(torch.zeros(2)))
            self.generator = nn.Module()
            self.generator.register_parameter("phase", nn.Parameter(torch.ones(2)))

    consumer = SimpleNamespace(runtime=Runtime())
    evaluators = [nn.Linear(1, 1) for _ in range(4)]
    optimizer = canary._freeze_models(consumer, *evaluators)
    generator = dict(consumer.runtime.named_parameters())["generator.phase"]
    for _ in range(4):
        generator.grad = torch.ones_like(generator)
        optimizer.step()
        optimizer.zero_grad(set_to_none=True)
    assert float(optimizer.state[generator]["step"].item()) == 4.0

    attempt_optimizer = canary._fresh_attempt_optimizer(consumer, *evaluators)

    assert attempt_optimizer is not optimizer
    assert attempt_optimizer.state == {}
    assert attempt_optimizer.param_groups[0]["params"] == [generator]
    assert generator.requires_grad
    assert not dict(consumer.runtime.named_parameters())["encoder.phase"].requires_grad
    assert canary._optimizer_internal_step(attempt_optimizer, generator) == 0
    generator.grad = torch.ones_like(generator)
    attempt_optimizer.step()
    assert canary._optimizer_internal_step(attempt_optimizer, generator) == 1


def test_scientific_runtime_rebuild_requires_disposed_preflight_state(
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    class Runtime(nn.Module):
        def __init__(self) -> None:
            super().__init__()
            self.encoder = nn.Module()
            self.encoder.register_parameter("phase", nn.Parameter(torch.tensor([2.0])))
            self.generator = nn.Module()
            self.generator.register_parameter("phase", nn.Parameter(torch.tensor([3.0])))

    preflight_consumer = SimpleNamespace(runtime=Runtime(), device=torch.device("cpu"))
    preflight_evaluators = [nn.Linear(1, 1) for _ in range(4)]
    preflight_optimizer = canary._freeze_models(preflight_consumer, *preflight_evaluators)
    preflight_data = {"sample": torch.ones(1)}
    preflight_refs = canary._preflight_state_references(
        preflight_consumer,
        preflight_optimizer,
        *preflight_evaluators,
        preflight_data,
    )
    with pytest.raises(canary.CanaryError, match="preflight mutable state is still alive"):
        canary._reconstruct_scientific_state(Path("."), {}, torch.device("cpu"), preflight_refs)

    del preflight_consumer, preflight_evaluators, preflight_optimizer, preflight_data
    gc.collect()
    assert all(reference() is None for reference in preflight_refs.values())

    scientific_consumer = SimpleNamespace(runtime=Runtime(), device=torch.device("cpu"))
    scientific_evaluators = [nn.Linear(1, 1) for _ in range(4)]
    scientific_optimizer = canary._freeze_models(scientific_consumer, *scientific_evaluators)
    expected_state = {
        "encoder.phase": torch.tensor([2.0]),
        "generator.phase": torch.tensor([3.0]),
    }
    monkeypatch.setattr(
        canary,
        "_load_run_models",
        lambda _root, _inputs, _device: (
            scientific_consumer,
            *scientific_evaluators,
            {"frozen": True},
            scientific_optimizer,
        ),
    )
    monkeypatch.setattr(canary, "_fixed_runtime_data", lambda _inputs, _consumer, _device: {"fixed": True})
    monkeypatch.setattr(canary.three, "_restore_rng", lambda _state: None)
    inputs = {"authorities": {"checkpoint": {"trainable_parameter_state": expected_state, "rng_state": {}}}}

    rebuilt, isolation = canary._reconstruct_scientific_state(Path("."), inputs, torch.device("cpu"), preflight_refs)

    assert rebuilt[0] is scientific_consumer
    assert rebuilt[6] is scientific_optimizer
    assert rebuilt[6].state == {}
    assert isolation["preflight_mutable_state_destroyed"] is True
    assert isolation["scientific_generator_matches_step600"] is True
    assert isolation["scientific_adam_step_before_attempt"] == 0
