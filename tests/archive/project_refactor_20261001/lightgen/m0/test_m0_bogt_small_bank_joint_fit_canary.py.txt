def test_small_bank_schedule_visits_every_row_50_times() -> None:
    rows = torch.cat([canary._row_indices(cursor) for cursor in range(canary.UPDATE_COUNT)])
    counts = torch.bincount(rows, minlength=canary.FIT_ROWS)

    assert counts.shape == (200,)
    assert torch.equal(counts, torch.full((200,), 50, dtype=torch.long))
    assert canary._expected_row_counts(1000) == [50] * 200
    assert canary._row_indices(20).tolist() == list(range(0, 10))


def test_optimizer_step_reader_uses_the_accepted_production_adapter() -> None:
    parameters = [torch.nn.Parameter(torch.tensor([1.0])), torch.nn.Parameter(torch.tensor([2.0]))]
    optimizer = torch.optim.Adam(parameters, lr=0.001)
    for parameter in parameters:

    optimizer.step()

    assert canary._optimizer_steps(optimizer) == [1, 1]

def test_fixed_population_binds_rows_labels_and_target_images() -> None:
    labels = torch.arange(10).repeat(32)
    images = torch.arange(320, dtype=torch.float32).reshape(320, 1, 1, 1).expand(320, 1, 28, 28) / 320
    context = SimpleNamespace(train_indices=torch.arange(320), labels=labels, images=images)
    fit = torch.arange(200)
    observation = torch.arange(200, 300)
    fit_targets = torch.nn.functional.adaptive_avg_pool2d(images[fit].clamp(0, 1), (28, 28))
    payload = {
        "fit_indices": fit,
        "observation_indices": observation,
        "fit_labels": labels[fit],
        "observation_labels": labels[observation],
        "fit_targets_28x28": fit_targets,
    }

    result = canary._validate_fixed_population_identity(
        context, payload, expected_fit=fit, expected_observation=observation
    )

    assert torch.equal(result["fit_indices"], fit)
    assert torch.equal(result["observation_indices"], observation)
    tampered = dict(payload, fit_indices=fit.roll(1))
    with pytest.raises(canary.CanaryError, match="row identities"):
        canary._validate_fixed_population_identity(
            context, tampered, expected_fit=fit, expected_observation=observation
        )
    tampered = dict(payload, fit_targets_28x28=fit_targets + 0.01)
    with pytest.raises(canary.CanaryError, match="target-image identity"):
        canary._validate_fixed_population_identity(
            context, tampered, expected_fit=fit, expected_observation=observation
        )

def test_historical_preflights_keep_their_original_authorization_identity() -> None:
    old_failure = {
        "schema": canary.PREFLIGHT_SCHEMA,
        "task_id": canary.TASK_ID,
        "outer_gate": canary.OUTER_GATE,
        "authorization_id": canary.HISTORICAL_AUTHORIZATION_ID,
        "status": canary.RUN_INVALID_HARNESS,
        "phase": "resource_extrapolation",
        "attempt_created": False,
        "scientific_optimizer_updates": 0,
    }
    budget_stop = {
        **old_failure,
        "status": "PREFLIGHT_BLOCKED(resource)",
        "phase": "cumulative_preflight_budget_admission",
    }

    canary._validate_historical_preflight(canary.Path("preflight_07.json"), old_failure)
    canary._validate_historical_preflight(canary.Path("preflight_08.json"), budget_stop)
    repair_failure = {
        **old_failure,
        "authorization_id": canary.PREVIOUS_AUTHORIZATION_ID,
        "phase": "resolve_authorities_and_fixed_populations",
    }
    canary._validate_historical_preflight(canary.Path("preflight_10.json"), repair_failure)
    ready = {
        **old_failure,
        "authorization_id": canary.PREVIOUS_AUTHORIZATION_ID,
        "status": canary.PREFLIGHT_READY,
        "phase": None,
        "scientific_optimizer_updates": None,
        "checks": {canary.CORRECTNESS_INVARIANTS: {"scientific_optimizer_updates": 0}},
    }
    canary._validate_historical_preflight(canary.Path("preflight_09.json"), ready)
    previous_budget_stop = {
        **old_failure,
        "authorization_id": canary.PREVIOUS_AUTHORIZATION_ID,
        "status": "PREFLIGHT_BLOCKED(resource)",
        "phase": "remaining_total_budget_admission",
    }
    canary._validate_historical_preflight(canary.Path("preflight_11.json"), previous_budget_stop)
    assert canary._has_zero_scientific_updates(ready)
    assert canary._has_zero_scientific_updates({**ready, "checks": {"scientific_optimizer_updates": 0}})
    with pytest.raises(canary.CanaryError, match="historical task identity"):
        canary._validate_historical_preflight(
            canary.Path("preflight_08.json"),
            {**budget_stop, "authorization_id": canary.AUTHORIZATION_ID},
        )

def test_recovery_adam_equivalence_uses_registered_float_tolerance_and_exact_step() -> None:
    uninterrupted = {
        "parameter_groups": [{"parameter_names": ["encoder.phase"], "configuration": {"lr": 1.0e-3}}],
        "states_by_parameter_name": {
            "encoder.phase": {
                "step": torch.tensor(1221.0),
                "exp_avg": torch.tensor([1.0, 0.5]),
                "exp_avg_sq": torch.tensor([0.1, 0.2]),
            }
        },
    }
    resumed = {
        "parameter_groups": [{"parameter_names": ["encoder.phase"], "configuration": {"lr": 1.0e-3}}],
        "states_by_parameter_name": {
            "encoder.phase": {
                "step": torch.tensor(1221.0),
                "exp_avg": torch.tensor([1.0 + 2.9802322e-8, 0.5]),
                "exp_avg_sq": torch.tensor([0.1, 0.2]),
            }
        },
    }

    assert canary._optimizer_state_recovery_equivalence(uninterrupted, resumed)["passed"] is True
    resumed["states_by_parameter_name"]["encoder.phase"]["step"] = torch.tensor(1222.0)
    report = canary._optimizer_state_recovery_equivalence(uninterrupted, resumed)
    assert report["passed"] is False
    assert report["state_by_parameter_name"]["encoder.phase"]["slots"]["step"]["exact"] is False
