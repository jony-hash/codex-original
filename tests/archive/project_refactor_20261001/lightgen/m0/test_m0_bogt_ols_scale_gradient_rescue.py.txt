class _ForwardResult:
    def __init__(self, raw_intensity: torch.Tensor) -> None:
        self.raw_intensity = raw_intensity


class _LinearRuntime(nn.Module):
    def forward_from_ols(self, coefficients: torch.Tensor) -> _ForwardResult:
        value = 0.1 + 100.0 * coefficients.real.sum(dim=1) + 70.0 * coefficients.imag.sum(dim=1)
        raw = value[:, None, None, None].expand(-1, 1, 200, 200)
        return _ForwardResult(raw)


def _fixture() -> tuple[SimpleNamespace, torch.Tensor, torch.Tensor]:
    coefficients = torch.full((rescue.FIT_ROWS, rescue.FEATURES), 1.0e-5 + 2.0e-5j, dtype=torch.complex64)
    targets = torch.zeros((rescue.FIT_ROWS, 1, 28, 28), dtype=torch.float32)
    targets[:, :, 7:21, 8:20] = 1.0
    return SimpleNamespace(runtime=_LinearRuntime()), coefficients, targets


def test_fixed_scale_and_projection_enforce_per_row_relative_radius() -> None:
    _, coefficients, _ = _fixture()
    scales = rescue._initial_scales(coefficients)
    assert scales.shape == (rescue.FIT_ROWS,)
    assert torch.allclose(scales, torch.linalg.vector_norm(coefficients.abs(), dim=1) / rescue.FEATURES**0.5)

    candidates = torch.zeros_like(coefficients)
    candidates[0] = 2.0 + 1.0j
    candidates[1] = 0.5 - 0.25j
    projected = rescue._project_dimensionless(candidates)
    norms = torch.linalg.vector_norm(projected.abs().to(torch.float64), dim=1)
    assert float(norms.max()) <= rescue.TRUST_RADIUS + 1.0e-6
    assert torch.allclose(projected[1], candidates[1])
    relative_offsets = scales * norms / torch.linalg.vector_norm(coefficients.abs().to(torch.float64), dim=1)
    assert float(relative_offsets.max()) <= 1.0 + 1.0e-6


def test_fixed_inputs_are_consumed_from_bound_train_artifact_only() -> None:
    indices = torch.arange(rescue.FIT_ROWS, dtype=torch.long)
    labels = torch.arange(rescue.FIT_ROWS, dtype=torch.long).remainder(10)
    targets = torch.zeros((rescue.FIT_ROWS, 1, 28, 28), dtype=torch.float32)
    targets[:, :, 8:20, 8:20] = 1.0
    ols = torch.full((rescue.FIT_ROWS, rescue.FEATURES), 1.0e-5 + 2.0e-5j, dtype=torch.complex64)
    data = rescue._fixed_data_from_prior(
        {"fit_indices": indices, "fit_labels": labels, "fit_targets_28x28": targets, "fit_ols_step600": ols},
        torch.device("cpu"),
    )
    assert torch.equal(data["fit_indices"], indices)
    assert torch.equal(data["fit_targets"], targets)
    assert torch.equal(data["fit_ols"], ols)
    unbalanced_labels = labels.clone()
    unbalanced_labels[0] = 1
    changed = {"fit_indices": indices, "fit_labels": unbalanced_labels, "fit_targets_28x28": targets, "fit_ols_step600": ols}
    with pytest.raises(rescue.ScaleRescueError, match="not class balanced"):
        rescue._fixed_data_from_prior(changed, torch.device("cpu"))


def test_schedule_updates_every_row_exactly_300_times() -> None:
    counts = [0] * rescue.FIT_ROWS
    for global_step in range(rescue.GLOBAL_UPDATES):
        for row in rescue._scheduled_rows(global_step):
            counts[row] += 1
    assert counts == [rescue.UPDATES_PER_ROW] * rescue.FIT_ROWS
    assert rescue._scheduled_rows(0) == list(range(10))
    assert rescue._scheduled_rows(rescue.CHUNKS_PER_PASS) == list(range(10))
    assert rescue._update_count_bounds(rescue.CHUNKS_PER_PASS - 1) == {"min": 0, "max": 1}
    assert rescue._update_count_bounds(rescue.CHUNKS_PER_PASS) == {"min": 1, "max": 1}
    assert rescue._update_count_bounds(rescue.GLOBAL_UPDATES) == {"min": 300, "max": 300}


def test_production_update_changes_only_selected_rows_and_zero_loss_control() -> None:
    consumer, coefficients, targets = _fixture()
    scales = rescue._initial_scales(coefficients)
    rows, optimizers = rescue._new_rows_and_optimizers(coefficients, torch.device("cpu"))
    before = torch.stack([row.detach().clone() for row in rows])
    result = rescue._apply_row_update(consumer, rows, optimizers, coefficients, scales, targets, list(range(rescue.BATCH_SIZE)))
    after = torch.stack([row.detach() for row in rows])
    assert result["zero_gradient_rows"] == 0
    assert not torch.equal(before[:rescue.BATCH_SIZE], after[:rescue.BATCH_SIZE])
    assert torch.equal(before[rescue.BATCH_SIZE:], after[rescue.BATCH_SIZE:])

    control_rows, control_optimizers = rescue._new_rows_and_optimizers(coefficients, torch.device("cpu"))
    control_before = torch.stack([row.detach().clone() for row in control_rows])
    control = rescue._apply_row_update(consumer, control_rows, control_optimizers, coefficients, scales, targets, list(range(rescue.BATCH_SIZE)), enabled=False)
    assert control["zero_gradient_rows"] == rescue.BATCH_SIZE
    assert torch.equal(control_before, torch.stack([row.detach() for row in control_rows]))

def test_retry_preflight_accepts_only_the_bound_first_signature(monkeypatch: pytest.MonkeyPatch, tmp_path) -> None:
    path = tmp_path / rescue.PREFLIGHT_01_PATH
    path.parent.mkdir(parents=True)
    path.write_text("record", encoding="utf-8")
    record = {
        "schema": rescue.PREFLIGHT_SCHEMA,
        "task_id": rescue.TASK_ID,
        "outer_gate": rescue.OUTER_GATE,
        "authorization_id": rescue.AUTHORIZATION_ID,
        **rescue.PREFLIGHT_RETRY_SIGNATURE,
    }
    monkeypatch.setattr(rescue, "file_sha256", lambda candidate: rescue.PREFLIGHT_01_SHA256)
    monkeypatch.setattr(rescue, "_read_json", lambda candidate: record)
    assert rescue._validate_retry_preflight(tmp_path)["recheck_sequence"] == 2
    record["reason"] = "KeyError: 'different field'"
    with pytest.raises(rescue.ScaleRescueError, match="exact first source-bound harness failure"):
        rescue._validate_retry_preflight(tmp_path)
