def test_joint_canonical_operator_normalizes_over_the_concatenated_parameters() -> None:
    gradients = {
        "E_S": (torch.tensor([3.0, 4.0]), torch.tensor([12.0])),
        "E_I": (torch.tensor([0.0, 2.0]), torch.tensor([0.0])),
        "E_C": (torch.tensor([1.0, 0.0]), torch.tensor([2.0])),
    }

    combined, metrics = canary._joint_canonical_gradient(gradients, delta_sic=1.0)

    expected_units = [
        (torch.tensor([3.0, 4.0]) / 13.0, torch.tensor([12.0]) / 13.0),
        (torch.tensor([0.0, 2.0]) / 2.0, torch.tensor([0.0])),
        (torch.tensor([1.0, 0.0]) / math.sqrt(5.0), torch.tensor([2.0]) / math.sqrt(5.0)),
    ]
    summed = tuple(sum(unit[index] for unit in expected_units) for index in range(2))
    rms = math.sqrt(sum(float(value.square().sum()) for value in summed) / 3.0)
    assert torch.allclose(combined[0], summed[0] / rms)
    assert torch.allclose(combined[1], summed[1] / rms)
    assert metrics["normalization_scope"] == "joint_encoder_generator_vector"
    assert "L_canonical" not in metrics


def test_kappa_is_the_median_of_exact_batchwise_weighted_reconstruction_ratios() -> None:
    rec = [(torch.tensor([3.0 * float(index + 1)]), torch.tensor([0.0])) for index in range(20)]
    canonical = [(torch.tensor([2.0]), torch.tensor([0.0])) for _ in range(20)]

    first, rows = canary._calibrate_kappa_from_gradients(rec, canonical, beta=3.0)
    second, rows_again = canary._calibrate_kappa_from_gradients(rec, canonical, beta=3.0)

    assert first == second
    assert rows == rows_again
    assert len(rows) == 20
    assert rows[0]["ratio"] == 1.5
    assert first == 15.75


def test_recovery_tensor_diff_reports_registered_float32_roundoff() -> None:
    expected = torch.tensor([0.25, 1.0], dtype=torch.float32)
    replayed = torch.tensor([0.25 + 2.9802322e-8, 1.0], dtype=torch.float32)

    report = canary._tensor_equivalence_metrics(
        expected,
        replayed,
        rtol=canary.accepted_canonical.RECOVERY_UPDATE_RTOL,
        atol=canary.accepted_canonical.RECOVERY_UPDATE_ATOL,
    )

    assert report["exact"] is False
    assert report["within_tolerance"] is True
    assert report["max_abs_diff"] == pytest.approx(2.9802322e-8)
    assert report["rtol"] == canary.accepted_canonical.RECOVERY_UPDATE_RTOL
    assert report["atol"] == canary.accepted_canonical.RECOVERY_UPDATE_ATOL
    outside = torch.tensor([0.25 + 2.0e-6, 1.0], dtype=torch.float32)
    assert not canary._tensor_equivalence_metrics(
        expected,
        outside,
        rtol=canary.accepted_canonical.RECOVERY_UPDATE_RTOL,
        atol=canary.accepted_canonical.RECOVERY_UPDATE_ATOL,
    )["within_tolerance"]

def test_optimizer_consumes_the_joint_gradient_sum_once() -> None:
    class Group(torch.nn.Module):
        def __init__(self) -> None:
            super().__init__()
            self.phase = torch.nn.Parameter(torch.tensor([1.0, -1.0]))

    class Runtime(torch.nn.Module):
        def __init__(self) -> None:
            super().__init__()
            self.encoder = Group()
            self.generator = Group()

    class Consumer:
        def __init__(self) -> None:
            self.runtime = Runtime()

    consumer = Consumer()
    parameters = [parameter for _, parameter in consumer.runtime.named_parameters()]
    optimizer = torch.optim.Adam(parameters, lr=1.0e-3)
    bogt_gradient = (torch.tensor([1.0, 2.0]), torch.tensor([-1.0, 3.0]))
    canonical_gradient = (torch.tensor([0.5, -0.5]), torch.tensor([2.0, -2.0]))

    result = canary._apply_update_vectors(consumer, optimizer, G_BOGT=bogt_gradient, G_canonical=canonical_gradient, kappa=2.0)

    assert torch.equal(result["G_total"][0], torch.tensor([2.0, 1.0]))
    assert torch.equal(result["G_total"][1], torch.tensor([3.0, -1.0]))
    assert all(int(state["step"].item()) == 1 for state in optimizer.state.values())
    assert all(value > 0.0 for value in result["group_update_norms"].values())

def _legacy_step600_checkpoint() -> dict[str, object]:
    # Mirrors the accepted m0_bogt_semantic_generation_checkpoint.v1 contract:
    # frozen_runtime_state_sha256 is absent, but the producer records runtime identity.
    return {
        "schema": "m0_bogt_semantic_generation_checkpoint.v1",
        "trainable_parameter_state": {
            "encoder.phase": torch.zeros(1),
            "generator.phase": torch.zeros(1),
        },
        "run_identity": {
            "config_identity": {
                "runtime": {
                    "frozen": ["OLS", "W_OLS", "optical geometry", "training evaluator", "independent E_D"],
                    "propagation_grid_size": 2048,
                    "trainable_parameter_names": ["encoder.phase", "generator.phase"],
                }
            },
            "model_identity": {
                "runtime": "LIGHTGEN_2D_CHEN_BOGT_COMPLEX_OLS_CONSUMER_V1",
                "trainable": ["encoder.phase", "generator.phase"],
                "training_evaluator": "FrozenMNISTEvaluator v1",
                "independent_evaluator": "SIC_INDEPENDENT_CONFIRMATION_EVALUATOR_D_V1",
            },
        },
    }


def _runtime_contract_consumer(frozen_hash: str = "frozen-digest") -> SimpleNamespace:
    encoder = torch.nn.Parameter(torch.zeros(1))
    generator = torch.nn.Parameter(torch.zeros(1))
    transport = torch.eye(100, dtype=torch.complex64)
    runtime = SimpleNamespace(
        ols=SimpleNamespace(transport=transport),
        named_parameters=lambda: (("encoder.phase", encoder), ("generator.phase", generator)),
        parameters=lambda: (encoder, generator),
        eval=lambda: None,
    )
    return SimpleNamespace(
        runtime=runtime,
        frozen_state_sha256=frozen_hash,
        runtime_identity=lambda: {
            "identity": "LIGHTGEN_2D_SEMANTIC_GENERATION_INPUT_OLS_ALIGNMENT_RUNTIME_V1",
            "module_sha256": "runtime-module-digest",
            "config": {"propagation_grid_size": 2048},
            "transport": "T=I_100 coefficient space",
            "transport_sha256": "transport-digest",
            "production_propagation_grid": 2048,
        },
        trainable_state_dict=lambda: {"encoder.phase": encoder, "generator.phase": generator},
        create_optimizer=lambda: pytest.fail("identity-only baseline must not create an optimizer"),
    )

@pytest.mark.parametrize(
    ("serialized_hash", "runtime_grid", "should_pass", "digest_source"),
    [
        (None, 2048, True, "reconstructed_from_source_bound_runtime_contract"),
        (None, 128, False, None),
        ("frozen-digest", 2048, True, "serialized_checkpoint_field"),
        ("wrong-digest", 2048, False, None),
    ],
)
def test_production_step600_dev_baseline_setup_uses_authoritative_runtime_identity(
    monkeypatch, tmp_path, serialized_hash, runtime_grid: int, should_pass: bool, digest_source: str | None
) -> None:
    checkpoint = _legacy_step600_checkpoint()
    checkpoint["optimizer_state"] = {"state": {"step": torch.tensor(600)}}
    checkpoint["rng_state"] = {}
    if serialized_hash is None:
        assert "frozen_runtime_state_sha256" not in checkpoint
    else:
        checkpoint["frozen_runtime_state_sha256"] = serialized_hash
    consumer = _runtime_contract_consumer()
    base_runtime_identity = consumer.runtime_identity

    def runtime_identity():
        result = base_runtime_identity()
        result["config"]["propagation_grid_size"] = runtime_grid
        result["production_propagation_grid"] = runtime_grid
        return result

    consumer.runtime_identity = runtime_identity
    parameters_before = [parameter.detach().clone() for _, parameter in consumer.runtime.named_parameters()]
    optimizer_state_before = copy.deepcopy(checkpoint.get("optimizer_state"))
    resolver_calls: list[dict[str, object]] = []
    measurement_calls: list[str] = []
    weights_reads: list[str] = []

    def resolve(_root, observed_checkpoint, _context):
        resolver_calls.append({"checkpoint": observed_checkpoint, "optimizer_created": False})
        return consumer, canary._validate_step600_runtime_contract(observed_checkpoint, consumer)

    def measure_optical(*args, **kwargs):
        measurement_calls.append("optical")
        assert kwargs["expected_frozen_runtime_state_sha256"] == "frozen-digest"
        assert all(torch.equal(before, parameter.detach()) for before, (_, parameter) in zip(parameters_before, consumer.runtime.named_parameters(), strict=True))
        return {"L_rec": 0.066799}, torch.ones(1)

    monkeypatch.setattr(canary, "_load_step600_runtime_with_identity", resolve, raising=False)
    monkeypatch.setattr(canary.readout, "_step600_external_weights", lambda *_args, **_kwargs: (weights_reads.append("read") or object(), object()))
    monkeypatch.setattr(canary.readout, "_measure_dev_optical", measure_optical)

    def measure_ed(evaluator, raw, labels, **_kwargs):
        assert all(not parameter.requires_grad and parameter.grad is None for parameter in evaluator.parameters())
        assert labels.shape == (canary.DEV_ROWS,)
        measurement_calls.append("E_D")
        return {"rows": canary.DEV_ROWS, "macro_recall": 0.1318}

    monkeypatch.setattr(canary.readout, "_measure_ed_on_raw", measure_ed)
    monkeypatch.setattr(canary, "_write_json_or_compare", lambda *_args, **_kwargs: None)
    monkeypatch.setattr(canary, "file_sha256", lambda _path: "step600-checkpoint")
    monkeypatch.setattr(canary.torch.cuda, "is_available", lambda: False)
    independent_evaluator = torch.nn.Linear(1, 1)
    for parameter in independent_evaluator.parameters():
        parameter.requires_grad_(False)

    class BaselineContext:
        def __init__(self, evaluator):
            self.data_identity = {"identity": "accepted"}
            self.independent_evaluator = evaluator
            self.dataset_accesses: list[str] = []

        @property
        def labels(self):
            self.dataset_accesses.append("labels")
            return torch.zeros(canary.DEV_ROWS, dtype=torch.long)

        @property
        def development_indices(self):
            self.dataset_accesses.append("development_indices")
            return torch.arange(canary.DEV_ROWS)

    context = BaselineContext(independent_evaluator)
    attempt = tmp_path / "attempt"
    attempt.mkdir()
    monkeypatch.setattr(canary, "_load_training_evaluators", lambda *_args, **_kwargs: (torch.nn.Identity(), torch.nn.Identity(), torch.nn.Identity()))
    monkeypatch.setattr(canary.bogt, "restore_rng_state", lambda _state: None)

    if should_pass:
        _consumer, optimizer, _evaluators = canary._load_training_runtime(tmp_path, checkpoint, context, create_optimizer=False)
        assert optimizer is None
        assert len(resolver_calls) == 1
        report = canary._measure_full_dev(
            tmp_path,
            checkpoint,
            context,
            {},
            SimpleNamespace(sample=lambda *_args, **_kwargs: None),
            attempt,
            "source-manifest",
            step=canary.START_STEP,
            task_started=0.0,
        )
        assert report["runtime_identity"]["frozen_runtime_digest_source"] == digest_source
        assert report["E_D"]["rows"] == canary.DEV_ROWS
        assert measurement_calls == ["optical", "E_D"]
        assert weights_reads == ["read"]
        assert context.dataset_accesses == ["labels", "development_indices"]
    else:
        with pytest.raises(canary.CanaryError):
            canary._measure_full_dev(
                tmp_path,
                checkpoint,
                context,
                {},
                SimpleNamespace(sample=lambda *_args, **_kwargs: None),
                attempt,
                "source-manifest",
                step=canary.START_STEP,
                task_started=0.0,
            )
        assert measurement_calls == []
        assert weights_reads == []
        assert context.dataset_accesses == []
    assert len(resolver_calls) == (2 if should_pass else 1)
    assert all(call == {"checkpoint": checkpoint, "optimizer_created": False} for call in resolver_calls)
    assert checkpoint.get("optimizer_state") == optimizer_state_before
    assert all(parameter.grad is None for parameter in context.independent_evaluator.parameters())
    assert all(torch.equal(before, parameter.detach()) for before, (_, parameter) in zip(parameters_before, consumer.runtime.named_parameters(), strict=True))

def _toy_gradient_case():
    class Group(torch.nn.Module):
        def __init__(self, value: float) -> None:
            super().__init__()
            self.phase = torch.nn.Parameter(torch.tensor([value], dtype=torch.float32))

    class Runtime(torch.nn.Module):
        def __init__(self) -> None:
            super().__init__()
            self.encoder = Group(0.3)
            self.generator = Group(-0.2)

    class Consumer:
        def __init__(self) -> None:
            self.runtime = Runtime()
            self.device = torch.device("cpu")
            self.weights = SimpleNamespace(alpha=0.7, beta=1.2, gamma=0.4)
            self.raw_refs: list[weakref.ReferenceType[torch.Tensor]] = []
            self.forward_count = 0

        def forward_loss(self, images: torch.Tensor, *, data_role: str):
            assert data_role == canary.bogt.TRAIN_ROLE
            self.forward_count += 1
            encoder = self.runtime.encoder.phase
            generator = self.runtime.generator.phase
            axis = torch.linspace(0.0, 1.0, 28 * 28, dtype=images.dtype).reshape(1, 1, 28, 28)
            raw = torch.sigmoid(encoder * axis + generator * axis.flip(-1) + 0.1).expand_as(images)
            self.raw_refs.append(weakref.ref(raw))
            l_dist = encoder.square().mean()
            l_rec = (raw - images).square().mean()
            l_op = generator.square().mean()
            result = SimpleNamespace(raw_intensity=raw)
            return SimpleNamespace(
                result=result,
                l_dist=l_dist,
                l_rec=l_rec,
                l_op=l_op,
                l_total=self.weights.alpha * l_dist + self.weights.beta * l_rec + self.weights.gamma * l_op,
            )

    class Evaluator(torch.nn.Module):
        def __init__(self, offset: float) -> None:
            super().__init__()
            weights = torch.arange(28 * 28 * 128, dtype=torch.float32).reshape(28 * 28, 128)
            self.register_buffer("projection", torch.sin(weights * 0.00017 + offset))

        def forward(self, images: torch.Tensor):
            features = images.flatten(1) @ self.projection
            return features, features

    consumer = Consumer()
    evaluators = tuple(Evaluator(offset).eval() for offset in (0.1, 0.4, 0.9))
    for evaluator in evaluators:
        for parameter in evaluator.parameters():
            parameter.requires_grad_(False)
            parameter.grad = None
    images = torch.rand((10, 1, 28, 28), generator=torch.Generator().manual_seed(123))
    return consumer, images, evaluators


def test_gradient_buffer_snapshot_reports_parameter_and_element_support() -> None:
    consumer, _, _ = _toy_gradient_case()
    parameters = canary._parameter_tuple(consumer)

    assert snapshot["first_nonzero_parameter_names"] == ["encoder.phase"]
    assert snapshot["clean"] is False


def test_disposable_restore_clears_gradients_without_changing_restored_state(tmp_path) -> None:
    class Group(torch.nn.Module):
        def __init__(self, value: float) -> None:
            super().__init__()
            self.phase = torch.nn.Parameter(torch.tensor([value], dtype=torch.float32))

    class Runtime(torch.nn.Module):
        def __init__(self) -> None:
            super().__init__()
            self.encoder = Group(0.25)
            self.generator = Group(-0.5)

    class Consumer:
        def __init__(self) -> None:
            self.runtime = Runtime()

        def trainable_state_dict(self):
            return {name: parameter.detach().cpu().clone() for name, parameter in self.runtime.named_parameters()}

        def load_trainable_state_dict(self, state):
            with torch.no_grad():
                for name, parameter in self.runtime.named_parameters():
                    parameter.copy_(state[name])

    class Sampler:
        def __init__(self) -> None:
            self.cursor = 600

        def state_dict(self):
            return {"cursor": self.cursor}

        def load_state_dict(self, state):
            self.cursor = int(state["cursor"])

    consumer = Consumer()
    parameters = canary._parameter_tuple(consumer)
    optimizer = torch.optim.Adam(parameters, lr=1.0e-2)
    sampler = Sampler()
    original_parameters = canary._trainable_state(consumer)
    original_optimizer = optimizer.state_dict()
    original_sampler = sampler.state_dict()
    original_rng = canary.bogt._rng_state_payload()

    canary._set_gradients(parameters, (torch.tensor([1.0]), torch.tensor([-2.0])))
    optimizer.step()
    sampler.cursor = 601
    assert all(parameter.grad is not None for parameter in parameters)

    # Reproduce the previous restore contract: model/Adam/cursor/RNG load leaves .grad behind.
    consumer.load_trainable_state_dict(original_parameters)
    optimizer.load_state_dict(original_optimizer)
    sampler.load_state_dict(original_sampler)
    canary.bogt.restore_rng_state(original_rng)
    assert all(parameter.grad is not None for parameter in parameters)

    ledger_path = tmp_path / "gradient_buffers.jsonl"
    recorder = canary.PhaseResourceRecorder(torch.device("cpu"), gradient_ledger_path=ledger_path)
    restored = canary._restore_disposable_preflight_state(
        consumer,
        optimizer,
        sampler,
        trainable_state=original_parameters,
        optimizer_state=original_optimizer,
        sampler_state=original_sampler,
        rng_state=original_rng,
        recorder=recorder,
        phase_prefix="negative_control_restore",
    )

    assert restored["pre_clear_grad_is_not_none_count"] == 2
    assert restored["post_clear_grad_is_not_none_count"] == 0
    assert restored["parameters_unchanged_by_clear"] is True
    assert restored["optimizer_state_unchanged_by_clear"] is True
    assert restored["clean_after_restore"] is True
    assert sampler.cursor == 600
    assert canary._tree_equal(canary._trainable_state(consumer), original_parameters)
    assert canary._tree_equal(optimizer.state_dict(), original_optimizer)

def test_sequential_canonical_gradient_matches_accepted_joint_operator() -> None:
    consumer, images, evaluators = _toy_gradient_case()
    parameters = canary._parameter_tuple(consumer)
    initial_parameters = [parameter.detach().clone() for parameter in parameters]
    evaluator_hashes = [canary._module_state_sha256(evaluator) for evaluator in evaluators]
    rng_before = torch.random.get_rng_state().clone()

    reference_forward = consumer.forward_loss(images, data_role=canary.bogt.TRAIN_ROLE)
    reference_losses = canary._canonical_losses_from_raw(
        reference_forward.result.raw_intensity, images, *evaluators
    )
    reference_components = {}
    for index, name in enumerate(("E_S", "E_I", "E_C")):
        gradients = torch.autograd.grad(
            reference_losses[name], parameters, retain_graph=index < 2, allow_unused=True
        )
        reference_components[name] = canary._gradient_tuple(gradients, parameters)
    reference, reference_metrics = canary._joint_canonical_gradient(
        reference_components, delta_sic=canary.three.DELTA_SIC
    )
    del reference_forward, reference_losses
    gc.collect()

    sequential, metrics, losses, components = canary._sequential_canonical_gradient(
        consumer, images, evaluators, keep_components=True
    )

    assert consumer.forward_count == 4  # one reference forward and one recompute per evaluator
    assert torch.allclose(sequential[1], reference[1], rtol=2.0e-5, atol=2.0e-7)
    assert torch.allclose(sequential[0], reference[0], rtol=2.0e-5, atol=2.0e-7)
    for name in ("E_S", "E_I", "E_C"):
        ref_unit = tuple(value / reference_metrics["evaluator_joint_gradient_norms"][name] for value in reference_components[name])
        assert all(torch.allclose(actual, expected, rtol=2.0e-5, atol=2.0e-7) for actual, expected in zip(components[name], ref_unit, strict=True))
        assert math.isclose(metrics["evaluator_joint_gradient_norms"][name], reference_metrics["evaluator_joint_gradient_norms"][name], rel_tol=2.0e-5, abs_tol=2.0e-7)
    assert math.isclose(canary._joint_norm(sequential), canary._joint_norm(reference), rel_tol=2.0e-5, abs_tol=2.0e-7)
    assert math.isclose(canary._joint_cosine(sequential, reference), 1.0, rel_tol=0.0, abs_tol=2.0e-6)
    assert set(losses) == {"E_S", "E_I", "E_C"}
    assert all(parameter.grad is None for parameter in parameters)
    assert all(torch.equal(parameter, previous) for parameter, previous in zip(parameters, initial_parameters, strict=True))
    assert evaluator_hashes == [canary._module_state_sha256(evaluator) for evaluator in evaluators]
    assert torch.equal(torch.random.get_rng_state(), rng_before)

def test_optimizer_recovery_equivalence_allows_registered_float_roundoff_but_requires_exact_adam_step() -> None:

def test_next_unconsumed_cursor_resolves_step604_not_previous_step603() -> None:
    context, saved, permutations = _toy_authoritative_train_context_and_sampler()
    sampler = canary.ExtendedTrainSampler(context, saved)
    history = []
    for position in range(canary.END_STEP):
        indices = torch.stack([permutation[position] for permutation in permutations])
        history.append({"step": position + 1, "train_indices_sha256": canary.tensor_sha256(indices)})

    for _ in range(3):
        sampler.next_batch()
    assert sampler.cursor == 603
    expected = canary._expected_train_record(history, sampler.cursor)
    actual_step604 = torch.stack([permutation[sampler.cursor] for permutation in permutations])

    assert expected["step"] == 604
    assert expected["train_indices_sha256"] == canary.tensor_sha256(actual_step604)
    assert history[sampler.cursor - 1]["step"] == 603
    assert history[sampler.cursor - 1]["train_indices_sha256"] != canary.tensor_sha256(actual_step604)


def test_authoritative_uninterrupted_and_saved_restored_rows_match_steps601_through608() -> None:
    context, saved, permutations = _toy_authoritative_train_context_and_sampler()
    history = []
    authoritative = {}
    for step in range(canary.START_STEP + 1, canary.START_STEP + 9):
        position = step - 1
        indices = torch.stack([permutation[position] for permutation in permutations])
        authoritative[step] = indices.tolist()
        history.append({"step": step, "train_indices_sha256": canary.tensor_sha256(indices)})
    # Prefix rows are represented by earlier checkpoint history; append placeholder
    # rows so the production history's global index remains aligned.
    full_history = [{"step": index + 1, "train_indices_sha256": "prior"} for index in range(canary.START_STEP)] + history

    uninterrupted = canary.ExtendedTrainSampler(context, saved)
    rows_b = {}
    for step in range(canary.START_STEP + 1, canary.START_STEP + 9):
        cursor_before = uninterrupted.cursor
        expected = canary._expected_train_record(full_history, cursor_before)
        indices, labels = uninterrupted.next_batch()
        assert expected["step"] == step
        assert labels.tolist() == list(range(10))
        rows_b[step] = indices.tolist()

    resumed = canary.ExtendedTrainSampler(context, saved)
    rows_c = {}
    for _ in range(3):
        step = resumed.cursor + 1
        indices, _labels = resumed.next_batch()
        rows_c[step] = indices.tolist()
    saved_state = resumed.state_dict()
    resumed = canary.ExtendedTrainSampler(context, saved)
    resumed.load_state_dict(saved_state)
    for step in range(canary.START_STEP + 4, canary.START_STEP + 9):
        expected = canary._expected_train_record(full_history, resumed.cursor)
        indices, _labels = resumed.next_batch()
        assert expected["step"] == step
        rows_c[step] = indices.tolist()

    assert rows_b == authoritative
    assert rows_c == authoritative
    assert uninterrupted.cursor == resumed.cursor == canary.START_STEP + 8
