def _fixed_population() -> tuple[torch.Tensor, torch.Tensor, torch.Tensor, torch.Tensor, torch.Tensor]:
    fit_indices = torch.arange(200, dtype=torch.long)
    fit_labels = torch.arange(10, dtype=torch.long).repeat(20)
    observation_indices = torch.arange(200, 300, dtype=torch.long)
    observation_labels = torch.arange(10, dtype=torch.long).repeat(10)
    fit_targets = torch.zeros((200, 1, 28, 28), dtype=torch.float32)
    return fit_indices, fit_labels, observation_indices, observation_labels, fit_targets


def test_fixed_population_accepts_exact_class_interleaved_rows_and_rejects_reordering() -> None:
    fit_indices, fit_labels, obs_indices, obs_labels, targets = _fixed_population()
    task._validate_fixed_population(fit_indices, fit_labels, obs_indices, obs_labels, targets)

    changed = fit_labels.clone()
    changed[0], changed[1] = changed[1].clone(), changed[0].clone()
    with pytest.raises(task.AttributionError, match="class-interleaved order"):
        task._validate_fixed_population(fit_indices, changed, obs_indices, obs_labels, targets)

def test_step2200_invalid_authority_uses_the_saved_string_failure_schema(tmp_path: Path) -> None:
    status_path = tmp_path / "status.json"
    checkpoint_path = tmp_path / "scientific_step_002200.pt"
    checkpoint_path.write_bytes(b"step2200")
    status = {
        "state": task.run_control.RUN_INVALID_HARNESS,
        "phase": "FINALIZING",
        "failure": "CanaryError: saved S1 confirmation artifact has stale lineage",
    }
    status_path.write_text(json.dumps(status), encoding="utf-8")
    report = {
        "task_id": task.small_bank.TASK_ID,
        "attempt": {
            "state": task.run_control.RUN_INVALID_HARNESS,
            "status_sha256": task.file_sha256(status_path),

        "formal_scientific_classification": None,
        "artifacts": {"attempt_artifact_sha256": {"scientific_step_002200.pt": task.file_sha256(checkpoint_path)}},
    }

    assert task._validate_step2200_evidence_binding(report, status, status_path, checkpoint_path) == task.file_sha256(checkpoint_path)

    nested_failure_status = {**status, "failure": {"message": status["failure"]}}
    status_path.write_text(json.dumps(nested_failure_status), encoding="utf-8")
    changed_report = {
        **report,
        "attempt": {**report["attempt"], "status_sha256": task.file_sha256(status_path)},

    with pytest.raises(task.AttributionError, match="step2200 invalid-attempt evidence"):

def test_canonical_gradient_telemetry_adapter_preserves_progress_fields() -> None:
    calls: list[tuple[str, dict[str, object]]] = []

    class Recorder:
        def sample(self, phase: str, *, progress: dict[str, object], started: float | None = None) -> dict[str, object]:
            calls.append((phase, progress))
            return {"phase": phase, "progress": progress}

    adapter = task._CanonicalGradientRecorderAdapter(Recorder())  # type: ignore[arg-type]
    result = adapter.sample("canonical_E_S_before_forward", batch_index=4, evaluator="E_S")

    expected = {"batch_index": 4, "evaluator": "E_S", "data_role": task.bogt.TRAIN_ROLE}
    assert result == {"phase": "canonical_E_S_before_forward", "progress": expected}
    assert calls == [("canonical_E_S_before_forward", expected)]

def test_preflight_persists_authority_resolution_failure_without_opening_attempt(
    tmp_path: Path,
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    run_dir = Path("runs")
    monkeypatch.setattr(task, "RUNS_DIR", run_dir)
    monkeypatch.setattr(task, "PREFLIGHT_PATH", run_dir / "preflight_04.json")
    monkeypatch.setattr(task, "SOURCE_PATH", run_dir / "source_manifest_preflight_04.json")
    monkeypatch.setattr(task, "RESOURCE_PATH", run_dir / "preflight_04_resource_telemetry.jsonl")
    monkeypatch.setattr(task.torch.cuda, "is_available", lambda: True)
    monkeypatch.setattr(task.torch.cuda, "device_count", lambda: 1)

    def reject_authorities(_root: Path) -> dict[str, object]:
        raise task.AttributionError("authority record mismatch")

    monkeypatch.setattr(task, "_load_authorities", reject_authorities)

    with pytest.raises(task.AttributionError, match="authority record mismatch"):
        task._preflight(tmp_path, token=task.AUTHORIZATION_TOKEN)

    receipt = json.loads((tmp_path / run_dir / "preflight_04.json").read_text(encoding="utf-8"))
    assert receipt["phase"] == "authority_resolution_and_runtime_load"
    assert receipt["failure"]["message"] == "authority record mismatch"
    assert receipt["source_manifest"] is None
    assert receipt["attempt_created"] is False
    assert receipt["optimizer_created"] is False
    assert receipt["parameter_updates"] == 0
    assert not (tmp_path / run_dir / "source_manifest_preflight_04.json").exists()


def test_cumulative_preflight_includes_preflight03_and_rejects_a_stale_total() -> None:
    history = {
        "preflight_01": {"authorization_id": task.HISTORICAL_AUTHORIZATION_ID, "elapsed_seconds": 10.7816133},
        "preflight_02": {"authorization_id": task.HISTORICAL_AUTHORIZATION_ID, "cumulative_preflight_elapsed_seconds": 25.2357177},
        "ready_consumer_failure": {"authorization_id": task.HISTORICAL_AUTHORIZATION_ID, "wall_elapsed_seconds": 7.5538133},
        "preflight_03": {"authorization_id": task.HISTORICAL_AUTHORIZATION_ID, "elapsed_seconds": 13.5367598, "cumulative_preflight_elapsed_seconds": 46.3262908},
    }
    historical_elapsed = task._validated_prior_preflight_elapsed(history)
    assert historical_elapsed == pytest.approx(46.3262908)
    assert task._cumulative_preflight_elapsed(
        historical_elapsed,
        task.PREFLIGHT_04_PRIOR_FAILED_INVOCATION_CHARGE_SECONDS,
    ) == pytest.approx(61.3262908)
    history["preflight_02"]["elapsed_seconds"] = 15.0
    with pytest.raises(task.ResourceLimitError, match="preflight_02 elapsed duration conflicts"):
        task._validated_prior_preflight_elapsed(history)
    del history["preflight_02"]["elapsed_seconds"]
    history["preflight_03"]["cumulative_preflight_elapsed_seconds"] = 32.789531
    with pytest.raises(task.AttributionError, match="preflight_03 cumulative"):
        task._validated_prior_preflight_elapsed(history)

def test_gradient_summary_keeps_encoder_generator_groups_and_structural_zero() -> None:

def _candidate_inputs(ols: tuple[float, float], cmos: tuple[float, float], ed: tuple[float, float]) -> tuple[dict[int, dict], dict[int, dict], dict[int, dict]]:
    def metrics(pair: tuple[float, float]) -> dict:
        recalls = [0.0] * 10
        recalls[1], recalls[7] = pair
        return {"per_class_recall": recalls}

    return (
        {1200: {}, 2200: {"observation_metrics": metrics(ols)}},
        {1200: {}, 2200: {"observation_metrics": metrics(cmos)}},
        {1200: {}, 2200: {"observation_metrics": metrics(ed)}},
    )


def test_classification_boundaries_include_pixel_probe_interpretation() -> None:
    ols, cmos, ed = _candidate_inputs((0.0, 0.0), (0.3, 0.4), (0.0, 0.0))
    label, detail = task._classification(ols, cmos, ed)
    assert label == "CONSISTENT_INPUT_SIDE_ATTRIBUTION_CANDIDATE"
    assert detail["CMOS_probe_interpretation"] == "both classes remain readable by the fixed pixel probe"

    ols, cmos, ed = _candidate_inputs((0.1, 0.2), (0.1, 0.0), (0.0, 0.0))
    label, detail = task._classification(ols, cmos, ed)
    assert label == "CONSISTENT_CMOS_REALIZATION_ATTRIBUTION_CANDIDATE"
    assert detail["class_7_cmos_probe_observation_recall_at_least_0_10"] is False

    ols, cmos, ed = _candidate_inputs((0.1, 0.0), (0.1, 0.1), (0.0, 0.0))
    label, _ = task._classification(ols, cmos, ed)
    assert label == "MIXED_OR_INSUFFICIENT_ATTRIBUTION"


def test_completed_report_requires_source_bound_artifact_and_rejects_tampering(tmp_path: Path) -> None:
    attempt = tmp_path / "attempt_01"
    attempt.mkdir()
    source = {"source_manifest_id": "source-1"}
    artifact = attempt / "rows.pt"
    artifact.write_bytes(b"fixed train rows")
    sidecar = {
        "source_manifest_id": source["source_manifest_id"],
        "artifact_kind": "fixed rows",
        "path": artifact.name,
        "sha256": task.file_sha256(artifact),
        "size_bytes": artifact.stat().st_size,
    }
    artifact.with_suffix(artifact.suffix + ".identity.json").write_text(json.dumps(sidecar), encoding="utf-8")
    report = {
        "schema": task.REPORT_SCHEMA,
        "task_id": task.TASK_ID,
        "outer_gate": task.OUTER_GATE,
        "attempt_id": task.ATTEMPT_ID,
        "source_manifest_id": source["source_manifest_id"],
        "result": "VALID_COMPLETED",
        "optimizer_created": False,
        "updates": 0,
        "artifacts": {"rows": {"path": artifact.name, "sha256": sidecar["sha256"], "size_bytes": sidecar["size_bytes"], "artifact_kind": "fixed rows"}},
    }

    task._validate_completed_report(attempt, report, source)
    artifact.write_bytes(b"changed")
    with pytest.raises(task.AttributionError, match="missing or changed"):
        task._validate_completed_report(attempt, report, source)
