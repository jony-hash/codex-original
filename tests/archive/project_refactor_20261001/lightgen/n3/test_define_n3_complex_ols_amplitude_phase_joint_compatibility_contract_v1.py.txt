"""Targeted tests for the definition-only complex OLS compatibility contract."""

from __future__ import annotations

import unittest

import torch

from scripts.research.lightgen.n3 import define_n3_complex_ols_amplitude_phase_joint_compatibility_contract_v1 as contract


class ComplexOlsCompatibilityContractTests(unittest.TestCase):
    def test_canonicalization_is_a_deterministic_existing_quotient_cross_section(self) -> None:
        z = torch.ones((1, contract.DIMENSION), dtype=torch.complex128)
        z[:, 1::2] = 1j
        canonical = contract.canonicalize_existing_global_phase(z)
        self.assertEqual(canonical.dtype, torch.complex128)
        self.assertEqual(tuple(canonical.shape), (1, contract.DIMENSION))
        self.assertEqual(float(canonical[0, 0].imag), 0.0)
        self.assertGreaterEqual(float(canonical[0, 0].real), 0.0)
        rotated = z * torch.exp(torch.tensor(0.731, dtype=torch.float64) * 1j)
        torch.testing.assert_close(
            canonical,
            contract.canonicalize_existing_global_phase(rotated),
            rtol=1e-12,
            atol=1e-12,
        )

    def test_phase_is_complete_principal_interval_and_composition_is_canonicalized(self) -> None:
        phase = torch.linspace(-torch.pi, torch.pi, contract.DIMENSION, dtype=torch.float64).reshape(1, -1)
        amplitude = torch.full_like(phase, 0.5)
        z = contract.compose_complex_from_amplitude_phase(amplitude, phase)
        extracted = contract.extract_complete_relative_phase(z)
        embedding = contract.complex_real_embedding(z)
        self.assertEqual(tuple(extracted.shape), (1, contract.DIMENSION))
        self.assertTrue(torch.all((extracted >= -torch.pi) & (extracted <= torch.pi)).item())
        self.assertEqual(tuple(embedding.shape), (1, 2 * contract.DIMENSION))
        self.assertEqual(embedding.dtype, torch.float64)

    def test_exact_zero_is_fail_closed_and_near_zero_is_not_masked(self) -> None:
        zero = torch.ones((1, contract.DIMENSION), dtype=torch.complex128)
        zero[0, 7] = 0
        with self.assertRaises(contract.InvalidHarnessError):
            contract.extract_complete_relative_phase(zero)
        near_zero = torch.ones((1, contract.DIMENSION), dtype=torch.complex128)
        near_zero[0, 7] = complex(1e-300, -1e-300)
        extracted = contract.extract_complete_relative_phase(near_zero)
        self.assertNotEqual(float(extracted[0, 7]), 0.0)

    def test_embedding_and_seed_rules_are_frozen(self) -> None:
        z = torch.full((2, contract.DIMENSION), 0.25 + 0.75j, dtype=torch.complex128)
        embedding = contract.complex_real_embedding(z)
        self.assertTrue(torch.isfinite(embedding).all().item())
        candidate_seed = contract.derive_seed("candidate", 0, "CANDIDATE_PSEUDO_DATA", 0, None)
        control_seed = contract.derive_seed("control", 0, "CONTROL_PSEUDO_DATA", 0, None)
        self.assertEqual(candidate_seed, contract.derive_seed("candidate", 0, "CANDIDATE_PSEUDO_DATA", 0, None))
        self.assertNotEqual(candidate_seed, control_seed)
        self.assertGreaterEqual(candidate_seed, 0)
        self.assertLessEqual(candidate_seed, contract.MAX_SIGNED_63)

    def test_definition_totality_is_definition_only(self) -> None:
        totality = contract._contract_totality_checks()
        self.assertEqual(totality["status"], "passed")
        self.assertFalse(totality["q_values_loaded"])
        self.assertFalse(totality["complex_ols_values_loaded"])
        self.assertFalse(totality["sealed_values_loaded"])
        self.assertFalse(totality["scientific_evaluation_executed"])
        self.assertEqual(contract.MODEL_BANK_COUNT, 4)
        self.assertEqual(contract.MODEL_BANK_SIZE, 200)
        self.assertEqual(contract.CALIBRATION_TRIAL_COUNT, 999)


if __name__ == "__main__":
    unittest.main()
