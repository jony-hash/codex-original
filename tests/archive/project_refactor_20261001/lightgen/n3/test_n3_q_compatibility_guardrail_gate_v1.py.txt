"""Focused tests for the frozen N3 q-compatibility guardrail calibration."""

from __future__ import annotations

import json
from pathlib import Path

from ldm_project.v2.evidence import file_sha256
from scripts.research.lightgen.n3 import define_n3_q_compatibility_guardrail_gate_v1 as contract, execute_n3_q_compatibility_guardrail_gate_v1 as execution


def _repo_root() -> Path:
    return Path(__file__).resolve().parents[4]


def test_contract_totality_freezes_allowed_two_hundred_trial_fallback() -> None:
    assert contract.PRODUCER_IDS == (
        "EMPIRICAL_MARGINAL_GAUSSIAN_COPULA_AMPLITUDE_V1",
        "EMPIRICAL_MARGINAL_FIXED_R_STUDENT_T_COPULA_V1",
    )
    assert contract.TRIAL_COUNT_PER_PRODUCER_FOLD == 200
    assert len(contract.PRODUCER_IDS) * contract.FOLD_COUNT * contract.TRIAL_COUNT_PER_PRODUCER_FOLD == 2000
    assert contract.SCALAR_METRICS[0] == "energy_distance"
    assert contract.NON_PRIMARY_METRICS == contract.SCALAR_METRICS[1:]
    assert contract.FIVE_FOLD_RELIABILITY_TARGET == 0.80


def test_clopper_pearson_bound_is_exact_and_not_tuned_to_zero_failures() -> None:
    upper_200 = execution.clopper_pearson_upper(0, 200)
    upper_2000 = execution.clopper_pearson_upper(0, 2000)
    assert upper_200 > 0.01
    assert upper_2000 < 0.01
    assert execution.clopper_pearson_upper(200, 200) == 1.0


def test_seed_namespaces_are_deterministic_and_disjoint() -> None:
    first = execution.validate_seed_namespace(execution.PRODUCER_IDS[0], 0, 0)
    second = execution.validate_seed_namespace(execution.PRODUCER_IDS[0], 0, 0)
    other_fold = execution.validate_seed_namespace(execution.PRODUCER_IDS[0], 1, 0)
    other_producer = execution.validate_seed_namespace(execution.PRODUCER_IDS[1], 0, 0)
    assert first == second
    assert set(first).isdisjoint(other_fold)
    assert set(first).isdisjoint(other_producer)


def test_published_contract_is_frozen_and_fail_closed() -> None:
    root = _repo_root()
    path = root / contract.OUTPUT_DIR / contract.CONTRACT_NAME
    payload = json.loads(path.read_text(encoding="utf-8"))
    assert file_sha256(path) == "a87979f9689159eee6d769f10eb6b7e30bcdc9803c4b72b108225a6ea22935b4"
    assert payload["schema"] == contract.CONTRACT_SCHEMA
    assert payload["status"] == "valid_completed"
    assert payload["result"] == "READY"
    assert payload["simulation"]["trials_per_producer_fold"] == 200
    assert payload["hard_guardrail_eligibility"]["criterion"]["no_arbitrary_subset_search"] is True
    assert payload["evaluator"]["primary_energy_distance_unchanged"] is True
    assert payload["firewalls"]["sealed_confirmation_values_loaded"] is False
    assert payload["firewalls"]["training_authorized"] is False


def test_published_report_covers_all_guardrails_and_direct_reliability() -> None:
    root = _repo_root()
    path = root / execution.OUTPUT_DIR / execution.REPORT_NAME
    report = json.loads(path.read_text(encoding="utf-8"))
    assert report["schema"] == execution.REPORT_SCHEMA
    assert report["status"] == "valid_completed"
    assert report["result"] == "VALID_COMPLETED"
    assert report["simulation"]["total_trials"] == 2000
    assert len(report["producer_fold_summaries"]) == 10
    assert set(report["guardrail_calibration"]) == set(execution.SCALAR_METRICS)
    assert set(report["v3"]["hard_guardrails"]).union(report["v3"]["descriptive_only_guardrails"]) == set(execution.SCALAR_METRICS)
    assert not set(report["v3"]["hard_guardrails"]).intersection(report["v3"]["descriptive_only_guardrails"])
    assert set(report["five_fold_reliability"]) == set(execution.PRODUCER_IDS)
    for item in report["five_fold_reliability"].values():
        assert item["direct_calibrated_gate"]["trials"] == 200
        assert 0.0 <= item["direct_calibrated_gate"]["all_five_fold_pass_rate"] <= 1.0
    assert report["evaluator"]["primary_energy_distance_unchanged"] is True
    assert report["firewalls"] == {
        "heldout_q_scientific_evaluation": False,
        "heldout_q_used_for_fit_or_sampling": False,
        "historical_gaussian_rerun": False,
        "historical_student_t_rerun": False,
        "new_producer_family_evaluated": False,
        "arbitrary_guardrail_subset_search": False,
        "post_simulation_threshold_tuning": False,
        "sealed_confirmation_values_loaded": False,
        "n4_semantic_selector": False,
        "optical_or_image_evaluation": False,
        "cuda_or_training": False,
        "selected_producer": None,
        "training_authorized": False,
        "formal_promotion": False,
        "generation_readiness": "not_ready",
    }


def test_v3_artifact_matches_report_when_ready() -> None:
    root = _repo_root()
    report_path = root / execution.OUTPUT_DIR / execution.REPORT_NAME
    report = json.loads(report_path.read_text(encoding="utf-8"))
    artifact = report["v3"]["evaluator_artifact"]
    if report["classification"] == execution.CLASSIFICATION_READY:
        assert artifact is not None
        path = root / artifact["path"]
        assert path.is_file()
        assert file_sha256(path) == artifact["sha256"]
        payload = json.loads(path.read_text(encoding="utf-8"))
        assert payload["schema"] == execution.V3_SCHEMA
        assert payload["result"] == "READY"
        assert payload["primary"]["metric"] == "energy_distance"
        assert payload["primary"]["role"] == "HARD_PRIMARY_UNCHANGED"
    else:
        assert artifact is None
