def _fit_rows() -> dict[str, list[int]]:
    return {str(label): [label * executor.FIT_ROWS_PER_CLASS + row for row in range(executor.FIT_ROWS_PER_CLASS)] for label in range(executor.CLASS_COUNT)}


def _candidate(structural: list[float], semantic: tuple[float, float, float]) -> dict[str, object]:
    return {
        "L_struct_by_class": structural,
        "semantic": dict(zip(("L_S", "L_I", "L_C"), semantic, strict=True)),
    }

def test_factor_initialization_uses_canonical_row_major_basis_and_nested_rank_order() -> None:
    b1, a1 = executor.initialize_factors(1)
    b2, a2 = executor.initialize_factors(2)
    expected_b2 = torch.zeros((executor.PARAMETER_DIMENSION, 2), dtype=torch.float32)
    expected_b2[0, 0] = 1.0
    expected_b2[1, 1] = 1.0
    classes = torch.arange(executor.CLASS_COUNT, dtype=torch.int64)
    expected_b1 = expected_b2[:, :1]
    expected_a1 = executor.INIT_ALPHA * (1.0 - 2.0 * (classes & 1).to(torch.float32)).unsqueeze(1)
    assert torch.equal(b1, expected_b1)
    assert torch.equal(b2, expected_b2)
    assert torch.equal(a1, expected_a1)
    assert torch.equal(b2[:, :1], b1)
    assert torch.equal(a2[:, :1], a1)
    assert b1.dtype == a1.dtype == b2.dtype == a2.dtype == torch.float32
    assert b1.device.type == a1.device.type == b2.device.type == a2.device.type == "cpu"

def test_model_order_prefers_semantic_feasibility_before_structural_gain() -> None:
    baseline = _candidate([10.0] * 10, (1.0, 1.0, 1.0))
    rank1 = _candidate([0.0] * 10, (1.1, 0.8, 0.8))
    rank2 = _candidate([9.0] * 10, (0.9, 0.9, 0.9))
    selected, trace = executor._select_rank({0: baseline, 1: rank1, 2: rank2})
    assert selected == 2
    assert not trace[1]["semantic_feasible"]
    assert trace[2]["semantic_feasible"]


def test_model_order_uses_favorable_class_fraction_after_equal_macro_gain() -> None:
    baseline = _candidate([0.0] * 10, (1.0, 1.0, 1.0))
    rank1 = _candidate([-0.125] * 9 + [0.125], (0.9, 0.9, 0.9))
    rank2 = _candidate([-0.125] * 6 + [-0.0625] * 4, (0.9, 0.9, 0.9))
    selected, trace = executor._select_rank({0: baseline, 1: rank1, 2: rank2})
    assert selected == 2
    assert trace[1]["structural_improvement"] == pytest.approx(trace[2]["structural_improvement"])
    assert trace[1]["favorable_class_fraction"] == 0.9
    assert trace[2]["favorable_class_fraction"] == 1.0


def test_model_order_uses_worst_class_degradation_then_lower_rank() -> None:
    baseline = _candidate([0.0] * 10, (1.0, 1.0, 1.0))
    rank1 = _candidate([-3.0 / 9.0] * 9 + [2.0], (0.9, 0.9, 0.9))
    rank2 = _candidate([-1.5 / 9.0] * 9 + [0.5], (0.9, 0.9, 0.9))
    selected, trace = executor._select_rank({0: baseline, 1: rank1, 2: rank2})
    assert selected == 2
    assert trace[1]["structural_improvement"] == pytest.approx(trace[2]["structural_improvement"])
    assert trace[1]["favorable_class_fraction"] == trace[2]["favorable_class_fraction"]
    assert trace[2]["worst_class_degradation"] < trace[1]["worst_class_degradation"]

    tied, _ = executor._select_rank({0: baseline, 1: rank2, 2: rank2})
    assert tied == 1

def test_windows_dll_bootstrap_prepends_deduplicated_active_environment_paths(
    tmp_path: Path, monkeypatch: pytest.MonkeyPatch
) -> None:
    prefix = tmp_path / "env"
    directories = [prefix, prefix / "Library" / "bin", prefix / "bin", prefix / "Scripts"]
    for directory in directories:
        directory.mkdir(parents=True, exist_ok=True)
    monkeypatch.setattr(
        executor,
        "sys",
        SimpleNamespace(platform="win32", prefix=str(prefix), executable=str(prefix / "Scripts" / "python.exe")),
    )
    monkeypatch.setattr(executor, "_DLL_DIRECTORY_HANDLES", [])
    monkeypatch.setattr(executor, "_DLL_DIRECTORY_KEYS", set())
    registered: list[str] = []

    def add_dll_directory(path: str) -> object:
        registered.append(path)
        return object()

    monkeypatch.setattr(executor.os, "add_dll_directory", add_dll_directory, raising=False)
    monkeypatch.setenv("PATH", os.pathsep.join((str(prefix / "bin"), str(prefix / "bin"), "C:\\Windows\\System32")))

    result = executor._bootstrap_windows_cuda_dll_search()

    assert result["python_prefix"] == str(prefix.resolve())
    assert str((prefix / "bin").resolve()) in result["process_path_prepended"]
    assert len(result["dll_search_directories"]) == len(set(os.path.normcase(path) for path in result["dll_search_directories"]))
    assert len(registered) == len(result["dll_search_directories"])
    current_path = os.environ["PATH"].split(os.pathsep)
    normalized_path = [os.path.normcase(os.path.normpath(os.path.abspath(path))) for path in current_path]
    assert len(normalized_path) == len(set(normalized_path))
    assert normalized_path.count(os.path.normcase(os.path.normpath(os.path.abspath(str(prefix / "bin"))))) == 1

def test_cuda_kernel_smoke_checks_complex_exp_and_runtime_failure_is_fail_closed(monkeypatch: pytest.MonkeyPatch) -> None:
    class FakeTensor:
        def item(self) -> complex:
            return complex(math.e, 0.0)

    synchronized: list[str] = []
    monkeypatch.setattr(executor.torch.cuda, "is_available", lambda: True)
    monkeypatch.setattr(executor.torch.cuda, "device_count", lambda: 1)
    monkeypatch.setattr(executor.torch.cuda, "get_device_name", lambda device: "NVIDIA GeForce RTX 3060 Laptop GPU")
    monkeypatch.setattr(executor.torch.cuda, "synchronize", lambda device: synchronized.append(str(device)))
    monkeypatch.setattr(executor.torch, "tensor", lambda *args, **kwargs: object())
    monkeypatch.setattr(executor.torch, "exp", lambda value: FakeTensor())
    smoke = executor._cuda_runtime_smoke({"dll_path": "env/bin/nvrtc-builtins64_121.dll"})
    assert smoke["status"] == "PASS"
    assert smoke["kernel_probe"] == "torch.exp(complex64[1])"
    assert synchronized == ["cuda:0"]

    monkeypatch.setattr(executor, "_bootstrap_windows_cuda_dll_search", lambda: {"status": "PASS"})
    monkeypatch.setattr(executor, "_resolve_and_probe_nvrtc_builtins", lambda runtime: {"status": "PASS", "dll_path": "test.dll"})
    monkeypatch.setattr(executor, "_cuda_runtime_smoke", lambda nvrtc: (_ for _ in ()).throw(RuntimeError("kernel failure")))
    with pytest.raises(executor.N3RuntimeInitializationError, match="CUDA_KERNEL_SMOKE") as captured:
        executor._prepare_cuda_runtime()
    assert captured.value.runtime_provenance["status"] == "FAILED"
    assert captured.value.runtime_provenance["failed_stage"] == "CUDA_KERNEL_SMOKE"

def test_recovery_roundtrip_binds_fit_history_and_cursor(tmp_path: Path, monkeypatch: pytest.MonkeyPatch) -> None:
    monkeypatch.setattr(
        executor,
        "_rng_snapshot",
        lambda: {"cpu": torch.arange(4, dtype=torch.uint8), "cuda": [], "cuda_available": False},
    )
    model = SimpleNamespace(rank=1, B=torch.nn.Parameter(torch.ones(3)), A=torch.nn.Parameter(torch.zeros(2, 1)))
    optimizer = torch.optim.Adam([model.B, model.A], lr=executor.LEARNING_RATE)
    for _ in range(executor.RECOVERY_INTERVAL):
        optimizer.zero_grad(set_to_none=True)
        (model.B.square().sum() + model.A.square().sum()).backward()
        optimizer.step()
    sampler = executor.N3FitSampler(_fit_rows(), cursor=executor.RECOVERY_INTERVAL)
    history = [{"update": update} for update in range(1, executor.RECOVERY_INTERVAL + 1)]
    manifest = _source_manifest("test-source")
    ledger = executor.SampleReadLedger()
    ledger.mark_backing_loaded()
    ledger.enter_fit()
    rank_dir = tmp_path / "rank_1"
    rank_dir.mkdir()

    recovery_path = executor._save_recovery(
        rank_dir,
        model,
        optimizer,
        sampler,
        executor.RECOVERY_INTERVAL,
        manifest,
        history,
        ledger=ledger,
        attempt_id="test-attempt",
    )
    recovered = executor._load_recovery(recovery_path, 1, manifest, attempt_id="test-attempt")
    assert recovered["sampler_cursor"] == executor.RECOVERY_INTERVAL
    assert len(recovered["history"]) == executor.RECOVERY_INTERVAL
    assert torch.equal(recovered["A"], model.A.detach())
    assert torch.equal(recovered["B"], model.B.detach())
    sidecar = executor._recovery_ledger_path(recovery_path)
    assert json.loads(sidecar.read_text(encoding="utf-8"))["ledger_state"] == ledger.snapshot()


def test_recovery_rejects_missing_or_tampered_ledger_credential(tmp_path: Path, monkeypatch: pytest.MonkeyPatch) -> None:
    monkeypatch.setattr(
        executor,
        "_rng_snapshot",
        lambda: {"cpu": torch.arange(4, dtype=torch.uint8), "cuda": [], "cuda_available": False},
    )
    model = SimpleNamespace(rank=1, B=torch.nn.Parameter(torch.ones(3)), A=torch.nn.Parameter(torch.zeros(2, 1)))

@pytest.mark.parametrize("selected_rank", [0, 2])
def test_select_and_confirm_recovery_credentials_resume_without_domain_reevaluation(
    tmp_path: Path,
    monkeypatch: pytest.MonkeyPatch,
    selected_rank: int,
) -> None:
    root = tmp_path
    attempt = root / executor.ATTEMPT_DIR
    attempt.mkdir(parents=True)
    manifest = _source_manifest(f"stage-source-{selected_rank}")
    attempt_id = f"n3-attempt-{selected_rank}"
    role_indices = {
        "N3_FIT": torch.arange(8, dtype=torch.long),
        "N3_SELECT": torch.arange(8, 16, dtype=torch.long),
        "N3_CONFIRM": torch.arange(16, 24, dtype=torch.long),
    }
    endpoint_paths: dict[int, Path] = {}
    for rank in (0, 1, 2):
        path = attempt / f"rank_{rank}" / f"fit_final_rank_{rank}_step_{executor.FIT_UPDATES:05d}.pt"
        path.parent.mkdir(parents=True)
        path.write_bytes(f"frozen-endpoint-{rank}".encode())
        endpoint_paths[rank] = path

    ledger = executor.SampleReadLedger()
    ledger.mark_backing_loaded()
    ledger.enter_fit()
    ledger.record_fit_endpoint(1, executor.FIT_UPDATES)
    ledger.record_fit_endpoint(2, executor.FIT_UPDATES)
    ledger.enter_selection()
    select_rows = role_indices["N3_SELECT"]
    for rank in (0, 1, 2):
        ledger.record_materialized("N3_SELECT", select_rows, rank=rank, purpose="model_order_candidate")
        ledger.record_evaluated("N3_SELECT", select_rows, rank=rank, purpose="model_order_candidate")
    ledger.verify_selection_evaluations(select_rows.numel())
    selection_document = {
        "schema": "n3_low_rank_model_order_selection.v1",
        "contract_sha256": executor.CONTRACT_SHA256,
        "protocol_sha256": executor.PROTOCOL_SHA256,
        "domain": "N3_SELECT",
        "sample_count": int(select_rows.numel()),
        "indices_sha256": executor.tensor_sha256(select_rows),
        "rank_results": {str(rank): {"rank": rank} for rank in (0, 1, 2)},
        "semantic_feasibility": {str(rank): {"L_S": True, "L_I": True, "L_C": True} for rank in (0, 1, 2)},
        "lexicographic_trace": [{"rank": rank} for rank in (0, 1, 2)],
        "selected_rank": selected_rank,
        "selection_count": 1,
    }
    selection_path = attempt / "selection.json"
    selection_path.write_text(json.dumps(selection_document), encoding="utf-8")
    selection_bytes = selection_path.read_bytes()
    executor._write_stage_recovery(root, attempt, "selection", selection_path, endpoint_paths, role_indices, ledger, manifest, attempt_id)
    monkeypatch.setattr(executor, "_evaluate_domain", lambda *args, **kwargs: pytest.fail("recovered N3_SELECT must not evaluate again"))
    selection_marker = executor._load_stage_recovery(root, attempt, "selection", selection_path, endpoint_paths, {"roles": role_indices}, manifest, attempt_id)
    assert selection_path.read_bytes() == selection_bytes
    assert selection_marker["ledger_state"]["phase"] == "SELECT"

    resumed = executor.SampleReadLedger()
    resumed.mark_backing_loaded()
    resumed.merge(selection_marker["ledger_state"])
    assert resumed.enter_confirmation(selection_path) == selected_rank
    confirm_rows = role_indices["N3_CONFIRM"]
    resumed.record_materialized("N3_CONFIRM", confirm_rows, rank=selected_rank, purpose="selected_candidate")
    resumed.record_evaluated("N3_CONFIRM", confirm_rows, rank=selected_rank, purpose="selected_candidate")
    if selected_rank != 0:
        resumed.record_materialized("N3_CONFIRM", confirm_rows, rank=0, purpose="shared_baseline_reference")
        resumed.record_evaluated("N3_CONFIRM", confirm_rows, rank=0, purpose="shared_baseline_reference")
    resumed.verify_confirmation_evaluations(confirm_rows.numel())
    confirmation_document = {
        "schema": "n3_low_rank_model_order_confirmation.v1",
        "contract_sha256": executor.CONTRACT_SHA256,
        "protocol_sha256": executor.PROTOCOL_SHA256,
        "domain": "N3_CONFIRM",
        "indices_sha256": executor.tensor_sha256(confirm_rows),
        "evaluations": 1,
        "selected_rank": selected_rank,
        "passed": False,
    }
    confirmation_path = attempt / "confirmation.json"
    confirmation_path.write_text(json.dumps(confirmation_document), encoding="utf-8")
    confirmation_bytes = confirmation_path.read_bytes()
    confirmation_endpoints = {0: endpoint_paths[0]}
    if selected_rank > 0:
        confirmation_endpoints[selected_rank] = endpoint_paths[selected_rank]
    executor._write_stage_recovery(root, attempt, "confirmation", confirmation_path, confirmation_endpoints, role_indices, resumed, manifest, attempt_id)
    confirmation_marker = executor._load_stage_recovery(
        root,

def test_recovery_roundtrip_checks_the_25_step_restart_and_next_update(
    tmp_path: Path, monkeypatch: pytest.MonkeyPatch
) -> None:
    class TinyModel:
        def __init__(self) -> None:
            self.rank = 1
            self.B = torch.nn.Parameter(torch.tensor([0.1, -0.2]))
            self.A = torch.nn.Parameter(torch.tensor([[0.3], [-0.4]]))

        def state_dict(self) -> dict[str, torch.Tensor]:
            return {"B": self.B.detach().clone(), "A": self.A.detach().clone()}

        def train(self) -> TinyModel:
            return self

        def load_trainable_state(self, state: dict[str, torch.Tensor]) -> None:
            with torch.no_grad():
                self.B.copy_(state["n3_correction.B"])
                self.A.copy_(state["n3_correction.A"])

    rng_state = {"cpu": torch.arange(4, dtype=torch.uint8), "cuda": [], "cuda_available": False}
    monkeypatch.setattr(executor, "_rng_snapshot", lambda: copy.deepcopy(rng_state))
    monkeypatch.setattr(executor, "_restore_rng", lambda state: None)
    monkeypatch.setattr(executor, "_new_model", lambda *args, **kwargs: TinyModel())
    monkeypatch.setattr(executor, "_optimizer", lambda model: torch.optim.Adam([model.B, model.A], lr=executor.LEARNING_RATE))

    def apply_update(model: TinyModel, optimizer: torch.optim.Adam, images: torch.Tensor, labels: torch.Tensor) -> dict[str, float]:
        optimizer.zero_grad(set_to_none=True)
        loss = (model.B.square().sum() + model.A.square().sum()) * images.sum().add(1.0)
        loss.backward()
        optimizer.step()
        return {"loss": float(loss.detach()), "label_sum": float(labels.sum())}

    monkeypatch.setattr(executor, "_apply_update", apply_update)
    model = TinyModel()
    optimizer = executor._optimizer(model)
    sampler = executor.N3FitSampler(_fit_rows())
    ledger = executor.SampleReadLedger()
    ledger.mark_backing_loaded()
    history = []
    images = torch.arange(2560 * 2, dtype=torch.float32).reshape(2560, 2) / 1000.0
    for _ in range(executor.RECOVERY_INTERVAL):
        indices, labels = sampler.next_batch()
        history.append(apply_update(model, optimizer, images.index_select(0, indices), labels))
    authorities = {
        "checkpoint": {},
        "indices_by_class": _fit_rows(),
        "images": images,
    }
    result = executor._verify_recovery_roundtrip(
        authorities=authorities,
        model=model,
        optimizer=optimizer,
        sampler=sampler,
        history=history,
        manifest=_source_manifest("roundtrip-source"),
        device=torch.device("cpu"),
        ledger=ledger,
    )
    assert result["passed"] is True
    assert result["state_equal"] is True
    assert result["optimizer_equal"] is True
    assert result["cursor_equal"] is True
    assert result["rng_equal"] is True
    assert result["history_equal"] is True
    assert result["next_update_equal"] is True

def test_preflight_temporary_state_roundtrip_compares_the_next_update(
    tmp_path: Path, monkeypatch: pytest.MonkeyPatch
) -> None:
    class TinyModel:
        def __init__(self) -> None:
            self.rank = 1
            self.B = torch.nn.Parameter(torch.tensor([0.1, -0.2]))
            self.A = torch.nn.Parameter(torch.tensor([[0.3], [-0.4]]))

        def state_dict(self) -> dict[str, torch.Tensor]:
            return {"B": self.B.detach().clone(), "A": self.A.detach().clone()}

        def train(self) -> TinyModel:
            return self

        def load_trainable_state(self, state: dict[str, torch.Tensor]) -> None:
            with torch.no_grad():
                self.B.copy_(state["n3_correction.B"])
                self.A.copy_(state["n3_correction.A"])

    monkeypatch.chdir(tmp_path)
    monkeypatch.setattr(executor, "_new_model", lambda *args, **kwargs: TinyModel())
    monkeypatch.setattr(
        executor,
        "_optimizer",
        lambda model: torch.optim.Adam([model.B, model.A], lr=executor.LEARNING_RATE),
    )

    def apply_update(
        model: TinyModel,
        optimizer: torch.optim.Adam,
        images: torch.Tensor,
        labels: torch.Tensor,
    ) -> dict[str, float]:
        optimizer.zero_grad(set_to_none=True)
        loss = (model.B.square().sum() + model.A.square().sum()) * images.sum().add(1.0)
        loss.backward()
        optimizer.step()
        return {"loss": float(loss.detach()), "label_sum": float(labels.sum())}

    monkeypatch.setattr(executor, "_apply_update", apply_update)
    model = TinyModel()
    optimizer = executor._optimizer(model)
    sampler = executor.N3FitSampler(_fit_rows())
    ledger = executor.SampleReadLedger()
    ledger.mark_backing_loaded()
    images = torch.arange(2560 * 2, dtype=torch.float32).reshape(2560, 2) / 1000.0
    history = []
    for _ in range(executor.PREFLIGHT_UPDATE_COUNT):
        indices, labels = sampler.next_batch()
        history.append(apply_update(model, optimizer, images.index_select(0, indices), labels))

    result = executor._verify_preflight_state_roundtrip(
        authorities={"checkpoint": {}, "indices_by_class": _fit_rows(), "images": images},
        model=model,
        optimizer=optimizer,
        sampler=sampler,
        history=history,
        manifest=_source_manifest("preflight-roundtrip-source"),
        device=torch.device("cpu"),
        ledger=ledger,
    )

    assert result["passed"] is True
    assert result["state_equal"] is True
    assert result["optimizer_equal"] is True
    assert result["cursor_equal"] is True
    assert result["rng_equal"] is True
    assert result["history_equal"] is True
    assert result["next_batch_equal"] is True
    assert result["next_update_equal"] is True
    assert result["temporary_state_removed"] is True

def test_preflight_no_grad_forward_does_not_leave_inference_transfer_kernel_for_backward() -> None:
    clear_production_transfer_cache()
    plane = PlaneSpec("source", (8, 8), (2e-6, 2e-6), 532e-9, 0.0, "unit")
    phase = torch.randn((1, 1, 8, 8), dtype=torch.float32, requires_grad=True)
    base = torch.ones((1, 1, 8, 8), dtype=torch.complex64)

    # This reproduces the previous preflight behavior: inference-mode creates
    # an inference tensor in the propagation cache, which backward cannot save.
    with torch.inference_mode():
        angular_spectrum_propagate(
            ComplexField(base, plane, OperationProvenance("source", None, None)),
            1e-3,
            "inference-cache",
        )
    assert next(iter(_PRODUCTION_TRANSFER_CACHE.values())).is_inference()
    with pytest.raises(RuntimeError, match="Inference tensors cannot be saved"):
        field = ComplexField(base * torch.exp(1j * phase), plane, OperationProvenance("source", None, None))
        angular_spectrum_propagate(field, 1e-3, "contaminated-backward").tensor.abs().square().sum().backward()

    clear_production_transfer_cache()
    with torch.no_grad():
        angular_spectrum_propagate(
            ComplexField(base, plane, OperationProvenance("source", None, None)),
            1e-3,
            "no-grad-cache",
        )
    field = ComplexField(base * torch.exp(1j * phase), plane, OperationProvenance("source", None, None))
    output = angular_spectrum_propagate(field, 1e-3, "valid-backward")
    output.tensor.abs().square().sum().backward()
    assert phase.grad is not None and bool(torch.isfinite(phase.grad).all().item())
    assert not next(iter(_PRODUCTION_TRANSFER_CACHE.values())).is_inference()
    clear_production_transfer_cache()

@pytest.mark.parametrize("dtype,atol", [(torch.float32, 1.0e-7), (torch.float64, 1.0e-12)])
def test_deterministic_cmos_view_matches_native_cpu_pooling(dtype: torch.dtype, atol: float) -> None:
    values = torch.linspace(-0.1, 1.1, 200 * 200, dtype=dtype).reshape(1, 1, 200, 200)
    values[0, 0, 0, :4] = torch.tensor([-0.1, 0.0, 1.0, 1.1], dtype=dtype)
    upstream = torch.linspace(-1.0, 1.0, 28 * 28, dtype=dtype).reshape(1, 1, 28, 28)

    native_input = values.clone().requires_grad_()
    native_view = executor.structural_exec.prior._cmos_view(native_input)
    native_gradient = torch.autograd.grad((native_view * upstream).sum(), native_input)[0]

    custom_input = values.clone().requires_grad_()
    custom_view = executor._deterministic_cmos_view(custom_input)
    custom_gradient = torch.autograd.grad((custom_view * upstream).sum(), custom_input)[0]

    assert torch.equal(custom_view, native_view)
    torch.testing.assert_close(custom_gradient, native_gradient, rtol=0.0, atol=atol)
    assert bool(torch.isfinite(custom_gradient).all().item())

    original_source = _source_manifest("old-source", "old-manifest")
    first_control = executor._preflight_identity(tmp_path, original_source)
    first = executor._write_preflight_failure(
        tmp_path,
        first_control,
        executor.N3ExecutionError("first harness failure"),
        "N3_STRICT_PREFLIGHT",
        manifest=original_source,
        runtime_provenance={"status": "FAILED", "failed_stage": "CUDA_KERNEL_SMOKE"},
    )
    assert first["decision"] == executor.PREFLIGHT_RECHECK_ALLOWED
    failure_path = tmp_path / executor.RUNS_ROOT / "preflight_failure_01" / "failure.json"
    failure_bytes = failure_path.read_bytes()
    original_failure = json.loads(failure_bytes)
    assert original_failure["execution_source_identity"] == executor._source_identity_provenance(original_source)
    assert original_failure["runtime_provenance"]["failed_stage"] == "CUDA_KERNEL_SMOKE"

    repaired_source = _source_manifest("repaired-source", "repaired-manifest")
    recheck_control = executor._preflight_identity(tmp_path, repaired_source)
    assert recheck_control.identity.execution_source_semantics_id == "repaired-source"
    assert recheck_control.authorization_id == executor.AUTHORIZATION_ID
    assert json.loads(failure_path.read_text(encoding="utf-8"))["decision"] == executor.PREFLIGHT_RECHECK_ALLOWED

        recheck_control.mark_preflight_ready(_ready_receipt("old-source"))

    )
    assert opened.execution_source_semantics_id == "repaired-source"
    assert opened.scientific_attempt_id == "n3-test-attempt-01"
    assert failure_path.read_bytes() == failure_bytes


def test_sample_read_ledger_enforces_phase_boundaries_and_reports_storage_separately(tmp_path: Path) -> None:
    ledger = executor.SampleReadLedger()
    select_indices = torch.arange(1280, dtype=torch.long)
    confirm_indices = torch.arange(2000, 3280, dtype=torch.long)

    with pytest.raises(executor.N3ExecutionError, match="backing load"):
        ledger.record_materialized("N3_FIT", torch.tensor([0]))
    ledger.mark_backing_loaded()
    with pytest.raises(executor.N3ExecutionError, match="PREFLIGHT"):
        ledger.record_materialized("N3_SELECT", select_indices, rank=0, purpose="model_order_candidate")

    ledger.enter_fit()
    ledger.record_materialized("N3_FIT", torch.tensor([0]), rank=1, purpose="fit_update")
    ledger.record_evaluated("N3_FIT", torch.tensor([0]), rank=1, purpose="fit_update")
    with pytest.raises(executor.N3ExecutionError, match="FIT"):
        ledger.record_evaluated("N3_SELECT", select_indices, rank=0, purpose="model_order_candidate")
    with pytest.raises(executor.N3ExecutionError, match="FIT"):
        ledger.record_evaluated("N3_CONFIRM", confirm_indices, rank=1, purpose="selected_candidate")
    ledger.record_fit_endpoint(1, executor.FIT_UPDATES)
    with pytest.raises(executor.N3ExecutionError, match="N3_SELECT preceded"):
        ledger.enter_selection()
    ledger.record_fit_endpoint(2, executor.FIT_UPDATES)
    ledger.enter_selection()
    for rank in (0, 1, 2):
        ledger.record_materialized("N3_SELECT", select_indices, rank=rank, purpose="model_order_candidate")
        ledger.record_evaluated("N3_SELECT", select_indices, rank=rank, purpose="model_order_candidate")
    ledger.verify_selection_evaluations(int(select_indices.numel()))

    selection_path = tmp_path / "selection.json"
    selection_path.write_text(
        json.dumps(
            {
                "domain": "N3_SELECT",
                "selection_count": 1,
                "contract_sha256": executor.CONTRACT_SHA256,
                "protocol_sha256": executor.PROTOCOL_SHA256,
                "selected_rank": 1,
            }
        ),
        encoding="utf-8",
    )
    assert ledger.enter_confirmation(selection_path) == 1
    with pytest.raises(executor.N3ExecutionError, match="selected candidate"):
        ledger.record_evaluated("N3_CONFIRM", confirm_indices, rank=2, purpose="selected_candidate")
    ledger.record_materialized("N3_CONFIRM", confirm_indices, rank=1, purpose="selected_candidate")
    ledger.record_evaluated("N3_CONFIRM", confirm_indices, rank=1, purpose="selected_candidate")
    # The frozen contract defines confirmation as a paired comparison against
    # theta_shared, so rank zero is allowed only as this named reference.
    ledger.record_materialized("N3_CONFIRM", confirm_indices, rank=0, purpose="shared_baseline_reference")
    ledger.record_evaluated("N3_CONFIRM", confirm_indices, rank=0, purpose="shared_baseline_reference")
    ledger.verify_confirmation_evaluations(int(confirm_indices.numel()))

    summary = ledger.summary()
    for role in ledger.roles:
        assert summary[role]["canonical_train_backing_loaded"] is True
    assert summary["N3_SELECT"]["role_rows_materialized"] == 3840
    assert summary["N3_SELECT"]["role_rows_evaluated"] == 3840
    assert summary["N3_CONFIRM"]["role_rows_materialized"] == 2560
    assert summary["N3_CONFIRM"]["role_rows_evaluated"] == 2560
    assert summary["phase_access"]["FIT"]["N3_SELECT"]["role_rows_evaluated"] == 0
    assert summary["phase_access"]["FIT"]["N3_CONFIRM"]["role_rows_evaluated"] == 0

def test_reconciliation_preserves_original_protocol_and_failure_artifacts() -> None:
    paths_and_hashes = (
        (ROOT / reconciler.CONTRACT_PATH, reconciler.CONTRACT_SHA256),
        (ROOT / reconciler.OLD_PROTOCOL_PATH, reconciler.OLD_PROTOCOL_SHA256),
        (ROOT / reconciler.READINESS_FAILURE_PATH, reconciler.READINESS_FAILURE_SHA256),
    )
    before = [hashlib.sha256(path.read_bytes()).hexdigest() for path, _ in paths_and_hashes]
    payload = reconciler.build_corrected_protocol(ROOT)
    after = [hashlib.sha256(path.read_bytes()).hexdigest() for path, _ in paths_and_hashes]
    assert before == after == [expected for _, expected in paths_and_hashes]
    assert payload["authority_reconciliation"]["active_for_execution"] is True
    assert payload["authority_reconciliation"]["superseded_protocol_active_for_execution"] is False
