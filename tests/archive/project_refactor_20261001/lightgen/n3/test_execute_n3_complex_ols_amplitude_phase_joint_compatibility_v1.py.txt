from ldm_project.v2.evidence import file_sha256, safe_torch_load, tensor_sha256
from scripts.research.lightgen.n3 import execute_n3_complex_ols_amplitude_phase_joint_compatibility_v1 as execution


def test_contract_and_gate_constants_are_frozen() -> None:
    assert execution.EXPECTED_BRANCH == "research/lightgen-ols-native-semantic-generation-v1"
    assert execution.EXPECTED_STARTING_HEAD == "a5c07512614944d4be6392ae29d48e0b189bea4b"
    assert execution.EXPECTED_CONTRACT_SHA256 == "52e07263343d765db13022fbcf56f74e7431a9dced16f8bfa31ca07ecc6bd726"
    assert execution.PRIMARY_METRIC == "energy_distance"
    assert execution.CALIBRATION_TRIAL_COUNT == 999
    assert execution.PER_FOLD_ALPHA == 0.01
    assert execution.FOLD_COUNT == 5


def test_seed_streams_are_deterministic_and_disjoint() -> None:
    seeds = [
        execution.contract_definition.derive_seed("candidate", fold, role, trial, bank)
        for fold in range(execution.FOLD_COUNT)
        for role, trial, bank in (
            ("CANDIDATE_MODEL_BANK", None, 0),
            ("CANDIDATE_PSEUDO_DATA", 0, None),
            ("CANDIDATE_Q_SAMPLE", 0, None),
            ("CANDIDATE_PHASE_DONOR", 0, None),
        )
    ]
    assert len(seeds) == len(set(seeds))
    assert seeds == [
        execution.contract_definition.derive_seed("candidate", fold, role, trial, bank)
        for fold in range(execution.FOLD_COUNT)
        for role, trial, bank in (
            ("CANDIDATE_MODEL_BANK", None, 0),
            ("CANDIDATE_PSEUDO_DATA", 0, None),
            ("CANDIDATE_Q_SAMPLE", 0, None),
            ("CANDIDATE_PHASE_DONOR", 0, None),
        )
    ]


def test_zero_complex_coordinate_is_rejected_fail_closed() -> None:
    zero = torch.zeros((1, execution.DIMENSION), dtype=torch.complex128)
    with pytest.raises(execution.contract_definition.InvalidHarnessError, match="exact-zero"):
        execution.contract_definition.canonicalize_existing_global_phase(zero)
    with pytest.raises(execution.contract_definition.InvalidHarnessError, match="exact-zero"):
        execution.contract_definition.compose_complex_from_amplitude_phase(
            torch.zeros((1, execution.DIMENSION), dtype=torch.float64),
            torch.zeros((1, execution.DIMENSION), dtype=torch.float64),
        )


def test_complete_row_bootstrap_is_deterministic_and_preserves_pairing() -> None:
    rows = torch.arange(400, dtype=torch.float64).reshape(4, execution.DIMENSION).to(torch.complex128)
    rows = rows + 1j * (rows + 100.0)
    first, first_indices = execution._sample_complete_rows(rows, 25, 17)
    second, second_indices = execution._sample_complete_rows(rows, 25, 17)
    assert torch.equal(first_indices, second_indices)
    assert torch.equal(first, second)
    assert torch.equal(first, rows.index_select(0, first_indices))
    assert tensor_sha256(first_indices) == tensor_sha256(second_indices)


def test_primary_aggregation_is_strict_five_of_five() -> None:
    folds = [
        {"primary": {"pass": True}}
        for _ in range(execution.FOLD_COUNT)
    ]
    aggregate = execution._aggregate_primary(folds)
    assert aggregate["passed_fold_count"] == 5
    assert aggregate["strict_five_of_five"] is True

    folds[3] = {"primary": {"pass": False}}
    aggregate = execution._aggregate_primary(folds)
    assert aggregate["passed_fold_count"] == 4
    assert aggregate["strict_five_of_five"] is False


def test_published_result_is_hash_bound_and_closed_when_available() -> None:
    repo_root = Path(__file__).resolve().parents[4]
    output_dir = repo_root / execution.OUTPUT_DIR
    report_path = output_dir / execution.REPORT_NAME
    if not report_path.is_file():
        pytest.skip("complex-OLS compatibility execution has not been published")

    report = json.loads(report_path.read_text(encoding="utf-8"))
    assert report["schema"] == execution.REPORT_SCHEMA
    assert report["result"] == "VALID_COMPLETED"
    assert report["classification"] in {
        execution.CLASSIFICATION_STABILITY,
        execution.CLASSIFICATION_SUPPORTED,
        execution.CLASSIFICATION_NOT_SUPPORTED,
    }
    assert report["contract"]["sha256"] == execution.EXPECTED_CONTRACT_SHA256
    assert report["control"]["aggregate"]["fold_count"] == execution.FOLD_COUNT
    assert report["control"]["aggregate"]["primary_metric"] == "energy_distance"
    assert report["governance"]["sealed_confirmation_consumed"] is True
    assert report["governance"]["sealed_confirmation_reused"] is False
    assert report["governance"]["training_authorized"] is False
    assert report["governance"]["formal_promotion"] is False
    assert report["governance"]["generation_readiness"] == "not_ready"

    lineage_ref = report["provenance"]["lineage_manifest"]
    lineage_path = repo_root / lineage_ref["path"]
    assert file_sha256(lineage_path) == lineage_ref["sha256"]
    lineage = json.loads(lineage_path.read_text(encoding="utf-8"))
    assert lineage["contract"]["sha256"] == execution.EXPECTED_CONTRACT_SHA256
    assert lineage["git"]["head_before"] == execution.EXPECTED_STARTING_HEAD
    assert lineage["git"]["head_after"] == execution.EXPECTED_STARTING_HEAD
    for reference in lineage["complex_materialization"]["artifact"], lineage["complex_materialization"]["manifest"], lineage["control"]["artifact"]:
        path = repo_root / reference["path"]
        assert file_sha256(path) == reference["sha256"]
    materialization_path = repo_root / lineage["complex_materialization"]["artifact"]["path"]
    payload = safe_torch_load(materialization_path, expected_sha256=file_sha256(materialization_path), map_location="cpu")
    assert payload["schema"] == execution.MATERIALIZATION_SCHEMA
    assert payload["r3_complex"].dtype == torch.complex64
    assert tuple(payload["r3_complex"].shape) == (execution.POOL_COUNT, execution.DIMENSION)
    assert payload["identity"]["q_equals_abs_r3_bitwise"] is True
