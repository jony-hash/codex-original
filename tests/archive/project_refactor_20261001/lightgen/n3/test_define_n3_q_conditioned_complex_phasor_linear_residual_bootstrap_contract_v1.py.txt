"""Targeted tests for the higher-expressivity phase contract definition."""

from __future__ import annotations

import unittest

import torch

from scripts.research.lightgen.n3 import define_n3_q_conditioned_complex_phasor_linear_residual_bootstrap_contract_v1 as contract


class HigherExpressivityPhaseContractTests(unittest.TestCase):
    def test_candidate_and_future_gate_are_frozen(self) -> None:
        self.assertEqual(contract.CANDIDATE_ID, "Q_CONDITIONED_COMPLEX_PHASOR_LINEAR_RESIDUAL_BOOTSTRAP_V1")
        self.assertEqual(contract.TASK_ID, "DEFINE_ONE_HIGHER_EXPRESSIVITY_Q_CONDITIONED_PHASE_MODEL_CONTRACT_V1")
        self.assertEqual(contract.NEXT_ACTION, "EXECUTE_N3_Q_CONDITIONED_COMPLEX_PHASOR_LINEAR_RESIDUAL_BOOTSTRAP_V1")
        self.assertEqual(contract.DESIGN_FEATURES, 101)
        self.assertEqual(contract.MODEL_BANK_COUNT, 4)
        self.assertEqual(contract.MODEL_BANK_SIZE, 200)
        self.assertEqual(contract.CALIBRATION_TRIAL_COUNT, 999)
        self.assertEqual(contract.PER_FOLD_ALPHA, 0.01)
        self.assertEqual(contract.OLS_RCOND, 1.0e-12)
        self.assertEqual(contract.PHASOR_PROJECTION_EPSILON, 1.0e-12)

    def test_complex_phasor_ols_has_fixed_intercept_and_orientation(self) -> None:
        x = torch.zeros((contract.DESIGN_FEATURES, contract.DIMENSION), dtype=torch.float64)
        x[1:, :] = torch.eye(contract.DIMENSION, dtype=torch.float64)
        target = torch.ones((contract.DESIGN_FEATURES, contract.DIMENSION), dtype=torch.complex128)
        coefficients, prediction, rank = contract._fit_complex_phasor_linear_ols(x, target)
        self.assertEqual(tuple(coefficients.shape), (contract.DESIGN_FEATURES, contract.DIMENSION))
        self.assertEqual(tuple(prediction.shape), tuple(target.shape))
        self.assertEqual(rank, contract.DESIGN_FEATURES)
        self.assertTrue(torch.isfinite(coefficients.real).all().item())
        self.assertTrue(torch.isfinite(coefficients.imag).all().item())

    def test_rank_deficiency_is_a_hard_stop(self) -> None:
        x = torch.zeros((contract.DESIGN_FEATURES, contract.DIMENSION), dtype=torch.float64)
        target = torch.ones((contract.DESIGN_FEATURES, contract.DIMENSION), dtype=torch.complex128)
        with self.assertRaises(contract.InvalidHarnessError):
            contract._fit_complex_phasor_linear_ols(x, target)

    def test_unit_projection_rejects_zero_and_keeps_complete_vector(self) -> None:
        value = torch.ones((1, contract.DIMENSION), dtype=torch.complex128) * (1.0 + 1.0j)
        projected = contract._project_unit_phasors(value)
        self.assertEqual(tuple(projected.shape), (1, contract.DIMENSION))
        self.assertTrue(torch.allclose(projected.abs(), torch.ones_like(projected.abs())))

        with self.assertRaises(contract.InvalidHarnessError):
            contract._project_unit_phasors(torch.zeros((1, contract.DIMENSION), dtype=torch.complex128))

    def test_definition_totality_has_no_scientific_or_governance_side_effect(self) -> None:
        totality = contract._contract_totality_checks()
        self.assertEqual(totality["status"], "passed")
        self.assertEqual(totality["candidate_id"], contract.CANDIDATE_ID)
        self.assertEqual(totality["rank_deficiency_probe"], "passed")
        self.assertEqual(totality["zero_or_below_epsilon_projection_probe"], "passed")
        self.assertFalse(totality["q_values_loaded"])
        self.assertFalse(totality["complex_ols_values_loaded"])
        self.assertFalse(totality["sealed_values_loaded"])
        self.assertFalse(totality["scientific_evaluation_executed"])
        self.assertFalse(totality["training_executed"])
        self.assertFalse(totality["cuda_used"])
        self.assertFalse(totality["governance_mutated"])


if __name__ == "__main__":
    unittest.main()
