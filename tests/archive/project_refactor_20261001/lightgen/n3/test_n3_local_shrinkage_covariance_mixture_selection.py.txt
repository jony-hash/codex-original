from __future__ import annotations

from pathlib import Path

import pytest
import torch

from scripts.research.lightgen.n3 import execute_n3_local_shrinkage_covariance_mixture_selection_v1 as selection
from scripts.research.lightgen.n3.execute_n3_local_shrinkage_covariance_mixture_selection_v1 import COMPONENT_INSUFFICIENT, hard_assignment_from_responsibilities, component_insufficiency_status


def test_hard_assignment_uses_lowest_canonical_index_on_tie() -> None:
    responsibilities = torch.tensor(
        [
            [0.5, 0.5, 0.0],
            [0.1, 0.7, 0.2],
            [0.0, 0.4, 0.6],
        ],
        dtype=torch.float64,
    )
    assert hard_assignment_from_responsibilities(responsibilities).tolist() == [0, 1, 2]


def test_component_insufficiency_is_fail_closed() -> None:
    assert component_insufficiency_status(1) == COMPONENT_INSUFFICIENT
    assert component_insufficiency_status(2) is None


def test_frozen_contract_lineage_loads_before_execution() -> None:
    repo_root = Path(__file__).resolve().parents[4]
    contract = selection._load_contract(repo_root)
    assert contract["classification"] == "DEFINE_N3_UNSUPERVISED_LOCAL_SHRINKAGE_COVARIANCE_MIXTURE_CONTRACT_V1_READY"
    assert contract["component_count_domain"]["candidate_K"] == [2, 4, 8]


def test_mismatched_contract_classification_is_rejected(monkeypatch: pytest.MonkeyPatch) -> None:
    repo_root = Path(__file__).resolve().parents[4]
    original = selection._read_json_exact

    def tampered_read(repo: Path, path: Path, expected_sha256: str, name: str) -> dict[str, object]:
        value = original(repo, path, expected_sha256, name)
        if path == selection.CONTRACT_PATH:
            value["classification"] = "TAMPERED_CONTRACT"
        return value

    monkeypatch.setattr(selection, "_read_json_exact", tampered_read)
    with pytest.raises(ValueError, match="contract is not ready"):
        selection._load_contract(repo_root)
