"""Focused tests for the N3 development-sampling stationarity diagnostic."""

from __future__ import annotations

import json
from pathlib import Path

import pytest
import torch

from scripts.research.lightgen.n3 import define_n3_development_sampling_stationarity_diagnostic_v1 as definition
from scripts.research.lightgen.n3 import execute_n3_development_sampling_stationarity_diagnostic_v1 as execution


def test_moment_metrics_cover_mean_std_and_radial_statistics() -> None:
    fit = torch.arange(800 * definition.DIMENSION, dtype=torch.float64).reshape(800, definition.DIMENSION)
    heldout = fit[:200].clone()
    metrics, details = execution._moment_metrics(fit, heldout)
    assert tuple(metrics) == definition.MOMENT_METRICS
    assert all(value >= 0.0 and torch.isfinite(torch.tensor(value)) for value in metrics.values())
    assert details["radial_mean_abs_difference"] >= 0.0
    assert details["radial_std_abs_difference"] >= 0.0


def test_reference_partitions_are_deterministic_and_class_balanced() -> None:
    labels = torch.arange(definition.POOL_COUNT, dtype=torch.int64).remainder(definition.CLASS_COUNT)
    heldout_first, fit_first = execution._build_reference_partitions(labels)
    heldout_second, fit_second = execution._build_reference_partitions(labels)
    assert torch.equal(heldout_first, heldout_second)
    assert torch.equal(fit_first, fit_second)
    assert tuple(heldout_first.shape) == (
        definition.PARTITION_REFERENCE_REPLICATES,
        definition.FOLD_COUNT,
        definition.HELDOUT_COUNT_PER_FOLD,
    )
    assert tuple(fit_first.shape) == (
        definition.PARTITION_REFERENCE_REPLICATES,
        definition.FOLD_COUNT,
        definition.FIT_COUNT_PER_FOLD,
    )
    for fold in range(definition.FOLD_COUNT):
        heldout_labels = labels.index_select(0, heldout_first[0, fold])
        fit_labels = labels.index_select(0, fit_first[0, fold])
        assert torch.equal(
            torch.bincount(heldout_labels, minlength=definition.CLASS_COUNT),
            torch.full((definition.CLASS_COUNT,), definition.HELDOUT_ROWS_PER_CLASS_PER_FOLD, dtype=torch.int64),
        )
        assert torch.equal(
            torch.bincount(fit_labels, minlength=definition.CLASS_COUNT),
            torch.full((definition.CLASS_COUNT,), definition.FIT_ROWS_PER_CLASS_PER_FOLD, dtype=torch.int64),
        )


def test_classification_rule_is_total_and_predeclared() -> None:
    assert execution._classification(0, False, False) == definition.CLASSIFICATION_SUPPORTED
    assert execution._classification(1, False, False) == definition.CLASSIFICATION_MIXED
    assert execution._classification(2, False, False) == definition.CLASSIFICATION_LIMITATION
    assert execution._classification(0, True, False) == definition.CLASSIFICATION_LIMITATION
    assert execution._classification(0, False, True) == definition.CLASSIFICATION_LIMITATION


def test_published_contract_replays_totality_when_available() -> None:
    path = (
        Path("outputs/20260909_v2/runs/n3_development_sampling_stationarity_diagnostic_v1_attempt_01")
        / definition.CONTRACT_NAME
    )
    if not path.is_file():
        pytest.skip("contract is published by the bounded implementation task")
    contract = json.loads(path.read_text(encoding="utf-8"))
    definition.validate_contract_payload(contract)
    assert contract["contract_totality"]["status"] == "passed"
    assert contract["source_position_contract"]["existing_authority_reused"] is True
    assert contract["finite_partition_reference_contract"]["replacement_of_official_folds"] is False
