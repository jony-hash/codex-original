from __future__ import annotations

import json
from pathlib import Path

import pytest

from ldm_project.v2.evidence import file_sha256
from scripts.research.lightgen.n3 import define_n3_empirical_marginal_fixed_r_student_t_copula_contract_v1 as contract


EXPECTED_ARTIFACT_SHA256 = "0f18dccf35758e590255fac07436eb968c866ddc11a3e19df7b02e25770be1d7"


def _artifact_path() -> Path:
    repo_root = Path(__file__).resolve().parents[4]
    return repo_root / contract.OUTPUT_DIR / contract.CONTRACT_NAME


def _load_artifact() -> dict[str, object]:
    path = _artifact_path()
    if not path.is_file():
        pytest.skip("contract artifact is published by the bounded definition task")
    return json.loads(path.read_text(encoding="utf-8"))


def test_contract_identity_and_totality_are_frozen() -> None:
    artifact = _load_artifact()
    path = _artifact_path()

    assert file_sha256(path) == EXPECTED_ARTIFACT_SHA256
    assert artifact["schema"] == contract.CONTRACT_SCHEMA
    assert artifact["task_id"] == contract.TASK_ID
    assert artifact["gate"] == contract.GATE
    assert artifact["status"] == "valid_completed"
    assert artifact["result"] == "READY"
    assert artifact["classification"] == contract.READY_CLASSIFICATION
    assert artifact["claim_classification"] == contract.CLAIM_CLASSIFICATION
    assert artifact["semantic_authority_id"] == contract.TASK_ID
    assert artifact["definition_only"] is True
    assert artifact["contract_totality"]["status"] == "passed"
    assert artifact["contract_totality"]["future_runtime_math_choices"] == []


def test_exactly_one_family_and_only_nu_changes_the_baseline() -> None:
    artifact = _load_artifact()
    family = artifact["family"]
    fit = artifact["student_t_copula_fit_contract"]

    assert family["family_id"] == contract.FAMILY_ID
    assert family["candidate_family_count"] == 1
    assert family["candidate_ids"] == [contract.FAMILY_ID]
    assert family["new_fitted_parameter"] == "nu"
    assert family["new_fitted_scalar_count_per_fold"] == 1
    assert family["changed_component"] == "copula radial law and one fitted global degrees-of-freedom scalar nu"
    assert family["unchanged_components"] == [
        "fit-only empirical marginal order statistics and inverse quantiles",
        "stable tie and interpolation convention",
        "Gaussianized-score construction used to derive R_shrunk",
        "analytic shrinkage formula and the fitting-derived matrix R_shrunk",
        "V2 development bank, fold assignment, count and class balance",
        "raw-q evaluator and all eight compatibility guardrails",
    ]
    assert fit["R_refit"] is False
    assert fit["marginal_refit_under_nu"] is False
    assert fit["fit_rows_only"] is True
    assert fit["nu_domain"] == {
        "lower": contract.NU_LOWER,
        "upper": contract.NU_UPPER,
        "closed_interval": True,
        "variance_finite": True,
        "upper_boundary_role": "finite Gaussian-limit approximation; legal fitted boundary, not infinity",
    }
    assert "log c_nu(u_i;R)=log f_t_d(x_i;nu,R)-sum_j log f_t_1(x_ij;nu)" in fit["copula_density"]

    assert optimizer["algorithm"] == "deterministic bounded golden-section maximization of ell(2+exp(eta))"

    assert optimizer["boundary_rule"].startswith("include both endpoints")

    assert optimizer["post_result_retuning"] is False

    assert numerical["device"] == "CPU only"
    assert numerical["t_cdf"]["max_iterations"] == contract.T_CDF_MAX_ITERATIONS
    assert numerical["t_cdf"]["nonconvergence"] == "INVALID_HARNESS"
    assert numerical["t_inverse_cdf"]["bracket_expansion"] is False
    assert numerical["t_inverse_cdf"]["max_iterations"] == contract.T_QUANTILE_MAX_ITERATIONS
    assert numerical["repair_policy"] == {
        "clamp": False,
        "abs": False,
        "relu": False,
        "softplus": False,
        "epsilon_repair": False,
        "adaptive_resampling": False,
        "seed_retry": False,
    }

    assert contract._run_contract_totality_checks() == artifact["contract_totality"]
