from __future__ import annotations

import json
from pathlib import Path

from ldm_project.v2.evidence import file_sha256
from scripts.research.lightgen.n3 import define_n3_copula_failure_attribution_contract_v2 as contract


def _load(path: Path) -> dict[str, object]:
    return json.loads(path.read_text(encoding="utf-8"))


def test_v2_binds_immutable_v1_and_records_delta_only_diff() -> None:
    repo_root = Path(__file__).resolve().parents[4]
    v1_path = repo_root / contract.BASE_CONTRACT
    v2_path = repo_root / contract.OUTPUT_DIR / contract.CONTRACT_NAME
    v1 = _load(v1_path)
    v2 = _load(v2_path)

    assert file_sha256(v1_path) == contract.BASE_CONTRACT_SHA256
    assert v2["schema"] == "n3_copula_failure_attribution_contract.v2"
    assert v2["semantic_authority_id"] == contract.TASK_ID
    assert v2["base_contract_sha256"] == contract.BASE_CONTRACT_SHA256
    assert v2["supersession"]["primary_scientific_semantics_changed"] is False
    assert v2["semantic_diff_validation"]["status"] == "passed"
    assert v2["semantic_diff_validation"]["primary_scientific_semantics_unchanged"] is True
    assert v1["schema"] == "n3_copula_failure_attribution_contract.v1"

    for path in contract.UNCHANGED_SEMANTIC_PATHS:
        assert contract._path_value(v1, path) == contract._path_value(v2, path)


def test_v2_freezes_negative_policy_and_secondary_bank_aggregation() -> None:
    repo_root = Path(__file__).resolve().parents[4]
    artifact = _load(repo_root / contract.OUTPUT_DIR / contract.CONTRACT_NAME)
    delta = artifact["v2_delta"]

    negative = delta["negative_nonfinite_policy"]
    assert negative["scale"] == "scale=2*abs(C)+abs(L)+abs(R)"
    assert negative["tolerance"] == "tau=1e-12*max(1.0,scale)"
    assert negative["roundoff_interval"]["condition"] == "-tau <= value < 0"
    assert negative["material_negative_interval"]["condition"] == "value < -tau"
    assert negative["nonfinite_terms_or_value"]["execution_outcome"] == "BLOCKED"

    spearman = delta["spearman_model_bank_semantics"]
    assert spearman["aggregate_model_matrix"]["source_role"] == "data-vs-model"
    assert spearman["aggregate_model_matrix"]["source_replicates"] == [0, 1, 2, 3]
    assert spearman["weighting"] == "equal arithmetic weights only; no DATA-derived, learned, best-bank, or selector weighting"
    assert len(spearman["historical_model_bank_order"]["exact_keys_by_fold"]["0"]) == 12

    spectral = delta["spectral_median_and_model_aggregation"]
    assert spectral["median_convention"]["central_indices_zero_based"] == [49, 50]
    assert spectral["median_convention"]["formula"] == "median=(lambda[49]+lambda[50])/2"
    assert spectral["aggregate_model_set"]["role"] == "data-vs-model"
    assert spectral["aggregate_model_set"]["replicates"] == [0, 1, 2, 3]
    assert spectral["aggregate_model_set"]["model_model_reference_banks_excluded"] is True


def test_v2_definition_is_metadata_only_and_does_not_load_q_tensors() -> None:
    source = Path(contract.__file__).read_text(encoding="utf-8")
    assert "import torch" not in source
    assert "torch.load" not in source
    assert "safe_torch_load" not in source
