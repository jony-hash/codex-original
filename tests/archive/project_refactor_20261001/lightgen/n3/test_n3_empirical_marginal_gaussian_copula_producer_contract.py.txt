from __future__ import annotations

from pathlib import Path

import pytest

from scripts.research.lightgen.n3 import define_n3_empirical_marginal_gaussian_copula_producer_contract_v1 as contract


def test_midrank_plotting_positions_are_deterministic_and_strictly_interior() -> None:
    positions = contract.midrank_plotting_positions([2.0, 1.0, 2.0, 3.0])
    assert positions == [0.5, 0.125, 0.5, 0.875]
    assert all(0.0 < value < 1.0 for value in positions)


def test_empirical_quantile_has_frozen_tail_and_linear_semantics() -> None:
    values = [0.0, 2.0, 6.0, 10.0]
    assert contract.piecewise_linear_empirical_quantile(values, 0.0) == 0.0
    assert contract.piecewise_linear_empirical_quantile(values, 1.0) == 10.0
    assert contract.piecewise_linear_empirical_quantile(values, 0.5) == 4.0


def test_contract_lineage_loads_without_loading_q_values() -> None:
    repo_root = Path(__file__).resolve().parents[4]
    local_contract, selection, location = contract._validate_previous_lineage(repo_root)
    assert local_contract["classification"].endswith("_READY")
    assert selection["classification"] == "N3_LOCAL_SHRINKAGE_COVARIANCE_MIXTURE_INCONCLUSIVE"
    assert location["classification"] == "N3_LOCAL_LOCATION_MIXTURE_NOT_SUPPORTED"


def test_mismatched_local_covariance_lineage_is_rejected(monkeypatch: pytest.MonkeyPatch) -> None:
    repo_root = Path(__file__).resolve().parents[4]
    original = contract._read_json_exact

    def tampered_read(repo: Path, path: Path, expected_sha256: str, name: str) -> dict[str, object]:
        value = original(repo, path, expected_sha256, name)
        if path == contract.PREVIOUS_LOCAL_COVARIANCE_SELECTION:
            value["classification"] = "TAMPERED_RESULT"
        return value

    monkeypatch.setattr(contract, "_read_json_exact", tampered_read)
    with pytest.raises(ValueError, match="previous local-covariance result changed"):
        contract._validate_previous_lineage(repo_root)
