"""Focused tests for the one-shot frozen-candidate V3 replay evidence."""

from __future__ import annotations

import json
from pathlib import Path

import pytest

from ldm_project.v2.evidence import file_sha256
from scripts.research.lightgen.n3 import execute_apply_n3_q_compatibility_evaluator_v3_to_frozen_candidates_v1 as execution


def _repo_root() -> Path:
    return Path(__file__).resolve().parents[4]


def _published() -> tuple[Path, dict, dict]:
    root = _repo_root()
    output_dir = root / execution.OUTPUT_DIR
    report_path = output_dir / execution.REPORT_NAME
    lineage_path = output_dir / execution.LINEAGE_NAME
    report = json.loads(report_path.read_text(encoding="utf-8"))
    lineage = json.loads(lineage_path.read_text(encoding="utf-8"))
    return root, report, lineage


def test_published_v3_replay_is_complete_and_preserves_project_boundaries() -> None:
    root, report, lineage = _published()
    report_path = root / execution.OUTPUT_DIR / execution.REPORT_NAME
    lineage_path = root / execution.OUTPUT_DIR / execution.LINEAGE_NAME

    assert file_sha256(report_path) == "cf6e6d62e6d2a7b35822d2d3727d08e5cbd59d51778bf794ec30ce297ffd0195"
    assert file_sha256(lineage_path) == "d49800c12ce5f7e4931aca301410d5f0c47ca301350324fdfae3a73526d19ee4"
    assert report["schema"] == execution.REPORT_SCHEMA
    assert report["status"] == "valid_completed"
    assert report["result"] == "VALID_COMPLETED"
    assert report["evaluator"]["hard_guardrails"] == ["energy_distance"]
    assert report["evaluator"]["descriptive_only_guardrails"] == list(execution.DESCRIPTIVE_METRICS)
    assert report["evaluator"]["primary_energy_distance_unchanged"] is True
    assert report["evaluator"]["threshold_semantics_unchanged"] is True
    evaluator_ref = report["evaluator"]["artifact"]
    assert file_sha256(root / evaluator_ref["path"]) == execution.V3_EVALUATOR_SHA256
    assert evaluator_ref["sha256"] == execution.V3_EVALUATOR_SHA256

    expected = {
        execution.GAUSSIAN_ID: (
            "N3_EMPIRICAL_GAUSSIAN_COPULA_V3_DEVELOPMENT_SUPPORTED",
            [True, True, True, True, True],
            "N4_Q_PRODUCER_COMPARISON_NO_CANDIDATE_SUPPORTED",
        ),
        execution.STUDENT_ID: (
            "N3_FIXED_R_STUDENT_T_COPULA_V3_NOT_SUPPORTED",
            [True, True, False, True, True],
            "N3_FIXED_R_STUDENT_T_COPULA_NOT_SUPPORTED",
        ),
    }
    assert {item["candidate_id"] for item in report["candidates"]} == set(expected)
    for candidate in report["candidates"]:
        classification, pattern, historical_classification = expected[candidate["candidate_id"]]
        assert candidate["classification"] == classification
        assert candidate["outer_classification"] == (
            execution.OUTER_SUPPORTED_CLASSIFICATION
            if all(pattern)
            else execution.OUTER_NOT_SUPPORTED_CLASSIFICATION
        )
        assert candidate["fold_pattern"] == pattern
        assert candidate["supported_fold_count"] == sum(pattern)
        assert candidate["fold_count"] == execution.FOLD_COUNT
        assert candidate["historical_v2_classification"] == historical_classification
        assert candidate["historical_v2_unchanged"] is True
        assert candidate["selection_performed"] is False
        assert len(candidate["fold_results"]) == execution.FOLD_COUNT
        for fold_result in candidate["fold_results"]:
            primary = fold_result["primary"]
            assert primary["role"] == "HARD_PRIMARY_UNCHANGED"
            assert primary["pass"] == (primary["data_value"] <= primary["q95_reference"])
            assert set(fold_result["descriptive_only"]) == set(execution.DESCRIPTIVE_METRICS)
            for metric, item in fold_result["descriptive_only"].items():
                assert metric in execution.DESCRIPTIVE_METRICS
                assert item["role"] == "DESCRIPTIVE_ONLY"
                assert item["historical_threshold_pass"] == (
                    item["data_value"] <= item["historical_threshold"]
                )
            assert fold_result["historical_v2"]["metrics_replayed_exactly"] is True
            assert fold_result["historical_v2"]["max_abs_delta_from_historical"] == 0.0
            assert fold_result["replay"] == {
                "recomputed_from_exact_frozen_data_and_model_banks": True,
                "new_sampling": False,
                "refit": False,
                "seed_changed": False,
                "evaluator_calls": 1,
            }

    assert report["development_supported_candidates"] == [execution.GAUSSIAN_ID]
    assert report["selection"] == {
        "development_supported_candidate": execution.GAUSSIAN_ID,
        "selected_producer": None,
        "posthoc_selection": False,
        "producer_freeze_performed": False,
        "next_action": "stop family search; next is producer-selection/freeze contract, no final fit/sealed",
    }
    assert report["zero_new_sampling_confirmation"] == {
        "new_sampling": False,
        "new_fit": False,
        "new_model_banks": 0,
        "rng_created_by_this_consumer": False,
        "historical_seeds_reused_as_identity_only": True,
        "model_q_banks_reused_without_resampling": True,
    }
    assert report["governance"]["selected_producer"] is None
    assert report["governance"]["training_authorized"] is False
    assert report["governance"]["formal_promotion"] is False
    assert report["governance"]["generation_readiness"] == "not_ready"
    assert report["governance"]["sealed_confirmation_consumed"] is False
    assert report["sealed_confirmation"] == {
        "status": "RESERVED_NOT_MATERIALIZED",
        "consumed": False,
        "values_loaded": False,
        "input_to_this_consumer": False,
    }
    assert report["historical_immutability"] == {
        "n4_classification_unchanged": True,
        "gaussian_v2_fold_pattern_unchanged": True,
        "student_v2_fold_pattern_unchanged": True,
        "historical_v2_reports_not_modified": True,
    }

    assert lineage["schema"] == execution.LINEAGE_SCHEMA
    assert lineage["status"] == "valid_completed"
    assert lineage["all_artifacts_write_once"] is True
    assert lineage["source_row_lineage_recorded"] is True
    assert lineage["consumer_report_is_not_self_referenced"] is True
    assert len(lineage["frozen_candidate_artifacts"]) == 20
    assert {item["role"] for item in lineage["frozen_candidate_artifacts"]} == {
        "fitted producer",
        "sampled model q",
    }
    for artifact in lineage["frozen_candidate_artifacts"]:
        assert file_sha256(root / artifact["path"]) == artifact["sha256"]
    assert lineage["evaluation"] == {
        "reused_exact_historical_fit_artifacts": True,
        "reused_exact_historical_sample_banks": True,
        "new_sampling": False,
        "refitting": False,
        "seed_change": False,
        "new_model_banks": 0,
        "new_evaluator_calls": 10,
        "sealed_input": False,
    }


def test_historical_v2_report_authorities_are_still_immutable() -> None:
    root, report, _lineage = _published()
    expected = {
        "n4_report": (
            execution.N4_REPORT_SHA256,
            "N4_Q_PRODUCER_COMPARISON_NO_CANDIDATE_SUPPORTED",
        ),
        "student_report": (
            execution.STUDENT_REPORT_SHA256,
            "N3_FIXED_R_STUDENT_T_COPULA_NOT_SUPPORTED",
        ),
    }
    for key, (sha256, classification) in expected.items():
        ref = report["input_authorities"][key]
        path = root / ref["path"]
        assert ref["sha256"] == sha256
        assert file_sha256(path) == sha256
        payload = json.loads(path.read_text(encoding="utf-8"))
        assert payload["classification"] == classification


def test_governance_alias_records_v3_support_without_downstream_authorization() -> None:
    root, _report, _lineage = _published()
    state = json.loads((root / execution.GOVERNANCE_STATE_PATH).read_text(encoding="utf-8"))
    assert state["active_objective"] == "N3_SELECTED_PRODUCER_FULL_DEVELOPMENT_FIT_COMPLETED"
    assert state["hard_blocker"] == "N3_SELECTED_PRODUCER_SEALED_CONFIRMATION_CONTRACT_REQUIRED"
    alias = state["evidence_aliases"]["N3_Q_COMPATIBILITY_V3_FROZEN_CANDIDATE_REPLAY_V1"]
    assert alias["sha256"] == "cf6e6d62e6d2a7b35822d2d3727d08e5cbd59d51778bf794ec30ce297ffd0195"
    assert alias["lineage_manifest_sha256"] == "d49800c12ce5f7e4931aca301410d5f0c47ca301350324fdfae3a73526d19ee4"
    assert alias["supported_candidates"] == [execution.GAUSSIAN_ID]
    assert alias["selected_producer"] is None
    assert alias["training_authorized"] is False
    assert alias["formal_promotion"] is False
    assert alias["generation_readiness"] == "not_ready"
    assert alias["sealed_confirmation_consumed"] is False


def test_frozen_lineage_accepts_valid_binding_and_rejects_stale_hash() -> None:
    root, report, _lineage = _published()
    historical_lineage_ref = report["input_authorities"]["n4_lineage"]
    historical_lineage = json.loads(
        (root / historical_lineage_ref["path"]).read_text(encoding="utf-8")
    )
    artifact = historical_lineage["artifacts"][0]
    execution._lineage_has_artifact(historical_lineage, artifact, "published frozen artifact")

    stale = dict(artifact)
    stale["sha256"] = "0" * 64
    with pytest.raises(execution.InvalidHarnessError, match="does not bind frozen artifact"):
        execution._lineage_has_artifact(historical_lineage, stale, "stale frozen artifact")

    with pytest.raises(execution.InvalidHarnessError, match="hash changed"):
        execution._read_json_exact(root, execution.V3_EVALUATOR_PATH, "0" * 64, "stale V3 evaluator")
