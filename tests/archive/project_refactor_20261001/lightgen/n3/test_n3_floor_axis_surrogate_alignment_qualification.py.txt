def _metrics(value: float = 2.0) -> dict[str, float]:
    return {
        "soft_floor_macro": value,
        "hard_floor_macro": value,
        "hard_floor_class8": value,
        "hard_floor_class1": value,
        "L_struct_336_macro": value,
        "nonfloor_macro": value,
        "hard_logrho_macro": value,
    }


def _controls() -> tuple[dict[str, object], dict[str, object]]:
    optical = {
        "passed": True,
        "reconstruction_mse_before": 0.2,
        "reconstruction_mse_after": 0.2,
        "raw_cmos_finite": True,
        "nonconstant_fraction": 1.0,
        "zero_image_count": 0,
    }
    semantic = {name: 0.2 for name in ("L_S", "L_I", "L_C")}
    return optical, qualification._semantic_control(semantic, semantic)


def test_v2_is_exact_immutable_v1_overlay() -> None:
    v1_path = ROOT / qualification.V1_PATH
    v2_path = ROOT / qualification.V2_PATH
    v1_bytes = v1_path.read_bytes()
    v2 = json.loads(v2_path.read_text(encoding="utf-8"))
    assert hashlib.sha256(v1_bytes).hexdigest() == qualification.V1_SHA256
    assert hashlib.sha256(v2_path.read_bytes()).hexdigest() == qualification.V2_SHA256
    assert v2["supersession"]["v1_sha256"] == qualification.V1_SHA256
    assert v2["supersession"]["V1_edited"] is False
    assert v2["inherited_scientific_definitions"]["frozen_floor_indices"] == [74, 165, 256, 285]
    assert v2["official_full_hard_criterion"]["qualification_control"] == "macro official 337D log-rho"
    assert v2["official_full_hard_criterion"]["raw_or_standardized_full_hard_distance"].startswith("SECONDARY_DIAGNOSTIC_ONLY")

def test_preflight_loads_runtime_and_checks_noop_rollback_before_attempt(tmp_path: Path, monkeypatch: pytest.MonkeyPatch) -> None:
    events: list[str] = []

    class FakeModule:
        def __init__(self, name: str):
            self.name = name
            self.training = True
            self.parameters_list: list[torch.nn.Parameter] = []

        def eval(self) -> None:
            self.training = False
            events.append(f"eval:{self.name}")

        def parameters(self) -> list[torch.nn.Parameter]:
            return self.parameters_list

    class FakeOptimizer:
        def __init__(self, model: FakeModel):
            self.state = {}
            self.param_groups = [{
                "params": [model.B, model.A],
                "lr": qualification.n3.LEARNING_RATE,
                "betas": qualification.n3.ADAM_BETAS,
                "eps": qualification.n3.ADAM_EPS,
                "weight_decay": qualification.n3.WEIGHT_DECAY,
                "amsgrad": False,
            }]

        def state_dict(self) -> dict[str, object]:
            return {"state": {}, "param_groups": []}

    class FakeSampler:
        def __init__(self, _indices: object):
            self.cursor = 0

    class FakeModel:
        def __init__(self):
            self.B = torch.nn.Parameter(torch.ones(1))
            self.A = torch.nn.Parameter(torch.ones(1))

        def train(self) -> FakeModel:
            events.append("model")
            return self

        def named_parameters(self) -> list[tuple[str, torch.nn.Parameter]]:
            return [("B", self.B), ("A", self.A)]

        def state_dict(self) -> dict[str, torch.Tensor]:
            return {"B": self.B.detach().clone(), "A": self.A.detach().clone()}

    def runtime_snapshot(_model: FakeModel) -> dict[str, object]:
        events.append("snapshot")
        return {"model": _model.state_dict(), "training": True}

    monkeypatch.setattr(qualification.n3, "_new_model", lambda *_args: FakeModel())
    monkeypatch.setattr(qualification.n3, "_optimizer", lambda model: FakeOptimizer(model))
    monkeypatch.setattr(qualification.n3, "N3FitSampler", FakeSampler)
    monkeypatch.setattr(
        qualification.n3,
        "_load_evaluators",
        lambda *_args: (FakeModule("S"), FakeModule("I"), FakeModule("C")),
    )
    monkeypatch.setattr(
        qualification.n3,
        "initialize_factors",
        lambda *_args, **_kwargs: (torch.ones(1), torch.ones(1)),
    )
    monkeypatch.setattr(qualification, "_load_reference_and_mean", lambda *_args: ({}, torch.ones(336), torch.ones(336)))
    monkeypatch.setattr(qualification.n3, "_runtime_snapshot", runtime_snapshot)
    monkeypatch.setattr(qualification.n3, "_tensor_state_sha256", lambda _state: "state-sha")
    monkeypatch.setattr(qualification, "_evaluator_grad_snapshot", lambda _evaluators: {"S": [], "I": [], "C": []})
    monkeypatch.setattr(qualification.n3, "_rng_snapshot", lambda: {"rng": 1})
    monkeypatch.setattr(qualification.n3, "_rng_equal", lambda _left, _right: True)
    monkeypatch.setattr(qualification.n3, "_restore_rng", lambda _state: events.append("rng-restore"))

    def rollback(*_args: object) -> dict[str, bool]:
        events.append("rollback")
        return {"passed": True, "model_runtime_exact": True, "optimizer_exact": True, "rng_exact": True}

    monkeypatch.setattr(qualification, "_rollback_transaction", rollback)
    previous_mean, previous_scale = qualification.n3._ACTIVE_MEAN, qualification.n3._ACTIVE_SCALE
    prepared = qualification._prepare_runtime_transaction(
        tmp_path,
        {"checkpoint": {}, "indices_by_class": {}},
        {},
        torch.device("cpu"),
    )

    assert prepared["rollback_probe"]["passed"] is True
    assert prepared["sampler"].cursor == 0
    assert prepared["optimizer"].state == {}
    assert events.index("model") < events.index("rollback")
    assert not any(event.startswith("forward:") or event == "optimizer-step" for event in events)
    assert qualification.n3._ACTIVE_MEAN is previous_mean
    assert qualification.n3._ACTIVE_SCALE is previous_scale

def test_fixed_population_selection_is_deterministic_disjoint_and_role_safe() -> None:
    labels = torch.arange(10, dtype=torch.long).repeat_interleave(2050)
    train_indices = torch.arange(labels.numel(), dtype=torch.long)
    one_per_class = torch.tensor([digit * 2050 + 17 for digit in range(10)], dtype=torch.long)
    roles = {
        "N3_FIT": one_per_class,
        "N3_SELECT": torch.empty(0, dtype=torch.long),
        "N3_CONFIRM": torch.empty(0, dtype=torch.long),
    }
    reference = torch.tensor([digit * 2050 + 20 for digit in range(10)], dtype=torch.long)
    first = qualification._select_fixed_populations(
        labels, train_indices, roles, [], torch.empty(0, dtype=torch.long), torch.empty(0, dtype=torch.long), reference
    )
    second = qualification._select_fixed_populations(
        labels, train_indices, roles, [], torch.empty(0, dtype=torch.long), torch.empty(0, dtype=torch.long), reference
    )
    assert torch.equal(first["qualification_indices"], second["qualification_indices"])
    assert torch.equal(first["future_paired_evaluation_indices"], second["future_paired_evaluation_indices"])
    assert first["qualification_indices"].numel() == 10_240
    assert first["future_paired_evaluation_indices"].numel() == 10_240
    assert not set(first["qualification_indices"].tolist()).intersection(roles["N3_FIT"].tolist())
    assert not set(first["qualification_indices"].tolist()).intersection(first["future_paired_evaluation_indices"].tolist())
    assert first["reference_independence_claim"] is False

def test_preflight_failure_allows_one_recheck_then_requires_root_cause_review(tmp_path: Path) -> None:
    source = {
        "branch": "test-branch",
        "head": "0" * 40,
        "dirty": True,
        "source_manifest_id": "source-id",
        "execution_source_semantics_id": "semantics-id",
        "dirty_patch_digest": "1" * 64,

    first = qualification._write_preflight_failure(
        tmp_path, RuntimeError("same stable failure"), source=source, read_ledger=ledger, elapsed=1.25
    )
    second = qualification._write_preflight_failure(
        tmp_path, RuntimeError("same stable failure"), source=source, read_ledger=ledger, elapsed=1.5
    )

    assert first["decision"] == "RECHECK_ALLOWED"
    assert second["decision"] == "ROOT_CAUSE_REVIEW_REQUIRED"
    assert first["scientific_attempt_created"] is second["scientific_attempt_created"] is False
    assert first["persistent_update_count"] == second["persistent_update_count"] == 0
    assert (tmp_path / qualification.RUNS_ROOT / "preflight_failure_01.json").is_file()
    assert (tmp_path / qualification.RUNS_ROOT / "preflight_failure_02.json").is_file()

def test_stale_authorization_is_rejected_before_train_authorities_or_attempt(tmp_path: Path, monkeypatch: pytest.MonkeyPatch) -> None:
    monkeypatch.setattr(qualification, "RUNS_ROOT", Path("runs"))
    monkeypatch.setattr(qualification, "AUTHORIZATION_PATH", Path("runs/authorization.json"))
    monkeypatch.setattr(qualification, "PREFLIGHT_DIR", Path("runs/preflight_01"))
    monkeypatch.setattr(qualification, "PREFLIGHT_FAILURE_PATH", Path("runs/preflight_failure_01.json"))
    monkeypatch.setattr(qualification, "ATTEMPT_DIR", Path("runs/attempt_01"))
    source = {
        "branch": "test-branch",
        "head": "0" * 40,
        "dirty": False,
        "source_manifest_id": "source-id",
        "execution_source_semantics_id": "semantics-id",
        "dirty_patch_digest": "1" * 64,
    }
    payload = {
        "schema": "n3_floor_axis_surrogate_alignment_execution_authorization.v1",
        "authorization_id": qualification.AUTHORIZATION_ID,
        "contract_v1_sha256": qualification.V1_SHA256,
        "contract_v2_sha256": "stale-contract-sha",
        "n3_contract_sha256": qualification.n3.CONTRACT_SHA256,
        "active_fit_protocol_sha256": qualification.n3.PROTOCOL_SHA256,
        "accepted_step9000_sha256": qualification.n3.START_CHECKPOINT_SHA256,
        "frozen_structural_reference_sha256": qualification.n3.STRUCTURAL_REFERENCE_SHA256,
        "surrogate_normalization_sha256": qualification.n3.SURROGATE_NORMALIZATION_SHA256,
        "source_manifest_id": source["source_manifest_id"],
        "execution_source_semantics_id": source["execution_source_semantics_id"],
    }
    auth_path = tmp_path / qualification.AUTHORIZATION_PATH
    auth_path.parent.mkdir(parents=True)
    auth_path.write_text(json.dumps(payload), encoding="utf-8")
    monkeypatch.setattr(
        qualification,
        "_authority",
        lambda _root: ({}, {"qualification_execution_authorization": {"id": qualification.AUTHORIZATION_ID}}, {}, object()),
    )
    monkeypatch.setattr(qualification, "_source_manifest", lambda _root: source)
    train_load_called = False

    def reject_train_load(*_args: object, **_kwargs: object) -> None:
        nonlocal train_load_called
        train_load_called = True
        raise AssertionError("TRAIN authority load must follow valid authorization binding")

    monkeypatch.setattr(qualification.n3, "_load_authorities", reject_train_load)
    with pytest.raises(qualification.QualificationError, match="strict preflight RECHECK_ALLOWED"):
        qualification.execute(repo_root=tmp_path)

    assert train_load_called is False
    assert not (tmp_path / qualification.PREFLIGHT_DIR).exists()
    assert not (tmp_path / qualification.ATTEMPT_DIR).exists()
    failure = json.loads((tmp_path / qualification.PREFLIGHT_FAILURE_PATH).read_text(encoding="utf-8"))
    assert failure["scientific_attempt_created"] is False
    assert failure["persistent_update_count"] == 0
