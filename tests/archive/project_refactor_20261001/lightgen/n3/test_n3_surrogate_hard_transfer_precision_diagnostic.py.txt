def test_fixed_population_is_hash_deterministic_and_uses_reference_disjoint_rows_when_possible() -> None:
    labels = torch.arange(60_000, dtype=torch.long).remainder(10)
    train = torch.arange(55_000, dtype=torch.long)
    roles = {"N3_FIT": torch.arange(0, 1_000, dtype=torch.long)}
    historical = torch.arange(1_000, 1_100, dtype=torch.long)
    prior = torch.arange(2_000, 2_100, dtype=torch.long)
    reference = torch.arange(3_000, 4_000, dtype=torch.long)

    first, details = precision.select_precision_indices(train, labels, roles, historical, prior, reference)
    second, _ = precision.select_precision_indices(train, labels, roles, historical, prior, reference)

    assert torch.equal(first, second)
    assert first.shape == (10_240,)
    assert torch.equal(torch.bincount(labels[first], minlength=10), torch.full((10,), 1_024))
    assert details["reference_disjoint_population_used"] is True
    assert details["reference_overlap_rows"] == 0
    assert not torch.isin(first, torch.cat((*roles.values(), historical, prior))).any()

    labels = torch.arange(60_000, dtype=torch.long).remainder(10)

    reference = torch.cat([torch.arange(digit, 49_000 + digit, 10) for digit in range(10)])

    indices, details = precision.select_precision_indices(

        torch.empty(0, dtype=torch.long),

    )

    assert indices.shape == (10_240,)
    assert details["reference_disjoint_population_used"] is False
    assert min(details["reference_disjoint_candidate_count_by_class"]) < 1_024
    assert details["reference_overlap_rows"] > 0
    assert details["reference_overlap_limitation"]

def test_stratified_bootstrap_and_primary_classification_rules_are_preregistered() -> None:
    labels = torch.arange(10).repeat_interleave(1_024)
    delta = torch.ones(10_240, dtype=torch.float64)
    bootstrap = precision._bootstrap_delta(labels, delta)
    coupling_gap = {"all": {"spearman_primary": -0.1, "strict_sign_concordance_fraction": 0.4, "p_hard_improvement_given_proxy_improvement": 0.4}}
    coupling_aligned = {"all": {"spearman_primary": 0.3, "strict_sign_concordance_fraction": 0.65, "p_hard_improvement_given_proxy_improvement": 0.7}}

    assert bootstrap["estimate"] == 1.0
    assert bootstrap["ci95"] == {"lower": 1.0, "upper": 1.0}
    assert precision._primary_classification(bootstrap, bootstrap, coupling_gap)["classification"] == "N3_SURROGATE_TO_HARD_TRANSFER_SUPPORTED"
    uncertain_hard = {**bootstrap, "estimate": 0.01, "ci95": {"lower": -0.02, "upper": 0.04}}
    assert precision._primary_classification(bootstrap, uncertain_hard, coupling_gap)["classification"] == "N3_SURROGATE_HARD_ALIGNMENT_GAP_SUPPORTED"
    assert precision._primary_classification(bootstrap, uncertain_hard, coupling_aligned)["classification"] == "N3_SURROGATE_HARD_TRANSFER_PRECISION_UNRESOLVED"

def test_fixed_reference_distance_decomposition_reconstructs_actual_delta() -> None:
    references = torch.zeros((2, 337), dtype=torch.float64)
    references[1, 0] = 2.0
    query0 = torch.zeros((2, 337), dtype=torch.float64)
    query0[1, 0] = 0.5
    query2 = torch.zeros((2, 337), dtype=torch.float64)
    query2[1, 0] = 1.0
    nearest0 = torch.tensor([0, 0])
    nearest2 = torch.tensor([0, 1])
    rank0_distance = precision._distance(query0, references, nearest0)
    rank2_distance = precision._distance(query2, references, nearest2)
    rank0 = {"standardized_signature": query0, "nearest_reference_index": nearest0, "distance": rank0_distance}
    rank2 = {"standardized_signature": query2, "nearest_reference_index": nearest2, "distance": rank2_distance}

    result = precision._distance_decomposition(rank0, rank2, {"standardized_signature": references})

    assert result["checks"]["fixed_reference_components_reconstruct_actual_distance_change"]["passed"] is True
    assert torch.allclose(result["feature_motion_component"] + result["reference_reassignment_component"], result["actual_total_distance_change"], rtol=1e-9, atol=1e-10)

def test_historical_and_nested_source_identity_normalize_to_same_form() -> None:
    source_identity = {
        "branch": "research/lightgen",
        "head": "500045d6a92843c81ffeda010e01351f559e453b",
        "dirty": True,
        "source_manifest_id": "a" * 64,
        "execution_source_semantics_id": "abd800346b0738206e5034ae3acf0d1415de52da39449e12a0dded6b2f6ded3a",
        "dirty_patch_digest": "b" * 64,
    }
    source = source_identity | {"config_identity": {"diagnostic_indices_sha256": "c" * 64}, "schema": "execution_source_manifest.v1"}
    historical = precision._normalise_execution_source_identity(
        source,
        semantic_aliases=(
            ("attempt_01_top_level", source_identity["execution_source_semantics_id"]),
            ("historical_ready_receipt", source_identity["execution_source_semantics_id"]),
        ),
        expected_semantics_id=source_identity["execution_source_semantics_id"],
    )
    nested = precision._normalise_execution_source_identity(
        source,
        nested_source_identity=source_identity,
        semantic_aliases=(("execution_manifest", source_identity["execution_source_semantics_id"]),),
    )

    assert historical == nested == source_identity


def test_source_identity_normalization_rejects_conflicting_manifest_and_wrong_expected_id() -> None:
    source = {
        "branch": "research/lightgen",
        "head": "500045d6a92843c81ffeda010e01351f559e453b",
        "dirty": True,
        "source_manifest_id": "a" * 64,
        "execution_source_semantics_id": "b" * 64,
        "dirty_patch_digest": "c" * 64,
    }
    conflicting_manifest = {**source, "source_manifest_id": "d" * 64}

    with pytest.raises(precision.PrecisionDiagnosticError, match="source identity conflicts"):
        precision._normalise_execution_source_identity(source, source_aliases=(("READY manifest", conflicting_manifest),))
    with pytest.raises(precision.PrecisionDiagnosticError, match="expected historical semantics ID"):
        precision._normalise_execution_source_identity(
            source,
            expected_semantics_id="e" * 64,
        )


@pytest.mark.parametrize(
    ("source", "semantic_aliases", "nested", "expected"),
    [
        ({"execution_source_semantics_id": "a" * 64}, (), None, None),

def test_output_filesystem_probe_rejects_insufficient_capacity_before_writing(tmp_path, monkeypatch) -> None:
    required_bytes = precision._minimum_output_free_bytes()
    monkeypatch.setattr(precision.shutil, "disk_usage", lambda _path: SimpleNamespace(free=required_bytes - 1))

        precision._output_filesystem_probe(tmp_path)

    assert list(tmp_path.iterdir()) == []

def test_invalid_harness_closeout_updates_heartbeat_to_terminal_state(tmp_path) -> None:
    class _Control:
        state = RUN_SCIENTIFIC_ACTIVE

        def invalid_harness(self) -> None:
            self.state = RUN_INVALID_HARNESS

    attempt = tmp_path / "attempt_01"
    attempt.mkdir()
    heartbeat = HeartbeatWriter(attempt, "invalid-run", total_steps=10, interval_seconds=0.0)
    heartbeat.update(state=RUN_SCIENTIFIC_ACTIVE, phase="STATISTICS", step=5, force=True)
    control = _Control()

    precision._mark_incomplete(attempt, control, OSError("injected artifact persistence failure"), heartbeat)

    latest = json.loads(heartbeat.latest_status_path.read_text(encoding="utf-8"))
    status = json.loads((attempt / "status.json").read_text(encoding="utf-8"))
    progress = (attempt / "progress.jsonl").read_text(encoding="utf-8").splitlines()
    assert control.state == RUN_INVALID_HARNESS
    assert latest["state"] == latest["phase"] == RUN_INVALID_HARNESS
    assert latest["step"] == 5
    assert status["state"] == status["phase"] == RUN_INVALID_HARNESS
    assert json.loads(progress[-1])["state"] == RUN_INVALID_HARNESS
    failure = json.loads((attempt / "diagnostic_failure.json").read_text(encoding="utf-8"))
    assert failure["classification"] is None
    assert failure["error_type"] == "OSError"
    assert not (attempt / "report.json").exists()
